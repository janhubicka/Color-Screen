34/* Return MTF of circular blur.
35 FREQ is the spatial frequency.
36 BLUR_CIRCLE_DIAMETER is the diameter of the blur circle.
37 This is model for small defocus that does not seem to work that well.
38 See Hopkins model below. */
39double
40defocus_mtf (double freq, double blur_circle_diameter)
41{
42 /* If perfectly in focus or frequency is almost 0, return 1 */
43 if (blur_circle_diameter < 1e-9 || freq < 1e-6)
44 return 1;
45
46 /* The transfer function of a circular blur is 2*J1(x)/x */
47 double arg = M_PI * freq * blur_circle_diameter;
48 return my_fabs (2.0 * get_j1 (arg) / arg);
49}

51/* Return MTF of gaussian blur.
52 FREQ is the spatial frequency.
53 SIGMA is the standard deviation of the Gaussian. */
54double
55gaussian_blur_mtf (double freq, double sigma)
56{
57 return std::exp (-2.0 * M_PI * M_PI * sigma * sigma * freq * freq);
58}

60/* Normalized sinc function: sin(pi * x) / (pi * x)
61 Used to model the sensor's pixel aperture effect. */
62double
63sinc (double x)
64{
65 x = my_fabs (x);
66 if (x < 1e-6)
67 return 1.0; /* Avoid division by zero (Taylor expansion limit) */
68 return my_sin (M_PI * x) / (M_PI * x);
69}

124/* Calculates the MTF of a lens including defocus effects.
125 FREQ Spatial frequency in lp/mm.
126 F_NUM The f-number of the lens.
127 WAVELENGTH_NM Wavelength in nanometers.
128 DEFOCUS_MM Displacement from the focal plane in millimeters.

130double calculate_defocused_mtf(double freq, double f_num, double wavelength_nm, double defocus_mm) {
131 /* Convert wavelength from nm to mm. */
132 double wavelength_mm = wavelength_nm / 1e6;
133 /* Calculate the cut-off frequency (fc) */
134 double cutoff_freq = 1.0 / (wavelength_mm * f_num);
135 /* Calculate the cut-off frequency (fc) */
136 double s = freq / cutoff_freq;

141 // 1. Diffraction-limited component

144 // 2. Defocus component (Geometric Approximation)

153 double mtf_defocus = std::abs(2.0 * /*std::*/j1(arg) / arg);

458 p.sigma = get_sigma (vals);

469 p.blur_diameter = get_blur_diameter (m, vals);
470 assert (diffraction == p.simulate_diffraction_p ());
471 /* Use double to avoid accumulation of an error */
472 double msum = 0;
473 for (size_t i = 0; i < measurement.size (); i++)
474 {
475 luminosity_t freq = measurement.get_freq (i);
476 /* Do not care about values above Nyquist. */
477 if (freq > 0.5)
478 continue;
479 luminosity_t contrast = measurement.get_contrast (i);
480 luminosity_t contrast2 = p.system_mtf (freq) * 100;
481 msum += (contrast - contrast2) * (contrast - contrast2);
482 if (f_vec)
483 {
484 assert (out_idx < n_observations);

486 }
487#if 0

492 if (be_verbose)

496 printf ("measurement %zu fill factor %f, f-stop %f (%f) gaussian blur sigma %f, wavelength %f, defocus %f, blur_diameter %f, sqsum %f\n",

498 if (m_progress)
499 m_progress->resume_stdout ();
500 }
501 sum += msum;
502 sums[m] = msum;
503 }
504 assert (!f_vec || out_idx == n_observations);
505
506 if (be_verbose)
507 {
508 if (m_progress)
509 m_progress->pause_stdout ();

545/* Determine PSF kernel radius. */
546static int
547get_psf_radius (mtf::psf_t *psf, int size, bool *ok = nullptr)
548{
549 double peak = 0;
550 for (int i = 0; i < size; i++)
551 {
552 // iprintf ("%i %f\n", i, psf[i]);
553 if (psf[i] > peak)
554 peak = psf[i];
555 }
556 int this_psf_radius = 1;
557 /* Center of the PSF kernel is at 0 */
558 for (int i = 1; i < size / 2 - 1; i++)
559 if (psf[i] > peak * 0.0001f)
560 this_psf_radius = i + 1;
561
562 /* This may be solved by iteratively reducing subsampling. */
563 if (this_psf_radius >= size / 2 - 2)
564 {
565 if (ok)
566 *ok = false;
567 }
568 else if (ok)
569 *ok = true;
570 return this_psf_radius;
571}

620/* Estimate lens blur using Hopkins model. */
621double
622mtf_parameters::hopkins_defocus_mtf (double pixel_freq) const
623{
624 double ef_stop = effective_f_stop ();
625 double s = nu (pixel_freq);
626 double freq = pixel_freq / (pixel_pitch * (1.0 / 1000));
627 double wavelength_mm = wavelength / 1e6;
628 /* Slight defocus model; not seem to be working in practice. */
629 if (0)
630 {
631 double blur_diameter = my_fabs (defocus) / effective_f_stop ();
632 return defocus_mtf (freq, blur_diameter);
633 }
634 if (s <= 0 || s >= 1)
635 return 1;
636
637 /* Calculate Wavefront Defocus (W_{020}). This converts physical
638 distance to wavefront error.
639
640 W_{20} = delta_z / (8 * f_number^2) */
641 double w20 = defocus / (8.0 * ef_stop * ef_stop);

643#if 0

663/* Calculates the Stokseth approximation for the MTF of a defocused lens.
664 PIXEL_FREQ is the spatial frequency in cycles per pixel.
665 Returns the MTF value as a product of diffraction and defocus effects. */
666
667double
668mtf_parameters::stokseth_defocus_mtf (double pixel_freq) const
669{
670 double ef_stop = effective_f_stop ();
671 double freq = pixel_freq / (pixel_pitch * (1.0 / 1000));
672 // 1. Calculate Diffraction Cutoff (fc)
673 double wavelength_mm = wavelength / 1e6;
674 double cutoff_freq = 1.0 / (wavelength_mm * ef_stop);
675
676 if (freq <= 0.0)
677 return 1.0;
678 if (freq >= cutoff_freq)
679 return 0.0;
680
681 double s = freq / cutoff_freq; // Normalized frequency (0 to 1)
682
683 // 2. Calculate Wavefront Aberration W20 (in mm)
684 // Formula: delta_z / (8 * N_eff^2)
685 double w20 = defocus / (8.0 * my_pow (ef_stop, 2));
686
687 // 3. Stokseth B Parameter
688 // Scales the defocus impact by frequency.
689 // Original approximation: B = 4 * pi * (w20 / lambda) * s * (1 - s)
690 // Standard approximation: B = 8 * pi * (w20 / lambda) * s * (1 - s)
691 double B
692 = (8.0 * M_PI * w20 * s) * (1.0 - s) / wavelength_mm;
693
694 // 4. Defocus MTF using Bessel J1
695 // 2*J1(B)/B is the optical transfer of a circular blur
696 double j_term = (my_fabs (B) < 1e-8) ? 1.0 : 2.0 * get_j1 (B) / B;
697
698 /* This seems to be incorrect. */
699 if (0)
700 {
701 // 5. Full Stokseth Polynomial Correction (1 - 0.6s + 0.4s^2)
702 double stokseth_poly = 1.0 - 0.6 * s + 0.4 * s * s;
703
704 // 6. Empirical Correction (1 - 0.6s)
705 // Interpolate between diffraction limit and Stokseth model based on defocus.
706 double phase_error_ratio = w20 / (wavelength_mm * 0.25);
707 double weight = std::clamp (phase_error_ratio, 0.0, 1.0);
708
709 // Apply correction: if weight is 0 (perfect focus), factor is 1.0.
710 // If weight is 1 (significant defocus), factor is the Stokseth polynomial.
711 double final_correction = (1.0 - weight) + (weight * stokseth_poly);
712
713 return my_fabs (j_term) * final_correction;
714 }
715 return my_fabs (j_term);
716}

718/* Simulate lens MTF.
719 PIXEL_FREQ is the spatial frequency in cycles per pixel.
720 Simulates lens as a combination of:
721 - diffraction limit (pixel_pitch, wavelength_mm, f_stop)
722 - gaussian blur (sigma)
723 - defocus (defocus_mm)
724 */
725
726double
727mtf_parameters::lens_mtf (double pixel_freq) const
728{
729 if (simulate_diffraction_p ())
730 return stokseth_defocus_mtf (pixel_freq) * lens_diffraction_mtf (pixel_freq)
731 * gaussian_blur_mtf (pixel_freq, sigma);
732 else
733 return gaussian_blur_mtf (pixel_freq, sigma)
734 * defocus_mtf (pixel_freq, blur_diameter);
735}

761/* Compute right half of LSF.
762 LSF is vector to store the result.
763 SUBSAMPLE is the spatial sampling step. */
764
765void
766mtf::compute_lsf (std::vector<psf_t, fft_allocator<psf_t>> &lsf,
767 luminosity_t subsample) const
768{
769 int size = lsf.size ();
770 if (size & 1)
771 {
772 lsf[size - 1] = 0;
773 size--;
774 }
775 std::vector<psf_t, fft_allocator<psf_t>> mtf_half (size);
776 auto plan = fft_plan_r2r_1d<psf_t> (size, FFTW_REDFT00, mtf_half.data (), lsf.data ());
777 psf_t scale = 1.0 / (size * subsample * 2);
778
779 /* Mirror mtf. */
780 for (int i = 0; i < size; i++)
781 mtf_half[i] = get_mtf (i * scale);
782
783 plan.execute_r2r (mtf_half.data (), lsf.data ());
784
785 /* Use double to avoid accumulation of error. */
786 double sum = 0;
787 for (int i = 0; i < size; i++)
788 sum += lsf[i];
789 psf_t fin_scale = 1.0 / sum;
790 for (int i = 0; i < size; i++)
791 lsf[i] *= fin_scale;
792}

794std::vector<mtf::psf_t, fft_allocator<mtf::psf_t>>
795mtf::compute_2d_psf (int psf_size, luminosity_t subscale,
796 progress_info *progress, bool parallel)
797{
798 int fft_size = psf_size / 2 + 1;
799 const psf_t psf_step = 1 / (psf_size * subscale);
800 // Use unique_ptr with FFTW allocator for fftw_complex array
801 auto mtf_kernel = fft_alloc_complex<psf_t> (psf_size * fft_size);
802 std::vector<psf_t, fft_allocator<psf_t>> psf_data (psf_size * psf_size);
803 auto plan = fft_plan_c2r_2d<psf_t> (psf_size, psf_size, mtf_kernel.get (), psf_data.data ());
804 for (int x = 0; x < fft_size; x++)
805 {
806 std::complex ker (std::clamp ((double)get_mtf (x, 0, psf_step),
807 0.0, 1.0),
808 0.0);
809 mtf_kernel.get ()[x][0] = std::real (ker);
810 mtf_kernel.get ()[x][1] = std::imag (ker);
811 }
812 //printf ("%i %i\n",fft_size, parallel);
813 // This loop is performance critical for focus finetuning.
814 // Bring it inline when parallelism is disabled or kernel is too small
815 if (!parallel || fft_size < 256)
816 {
817 for (int y = 1; y < fft_size; y++)
818 for (int x = 0; x < fft_size; x++)
819 {
820 std::complex ker (std::clamp ((double)get_mtf (x, y, psf_step),
821 0.0, 1.0),
822 0.0);
823 mtf_kernel.get ()[y * fft_size + x][0] = std::real (ker);
824 mtf_kernel.get ()[y * fft_size + x][1] = std::imag (ker);
825 mtf_kernel.get ()[(psf_size - y) * fft_size + x][0] = std::real (ker);
826 mtf_kernel.get ()[(psf_size - y) * fft_size + x][1] = std::imag (ker);
827 }
828 }
829 else
830 {
831#pragma omp parallel for default(none) schedule(dynamic) collapse(2) \
832 shared(fft_size, psf_step, mtf_kernel, psf_size) //if (parallel && fft_size > 256)
833 for (int y = 1; y < fft_size; y++)
834 for (int x = 0; x < fft_size; x++)
835 {
836 std::complex ker (std::clamp ((double)get_mtf (x, y, psf_step),

839 mtf_kernel.get ()[y * fft_size + x][0] = std::real (ker);
840 mtf_kernel.get ()[y * fft_size + x][1] = std::imag (ker);
841 mtf_kernel.get ()[(psf_size - y) * fft_size + x][0] = std::real (ker);
842 mtf_kernel.get ()[(psf_size - y) * fft_size + x][1] = std::imag (ker);

844 }
845 plan.execute_c2r (mtf_kernel.get (), psf_data.data ());
846
847 return psf_data;
848}

871 lsf_size *= 2;
872 else
873 subscale *= 2;
874 continue;
875 }
876 int radius = 1;
877 for (int v = 2; v < lsf_size; v++)
878 if (lsf[v] > max * min_threshold)

880#if 0
881 if (sum_threshold)

902/* Compute PSF as 2D FFT of circular MTF.
903 MAX_RADIUS is an estimate of radius. SUBSCALE is a size of
904 pixel we compute at (smaller pixel means more precise PSF) */
905bool
906mtf::compute_psf (luminosity_t max_radius, luminosity_t subscale, const char *filename,
907 const char **error, bool parallel)
908{
909 bool verbose = false;
910 /* Cap size of FFT to solve. */
911 while (my_ceil (max_radius / subscale) * 2 + 1 > 1024)
912 subscale *= 2;
913 int psf_size = my_ceil (max_radius / subscale) * 2 + 1;
914 int iterations = 0;
915
916 while (true)
917 {
918 /* Determine PSF radius. */
919
920 bool ok;
921 auto psf_data = mtf::compute_2d_psf (psf_size, subscale, nullptr, parallel);
922 if (psf_data.empty ())
923 return false;
924 int radius = get_psf_radius (psf_data.data (), psf_size, &ok);
925 /* If FFT size was to small for the PSF, increase it and restart. */
926 if (!ok && iterations < 10)
927 {
928 if (psf_size < 1024)
929 psf_size *= 2;
930 else
931 subscale /= 2;
932 //printf ("Iterating PSF computation %i %i %f\n", iterations, psf_size, subscale);
933 iterations++;
934 continue;
935 }
936 m_psf_radius = radius * subscale;
937
938 /* Compute LSF. Circular LSF is PSF. */
939 /* Make sure PSF also trails by 0. */
940 luminosity_t d1 = psf_data[radius];
941 luminosity_t d2 = psf_data[radius + 1];
942 psf_data[radius] = 0;
943 psf_data[radius + 1] = 0;
944 m_psf.set_range (0, (radius + 2) * subscale);

958 pp.filename = filename;
959 tiff_writer renderedu (pp, &error);
960 if (error)
961 return false;

963 for (int y = 0; y < psf_size / 2; y++)
964 for (int x = 0; x < psf_size / 2; x++)
965 {
966 luminosity_t val = get_psf (x, y, 1 / subscale);
967 luminosity_t diff = my_fabs (val - psf_data[y * psf_size + x]);
968 if (val > m)

970 if (diff > err)
971 err = diff;
972 // psf_data[y * psf_size + x] = val;
973 }
974 for (int y = 0; y < height; y++)
975 {
976 for (int x = 0; x < width; x++)
977 {
978 int xp = nearest_int (x * subscale);
979 int yp = nearest_int (y * subscale);
980 int xx = ((x + psf_size / 2 - radius) + psf_size / 2)
981 % psf_size;
982 int yy = ((y + psf_size / 2 - radius) + psf_size / 2)
983 % psf_size;
984 int v = std::clamp (
985 (int)(invert_gamma (psf_data[yy * psf_size + xx] / m, -1)
986 * (65535)
987 + 0.5),
988 0, 65535);
989 int vv
990 = std::clamp (v + 100 * 256 * ((xp + yp) % 2), 0, 65535);
991 renderedu.put_pixel (x, v, v, vv);
992 }
993 if (!renderedu.write_row ())
994 return false;
995 }
996 std::vector<psf_t, fft_allocator<psf_t>> lsf (radius);
997 compute_lsf (lsf, subscale);
998 psf_t lsf_max = lsf[0], collected_lsf_max = 0;
999 std::vector<psf_t, fft_allocator<psf_t>> collected_lsf (width);
1000 for (int x = 0; x < width; x++)
1001 {
1002 int xx = (x < radius ? radius - x - 1 : x - radius);
1003 for (int y = 0; y < psf_size; y++)
1004 collected_lsf[x] += psf_data[xx * psf_size + y];
1005 if (collected_lsf[x] > collected_lsf_max)
1006 collected_lsf_max = collected_lsf[x];
1007 }
1008 for (int y = 0; y < lsf_size; y++)
1009 {
1010 for (int x = 0; x < width; x++)
1011 {
1012 int idx = lsf_size
1013 - nearest_int (
1014 lsf[(x < radius ? radius - x - 1 : x - radius)]
1015 * lsf_size / lsf_max);
1016 int idx2
1017 = lsf_size
1018 - nearest_int (psf_data[((x + psf_size / 2 - radius)
1019 + psf_size / 2)
1020 % psf_size]
1021 * lsf_size / m);
1022 int idx3 = lsf_size
1023 - nearest_int (collected_lsf[x] * lsf_size
1024 / collected_lsf_max);
1025 int r = (idx == y) * 65535;
1026 int g = (idx2 == y) * 65535;
1027 int b = (idx3 == y) * 65535;
1028 renderedu.put_pixel (x, r, g, b);
1029 }
1030 if (!renderedu.write_row ())
1031 return false;
1032 }
1033 if (verbose)
1034 printf ("Max %f, err %f normalized %f\n", m, err, err / m);
1035 }
1036 return true;
1037 }

1041mtf::precompute (progress_info *progress, bool parallel)
1042{
1043 std::lock_guard<std::mutex> lock (m_lock);
1044 if (m_precomputed)
1045 {
1046 return true;
1047 }
1048
1049 /* Use actual MTF data. */
1050 if (m_params.use_measured_mtf ())
1051 {
1052 bool monotone = true;
1053 mtf_measurement &measurement = m_params.measurements[m_params.measured_mtf_idx];
1054 for (size_t i = 1; i < measurement.size () && monotone; i++)
1055 {
1056 if (!(measurement.get_freq (i - 1) < measurement.get_freq (i)))
1057 {
1058 printf ("Sorry, unimplemented: MTF freqs are not in increasing "
1059 "order.\n");
1060 monotone = false;
1061 abort ();
1062 }
1063 }
1064
1065 /* See if the MTF table has regular steps. In this case we can initialize
1066 precomputed function exactly. */
1067 bool regular_steps = true;
1068 luminosity_t step
1069 = (measurement.get_freq (measurement.size () - 1) - measurement.get_freq (0)) / (measurement.size () - 1);
1070 for (size_t i = 1; i < measurement.size () - 2 && regular_steps; i++)
1071 if (my_fabs (measurement.get_freq (i) - measurement.get_freq (0) - i * step) > 0.0006)
1072 regular_steps = false;
1073
1074 if (!regular_steps)
1075 {
1076 std::vector<luminosity_t> contrasts (measurement.size ()
1077 + 2);
1078 std::vector<luminosity_t> freqs (measurement.size ()
1079 + 2);
1080 for (size_t i = 0; i < measurement.size (); i++)
1081 {
1082 luminosity_t freq = measurement.get_freq (i);
1083 contrasts[i] = measurement.get_contrast (i) * 0.01
1084 * m_params.measured_mtf_correction (freq);
1085 freqs[i] = freq;
1086 }
1087 contrasts[measurement.size ()] = 0;
1088 contrasts[measurement.size () + 1] = 0;
1089 freqs[measurement.size ()] = freqs[measurement.size () - 1] + step;
1090 freqs[measurement.size () + 1] = freqs[measurement.size () - 1] + 2 * step;
1091 m_mtf.set_range (measurement.get_freq (0), measurement.get_freq (measurement.size () - 1) + 2 * step);
1092 m_mtf.init_by_x_y_values (freqs.data (), contrasts.data (),
1093 measurement.size () + 2, 1024);
1094 }
1095 else
1096 {
1097 std::vector<luminosity_t> contrasts (measurement.size ()
1098 + 2);
1099 for (size_t i = 0; i < measurement.size (); i++)
1100 {
1101 luminosity_t freq = measurement.get_freq (i);
1102 contrasts[i] = measurement.get_contrast (i) * 0.01
1103 * m_params.measured_mtf_correction (freq);
1104 }
1105 /* Be sure that MTF trails in 0. */
1106 contrasts[measurement.size ()] = 0;
1107 contrasts[measurement.size () + 1] = 0;
1108 m_mtf.set_range (measurement.get_freq (0), measurement.get_freq (measurement.size () - 1) + 2 * step);
1109 m_mtf.init_by_y_values (contrasts.data (), measurement.size () + 2);
1110 }
1111 if (colorscreen_checking)
1112 for (size_t i = 0; i < measurement.size (); i++)
1113 {
1114 luminosity_t freq = measurement.get_freq (i);
1115 if (my_fabs (measurement.get_contrast (i) * 0.01
1116 * m_params.measured_mtf_correction (freq)
1117 - get_mtf (freq))
1118 > 0.01)
1119 {
1120 printf (
1121 "Mismatch (measured) %i freq %f table %f precomputed %f\n",
1122 (int)i, freq,
1123 measurement.get_contrast (i) * 0.01
1124 * m_params.measured_mtf_correction (freq),
1125 m_mtf.apply (freq));
1126 abort ();
1127 }
1128 }
1129
1130 if (progress)
1131 progress->set_task ("computing point spread function", 1);
1132 }
1133 /* Use lens model. */
1134 else
1135 {
1136 const int entries = 512;
1137 std::vector<luminosity_t> contrasts (entries);
1138 double step = 1.0 / (entries - 2);
1139 for (int i = 0; i < entries - 2; i++)
1140 contrasts[i] = m_params.system_mtf (i * step);
1141 contrasts[entries - 2] = contrasts[entries - 1] = 0;
1142 m_mtf.set_range (0, 1 + step);
1143 m_mtf.init_by_y_values (contrasts.data (), entries);
1144
1145 if (colorscreen_checking)
1146 for (int i = 0; i < entries - 1; i++)
1147 if (my_fabs (m_params.system_mtf (i * step)
1148 - m_mtf.apply (i * step))
1149 > 0.0001)
1150 {
1151 printf ("Mismatch (model) %f %f %f\n",
1152 m_params.system_mtf (i * step),
1153 m_mtf.apply (i * step), step);
1154 abort ();
1155 }
1156
1157 if (progress)
1158 progress->set_task ("computing point spread function", 1);
1159 }
1160 m_psf_radius = estimate_psf_size ();
1161 // print_lsf (stdout);
1162
1163 // m_mtf.plot (0, 1);
1164 // m_psf.plot (0, 5);
1165 m_precomputed = true;
1166 return true;
