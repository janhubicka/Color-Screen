14/* Hurter–Driffield characteristic curve based on data points
15 with simple linear interpolation. */
16struct hd_curve
17{
18 /* x coordinates of measured values */
19 luminosity_t *xs = nullptr;
20 /* y coordinates of measured values */
21 luminosity_t *ys = nullptr;
22 /* Number of points. */
23 int n = 0;
24 /* If true, memory for XS and YS is owned and should be freed. */
25 bool m_owns_memory = false;
26
27 /* Constructor for HD_CURVE. Initialize with arrays of NEW_XS and NEW_YS of
28 length NEW_N. If OWNS is true, the arrays are owned by the class. */
29 constexpr hd_curve (luminosity_t *new_xs, luminosity_t *new_ys, int new_n,
30 bool owns = false)
31 : xs (new_xs), ys (new_ys), n (new_n), m_owns_memory (owns)
32 {
33 }
34 /* Default constructor for HD_CURVE. */
35 constexpr hd_curve () = default;
36
37 /* Destructor for HD_CURVE. DELETE[] the arrays if they are owned. */
38 DLL_PUBLIC_EXP virtual ~hd_curve ()
39 {
40 if (m_owns_memory)
41 {
42 delete[] xs;
43 delete[] ys;
44 }
45 }
46
47 /* Return density for given exposure IN. */
48 luminosity_t
49 apply (luminosity_t in)
50 {
51 if (in <= xs[0])
52 return ys[0];
53 for (int i = 1; i < n; i++)
54 {
55 if (in <= xs[i])
56 {
57 return (ys[i - 1]
58 + (in - xs[i - 1])
59 * ((ys[i] - ys[i - 1]) / (xs[i] - xs[i - 1])));
60 }
61 }
62 return ys[n - 1];
63 }
64 /* Get minimal value (base fog) multiplied by BOOST. */
65 constexpr luminosity_t
66 get_fog (luminosity_t boost) const
67 {
68 luminosity_t min = ys[0];
69 for (int i = 1; i < n; i++)
70 min = std::min (min, ys[i]);
71 return min * boost;
72 }
73 /* Get maximal value (saturation density). */
74 constexpr luminosity_t
75 get_max () const
76 {
77 luminosity_t max = ys[0];
78 for (int i = 1; i < n; i++)
79 max = std::max (max, ys[i]);
80 return max;
81 }
82 /* Output GNU plottable data to file F. */
83 void
84 print (FILE *f)
85 {
86 for (int i = 0; i < n; i++)
87 fprintf (f, "%f %f\n", xs[i], ys[i]);
88 }

99/* The generalized logistic function, also known as Richards' curve,
100 is a flexible growth model that provides an alternative to the standard
101 logisitic or sigmoid functions. It is used here to model the H&D
102 (Hurter-Driffield) characteristic curve of film.
103
104 The formula used is:
105 Y(x) = A + (K - A) / (1 + exp(-B * (x - M)))^(1/v)
106
107 If is_inverse is true, we calculate the curve as X(y) instead of Y(x)
108 to handle very steep regions (high gamma). */
109struct richards_curve_parameters
110{
111 /* Lower asymptote. In direct mode, this represents the minimal density. */
112 luminosity_t A = 0;
113 /* Upper asymptote. In direct mode, this represents the maximal density. */
114 luminosity_t K = 0;
115 /* Growth rate or slope factor. Controls the steepness of the curve. */
116 luminosity_t B = 0;
117 /* Horizontal shift or offset. Corresponds to the center of the linear
118 * region. */
119 luminosity_t M = 0;
120 /* Asymmetry parameter. Controls where the inflection point occurs relative
121 to the asymptotes. v=1 gives the standard symmetric logistic curve. */
122 luminosity_t v = 1;
123 /* If true, the curve defines X as a function of Y. This helps maintain
124 numerical stability for extremely steep characteristic curves. */
125 bool is_inverse = false;
126
127 constexpr richards_curve_parameters (luminosity_t new_A, luminosity_t new_K,
128 luminosity_t new_B, luminosity_t new_M,
129 luminosity_t new_v, bool new_inverse)
130 : A (new_A), K (new_K), B (new_B), M (new_M), v (new_v),
131 is_inverse (new_inverse)
132 {
133 }
134 constexpr richards_curve_parameters () = default;
135
136 bool
137 operator== (const richards_curve_parameters &o) const
138 {
139 return A == o.A && K == o.K && B == o.B && M == o.M && v == o.v
140 && is_inverse == o.is_inverse;
141 }
142};

144/* Description of a typical HD curve. */
145struct hd_curve_parameters
146{
147 /* Point where minimal density is reached. */
148 luminosity_t minx, miny;
149 /* First point of the linear segment. */
150 luminosity_t linear1x, linear1y;
151 /* Last point of the linear segment. */
152 luminosity_t linear2x, linear2y;
153 /* Point where the maximal density is reached. */
154 luminosity_t maxx, maxy;
155
156 constexpr hd_curve_parameters ()
157 : minx (-6), miny (5.5), linear1x (-5), linear1y (5), linear2x (5),
158 linear2y (-5), maxx (6), maxy (-5.5)
159 {
160 }
161 constexpr hd_curve_parameters (luminosity_t new_minx, luminosity_t new_miny,
162 luminosity_t new_linear1x,
163 luminosity_t new_linear1y,
164 luminosity_t new_linear2x,
165 luminosity_t new_linear2y,
166 luminosity_t new_maxx, luminosity_t new_maxy, bool normalize = false)
167 : minx (new_minx + (normalize ? -0.3-(new_linear1x + new_linear2x) : 0)),
168 miny (new_miny),
169 linear1x (new_linear1x + (normalize ? -0.3-(new_linear1x + new_linear2x) : 0)),
170 linear1y (new_linear1y),
171 linear2x (new_linear2x + (normalize ? -0.3-(new_linear1x + new_linear2x) : 0)),
172 linear2y (new_linear2y),
173 maxx (new_maxx + (normalize ? -0.3-(new_linear1x + new_linear2x) : 0)),
174 maxy (new_maxy)
175 {
176 }
177 bool
178 operator== (const hd_curve_parameters &other) const
179 {
180 return minx == other.minx && miny == other.miny
181 && linear1x == other.linear1x && linear1y == other.linear1y
182 && linear2x == other.linear2x && linear2y == other.linear2y
183 && maxx == other.maxx && maxy == other.maxy;
184 }

186 bool
187 is_inverted_p () const
188 {
189 luminosity_t gamma
190 = (linear2x != linear1x)
191 ? std::abs ((linear2y - linear1y) / (linear2x - linear1x))
192 : 1e10;
193 luminosity_t toe = (linear1x != minx)
194 ? std::abs ((linear1y - miny) / (linear1x - minx))
195 : 1e10;
196 luminosity_t shoulder
197 = (maxx != linear2x) ? std::abs ((maxy - linear2y) / (maxx - linear2x))
198 : 1e10;
199 return (toe > gamma || shoulder > gamma
200 || (gamma > 10.0 && gamma != 1e10));
201 }

203 bool
204 is_valid_for_richards_curve () const
205 {
206 auto monotonic = [] (double a, double b, double c, double d)
207 {
208 if (std::abs (a - b) < 1e-6 || std::abs (b - c) < 1e-6
209 || std::abs (c - d) < 1e-6)
210 return false;
211 return (a < b && b < c && c < d) || (a > b && b > c && c > d);
212 };
213 return monotonic (minx, linear1x, linear2x, maxx)
214 && monotonic (miny, linear1y, linear2y, maxy);
215 }

229 void
230 adjust_M (double old_M, double new_M)
231 {
232 double delta = new_M - old_M;
233 if (is_inverted_p ())
234 {
235 miny += delta;
236 linear1y += delta;
237 linear2y += delta;
238 maxy += delta;
239 }
240 else
241 {
242 minx += delta;
243 linear1x += delta;
244 linear2x += delta;
245 maxx += delta;
246 }
247 }
248
249 void
250 adjust_B (double old_B, double new_B, double M)
251 {
252 double ratio = old_B / new_B;
253 if (is_inverted_p ())
254 {
255 miny = M + (miny - M) * ratio;
256 linear1y = M + (linear1y - M) * ratio;
257 linear2y = M + (linear2y - M) * ratio;
258 maxy = M + (maxy - M) * ratio;
259 }
260 else
261 {
262 minx = M + (minx - M) * ratio;
263 linear1x = M + (linear1x - M) * ratio;
264 linear2x = M + (linear2x - M) * ratio;
265 maxx = M + (maxx - M) * ratio;
266 }
267 }
268
269 void
270 adjust_A (double old_A, double new_A, double K)
271 {
272 auto f = [&] (double v)
273 {
274 if (std::abs (K - old_A) < 1e-8)
275 return v;
276 return new_A + (v - old_A) * (K - new_A) / (K - old_A);
277 };
278 if (is_inverted_p ())
279 {
280 minx = f (minx);
281 linear1x = f (linear1x);
282 linear2x = f (linear2x);
283 maxx = f (maxx);
284 }
285 else
286 {
287 miny = f (miny);
288 linear1y = f (linear1y);
289 linear2y = f (linear2y);
290 maxy = f (maxy);
291 }
292 }
293
294 void
295 adjust_K (double old_K, double new_K, double A)
296 {
297 auto f = [&] (double v)
298 {
299 if (std::abs (old_K - A) < 1e-8)
300 return v;
301 return A + (v - A) * (new_K - A) / (old_K - A);
302 };
303 if (is_inverted_p ())
304 {
305 minx = f (minx);
306 linear1x = f (linear1x);
307 linear2x = f (linear2x);
308 maxx = f (maxx);
309 }
310 else
311 {
312 miny = f (miny);
313 linear1y = f (linear1y);
314 linear2y = f (linear2y);
315 maxy = f (maxy);
316 }
317 }

408 /* Precompute values used for scaling. */
409 void
410 precompute ()
411 {
412 if (m_clip)
413 Dfog = m_curve->get_fog (m_boost);
414 else
415 Dfog = 0;
416 if (m_curve->n < 2)
417 abort ();
418 }
419 /* Apply simulated photographic process to scanned transmittance Y.
420 Models the chain: Scanned Transmittance -> Exposure -> Density ->
421 Output Transmittance. */
422 luminosity_t
423 apply (luminosity_t y)
424 {
425 /* Preflash. */
426 y += m_preflash / 100;
427
428 /* Apply exposure */
429 y *= m_exposure;
430
431 /* Scale emulsion response and logarithmize. */
432
433 if (y > 0)
434 {
435 y = std::log10 (y);
436 /* Apply HD curve. */
437 y = m_curve->apply (y);
438 }
439 else
440 y = m_curve->ys[0];
441
442 /* Adjust contrast. */
443 // y = (y - mid) * m_contrast + mid;
444
445 /* Apply density boost. */
446 y *= m_boost;
447
448 /* Compensate fog */
449 y -= Dfog;
450
451 /* Get back to linear. */
452 y = pow (10, -y);
453 return y;
454 }
455 /* Return original scanned transmittance for given output transmittance Y.
456 Uses binary search to find the inverse of the simulated process. */
457 luminosity_t
458 unapply (luminosity_t y)
459 {
460 float min = 0;
461 float max = 1;
462 while (true)
463 {
464 luminosity_t yy = (min + max) / 2;
465 luminosity_t ap = apply (yy);
466 // fprintf (stderr, "%f:%f %f %f %f\n", y, min, max, yy, ap);
467 if (fabs (ap - y) < 0.000001 || max - min < 0.000001)
468 return yy;
469 if (ap < y)
470 max = yy;
471 else
472 min = yy;
473 }
474 }
