416void ColorPanel::updateSpectraChart() {
417 ParameterState state = m_stateGetter();
418 render_parameters::transmission_data data;
419 data.min_freq = 400;
420 data.max_freq = 720;
421 // 1nm resolution
422 size_t size = (size_t)((data.max_freq - data.min_freq) / 5 + 1);
423 data.red.resize(size);
424 data.green.resize(size);
425 data.blue.resize(size);
426 data.backlight.resize(size);
427
428 if (state.rparams.get_transmission_data(data)) {
429 bool absorbance = (m_spectraMode->currentIndex() == 1);
430
431 if (absorbance) {
432 // Convert to Absorbance
433 for (auto &v : data.red)
434 v = colorscreen::transmitance_to_absorbance(v);
435 for (auto &v : data.green)
436 v = colorscreen::transmitance_to_absorbance(v);
437 for (auto &v : data.blue)
438 v = colorscreen::transmitance_to_absorbance(v);
439 for (auto &v : data.backlight)
440 v = colorscreen::transmitance_to_absorbance(v);
441
442 m_spectraChart->setYAxis(0.0, 4.0, "Absorbance", "");
443 } else {
444 m_spectraChart->setYAxis(0.0, 1.0, "Transmittance", "%");
445 }
446
447 m_spectraChart->setSpectraData(data.red, data.green, data.blue,
448 data.backlight);
449 if (m_spectraSection)
450 m_spectraSection->show();
451 } else {
452 m_spectraChart->clear();
453 if (m_spectraSection)
454 m_spectraSection->hide();
455 }
456}
