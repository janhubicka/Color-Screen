307void TilePreviewPanel::onTriggerRender(int reqId, std::shared_ptr<colorscreen::progress_info> progress, const QVariant &userData) {
308 if (m_tileLabels.empty() || !userData.canConvert<RenderRequest>()) {
309 m_renderQueue.reportFinished(reqId, true);
310 return;
311 }
312
313 RenderRequest req = userData.value<RenderRequest>();
314
315 if ((!req.scan && requiresScan()) || !isTileRenderingEnabled(req.state)) {
316 for (auto l : m_tileLabels)
317 l->setPixmap(QPixmap());

319 return;
320 }
321
322 // No fixed size enforcement needed with ScalableImageLabel
323 // for (auto l : m_tileLabels)
324 // l->setFixedSize(req.tileSize, req.tileSize);
325
326 // Subclass check
327 bool needsUpdate = shouldUpdateTiles(req.state);
328
329 // Also check if tileSize changed significantly or first run
330 if (req.tileSize != m_lastRenderedTileSize) {
331 needsUpdate = true;
332 m_lastRenderedTileSize = req.tileSize;
333 }
334
335 // Force update if tiles are empty/not rendered yet
336 if (!m_tileLabels.empty() && m_tileLabels[0]->pixmap().isNull()) {
337 needsUpdate = true;
338 }

341 m_renderQueue.reportFinished(reqId, true);
342 return;
343 }

345 onTileUpdateScheduled();

347 // Start background render
348 QFuture<TileRenderResult> future = QtConcurrent::run(
349 renderTilesGeneric,
350 req.state,
351 req.scanWidth,
352 req.scanHeight,
353 reqId, // Pass reqId as generation
354 req.tileSize,
355 req.pixelSize,
356 req.tileTypes,
357 progress
358 );
359
360 // Monitor it
361 // We create a new watcher for each job to support concurrency handled by queue
362 QFutureWatcher<TileRenderResult> *watcher = new QFutureWatcher<TileRenderResult>(this);
363 connect(watcher, &QFutureWatcher<TileRenderResult>::finished, this, [this, watcher, reqId](){
364 TileRenderResult result = watcher->result();
365
366 // Update UI if successful
367 if (result.success) {
368 if (result.tiles.size() == m_tileLabels.size()) {
369 for (size_t i = 0; i < m_tileLabels.size(); ++i) {
370 const QImage& img = result.tiles[i];
371 if (!img.isNull())
372 m_tileLabels[i]->setPixmap(QPixmap::fromImage(img));
373 }
374 }
375 } else {
376 // Failed or cancelled
377 m_lastRenderedTileSize = 0; // Force retry
