382backlight_correction::backlight_correction (
383 backlight_correction_parameters &params, int width, int height,
384 luminosity_t black, bool white_balance, progress_info *progress)
385 : m_params (params),
386 /*m_img_width (width), m_img_height (height),*/ m_width (params.m_width),
387 m_height (params.m_height),
388 m_img_width_rec (params.m_width / (coord_t)width),
389 m_img_height_rec (params.m_height / (coord_t)height), m_weights (NULL)
390{
391 if (progress)
392 progress->set_task ("computing backlight correction", 1);
393 m_weights = (entry *)malloc (m_width * m_height * sizeof (entry));
394 m_black = black;
395 const luminosity_t epsilon = 1.0 / 256;
396
397 if (!m_weights)
398 return;
399 black_correction = m_params.black_correction;
400 luminosity_t sum[4] = { 0, 0, 0, 0 };
401 for (int x = 0; x < m_width * m_height; x++)
402 for (int i = 0; i < 4; i++)
403 sum[i] += m_params.m_luminosities[x].lum[i] - black;
404 if (white_balance)
405 sum[0] = sum[1] = sum[2] = (sum[0] + sum[1] + sum[2]) / 3;
406 luminosity_t correct[4];
407 for (int i = 0; i < 4; i++)
408 if (sum[i] > epsilon * (m_width * (int64_t)m_height))
409 correct[i] = sum[i] / (m_width * (int64_t)m_height);
410 else
411 correct[i] = 1;
412
413 for (int x = 0; x < m_width * m_height; x++)
414 for (int i = 0; i < 4; i++)
415 {
416 if ((m_params.m_luminosities[x].lum[i] - black) > epsilon)
417 m_weights[x].mult[i]
418 = correct[i] / (m_params.m_luminosities[x].lum[i] - black);
419 else
420 m_weights[x].mult[i] = correct[i];
421 m_weights[x].sub[i] = m_params.m_luminosities[x].sub[i];
422 }
