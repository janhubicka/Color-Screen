173void ParameterPanel::addSliderParameter(

193 slider->setRange(0, SLIDER_MAX);

201 spin->setRange(min, max);
202 spin->setDecimals(decimals);
203 spin->setSingleStep(1.0 / scale);

206 if (!specialValueText.isEmpty())

218 // Helper to map Slider -> Value
219 auto sliderToValue = [min, max, scale, gamma, SLIDER_MAX,
220 logarithmic](int s) -> double {
221 if (!logarithmic && gamma == 1.0)
222 return (double)s / scale;
223
224 double t = (double)s / SLIDER_MAX; // 0..1
225
226 if (logarithmic) {
227 if (min <= 0) {
228 // v = (max + 1)^t - 1
229 return std::pow(max + 1.0, t) - 1.0;
230 } else {
231 // v = min * (max/min)^t
232 return min * std::pow(max / min, t);
233 }
234 }
235
236 // v = min + (max-min) * t^gamma
237 return min + (max - min) * std::pow(t, gamma);
238 };
239
240 // Helper to map Value -> Slider
241 auto valueToSlider = [min, max, scale, gamma, SLIDER_MAX,
242 logarithmic](double v) -> int {
243 if (!logarithmic && gamma == 1.0)
244 return qRound(v * scale);
245
246 double t = 0;
247 if (logarithmic) {
248 if (min <= 0) {
249 // t = log(v + 1) / log(max + 1)
250 if (v <= 0)
251 t = 0;
252 else
253 t = std::log(v + 1.0) / std::log(max + 1.0);
254 } else {
255 // t = log(v/min) / log(max/min)
256 if (v <= min)
257 t = 0;
258 else
259 t = std::log(v / min) / std::log(max / min);
260 }
261 } else {
262 // t = ((v - min) / (max - min)) ^ (1/gamma)
263 double ratio = (v - min) / (max - min);
264 if (ratio <= 0)
265 t = 0;
266 else if (ratio >= 1)
267 t = 1;
268 else
269 t = std::pow(ratio, 1.0 / gamma);
270 }
271 return std::clamp((int)qRound(t * SLIDER_MAX), 0, SLIDER_MAX);
272 };

297 });

301 [slider, spin, getter, valueToSlider](const ParameterState &state) {
302 double val = getter(state);
303 spin->blockSignals(true);
304 spin->setValue(val);
305 spin->blockSignals(false);
306
307 slider->blockSignals(true);
308 slider->setValue(valueToSlider(val));
309 slider->blockSignals(false);
310 });
311
312 // Enable Update
313 if (enabledCheck) {
314 m_widgetStateUpdaters.push_back(
315 [this, slider, spin, container, enabledCheck]() {
316 ParameterState state = m_stateGetter();
317 bool en = enabledCheck(state);
318 slider->setEnabled(en);
319 spin->setEnabled(en);

732
733 QSlider *slider = new QSlider(Qt::Horizontal);
734 slider->setRange(min * scale, max * scale);
735
736 QDoubleSpinBox *spin = new QDoubleSpinBox();
737 spin->setRange(min, max);
738 spin->setDecimals(decimals);
739 spin->setSingleStep(1.0 / scale);
740 if (!suffix.isEmpty())
741 spin->setSuffix(QString(" %1").arg(suffix));
742 if (!tooltip.isEmpty()) {
743 slider->setToolTip(tooltip);
744 spin->setToolTip(tooltip);
745 }
746
747 hLayout->addWidget(slider, 1);
748 hLayout->addWidget(spin, 0);
749
750 if (m_currentGroupForm)
751 m_currentGroupForm->addRow(QString("%1 %2").arg(names[i]).arg(label),
752 container);
753 else
754 m_form->addRow(QString("%1 %2").arg(names[i]).arg(label), container);
755
756 channels.push_back({slider, spin});
757
758 // Internal Sync for each channel
759 connect(slider, &QSlider::valueChanged, this, [spin, scale](int val) {
760 spin->blockSignals(true);
761 spin->setValue((double)val / scale);
762 spin->blockSignals(false);
763 });
764 connect(spin, QOverload<double>::of(&QDoubleSpinBox::valueChanged), this,
765 [slider, scale](double val) {
766 slider->blockSignals(true);
767 slider->setValue(qRound(val * scale));
768 slider->blockSignals(false);
769 });
770 }
771
772 // Link Checkbox row
773 if (m_currentGroupForm)
774 m_currentGroupForm->addRow("", linkCheck);
775 else
776 m_form->addRow("", linkCheck);
777
778 // Interaction Logic
779 auto handleValueChange = [this, channels, linkCheck, getter, setter,
780 scale, label](int changedIdx, double newVal) {
781 ParameterState s = m_stateGetter();
782 colorscreen::rgbdata current = getter(s);
783 double oldVal = current[changedIdx];
784 double delta = newVal - oldVal;
785
786 colorscreen::rgbdata next = current;
787 next[changedIdx] = newVal;
788
789 if (linkCheck->isChecked()) {
790 for (int i = 0; i < 3; ++i) {
791 if (i != changedIdx) {
792 next[i] += delta;
793 }
794 }
795 }
796
797 applyChange([setter, next](ParameterState &state) { setter(state, next); }, label);
798
799 // Optimistic UI update for linked sliders
800 if (linkCheck->isChecked()) {
801 for (int i = 0; i < 3; ++i) {
802 if (i != changedIdx) {
803 channels[i].spin->blockSignals(true);
804 channels[i].spin->setValue(next[i]);
805 channels[i].spin->blockSignals(false);
806
807 channels[i].slider->blockSignals(true);
808 channels[i].slider->setValue(qRound(next[i] * scale));
809 channels[i].slider->blockSignals(false);
810 }
811 }
812 }
813 };
814
815 // Connect user interaction
816 for (int i = 0; i < 3; ++i) {
817 int idx = i;
818 connect(channels[i].spin, QOverload<double>::of(&QDoubleSpinBox::valueChanged),
819 this, [handleValueChange, idx](double v) {
820 handleValueChange(idx, v);
821 });
822 connect(channels[i].slider, &QSlider::valueChanged, this,
823 [handleValueChange, idx, scale](int v) {
824 handleValueChange(idx, (double)v / scale);
825 });
826 }
827
828 // Updater State -> UI
829 m_paramUpdaters.push_back([channels, getter, scale](const ParameterState &s) {
830 colorscreen::rgbdata v = getter(s);
831 for (int i = 0; i < 3; ++i) {
832 channels[i].spin->blockSignals(true);
833 channels[i].spin->setValue(v[i]);
834 channels[i].spin->blockSignals(false);
835
836 channels[i].slider->blockSignals(true);
837 channels[i].slider->setValue(qRound(v[i] * scale));
838 channels[i].slider->blockSignals(false);
839 }
840 });
841
842 // Enablement
843 if (enabledCheck) {
844 m_widgetStateUpdaters.push_back(
845 [this, channels, linkCheck, enabledCheck, label]() {
846 ParameterState s = m_stateGetter();
847 bool en = enabledCheck(s);
848 linkCheck->setEnabled(en);
849 for (auto &c : channels) {
850 c.slider->setEnabled(en);
851 c.spin->setEnabled(en);
852 QWidget *labelWidget = m_form->labelForField(c.slider->parentWidget());
853 if (labelWidget)
854 labelWidget->setEnabled(en);
855 }
856 });
857 }
858}

907 arrowBtn->setArrowType(checked ? Qt::DownArrow : Qt::RightArrow);
908
909 auto setVisibleRecursive = [](QLayoutItem *item, bool visible) {
910 auto recurse = [](auto self, QLayoutItem *itm, bool vis) -> void {
911 if (!itm) return;
912 if (itm->widget()) {
913 itm->widget()->setVisible(vis);
914 } else if (itm->layout()) {
915 QLayout *subLayout = itm->layout();
916 for (int j = 0; j < subLayout->count(); ++j) {
917 self(self, subLayout->itemAt(j), vis);
918 }
919 }
920 };
921 recurse(recurse, item, visible);
922 };
923
924 // Hide all widgets in the form layout
925 for (int i = 0; i < groupForm->rowCount(); ++i) {
926 QLayoutItem *labelItem = groupForm->itemAt(i, QFormLayout::LabelRole);
927 QLayoutItem *fieldItem = groupForm->itemAt(i, QFormLayout::FieldRole);
928 // Check SpanningRole as well just in case, though usually FieldRole covers it in 2-arg addRow
929 QLayoutItem *spanningItem = groupForm->itemAt(i, QFormLayout::SpanningRole);
930
931 setVisibleRecursive(labelItem, checked);
932 setVisibleRecursive(fieldItem, checked);
933 if (spanningItem && spanningItem != fieldItem && spanningItem != labelItem) {
934 setVisibleRecursive(spanningItem, checked);
935 }
936 }
937 });
938
939 m_form->addRow(group);
940 m_currentGroupForm = groupForm;
941
942 return arrowBtn; // Return the toggle button
943}
