468template <typename T, const char *const names[], int max>
469T
470parse_enum (const char *arg, const char *errmsg)
471{
472 for (int i = 0; i < max; i++)
473 if (!strcmp (arg, names[i]))
474 return (T)i;
475 fprintf (stderr, errmsg, arg);
476 fprintf (stderr, "Possible values are: ");
477 for (int i = 0; i < max; i++)
478 fprintf (stderr, " %s", names[i]);
479 fprintf (stderr, "\n");
480 exit (1);
481 return (T)max;
482}

951 rfparams.scale = scale;
952 if (screen_scale)
953 rfparams.screen_scale = screen_scale;
954 if (ignore_infrared)
955 rparam.ignore_infrared = true;
956
957 /* ... and render! */
958 rfparams.verbose = verbose;
959 if (!render_to_file (scan, param, dparam, rparam, rfparams, rtparam,
960 &progress, &error))
961 {
962 progress.pause_stdout ();
963 fprintf (stderr, "Can not save %s: %s\n", rfparams.filename, error);
964 return 1;
965 }
966 progress.pause_stdout ();
967 return 0;
968}
