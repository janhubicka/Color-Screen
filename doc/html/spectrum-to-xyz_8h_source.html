282 /* dyes_rgb_to_xyz converts dyes (R, G, B) to XYZ color space.
283 OBSERVER is the observer to use. */
284 struct xyz pure_attr __attribute__ ((noinline))
285 dyes_rgb_to_xyz (luminosity_t r, luminosity_t g, luminosity_t b, int observer = default_observer)
286 {
287 spectrum s;
288 if (!subtractive)
289 {
290 for (int i = 0; i < SPECTRUM_SIZE; i++)
291 s[i] = red [i] * r * rscale + green [i] * g * gscale + blue [i] * b * bscale;
292 }
293 else
294 {
295 r = transmitance_to_absorbance (std::max (r, (luminosity_t)0) * rscale + 1.0/65535);
296 g = transmitance_to_absorbance (std::max (g, (luminosity_t)0) * gscale + 1.0/65535);
297 b = transmitance_to_absorbance (std::max (b, (luminosity_t)0) * bscale + 1.0/65535);
298 for (int i = 0; i < SPECTRUM_SIZE; i++)
299 s[i] = absorbance_to_transmitance (cyan [i] * r + magenta [i] * g + yellow [i] * b);
300 }
301 struct xyz ret = get_xyz (s, observer) - dark;
302 ret.x *= xscale;
303 ret.y *= yscale;
304 ret.z *= zscale;
305 return ret;
306 }

432 {

435 //assert (observer == 1931 || observer == 1964);

443 ret.y += cie_cmf_y[i] * s[i] * backlight[i];
