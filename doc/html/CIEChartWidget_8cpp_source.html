32void CIEChartWidget::updateLocus() {
33 m_locus.clear();
34 // Build spectral locus from CMF data
35 // Assuming cie_cmf_* arrays are available and linked
36 for (int i = 0; i < SPECTRUM_SIZE; ++i) {
37 double x = cie_cmf_x[i];
38 double y = cie_cmf_y[i];
39 double z = cie_cmf_z[i];
40 double sum = x + y + z;
41 if (sum > 0) {
42 m_locus << QPointF(x / sum, y / sum);
43 }
44 }
45 // Close polygon
46 if (!m_locus.isEmpty())
47 m_locus << m_locus.first();
48}

229void CIEChartWidget::generateCache() {
230 m_cache = QImage(size(), QImage::Format_ARGB32_Premultiplied);
231 m_cache.fill(Qt::transparent); // Match widget background by being transparent
232
233 QPolygonF screenLocus;
234 for (const auto &pt : m_locus) {
235 screenLocus << mapToWidget(pt.x(), pt.y());
236 }
237
238 // Bounding rect for optimization
239 QRect bounds = screenLocus.boundingRect().toRect().intersected(rect());
240
241 // Pixel generation
242 for (int y = bounds.top(); y <= bounds.bottom(); ++y) {
243 QRgb *scanLine = (QRgb *)m_cache.scanLine(y);
244 for (int x = bounds.left(); x <= bounds.right(); ++x) {
245 QPointF p(x, y);
246 if (screenLocus.containsPoint(p, Qt::OddEvenFill)) {
247 auto val = mapFromWidget(p);
248 // Convert xy to RGB
249 // Assume Y = 1.0 (maximum brightness)
250 double X = val.first;
251 double Y = 1.0;
252 double Z = (1.0 - val.first - val.second);
253
254 if (val.second <= 0.001)
255 continue; // Avoid div by zero
256 X = X / val.second;
257 Z = Z / val.second;
258
259 luminosity_t r, g, b;
260 xyz_to_srgb(X, Y, Z, &r, &g, &b);
261
262 // Simple gamut mapping/clipping
263 // Find max component to normalize brightness if it exceeds 1
264 double maxComp = std::max({r, g, b});
265 if (maxComp > 1.0) {
266 r /= maxComp;
267 g /= maxComp;
268 b /= maxComp;
269 }
270
271 // Clamp to 0-1
272 r = std::clamp((double)r, 0.0, 1.0) * 0.3;
273 g = std::clamp((double)g, 0.0, 1.0) * 0.3;
274 b = std::clamp((double)b, 0.0, 1.0) * 0.3;
275
276 scanLine[x] = qRgb(r * 255, g * 255, b * 255);
277 }
278 }
279 }

338QRectF CIEChartWidget::getChartRect() const {
339 double w = width();
340 double h = height();
341 // Keep 1:1 aspect ratio for the plot area (0.8 width vs 0.9 height in data)
342 double rangeX = m_maxX - m_minX;
343 double rangeY = m_maxY - m_minY;
344
345 // Add margins for axes
346 double marginL = 40;
347 double marginR = 20;
348 double marginT = 20;
349 double marginB = 40;
350
351 double availW = w - marginL - marginR;
352 double availH = h - marginT - marginB;
353
354 if (availW < 10) availW = 10;
355 if (availH < 10) availH = 10;
356
357 double scale = std::min(availW / rangeX, availH / rangeY);
358
359 double plotW = rangeX * scale;
360 double plotH = rangeY * scale;
361
362 // Center it
363 double x = marginL + (availW - plotW) / 2.0;
364 double y = marginT + (availH - plotH) / 2.0;
365
366 return QRectF(x, y, plotW, plotH);
