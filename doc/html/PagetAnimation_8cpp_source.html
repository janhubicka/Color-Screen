70void PagetAnimation::launchFirework() {
71 if (width() < 100 || height() < 100) return;
72
73 std::uniform_real_distribution<> xDist(width() * LAUNCH_X_MIN, width() * LAUNCH_X_MAX);
74 std::uniform_real_distribution<> yDist(height() * TARGET_HEIGHT_MIN, height() * TARGET_HEIGHT_MAX);
75 std::uniform_real_distribution<> angleDist(LAUNCH_ANGLE_MIN, LAUNCH_ANGLE_MAX);
76
77 QPointF launchPos(xDist(m_rng), height()); // Bottom of screen
78 QPointF targetPos(xDist(m_rng), yDist(m_rng)); // Random height
79 double launchAngle = angleDist(m_rng) * M_PI / 180.0; // Convert to radians
80
81 Firework fw(launchPos, targetPos);
82 fw.launchTime = m_time;
83
84 // Bayer pattern repeats: GRBG pattern, rotated 45 degrees
85 // With swapped colors: Blue-Red-Green pattern
86 QColor colors[2][2] = {
87 {QColor(50, 100, 220), QColor(220, 50, 50)}, // Blue, Red (was Green, Red)
88 {QColor(50, 200, 50), QColor(50, 100, 220)} // Green, Blue (was Blue, Green)
89 };
90
91 double spacing = PARTICLE_SPACING;
92 double size = PARTICLE_SIZE;
93 double rotation = PARTICLE_ROTATION * M_PI / 180.0;
94
95 for (int row = 0; row < GRID_SIZE; ++row) {
96 for (int col = 0; col < GRID_SIZE; ++col) {
97 double offsetX = (col - 2.0) * spacing; // Center around 0
98 double offsetY = (row - 2.0) * spacing;
99
100 // Rotate by 45 degrees
101 double rotatedX = offsetX * std::cos(rotation) - offsetY * std::sin(rotation);
102 double rotatedY = offsetX * std::sin(rotation) + offsetY * std::cos(rotation);
103
104 QPointF pos = launchPos + QPointF(rotatedX, rotatedY);
105
106 // Bayer pattern repeats every 2×2
107 QColor color = colors[row % 2][col % 2];
108
109 // Launch velocity with angle
110 double vx = std::sin(launchAngle) * LAUNCH_SPEED;
111 double vy = -std::cos(launchAngle) * LAUNCH_SPEED;
112
113 Particle p(pos, QPointF(vx, vy), color, size);
114 fw.particles.append(p);
115 }
116 }
117
118 m_fireworks.append(fw);

121void PagetAnimation::explodeFirework(Firework &fw) {
122 fw.exploded = true;
123
124 // Calculate center from average position of launch particles
125 QPointF center(0, 0);
126 for (const Particle &p : fw.particles) {
127 center += p.pos;
128 }
129 center /= fw.particles.size();
130
131 // Keep the original 25 particles and give them outward velocities
132 std::uniform_real_distribution<> fuzzDist(-EXPLOSION_FUZZ, EXPLOSION_FUZZ);
133
134 for (Particle &p : fw.particles) {
135 // Calculate direction from center to particle
136 QPointF dir = p.pos - center;
137 double dist = std::sqrt(dir.x() * dir.x() + dir.y() * dir.y());
138
139 if (dist > 0.1) {
140 dir /= dist; // Normalize
141 } else {
142 dir = QPointF(1, 0); // Default direction if particle is at center
143 }
144
145 // Base outward speed with random fuzz
146 double speed = EXPLOSION_SPEED_BASE + fuzzDist(m_rng);
147
148 // Set velocity: outward from center with fuzz
149 p.velocity = QPointF(
150 dir.x() * speed + fuzzDist(m_rng) * 0.5,
151 dir.y() * speed + fuzzDist(m_rng) * 0.5
152 );
153 }
154}

156void PagetAnimation::updateAnimation() {
157 double dt = 0.016; // Fixed physics step
158
159 // Add elapsed time to accumulator
160 if (m_elapsedTimer.isValid()) {
161 double elapsed = m_elapsedTimer.restart() / 1000.0;
162 // Cap max elapsed time to prevent spiral of death if rendering freezes
163 if (elapsed > 0.1) elapsed = 0.1;
164 m_physicsAccumulator += elapsed;
165 }
166
167 // Consume accumulated time in fixed steps
168 while (m_physicsAccumulator >= dt) {
169 stepAnimation(dt);
170 m_physicsAccumulator -= dt;
171 }

176void PagetAnimation::stepAnimation(double dt) {
177 m_time += dt;
178
179 // Update all fireworks
180 for (Firework &fw : m_fireworks) {
181 if (!fw.exploded) {
182 // Update ALL launch particles together
183 if (!fw.particles.isEmpty()) {
184 // Update all particles with same physics
185 for (Particle &p : fw.particles) {
186 p.velocity.setY(p.velocity.y() + GRAVITY * dt);
187 p.pos += p.velocity * dt;
188 }
189
190 // Check first particle to see if we should explode
191 // Wait until rocket has fallen ~50 pixels past its peak before exploding
192 Particle &first = fw.particles.first();
193 if (first.velocity.y() > 0 && first.pos.y() > fw.targetPos.y() + EXPLOSION_DELAY) {
194 explodeFirework(fw);
195 }
196 }
197 } else {
198 // Update explosion particles
199 for (Particle &p : fw.particles) {
200 p.velocity.setY(p.velocity.y() + GRAVITY * dt);
201 p.pos += p.velocity * dt;
202 p.life -= PARTICLE_FADE_RATE;
203 }
204 }
205 }
206
207 // Remove dead fireworks
208 m_fireworks.erase(
209 std::remove_if(m_fireworks.begin(), m_fireworks.end(),
210 [](const Firework &fw) {
211 if (!fw.exploded) return false;
212 for (const Particle &p : fw.particles) {
213 if (p.life > 0.0) return false;
214 }
215 return true;
216 }),
217 m_fireworks.end()
218 );
219
220 m_subtitles.update(dt);
221}
