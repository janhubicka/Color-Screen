84inline rgbdata
85out_color_adjustments::linear_hdr_color (rgbdata c) const noexcept
86{
87 luminosity_t r = c.red, g = c.green, b = c.blue;
88 m_color_matrix.apply_to_rgb (r, g, b, &r, &g, &b);
89 if (m_spectrum_dyes_to_xyz)
90 {
91 if (m_spectrum_dyes_to_xyz->red_characteristic_curve)
92 r = m_spectrum_dyes_to_xyz->red_characteristic_curve->apply (r);
93 if (m_spectrum_dyes_to_xyz->green_characteristic_curve)
94 g = m_spectrum_dyes_to_xyz->green_characteristic_curve->apply (g);
95 if (m_spectrum_dyes_to_xyz->blue_characteristic_curve)
96 b = m_spectrum_dyes_to_xyz->blue_characteristic_curve->apply (b);
97 xyz c_xyz = m_spectrum_dyes_to_xyz->dyes_rgb_to_xyz (r, g, b);
98 m_color_matrix2.apply_to_rgb (c_xyz.x, c_xyz.y, c_xyz.z, &r, &g, &b);
99 }
100
101 /* Apply DNG-style tone curve correction. */
102 if (m_tone_curve)
103 {
104 rgbdata c_tc = { r, g, b };
105 assert (!m_tone_curve->is_linear ());
106 c_tc = m_tone_curve->apply_to_rgb (c_tc);
107 color_matrix cm;
108 pro_photo_rgb_xyz_matrix m1;
109 cm = m1 * cm;
110 bradford_d50_to_d65_matrix m2;
111 cm = m2 * cm;
112 xyz_srgb_matrix m;
113 cm = m * cm;
114 cm.apply_to_rgb (c_tc.red, c_tc.green, c_tc.blue, &c_tc.red, &c_tc.green,
115 &c_tc.blue);
116 return c_tc;
117 }
118
119 return { r, g, b };
120}
