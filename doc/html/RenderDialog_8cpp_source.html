19// ── Scale <-> slider mapping (log scale 0.01 .. 20) ──────────────────────────
20static int scaleToSlider(double s) {
21 double t = (std::log(s) - std::log(0.01)) / (std::log(20.0) - std::log(0.01));
22 return (int)std::round(std::clamp(t, 0.0, 1.0) * 1000);
23}
24static double sliderToScale(int v) {
25 double t = v / 1000.0;
26 return std::exp(std::log(0.01) + t * (std::log(20.0) - std::log(0.01)));
27}

316void RenderDialog::updateControlStates()
317{
318 bool widthOrHeight = (m_widthSpin && m_widthSpin->value() > 0) ||
319 (m_heightSpin && m_heightSpin->value() > 0);
320 bool screenScaleSet = m_screenScaleSpin && m_screenScaleSpin->value() > 0;
321
322 // Width or height overrides everything → gray out scale and screen_scale
323 if (m_scaleSpin) m_scaleSpin->setEnabled(!widthOrHeight);
324 if (m_scaleSlider) m_scaleSlider->setEnabled(!widthOrHeight);
325 if (m_screenScaleSpin) m_screenScaleSpin->setEnabled(!widthOrHeight);
326 if (m_screenScaleSlider) m_screenScaleSlider->setEnabled(!widthOrHeight);
327
328 // screen_scale set → gray out scale (but only if w/h not already disabling)
329 if (!widthOrHeight) {
330 if (m_scaleSpin) m_scaleSpin->setEnabled(!screenScaleSet);
331 if (m_scaleSlider) m_scaleSlider->setEnabled(!screenScaleSet);

335void RenderDialog::updateSizePreview()
336{
337 if (!m_scan || !m_sizePreviewLabel) return;
338
339 render_to_file_params rfp;
340 rfp.scale = m_scaleSpin ? m_scaleSpin->value() : 1.0;
341 rfp.screen_scale = (screenGeometryActive() && m_screenScaleSpin)
342 ? m_screenScaleSpin->value() : 0.0;
343 rfp.width = m_widthSpin ? m_widthSpin->value() : 0;
344 rfp.height = m_heightSpin ? m_heightSpin->value() : 0;
345 rfp.antialias = m_antialiasSpin ? m_antialiasSpin->value() : 0;
346 rfp.geometry = geometry();
347
348 render_type_parameters rt = renderTypeParams();
349 scr_to_img_parameters sc = m_scrParams;
350
351 if (!complete_rendered_file_parameters(rt, sc, const_cast<image_data &>(*m_scan), &rfp)) {
352 m_sizePreviewLabel->setText(tr("(cannot compute)"));
353 return;
354 }
355
356 int depth = m_depthCombo ? m_depthCombo->currentData().toInt() : 16;
357 double mpx = rfp.width * (double)rfp.height / 1e6;
358 double bytes = rfp.width * (double)rfp.height * 3.0 * (depth / 8);
359 double mb = bytes / (1024.0 * 1024.0);
360 QString sizeStr;
361 if (mb >= 1024.0)
362 sizeStr = tr("%1 GB").arg(mb / 1024.0, 0, 'f', 2);
363 else
364 sizeStr = tr("%1 MB").arg(mb, 0, 'f', 1);
365
366 m_sizePreviewLabel->setText(
367 tr("%1 × %2 px (%3 Mpx, %4)")
368 .arg(rfp.width)
369 .arg(rfp.height)
370 .arg(mpx, 0, 'f', 2)
371 .arg(sizeStr));
372}
