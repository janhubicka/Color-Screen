66 /* Apply mesh transformation to point P. */
67 point_t pure_attr
68 apply (point_t p) const
69 {
70 int ix, iy;
71 p.x = my_modf ((p.x + m_xshift) * m_xstepinv, &ix);
72 p.y = my_modf ((p.y + m_yshift) * m_ystepinv, &iy);
73 if (ix < 0)
74 {
75 p.x = 0;
76 ix = 0;
77 }
78 if (iy < 0)
79 {
80 p.y = 0;
81 iy = 0;
82 }
83 if (ix >= m_width - 1)
84 {
85 p.x = 1;
86 ix = m_width - 2;
87 }
88 if (iy >= m_height - 1)
89 {
90 p.y = 1;
91 iy = m_height - 2;
92 }
93 mesh_point np = { (mesh_coord_t)p.x, (mesh_coord_t)p.y };
94 np = interpolate (m_data[iy * m_width + ix], m_data[iy * m_width + ix + 1],
95 m_data[(iy + 1) * m_width + ix],
96 m_data[(iy + 1) * m_width + ix + 1], np);
97 return { np.x, np.y };
98 }

241 /* Compute A, B such that
242 X1 + DX1 * A = X2 + DX2 * B
243 Y1 + DY1 * A = Y2 + DY2 * B */
244 static void
245 intersect_vectors (mesh_coord_t x1, mesh_coord_t y1, mesh_coord_t dx1,
246 mesh_coord_t dy1, mesh_coord_t x2, mesh_coord_t y2,
247 mesh_coord_t dx2, mesh_coord_t dy2, mesh_coord_t *a,
248 mesh_coord_t *b)
249 {
250 matrix2x2<mesh_coord_t> m (dx1, -dx2, dy1, -dy2);

255 /* Smoothly map triangle (0,0), (1,0), (1,1) to triangle Z, X, Y. */
256 static mesh_point const_attr
257 triangle_interpolate (mesh_point z, mesh_point x, mesh_point y, mesh_point p)
258 {
259 mesh_point ret;
260 if (p.x != 0)
261 {
262 mesh_coord_t yp = p.y / p.x;
263 mesh_coord_t x1 = x.x * (1 - yp) + y.x * yp;
264 mesh_coord_t y1 = x.y * (1 - yp) + y.y * yp;
265 ret.x = z.x * (1 - p.x) + x1 * p.x;
266 ret.y = z.y * (1 - p.x) + y1 * p.x;
267 }
268 else
269 {
270 ret.x = z.x;
271 ret.y = z.y;
272 }
273 return ret;
274 }

276 /* Bilinearly interpolate point P within square defined by TL, TR, BL, BR. */
277 static mesh_point const_attr
278 interpolate (mesh_point tl, mesh_point tr, mesh_point bl, mesh_point br,
279 mesh_point p)
280 {
281 bool swap = (p.x < p.y);
282 if (swap)
283 {
284 std::swap (p.x, p.y);
285 std::swap (tr, bl);
286 }
287 p = triangle_interpolate (tl, tr, br, p);
288 return p;
289 }
