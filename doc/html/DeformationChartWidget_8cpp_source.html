105int DeformationChartWidget::heightForWidth(int width) const
106{
107 if (!m_hasData || m_scanWidth <= 0 || m_scanHeight <= 0)
108 return 350;
109
110 // Calculate desired height based on aspect ratio
111 // w_chart / h_chart = w_scan / h_scan
112
113 // Account for margins and slider height
114 const int sliderHeight = 50;
115 const int verticalMargins = 20;
116 const int horizontalMargins = 20;
117
118 int availableWidth = width - horizontalMargins;
119 if (availableWidth <= 0) availableWidth = 10;
120
121 double aspectRatio = getAspectRatio();
122 int desiredContentHeight = static_cast<int>(availableWidth / aspectRatio);
123
124 // Limit height for portrait images to prevent excessive vertical space usage.

128 // Ensure at least some visibility
129 if (desiredContentHeight < 50) desiredContentHeight = 50;
130
131 return desiredContentHeight + sliderHeight + verticalMargins;

136double DeformationChartWidget::getAspectRatio() const
137{
138 if (m_scanHeight > 0) {
139 double w = m_scanWidth;
140 double h = m_scanHeight;
141
142 // Use scan_rotation (0, 1, 2, 3) where 1=90deg
143 int angle = (m_rotation * 90) % 360;
144 if (angle < 0) angle += 360;
145
146 if (angle == 90 || angle == 270) {
147 std::swap(w, h);
148 }
149
150 return w / h;
151 }
152 return 1.0;

155float DeformationChartWidget::getExaggerationFactor() const
156{
157 if (!m_exaggerateSlider)
158 return 1.0f;
159
160 int sliderValue = m_exaggerateSlider->value();
161
162 // Logarithmic scale: 0-10000 slider maps to 1.0-10000.0 exaggeration
163 // Formula: factor = 10000.0 ^ (sliderValue / 10000.0)
164
165 float normalizedValue = sliderValue / 10000.0f; // 0.0 to 1.0
166 float factor = std::pow(10000.0f, normalizedValue); // 1.0 to 10000.0
167
168 return factor;
169}
