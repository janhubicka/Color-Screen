19/* Allocate aligned memory so vectorized loops works fast.

22template <class T> struct fft_allocator
23{
24 typedef T value_type;

26 allocate (std::size_t n)

28#ifdef HAVE_MEMALIGN

33 if (!ptr)
34 throw std::bad_alloc ();
