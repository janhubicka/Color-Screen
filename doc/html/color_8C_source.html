94/* Computatoin of DeltaE 2000
95 Taken from ArgyllCMS. */
96static luminosity_t
97deltaE2000_squared (cie_lab c1, cie_lab c2)
98{
99 double C1, C2;
100 double h1, h2;
101 double dL, dC, dH;
102 double dsq;
103
104 /* The trucated value of PI is needed to ensure that the */
105 /* test cases pass, as one of them lies on the edge of */
106 /* a mathematical discontinuity. The precision is still */
107 /* enough for any practical use. */
108#define RAD2DEG(xx) (180.0/M_PI * (xx))
109#define DEG2RAD(xx) (M_PI/180.0 * (xx))
110
111 /* Compute Cromanance and Hue angles */
112 {
113 double C1ab, C2ab;
114 double Cab, Cab7, G;
115 double a1, a2;
116
117 C1ab = my_sqrt (c1.a * c1.a + c1.b * c1.b);
118 C2ab = my_sqrt (c2.a * c2.a + c2.b * c2.b);
119 Cab = 0.5 * (C1ab + C2ab);
120 Cab7 = pow (Cab, 7.0);
121 G = 0.5 * (1.0 - my_sqrt (Cab7 / (Cab7 + 6103515625.0)));
122 a1 = (1.0 + G) * c1.a;
123 a2 = (1.0 + G) * c2.a;
124 C1 = my_sqrt (a1 * a1 + c1.b * c1.b);
125 C2 = my_sqrt (a2 * a2 + c2.b * c2.b);
126
127 if (C1 < 1e-9)
128 h1 = 0.0;
129 else
130 {
131 h1 = RAD2DEG (atan2 (c1.b, a1));
132 if (h1 < 0.0)
133 h1 += 360.0;
134 }
135
136 if (C2 < 1e-9)
137 h2 = 0.0;
138 else
139 {
140 h2 = RAD2DEG (atan2 (c2.b, a2));
141 if (h2 < 0.0)
142 h2 += 360.0;
143 }
144 }
145
146 /* Compute delta L, C and H */
147 {
148 double dh;
149
150 dL = c2.l - c1.l;
151 dC = C2 - C1;
152 if (C1 < 1e-9 || C2 < 1e-9)
153 {
154 dh = 0.0;
155 }
156 else
157 {
158 dh = h2 - h1;
159 if (dh > 180.0)
160 dh -= 360.0;
161 else if (dh < -180.0)
162 dh += 360.0;
163 }
164
165 dH = 2.0 * my_sqrt (C1 * C2) * sin (DEG2RAD (0.5 * dh));
166 }
167
168 {
169 double L, C, h, T;
170 double hh, ddeg;
171 double C7, RC, L50sq, SL, SC, SH, RT;
172 double dLsq, dCsq, dHsq, RCH;
173
174 L = 0.5 * (c1.l + c2.l);
175 C = 0.5 * (C1 + C2);
176
177 if (C1 < 1e-9 || C2 < 1e-9)
178 {
179 h = h1 + h2;
180 }
181 else
182 {
183 h = h1 + h2;
184 if (fabs (h1 - h2) > 180.0)
185 {
186 if (h < 360.0)
187 h += 360.0;
188 else if (h >= 360.0)
189 h -= 360.0;
190 }
191 h *= 0.5;
192 }
193
194 T = 1.0 - 0.17 * cos (DEG2RAD (h - 30.0)) + 0.24 * cos (DEG2RAD (2.0 * h))
195 + 0.32 * cos (DEG2RAD (3.0 * h + 6.0)) -
196 0.2 * cos (DEG2RAD (4.0 * h - 63.0));
197 L50sq = (L - 50.0) * (L - 50.0);
198
199 SL = 1.0 + (0.015 * L50sq) / my_sqrt (20.0 + L50sq);
200 SC = 1.0 + 0.045 * C;
201 SH = 1.0 + 0.015 * C * T;
202
203 dLsq = dL / SL;
204 dCsq = dC / SC;
205 dHsq = dH / SH;
206
207 hh = (h - 275.0) / 25.0;
208 ddeg = 30.0 * exp (-hh * hh);
209 C7 = pow (C, 7.0);
210 RC = 2.0 * my_sqrt (C7 / (C7 + 6103515625.0));
211 RT = -sin (DEG2RAD (2 * ddeg)) * RC;
212
213 RCH = RT * dCsq * dHsq;
214
215 dLsq *= dLsq;
216 dCsq *= dCsq;
217 dHsq *= dHsq;
218
219 dsq = dLsq + dCsq + dHsq + RCH;
220 }
221
222 return dsq;
223
224#undef RAD2DEG
225#undef DEG2RAD
226}

257/* Compute intersection vector (x1, y1)->(x2, y2) and line segment (x3,y3)-(x4,y4).
258 Return true if it lies on the line sergment and in positive direction of the vector.
259 If true is returned, t is initialized to position of the point in the line. */
260static bool
261intersect (luminosity_t x1, luminosity_t y1, luminosity_t x2, luminosity_t y2,
262 luminosity_t x3, luminosity_t y3, luminosity_t x4, luminosity_t y4, luminosity_t *t)
263{
264 luminosity_t a = x1;
265 luminosity_t b = x2-x1;
266 luminosity_t c = x3;
267 luminosity_t d = x4-x3;
268 luminosity_t e = y1;
269 luminosity_t f = y2-y1;
270 luminosity_t g = y3;
271 luminosity_t h = y4-y3;
272
273 luminosity_t rec = 1/(d*f-b*h);
274 if (d == 0)
275 return false;
276 luminosity_t p1 = (a*h-c*h-d*e+d*g) * rec;
277 luminosity_t p2 = (e*f - b*e + b*g - c*f) * rec;
278 if (p1 < 0 || p2 < 0 || p2 > 1)
279 return false;
280 *t = p2;
281
282 return true;
283}

285/* Dominant wavelength is computed as intersection of the line from whitepoint to a given
286 color with the horsehoe of CIE1931.
287 Not all colors have dominant wavelengths; return 0 if they does not */
288
289luminosity_t
290dominant_wavelength (xy_t color, xy_t whitepoint)
291{
292 for (int i = 0; i < SPECTRUM_SIZE - 1; i++)
293 {
294 /* Compute points of the horsehose observer. */
295 xy_t c1 ((xyz){cie_cmf_x[i], cie_cmf_y[i], cie_cmf_z[i]});
296 xy_t c2 ((xyz){cie_cmf_x[i+1], cie_cmf_y[i+1], cie_cmf_z[i+1]});
297 luminosity_t t;
298 /* Compute the intersection. */
299 if (intersect (whitepoint.x, whitepoint.y, color.x, color.y,
300 c1.x, c1.y, c2.x, c2.y, &t))
301 return SPECTRUM_START + (i + t) * SPECTRUM_STEP;
302 }
303 return 0;
304}

306/* Find whitepoint so that red, green and blue dominant wavelengths matches
307 as well as possible to the specified values. */
308
309xy_t
310find_best_whitepoint (xyz red, xyz green, xyz blue,
311 luminosity_t red_dominating_wavelength,
312 luminosity_t green_dominating_wavelength,
313 luminosity_t blue_dominating_wavelength)
314{
315 xy_t best_white(-1, -1);
316 bool best_white_found = 0;
317 luminosity_t best_white_dist = 0;
318 for (luminosity_t x = 0.01; x < 1; x += 0.01)
319 for (luminosity_t y = 0.01; y < 1; y += 0.01)
320 {
321 xy_t w (x, y);
322 luminosity_t d1 = dominant_wavelength (red, w);
323 luminosity_t d2 = dominant_wavelength (green, w);
324 luminosity_t d3 = dominant_wavelength (blue, w);
325 if (!d1 || !d2 || !d3)
326 continue;
327 luminosity_t d = fabs (red_dominating_wavelength - d1) + fabs (green_dominating_wavelength - d2) + fabs (blue_dominating_wavelength - d3);
328 if (!best_white_found || d < best_white_dist)
329 {
330 best_white = w;
331 best_white_dist = d;
332 best_white_found = true;
333 }
334 }
335 return best_white;
336}
