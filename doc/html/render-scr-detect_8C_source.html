505int cmp_entry(const void *p1, const void *p2)
506{
507 distance_list::entry *e1 = (distance_list::entry *)p1;
508 distance_list::entry *e2 = (distance_list::entry *)p2;
509 if (e1->fdist < e2->fdist)

523 if (dist > max_distance)

525 list[num].x = x;
526 list[num].y = y;
