30tiff_writer::tiff_writer (tiff_writer_params &p, const char **error)
31{
32 *error = nullptr;
33 if (p.hdr)
34 {
35 if (p.depth == 16)
36 pixel_format = !p.alpha ? pixel_16bit_hdr : pixel_16bit_hdr_alpha;
37 else if(p.depth == 32)
38 pixel_format = !p.alpha ? pixel_32bit_hdr : pixel_16bit_hdr_alpha;
39 else
40 {
41 *error = "unsupported bit depth in HDR tiff file";
42 return;
43 }
44 }
45 else
46 {
47 if (p.depth == 8)
48 pixel_format = !p.alpha ? pixel_8bit : pixel_8bit_alpha;
49 else if (p.depth == 16)
50 pixel_format = !p.alpha ? pixel_16bit : pixel_16bit_alpha;
51 else
52 {
53 *error = "unsupported bit depth in TIFF file";
54 return;
55 }
56 }
57 out = TIFFOpen (p.filename, "wb");
58 y = 0;
59
60 if (!out)
61 {
62 *error = "cannot open tiff file";
63 return;
64 }
65 static uint16_t extras[] = {EXTRASAMPLE_UNASSALPHA};
66 int samplesperpixel = p.alpha ? 4 : 3;
67 /* We must set some DPI since offset is specified relative to it. */
68 if (p.tile && !p.xdpi)
69 p.xdpi = p.ydpi = 300;
70 if (!TIFFSetField (out, TIFFTAG_IMAGEWIDTH, (uint32_t) p.width)
71 || !TIFFSetField (out, TIFFTAG_IMAGELENGTH, (uint32_t) p.height)
72 || !TIFFSetField (out, TIFFTAG_SAMPLESPERPIXEL, samplesperpixel)
73 || !TIFFSetField (out, TIFFTAG_BITSPERSAMPLE, p.depth)
74 || !TIFFSetField (out, TIFFTAG_SAMPLEFORMAT, p.hdr ? SAMPLEFORMAT_IEEEFP : SAMPLEFORMAT_UINT)
75 || !TIFFSetField (out, TIFFTAG_ORIENTATION, ORIENTATION_TOPLEFT)
76 || !TIFFSetField (out, TIFFTAG_PLANARCONFIG, PLANARCONFIG_CONTIG)
77 || !TIFFSetField (out, TIFFTAG_PHOTOMETRIC, PHOTOMETRIC_RGB)
78 || (p.alpha && !TIFFSetField (out, TIFFTAG_EXTRASAMPLES, 1, extras))
79 || !TIFFSetField (out, TIFFTAG_SOFTWARE, "Color-screen")
80 || (p.xdpi && !TIFFSetField (out, TIFFTAG_XRESOLUTION, (double)p.xdpi))
81 || (p.ydpi && !TIFFSetField (out, TIFFTAG_YRESOLUTION, (double)p.ydpi)))
82 {
83 *error = "write error";
84 TIFFClose (out);
85 out = nullptr;
86 return;
87 }
88 if (!p.dng)
89 {
90 if (!TIFFSetField (out, TIFFTAG_ICCPROFILE, p.icc_profile ? (uint32_t)p.icc_profile_len : (uint32_t) sRGB_icc_len, p.icc_profile ? p.icc_profile : sRGB_icc)
91 || !TIFFSetField (out, TIFFTAG_COMPRESSION, COMPRESSION_LZW))
92 {
93 *error = "write error";
94 TIFFClose (out);
95 out = nullptr;
96 return;
97 }
98 }
99 if (p.tile)
100 {
101 if (!TIFFSetField (out, TIFFTAG_XPOSITION, ((double)p.xoffset / p.xdpi))
102 || !TIFFSetField (out, TIFFTAG_YPOSITION, ((double)p.yoffset / p.ydpi))
103 || !TIFFSetField (out, TIFFTAG_PIXAR_IMAGEFULLWIDTH, (long)(p.width + p.xoffset))
104 || !TIFFSetField (out, TIFFTAG_PIXAR_IMAGEFULLLENGTH, (long)(p.height + p.yoffset)))
105 {
106 *error = "write error";
107 TIFFClose (out);
108 out = nullptr;
109 return;
110 }
111 }
112 if (p.dng)
113 {
114 TIFFMergeFieldInfo (out, tiffFields, sizeof (tiffFields) / sizeof (TIFFFieldInfo));
115 long white[] = {0xffff, 0xffff, 0xffff, 0xffff};
116 float forward_matrix[] = {
117 (float)p.dye_to_xyz(0, 0), (float)p.dye_to_xyz(0, 1), (float)p.dye_to_xyz(0, 2),
118 (float)p.dye_to_xyz(1, 0), (float)p.dye_to_xyz(1, 1), (float)p.dye_to_xyz(1, 2),
119 (float)p.dye_to_xyz(2, 0), (float)p.dye_to_xyz(2, 1), (float)p.dye_to_xyz(2, 2)};
120 color_matrix m = p.dye_to_xyz.invert ();
121 float cam_xyz[] = {
122 (float)m(0, 0), (float)m(0, 1), (float)m(0, 2),
123 (float)m(1, 0), (float)m(1, 1), (float)m(1, 2),
124 (float)m(2, 0), (float)m(2, 1), (float)m(2, 2)};
125 float neutral[3] = {1, 1, 1};
126 float tonecurve[] = {0, 0, 1, 1};
127 short black_level[] = {(short)p.black, (short)p.black, (short)p.black, (short)p.black};
128 if (p.parallel)
129 printf ("Black level: %i\n", p.black);
130 /* TODO: Thumbnail should be subfiletype 1. */
131 if (!TIFFSetField (out, TIFFTAG_SUBFILETYPE, 0)
132 || !TIFFSetField (out, TIFFTAG_DNGVERSION, "\001\004\0\0")
133 || !TIFFSetField (out, TIFFTAG_DNGBACKWARDVERSION, "\001\0\0\0")
134 || !TIFFSetField (out, TIFFTAG_COLORMATRIX1, 9, cam_xyz)
135 || !TIFFSetField (out, TIFFTAG_FORWARDMATRIX1, 9, forward_matrix)
136 || !TIFFSetField (out, TIFFTAG_ASSHOTNEUTRAL, 3, neutral)
137 || !TIFFSetField (out, TIFFTAG_MAKE, "Early color photography")
138 || !TIFFSetField (out, TIFFTAG_MODEL, "Really old camera")
139 || !TIFFSetField (out, TIFFTAG_UNIQUECAMERAMODEL, "Long forgotten")
140 || !TIFFSetField (out, TIFFTAG_CALIBRATIONILLUMINANT1, 21 /*21 is D65, 17 is IL A*/)
141 || !TIFFSetField (out, TIFFTAG_BLACKLEVEL, samplesperpixel, &black_level)
142 || !TIFFSetField (out, TIFFTAG_WHITELEVEL, samplesperpixel, &white)
143 || !TIFFSetField (out, TIFFTAG_PROFILETONECURVE, 4, tonecurve))
144 {
145 *error = "write error";
146 TIFFClose (out);
147 out = nullptr;
148 return;
149 }
150 }
151 height = p.height;
152 stride = p.width * (p.alpha ? 4 : 3);
153 bytestride = (size_t)p.depth * stride / 8;
154 /* If parallelism is supported, try to write blocks of at least
155 one megapixel. */
156 if (p.parallel)
157 {
158 n_rows = (1024*1024 + p.width - 1) / p.width;
159 if (n_rows > height)
160 n_rows = height;
161 }
162 else
163 n_rows = 1;
164 outrow = std::make_unique<uint8_t[]> (bytestride * n_rows);
165}

168bool
169tiff_writer::write_rows (progress_info *progress)
170{
171 for (int i = 0; i < n_rows; i++)
172 {
173 if (progress && progress->cancel_requested ())
174 return false;
175 if (TIFFWriteScanline (out, (void *)(outrow.get () + bytestride * i), y++, 0) < 0)
176 {
177 TIFFClose (out);
178 out = nullptr;
179 return false;
180 }
181 if (progress)
182 progress->inc_progress ();
183 }
184 assert (y <= height);
185 if (y + n_rows > height)
186 n_rows = height - y;
187 return true;
188}
