182// ============================================================================
183// LANDING Y HELPER
184// Returns the Y screen coordinate of the surface of the 2nd-from-bottom
185// foreground dune at screen X SCREENX, given current camera offset.
186// ============================================================================
187
188double HurleyAnimation::duneLandingY(double worldX) const {
189 // 2nd from bottom = index DUNE_FG_COUNT - 2
190 int i = DUNE_FG_COUNT - 2;
191 int h = height();
192 double zoneTop = h * 0.45;
193 double zoneH = h * 0.58; // Match drawDuneFgStrips
194 double stripH = zoneH / DUNE_FG_COUNT;
195
196 double yBase = zoneTop + i * stripH;
197 double perspect = 0.5 + 0.5 * ((double)(i + 1) / DUNE_FG_COUNT);
198 double amp = stripH * 0.65 * m_duneFg[i].currentAmpFactor * perspect;
199 double freq = DUNE_WAVE_FREQ_FG * (0.85 + 0.3 * qCos(i * 0.5));
200
201 double angle = worldX * freq + m_duneFg[i].phase;
202 return yBase + amp * qSin(angle);
203}

213int HurleyAnimation::findBestPlaneSlot(bool isHero) {
214 // 1. Look for truly free slot
215 for (int i = 0; i < MAX_PLANES; ++i) {
216 if (!m_planes[i].active) return i;
217 }
218
219 // 2. Pool Saturation: Hijack oldest ground wreckage
220 double maxTime = -1.0;
221 int bestSlot = -1;
222 for (int i = 0; i < MAX_PLANES; ++i) {
223 // Hero can hijack any non-hero debris.
224 // Allied/Enemy only hijack very old debris to avoid "popping" too many wrecks.
225 if (m_planes[i].active && !m_planes[i].isHero && m_planes[i].state == PlaneState::Landed) {
226 if (m_planes[i].landedTimer > maxTime) {
227 maxTime = m_planes[i].landedTimer;
228 bestSlot = i;
229 }
230 }
231 }
232
233 // If still no slot (all flying or pool full of fresh wreckage),
234 // Hero MUST have a slot; take any non-hero.
235 if (bestSlot < 0 && isHero) {
236 for (int i = 0; i < MAX_PLANES; ++i) {
237 if (!m_planes[i].isHero) return i;
238 }
239 }
240
241 if (bestSlot >= 0) {
242 m_planes[bestSlot].active = false; // Mark for reuse
243 }
244 return bestSlot;
245}

384void HurleyAnimation::fireFromPlane(int planeIdx) {
385 const Airplane &shooter = m_planes[planeIdx];
386
387 // Current forward vector in global space
388 double fxHead = shooter.facingRight ? 1.0 : -1.0;
389 double rad = qDegreesToRadians(shooter.angle);
390 double fwdX = fxHead * qCos(rad);
391 double fwdY = qSin(rad);
392
393 // Vector pointing "Local Down" (under the wings) relative to flight path
394 // For mirrored planes, clockwise rotation from fwd towards down is still dot > 0
395 // consistently if we use: downX = -faceScale * fwdY, downY = faceScale * fwdX.
396 double downX = -fxHead * fwdY;
397 double downY = fxHead * fwdX;
398
399 // Find nearest enemy in range and NOT "under" the aircraft
400 double bestDist = PLANE_SHOOT_RANGE;
401 int bestTarget = -1;
402 for (int j = 0; j < MAX_PLANES; ++j) {
403 if (j == planeIdx) continue;
404 const Airplane &t = m_planes[j];
405 if (!t.active || t.state != PlaneState::Flying) continue;
406 if (t.team == shooter.team) continue;
407
408 double dx = t.x - shooter.x;
409 double dy = t.y - shooter.y;
410 double dist = qSqrt(dx*dx + dy*dy);
411
412 if (dist < bestDist) {
413 // Check if target is "under" the nose-line using dot product with local down
414 double dot = dx * downX + dy * downY;
415 if (dot > 0) continue; // Forbidden zone (under the airplane)
416
417 bestDist = dist;
418 bestTarget = j;
419 }
420 }
421 if (bestTarget < 0) return;
422
423 // Find free bullet slot
424 for (int k = 0; k < MAX_BULLETS; ++k) {
425 if (!m_bullets[k].active) {
426 Bullet &b = m_bullets[k];
427 b.active = true;
428 b.team = shooter.team;
429 b.shooterIdx = planeIdx;
430 b.x = shooter.x;
431 b.y = shooter.y;
432
433 // Aim at predicted position
434 const Airplane &tgt = m_planes[bestTarget];
435 double T = bestDist / BULLET_SPEED;
436 double predX = tgt.x + tgt.vx * T * 0.6;
437 double predY = tgt.y + tgt.vy * T * 0.3;
438
439 // Final verification: Is predicted trajectory under the nose-line?
440 double pdx = predX - shooter.x;
441 double pdy = predY - shooter.y;
442 if (pdx * downX + pdy * downY > 0) return;
443
444 double bdx = predX - b.x;
445 double bdy = predY - b.y;
446 double blen = qSqrt(bdx*bdx + bdy*bdy);
447 if (blen < 1.0) blen = 1.0;
448 b.vx = BULLET_SPEED * bdx / blen;
449 b.vy = BULLET_SPEED * bdy / blen;
450 break;
451 }
452 }
453}

537void HurleyAnimation::updatePlanes(double dt) {
538 double h = height();
539 double airYMin = h * 0.04;
540 double airYMax = h * 0.65; // Expanded downwards to allow flight behind FG dunes
541
542 for (int i = 0; i < MAX_PLANES; ++i) {
543 Airplane &a = m_planes[i];
544 if (!a.active) continue;
545
546 if (a.state == PlaneState::Flying) {
547 // ------------------------------------------------------------------
548 // 1. AI & Control
549 // ------------------------------------------------------------------
550 double groundY = getGroundY(a.x);
551
552 if (!a.pilotEjected) {
553 bool isManual = (a.isHero && m_heroManualControl);
554
555 if (!isManual) {
556 a.jinkTimer -= dt;
557 if (a.jinkTimer <= 0) {
558 // Default: wander and maintain altitude
559 double midY = (airYMin + airYMax) * 0.5;
560 double altFactor = (midY - a.y) / (h * 0.15);
561 double baseA = -altFactor * 40.0;
562 double wander = randRange(-PLANE_JINK_ANGLE, PLANE_JINK_ANGLE);
563 a.targetAngle = baseA + wander;
564
565 // Pursuit: Look for nearest enemy
566 double bestDist = PLANE_CHASE_RANGE;
567 int bestTarget = -1;
568 for (int j = 0; j < MAX_PLANES; ++j) {
569 const Airplane &t = m_planes[j];
570 if (!t.active || t.state != PlaneState::Flying || t.team == a.team) continue;
571 double dist = qSqrt(qPow(t.x - a.x, 2) + qPow(t.y - a.y, 2));
572 if (dist < bestDist) {
573 // Check if in front quadrant (roughly)
574 double dx = (t.x - a.x) * (a.facingRight ? 1 : -1);
575 if (dx > 0) { // Target is in front
576 bestDist = dist;
577 bestTarget = j;
578 }
579 }
580 }
581
582 // Apply Pitch Authority: At low speeds, restrict target angle to maintain lift
583 double speed = qSqrt(a.vx * a.vx + a.vy * a.vy);
584 double speedRel = speed / PLANE_CRUISE_SPEED;
585 // Authority is 1.0 at cruise speed, falls off squared below cruise
586 double authority = qBound(0.15, speedRel * speedRel, 1.2);
587
588 if (bestTarget >= 0) {
589 const Airplane &t = m_planes[bestTarget];
590 double dx = (t.x - a.x) * (a.facingRight ? 1 : -1);
591 double dy = t.y - a.y;
592 double rawAngle = qRadiansToDegrees(qAtan2(dy, dx));
593 // Clamp pursuit angle based on authority
594 a.targetAngle = rawAngle * authority;
595 a.jinkTimer = 0.5; // Frequent updates while chasing
596 } else {
597 // 10% chance: do a full loop (Only if speed is high enough!)
598 if (authority > 0.8 && QRandomGenerator::global()->bounded(100) < 10) {
599 a.targetAngle = (randRange(0, 1) > 0.5 ? 360.0 : -360.0);
600 a.jinkTimer = 2.5;
601 } else {
602 // Regular wandering, also scaled by authority
603 double limitedWander = randRange(-PLANE_JINK_ANGLE * authority,
604 PLANE_JINK_ANGLE * authority);
605 a.targetAngle = baseA * authority + limitedWander;
606 a.jinkTimer = randRange(PLANE_JINK_INTERVAL_MIN, PLANE_JINK_INTERVAL_MAX);
607 }
608 }
609 }
610
611 // Ceiling Awareness: Every frame penalty if too high
612 if (a.y < airYMin + 100.0) {
613 double danger = (airYMin + 100.0 - a.y) / 100.0;
614 // Force target angle to be at least N degrees down
615 double minDown = danger * 60.0;
616 if (a.targetAngle < minDown) a.targetAngle = minDown;
617 }
618 }
619
620 // Pitch control: AngularVel accelerates toward targetAngle
621 double err = a.targetAngle - a.angle;
622 // Shortest path wrap
623 while (err > 180) err -= 360;
624 while (err < -180) err += 360;
625
626 double wantedAngVel = qBound(-PLANE_MAX_PITCH, err * 5.0, PLANE_MAX_PITCH);
627 double acc = (wantedAngVel - a.angularVel) * 10.0;
628 acc = qBound(-PLANE_PITCH_ACCEL, acc, PLANE_PITCH_ACCEL);
629 a.angularVel += acc * dt;
630 a.angle += a.angularVel * dt;
631
632 if (!isManual) {
633 // Throttle control: adaptive to help with altitude
634 double targetCruise = PLANE_CRUISE_SPEED;
635 double midY = (airYMin + airYMax) * 0.5;
636 double speed = qSqrt(a.vx * a.vx + a.vy * a.vy);
637
638 // 1. Predictive Panic Recovery: If imminent crash, pull up and throttle up
639 double predictedY = a.y + a.vy * 1.5; // Look ahead 1.5 seconds
640 bool panic = (predictedY > groundY) && (a.takeoffTimer <= 0);
641
642 if (panic) {
643 targetCruise = 260.0; // Panic speed!
644 a.targetAngle = -15.0; // Force a horizontal-to-climb attitude
645 } else if (a.y > midY + 50.0) {
646 // Regular recovery logic
647 targetCruise += 120.0;
648 a.targetAngle = a.targetAngle * 0.65;
649 }
650
651 // 2. Speed Governor: If too fast, kill engine to stabilize
652 if (speed > 240.0 && !panic) {
653 targetCruise = 100.0;
654 }
655
656 // Standard altitude-based throttle bias
657 if (a.y > midY + 40.0) targetCruise += 40.0;
658 if (a.y < midY - 40.0) targetCruise -= 40.0;
659
660 // 3. Falling Flat recovery: if losing height and nose is near horizontal, boost!
661 if (!panic && a.vy > 15.0 && std::abs(a.angle) < 20.0) {
662 targetCruise += 100.0;
663 }
664
665 double speedErr = targetCruise - speed;
666 double responseK = panic ? 0.15 : 0.02; // React much faster if crashing!
667 a.throttle += speedErr * responseK * dt;
668 }
669
670 a.throttle = qBound(0.2, a.throttle, 1.0);
671
672 // Advance propeller phase based on engine speed
673 a.propPhase += dt * (25.0 + a.throttle * 90.0);
674
675 // Take-off logic: keep nose horizontal and handle skimming
676 if (a.takeoffTimer > 0) {
677 a.takeoffTimer -= dt;
678 if (!isManual) a.targetAngle = 0; // Forced horizontal run
679 }
680 }
681
682 // ------------------------------------------------------------------
683 // 2. Physics Simulation: Lateral Stability & Aerodynamics
684 // ------------------------------------------------------------------
685 // ------------------------------------------------------------------
686 // 2. Physics Simulation: Vector-Based Inertia & Uplift
687 // ------------------------------------------------------------------
688 double rad = qDegreesToRadians(a.angle);
689 double cosA = cos(rad);
690 double sinA = sin(rad);
691
692 double fxHead = a.facingRight ? 1.0 : -1.0;
693 // The sprite transformation is Rotate(A) * Scale(fxHead, 1)
694 // Forward (Global) = R(A) * S(fxHead, 1) * (1, 0) = (fxHead*cosA, fxHead*sinA)
695 // Up (Global) = R(A) * S(fxHead, 1) * (0, -1) = (sinA, -cosA)
696 a.phys.fwdX = fxHead * cosA;
697 a.phys.fwdY = fxHead * sinA;
698 a.phys.upX = sinA;
699 a.phys.upY = -cosA;
700
701 // Inertia only in the direction of airplane nose
702 double v_para = std::max (a.vx * a.phys.fwdX + a.vy * a.phys.fwdY, 0.0);
703
704 // Thrust and Lift vectors (Quadratic Lift)
705 double thrust = a.throttle * PLANE_THRUST_CONST;
706 double lift = (v_para * v_para) * PLANE_LIFT_CONST;
707
708 a.phys.thrustX = a.phys.fwdX * thrust;
709 a.phys.thrustY = a.phys.fwdY * thrust;
710 a.phys.liftX = a.phys.upX * lift;
711 a.phys.liftY = a.phys.upY * lift;
712
713 // Movement is combination of inertia * slowdown + forces
714 double dvx = a.phys.thrustX + a.phys.liftX;
715 double dvy = a.phys.thrustY + a.phys.liftY + GRAVITY;
716
717 // V_new = (V_old + Force * dt)
718 a.vx += dvx * dt;
719 a.vy += dvy * dt;
720
721 // --- Dynamic Slowdown (Parasitic + Induced Drag) ---
722 // 1. Parasitic: Increases with speed
723 double currentSpeed = qSqrt(a.vx * a.vx + a.vy * a.vy);
724 double dynamicSlowdown = PLANE_SLOWDOWN * (currentSpeed / PLANE_CRUISE_SPEED);
725
726 // 2. Induced: Increases with lift (more maneuvering = more bleed)
727 double inducedSlowdown = lift * PLANE_INDUCED_DRAG_CONST;
728 double totalSlowdown = dynamicSlowdown + inducedSlowdown;
729
730 // Apply exponential decay using the user's stable pow() formula.
731 // We cap the coefficient to prevent numerical instability at extreme speeds.
732 double dragFactor = pow(1.0 - qMin(0.99, totalSlowdown), dt);
733 a.vx *= dragFactor;
734 a.vy *= dragFactor;
735
736 // --- Directional Stability (Vaning Effect / Carving) ---
737 // We strictly dampen the "sideways" velocity (sideslip) relative to the nose.
738 // This forces the aircraft to track its longitudinal axis like a vane.
739 double n_para = a.vx * a.phys.fwdX + a.vy * a.phys.fwdY;
740 double n_perp = a.vx * a.phys.upX + a.vy * a.phys.upY;
741
742#if 1
743 // Kill 95% of lateral speed per second
744 n_perp *= pow (1.0 - 0.95, dt);
745
746 a.vx = n_para * a.phys.fwdX + n_perp * a.phys.upX;
747 a.vy = n_para * a.phys.fwdY + n_perp * a.phys.upY;
748#endif
749
750
751 // Integrate Position
752 a.x += a.vx * dt;
753 a.y += a.vy * dt;
754
755 // --- Engine Damage Smoke ---
756 int maxH = a.isHero ? PLANE_HEALTH_HERO : PLANE_HEALTH_NORMAL;
757 if (a.health < maxH && a.state == PlaneState::Flying) {
758 double damage = 1.0 - (double)a.health / maxH;
759 // Emit up to 14 particles/sec at near-death
760 a.smokeAccum += dt * (damage * 14.0);
761 if (a.smokeAccum >= 1.0) {
762 double scale = a.isHero ? PLANE_HERO_SCALE : PLANE_NORMAL_SCALE;
763 // Engine cowling is just behind propeller (prop is at 41.4)
764 double localE = 37.0;
765 double ex = a.x + a.phys.fwdX * (localE * scale);
766 double ey = a.y + a.phys.fwdY * (localE * scale);
767 // Spawn single puff with reduced velocity
768 spawnSmoke(ex, ey, a.vx * 0.4, a.vy * 0.4, 1);
769 a.smokeAccum -= 1.0;
770 }
771 }
772
773
774 // No hard bounce at top anymore, handled by AI awareness
775 // if (a.y < airYMin) { a.y = airYMin; a.vy = qAbs(a.vy) * 0.4; }
776
777 // Ground crash check
778
779 // Predictive Emergency Catapult: Bailing out just before a high-speed impact
780 // Note: During takeoff skimming, we are immune to this impact-panic.
781 if (!a.pilotEjected && a.vy > 50.0 && a.takeoffTimer <= 0) {
782 if (a.y + a.vy * (dt * 1.5) >= groundY) { // Look ahead slightly
783 spawnPilotGuaranteed(a);
784 }
785 }
786
787 if (a.y >= groundY) {
788 if (a.takeoffTimer > 0) {
789 // Ground-skimming immunity: slide along the sand
790 a.y = groundY;
791 if (a.vy > 0) a.vy = 0;
792 } else {
793 // High-speed collision or failed take-off
794 if (a.isHero) {
795 m_score -= 50;
796 setSunExpression(SunExpression::Sad, 2.5);
797 }
798 a.y = groundY;
799 a.state = PlaneState::Falling;
800 a.fallY = groundY;
801 a.health = 0;
802 // Immediate explosion puff
803 spawnSmoke(a.x, a.y, 0, 0, 15);
804 }
805 }
806
807 // ---- Shooting ----
808 if (!a.pilotEjected) {
809 a.shootCooldown -= dt;
810
811 if (a.isHero && m_heroManualFire) {
812 if (m_isMouseDown && a.shootCooldown <= 0) {
813 // Manual firing logic towards mouse
814 double dx = (m_mousePos.x() + m_cameraX) - a.x;
815 double dy = m_mousePos.y() - a.y;
816 double dist = qSqrt(dx*dx + dy*dy);
817
818 if (dist > 5.0) {
819 // Upper hemisphere check (robust dot-product)
820 double fxHead = a.facingRight ? 1.0 : -1.0;
821 double rad = qDegreesToRadians(a.angle);
822 double fwdX = fxHead * qCos(rad);
823 double fwdY = qSin(rad);
824 double downX = -fxHead * fwdY;
825 double downY = fxHead * fwdX;
826
827 if (dx * downX + dy * downY <= 0) {
828 // Find free bullet slot
829 for (int k = 0; k < MAX_BULLETS; ++k) {
830 if (!m_bullets[k].active) {
831 Bullet &b = m_bullets[k];
832 b.active = true;
833 b.team = a.team;
834 b.shooterIdx = i;
835 b.x = a.x;
836 b.y = a.y;
837 b.vx = BULLET_SPEED * dx / dist;
838 b.vy = BULLET_SPEED * dy / dist;
839 break;
840 }
841 }
842 a.shootCooldown = 0.08; // Machine gun rate
843 }
844 }
845 }
846 } else if (a.shootCooldown <= 0) {
847 double cd = a.isHero ? PLANE_HERO_SHOOT_CD : PLANE_SHOOT_COOLDOWN;
848 fireFromPlane(i);
849 a.shootCooldown = cd * randRange(0.7, 1.3);
850 }
851 }
852
853 // ---- Despawn / Respawn Check ----
854 double sx = toScreenX(a.x);
855 if (sx < -150 || sx > width() + 150 || a.y < -400 || a.y > height() + 400) {
856 // Non-hero planes simply deactivate.
857 // Hero plane deactivation is handled by the dedicated monitor below.
858 if (!a.isHero) a.active = false;
859 }
860
861 } else if (a.state == PlaneState::Falling) {
862 // Tumble and fall (simple gravity + drag)
863 a.vy += GRAVITY * dt;
864 a.vx *= (1.0 - 0.5 * dt);
865 a.x += a.vx * dt;
866 a.y += a.vy * dt;
867 a.spinAngle += PLANE_FALL_SPIN_SPEED * dt;
868 a.angle = a.spinAngle;
869
870 a.smokeAccum += dt * SMOKE_EMIT_RATE;
871 while (a.smokeAccum >= 1.0) {
872 spawnSmoke(a.x, a.y, a.vx, a.vy, 1);
873 a.smokeAccum -= 1.0;
874 }
875
876 if (a.y >= a.fallY) {
877 a.y = a.fallY; a.vy = 0; a.vx = 0; a.state = PlaneState::Landed;
878 if (!a.pilotEjected) {
879 a.pilotEjected = true;
880 spawnPilotGuaranteed(a);
881 }
882 }
883
884 } else if (a.state == PlaneState::Landed) {
885 // No state-specific deactivation here anymore; handled above.
886 }
887 }
888
889 // -----------------------------------------------------------------------
890 // DEDICATED HERO MONITOR
891 // -----------------------------------------------------------------------
892 bool heroActiveInPool = false;
893 for (int i = 0; i < MAX_PLANES; ++i) {
894 if (m_planes[i].active && m_planes[i].isHero) {
895 heroActiveInPool = true;
896 Airplane &hero = m_planes[i];
897 m_heroIdx = i; // Re-sync index
898
899 if (hero.state != PlaneState::Flying) {
900 // Hero aircraft is downed (Falling or Landed).
901 // Check if the camera pilot is still descending.
902 bool cameraActive = false;
903 for (int pIdx = 0; pIdx < MAX_PILOTS; ++pIdx) {
904 if (m_pilots[pIdx].active && m_pilots[pIdx].hasCamera) {
905 cameraActive = true;
906 break;
907 }
908 }
909
910 // If the pilot has vanished, the mission continues!
911 if (!cameraActive) {
912 // Detach old wreckage so it cleans up naturally
913 hero.isHero = false;
914 m_heroIdx = -1;
915 spawnHero();
916 }
917 }
918 break;
919 }
920 }
921
922 // Fallback: If no Hero plane is active in the pool at all, spawn one.
923 if (!heroActiveInPool) {
924 m_heroIdx = -1;
925 spawnHero();
926 }
927}

972

975 m_score -= 50;
976 setSunExpression(SunExpression::Sad, 2.5);

1006 setSunExpression(SunExpression::Sad, 2.0);
1007 }
1008
1009 p.parachuteDestroyed = true;
1010 p.parachuteOpen = false;
1011 b.active = false;
1012 spawnSmoke(p.x, p.y, p.vx, p.vy, 3);
1013 break;
1014 }
1015 }
1016 }

1380void HurleyAnimation::drawDuneBgStrips(QPainter &p) {

1383 double zoneH = h * 0.22; // How tall the BG zone is
1384 double stripH = zoneH / DUNE_BG_COUNT;
1385
1386 // Integrated Parallax Index mapping: Total 9 strips (6 FG, 3 BG)
1387 // i_from_bottom: 0=closest FG, 5=furthest FG, 6=closest BG, 8=furthest BG

1389 constexpr double DUNE_DISTANCE = 0.5;

1392 // Index mapping for BG: m_duneBg[0] is top (furthest), m_duneBg[2] is bottom (closest BG)
1393 int i_from_bottom = 8 - i;
1394 double F = (MIN_DISTANCE + i_from_bottom * DUNE_DISTANCE) / (MIN_DISTANCE + 2 * DUNE_DISTANCE);
1395 double bgOffsetX = m_cameraX / F;
1396
1397 double yBase = zoneTop + i * stripH;
1398
1399 // Perspective: strips closer to bottom are slightly larger
1400 double perspect = 0.35 + 0.65 * ((double)(i + 1) / DUNE_BG_COUNT);
1401 double amp = stripH * 0.55 * m_duneBg[i].currentAmpFactor * perspect;
1402 double freq = DUNE_WAVE_FREQ_BG * (0.85 + 0.3 * qCos(i * 0.6));
1403
1404 // Sandy gold gradient per strip: lighter at top, richer at bottom
1405 double t = (double)i / (DUNE_BG_COUNT - 1);
1406 QColor sandTop(220 - (int)(t*20), 195 - (int)(t*40), 100 + (int)(t*30));
1407 QColor sandBot(195 - (int)(t*20), 165 - (int)(t*30), 70 + (int)(t*20));
1408
1409 QPolygonF poly;
1410 for (int sx = 0; sx <= w; sx += 4) {
1411 double worldX = sx + bgOffsetX;
1412 double angle = worldX * freq; // No procedural phase
1413 double waveY = amp * qSin(angle);
1414 poly << QPointF(sx, yBase + waveY);
1415 }
1416 double bottomY = yBase + stripH * 1.2 + amp;
1417 poly << QPointF(w, bottomY) << QPointF(0, bottomY);
1418
1419 // Fill with sandy gradient
1420 QLinearGradient grad(0, yBase, 0, bottomY);
1421 grad.setColorAt(0.0, sandTop);
1422 grad.setColorAt(1.0, sandBot);
1423 p.setBrush(grad);
1424 p.setPen(Qt::NoPen);
1425 p.drawPolygon(poly);
1426
1427 // Subtle shadow at crest
1428 QPainterPath path;
1429 path.addPolygon(poly);
1430 QLinearGradient shadow(0, yBase, 0, bottomY);
1431 shadow.setColorAt(0.0, QColor(0, 0, 0, 0));
1432 shadow.setColorAt(0.3, QColor(0, 0, 0, 20));
