22/* Parameters for tile-based analysis caching. */
23struct analyzer_params
24{
25 uint64_t img_id;
26 uint64_t graydata_id;
27 uint64_t simulated_screen_id;
28 uint64_t screen_id;
29 luminosity_t gamma;
30 enum analyze_base::mode mode;
31 luminosity_t collection_threshold;
32 uint64_t mesh_trans_id;
33 scr_to_img_parameters params;
34
35 const image_data *img;
36 const screen *scr;
37 class render_to_scr *render;
38 class scr_to_img *scr_to_img_map;
39 struct simulated_screen *simulated_screen_ptr;
40
41 /* Return true if this structure is equal to O. */
42 bool
43 operator== (const analyzer_params &o) const
44 {
45 if (mode != o.mode || mesh_trans_id != o.mesh_trans_id
46 || simulated_screen_id != o.simulated_screen_id
47 || (!mesh_trans_id && params != o.params)
48 || params.type != o.params.type)
49 return false;
50 if (mode == analyze_base::color || mode == analyze_base::precise_rgb)
51 {
52 if (img_id != o.img_id || gamma != o.gamma)
53 return false;
54 }
55 else if (graydata_id != o.graydata_id)
56 return false;
57 if (mode == analyze_base::fast || mode == analyze_base::color)
58 return true;
59 return screen_id == o.screen_id
60 && collection_threshold == o.collection_threshold;
61 }
62};

90/* Factory function for Dufay color analysis. */
91std::unique_ptr<analyze_dufay>
92get_new_dufay_analysis (struct analyzer_params &p, int_image_area area,
93 progress_info *progress)
94{
95 auto ret = std::make_unique<analyze_dufay> ();
96 {
97 const screen *s = p.scr;
98 screen adapted;
99 if (!p.scr)
100 ;
101 else if (p.params.type == DioptichromeB)
102 {
103 s = &adapted;
104 for (int y = 0; y < screen::size; y++)
105 for (int x = 0; x < screen::size; x++)
106 {
107 adapted.mult[y][x][0] = p.scr->mult[y][x][1];
108 adapted.mult[y][x][1] = p.scr->mult[y][x][0];
109 adapted.mult[y][x][2] = p.scr->mult[y][x][2];
110 }
111 }
112 else if (p.params.type == ImprovedDioptichromeB
113 || p.params.type == Omnicolore)
114 {
115 s = &adapted;
116 for (int y = 0; y < screen::size; y++)
117 for (int x = 0; x < screen::size; x++)
118 {
119 adapted.mult[y][x][0] = p.scr->mult[y][x][2];
120 adapted.mult[y][x][1] = p.scr->mult[y][x][1];
121 adapted.mult[y][x][2] = p.scr->mult[y][x][0];
122 }
123 }
124 if (ret->analyze (p.render, p.img, p.scr_to_img_map, s, p.simulated_screen_ptr,
125 area, p.mode,
126 p.collection_threshold, progress))
127 return ret;
128 }
129 return nullptr;
130}

247/* Precompute internal data structures for AREA. PROGRESS is used for progress
248 reporting. */
249
250bool
251render_interpolate::precompute (int_image_area area, progress_info *progress)
252{
253 uint64_t screen_id = 0;
254 if (m_scr_to_img_param.type == Random)
255 return false;
256 /* When doing profiled matrix, we need to pre-scale the profile so black
257 point correction goes right. Without doing so, for example black from red
258 pixels would be subtracted too aggressively, since we account for every
259 pixel in image, not only red patch portion. */
260 if (!render_to_scr::precompute_img_range (!m_original_color && !m_precise_rgb,
261 !m_original_color || m_profiled, area, progress))
262 return false;
263 if (m_screen_compensation
264 || m_params.collection_quality != render_parameters::fast_collection
265 || m_precise_rgb)
266 {
267 coord_t psize = pixel_size ();
268 sharpen_parameters sharpen = m_params.sharpen;
269 sharpen.usm_radius = m_params.screen_blur_radius * psize;
270 sharpen.scanner_mtf_scale *= psize;
271
272 if (sharpen.get_mode () != sharpen_parameters::none
273 && m_params.collection_quality
274 == render_parameters::simulated_screen_collection)
275 simulate_screen (progress);
276
277 m_screen = get_screen (m_scr_to_img.get_type (), false,
278 sharpen.deconvolution_p (), sharpen,
279 m_params.red_strip_width,
280 m_params.green_strip_width, progress, &screen_id);
281 if (!m_screen)
282 return false;
283 if (!m_original_color && !m_precise_rgb)
284 {
285 if (m_params.scanner_blur_correction)
286 compute_saturation_loss_table (m_screen.get (), screen_id,
287 m_params.collection_threshold,
288 m_params.sharpen, progress);
289 else
290 {
291 rgbdata cred, cgreen, cblue;
292 sharpen_parameters sharpen = m_params.sharpen;
293 sharpen.usm_radius = m_params.screen_blur_radius * psize;
294 sharpen.scanner_mtf_scale *= psize;
295 std::shared_ptr<screen> scr = get_screen (m_scr_to_img.get_type (), false, false,
296 sharpen, m_params.red_strip_width,
297 m_params.green_strip_width, progress,
298 &screen_id);
299 if (determine_color_loss (
300 &cred, &cgreen, &cblue, *scr, *m_screen,
301 m_simulated_screen.get (), m_params.collection_threshold,
302 m_params.sharpen, m_scr_to_img,
303 {m_img.width / 2 - 100, m_img.height / 2 - 100, 200, 200}))
304 {
305 color_matrix sat (cred.red, cgreen.red, cblue.red,
306 (luminosity_t)0,
307 cred.green, cgreen.green, cblue.green,
308 (luminosity_t)0,
309 cred.blue, cgreen.blue, cblue.blue,
310 (luminosity_t)0,
311 (luminosity_t)0, (luminosity_t)0, (luminosity_t)0, (luminosity_t)1);
312 /* cred is result of data collection of red element; this means cred.blue is amount
313 of blue collected to red. So when screen is fully red, we will account
314 cred.red, cgreen.red, cblue.red. This is why we need to transpose. */
315 sat.transpose ();
316 m_saturation_matrix = sat.invert ();
317#if 0
318 m_saturation_matrix.apply_to_rgb (rgbdata(1,0,0)).print (stdout);
319 m_saturation_matrix.print (stdout);
320 m_saturation_matrix.apply_to_rgb (cred).print (stdout);
321 m_saturation_matrix.apply_to_rgb (cgreen).print (stdout);
322 m_saturation_matrix.apply_to_rgb (cblue).print (stdout);
323#endif
324 }
325 }
326 }
327 }
328 int_image_area full_range (m_scr_to_img.get_range (m_img.width, m_img.height));
329 int_image_area analysis_area = area;
330 //printf ("full %i %i %i %i\n", full_range.x, full_range.y, full_range.width, full_range.height);
331 //printf ("area %i %i %i %i\n", area.x, area.y, area.width, area.height);
332
333 /* If area is significantly larger then half of image, just compute whole image. */
334 if (analysis_area.width * analysis_area.height > full_range.width * full_range.height / 2)
335 analysis_area = full_range;
336
337 /* For UI response,
338 it is better to compute whole image then significant portion of it. */
339 /* We need to compute bit more to get interpolation right.
340 TODO: figure out how much. */
341 analysis_area.x -= 5;
342 analysis_area.y -= 5;
343 analysis_area.width += 9;
344 analysis_area.height += 9;
345
346 struct analyzer_params p{
347 m_img.id,
348 m_gray_data_id,
349 m_simulated_screen_id,
350 screen_id,
351 m_params.gamma,
352 m_original_color
353 ? analyze_base::precise_rgb
354 : (m_precise_rgb ? analyze_base::precise_rgb
355 : (m_params.collection_quality
356 == render_parameters::fast_collection
357 ? analyze_base::fast
358 : analyze_base::precise)),
359 m_params.collection_threshold,
360 m_scr_to_img.get_param ().mesh_trans
361 ? m_scr_to_img.get_param ().mesh_trans->id
362 : 0,
363 m_scr_to_img.get_param (),
364 &m_img,
365 m_screen.get (),
366 this,
367 &m_scr_to_img,
368 m_simulated_screen.get ()
369 };
370 if (paget_like_screen_p (m_scr_to_img.get_type ()))
371 {
372 uint64_t id;
373 m_paget = paget_analyzer_cache.get (p, analysis_area, progress, &id);
374 if (!m_paget)
375 return false;
376 if ((int)m_params.screen_demosaic >= (int)render_parameters::hamilton_adams_demosaic
377 || m_params.screen_demosaic == render_parameters::default_demosaic)
378 {
379 struct demosaiced_params<analyze_paget> pp = {
380 id, m_params.dark_point, m_params.scan_exposure, m_params.contact_copy,
381 m_params.screen_demosaic == render_parameters::default_demosaic
382 ? (m_screen_compensation ? render_parameters::rcd_demosaic : render_parameters::amaze_demosaic)
383 : m_params.screen_demosaic,
384 m_paget.get (), this
385 };
386 m_demosaic_paget = demosaic_paget_cache.get (pp, progress);
387 if (!m_demosaic_paget)
388 return false;
389 }
390 }
391 else if (dufay_like_screen_p (m_scr_to_img.get_type ()))
392 {
393 m_dufay = dufay_analyzer_cache.get (p, analysis_area, progress);
394 if (!m_dufay)
395 return false;
396 }
397 else if (screen_with_vertical_strips_p (m_scr_to_img.get_type ()))
398 {
399 m_strips = strips_analyzer_cache.get (p, analysis_area, progress);
400 if (!m_strips)
401 return false;
402 }
403 else
404 return false;
405 m_interpolation_proportions = m_scr_to_img.patch_proportions (&m_params);
406 if (!m_original_color)
407 {
408 if (m_scr_to_img.get_type () == DioptichromeB)
409 std::swap (m_interpolation_proportions.red,
410 m_interpolation_proportions.green);
411 else if (m_scr_to_img.get_type () == ImprovedDioptichromeB
412 || m_scr_to_img.get_type () == Omnicolore)
413 std::swap (m_interpolation_proportions.red,
414 m_interpolation_proportions.blue);
415 else if (m_scr_to_img.get_type () == Joly)
416 std::swap (m_interpolation_proportions.red,
417 m_interpolation_proportions.blue);
418 }
419 return !progress || !progress->cancelled ();
420}

422/* Sample pixel at screen coordinate P and return its color. */
423
424pure_attr rgbdata
425render_interpolate::sample_pixel_scr (point_t p) const
426{
427 coord_t x = p.x, y = p.y;
428 rgbdata c;
429 bool adjusted = false;
430
431 if (paget_like_screen_p (m_scr_to_img.get_type ()))
432 {
433 if (m_demosaic_paget)
434 {
435 c = m_demosaic_paget->interpolate ({x, y}, m_interpolation_proportions,
436 m_params.demosaiced_scaling == render_parameters::default_scaling
437 ? (m_screen_compensation ? render_parameters::bspline_scaling : render_parameters::lanczos3_scaling)
438 : m_params.demosaiced_scaling);
439 adjusted = true;
440 }
441 else
442 c = m_paget->interpolate ({ x, y }, m_interpolation_proportions, m_params.screen_demosaic);
443 }
444 else if (screen_with_vertical_strips_p (m_scr_to_img.get_type ()))
445 {
446 c = m_strips->interpolate ({ x, y },
447 m_interpolation_proportions, m_params.screen_demosaic);
448 if (m_original_color)
449 ;
450 else if (m_scr_to_img.get_type () == Joly)
451 std::swap (c.red, c.blue);
452 }
453 else
454 {
455 c = m_dufay->interpolate ({ x, y }, m_interpolation_proportions, m_params.screen_demosaic);
456 if (m_original_color)
457 ;
458 else if (m_scr_to_img.get_type () == DioptichromeB)
459 std::swap (c.red, c.green);
460 else if (m_scr_to_img.get_type () == ImprovedDioptichromeB
461 || m_scr_to_img.get_type () == Omnicolore)
462 std::swap (c.red, c.blue);
463 }
464 /* TODO: With demosaicing we incorrectly first apply adjust_luminosity_ir
465 (when constructing demosaiced data) and only later compensate saturation loss.
466 This seems unavoidable, since we can only compensate after demosaicing.
467 It seems that in this case we may need to build more complex profile? */
468 if (!m_original_color)
469 c = compensate_saturation_loss_scr (p, c);
470 if (m_unadjusted || adjusted)
471 ;
472 else if (!m_original_color)
473 {
474 c.red = adjust_luminosity_ir (c.red);
475 c.green = adjust_luminosity_ir (c.green);
476 c.blue = adjust_luminosity_ir (c.blue);
477 }
478 else if (m_profiled)
479 {
480 profile_matrix.apply_to_rgb (c.red, c.green, c.blue, &c.red, &c.green,
481 &c.blue);
482 c.red = adjust_luminosity_ir (c.red);
483 c.green = adjust_luminosity_ir (c.green);
484 c.blue = adjust_luminosity_ir (c.blue);
485 }
486 else
487 c = adjust_rgb (c);
488 if (m_screen_compensation)
489 {
490 point_t pi = m_scr_to_img.to_img (p);
491 coord_t lum = get_img_pixel (pi);
492 rgbdata s;
493 if (!m_simulated_screen)
494 s = m_screen->interpolated_mult (p);
495 else
496 s = get_simulated_screen_pixel (pi);
497
498 /* This is clamping logic trying to avoid too colorful artifacts
499 alongs the edges. */
500#if 0
501 c.red = std::max (c.red, (luminosity_t)0);
502 c.green = std::max (c.green, (luminosity_t)0);
503 c.blue = std::max (c.blue, (luminosity_t)0);
504#endif
505
506 luminosity_t llum = c.red * s.red + c.green * s.green + c.blue * s.blue;
507 luminosity_t correction = llum ? lum / llum : lum * (luminosity_t)100;
508
509#if 1
510 luminosity_t redmin = lum - (1 - std::min (s.red, (luminosity_t)1));
511 luminosity_t redmax = lum + (1 - std::min (s.red, (luminosity_t)1));
512 if (c.red * correction < redmin)
513 correction = redmin / c.red;
514 else if (c.red * correction > redmax)
515 correction = redmax / c.red;
516
517 luminosity_t greenmin = lum - (1 - std::min (s.green, (luminosity_t)1));
518 luminosity_t greenmax = lum + (1 - std::min (s.green, (luminosity_t)1));
519 if (c.green * correction < greenmin)
520 correction = greenmin / c.green;
521 else if (c.green * correction > greenmax)
522 correction = greenmax / c.green;
523
524 luminosity_t bluemin = lum - (1 - std::min (s.blue, (luminosity_t)1));
525 luminosity_t bluemax = lum + (1 - std::min (s.blue, (luminosity_t)1));
526 if (c.blue * correction < bluemin)
527 correction = bluemin / c.blue;
528 else if (c.blue * correction > bluemax)
529 correction = bluemax / c.blue;
530#endif
531 correction = std::clamp (correction, (luminosity_t)0.0, (luminosity_t)5.0);
532
533 return c * correction;
534 }
535 else if (m_adjust_luminosity)
536 {
537 luminosity_t l = get_img_pixel_scr (p);
538 luminosity_t red2, green2, blue2;
539 out_color.m_color_matrix.apply_to_rgb (c.red, c.green, c.blue, &red2, &green2,
540 &blue2);
541 // TODO: We really should convert to XYZ and determine just Y.
542 luminosity_t gr = (red2 * rwght + green2 * gwght + blue2 * bwght);
543 if (gr <= (luminosity_t)1e-5 || l <= (luminosity_t)1e-5)
544 red2 = green2 = blue2 = l;
545 else
546 {
547 gr = l / gr;
548 red2 *= gr;
549 green2 *= gr;
550 blue2 *= gr;
551 }
552 // TODO: Inverse color matrix can be stored.
553 out_color.m_color_matrix.invert ().apply_to_rgb (red2, green2, blue2, &c.red,
554 &c.green, &c.blue);
555 return c;
556 }
557 else
558 return c;
559}

584/* Run ANALYZE on every screen point in the given AREA. If SCREEN is true,
585 AREA is in screen coordinates, otherwise it is in image coordinates.
586 PROGRESS is used for progress reporting. */
587
588bool
589render_interpolate::analyze_patches (analyzer analyze, const char *task,
590 bool screen, int_image_area area,
591 progress_info *progress)
592{
593 assert (!m_precise_rgb && !m_original_color);
594 if (dufay_like_screen_p (m_scr_to_img.get_type ()))
595 {
596 if (progress)
597 progress->set_task (task, m_dufay->get_height ());
598 for (int y = 0; y < m_dufay->get_height (); y++)
599 {
600 if (!progress || !progress->cancel_requested ())
601 for (int x = 0; x < m_dufay->get_width (); x++)
602 {
603 coord_t xs = x - m_dufay->get_xshift (),

997/* Return deltaE 2000 difference between colors FC1 and FC2. */
998
999static double
1000get_deltae (xyz fc1, xyz fc2, long *cln = NULL, xyz *mins = NULL,
1001 xyz *maxs = NULL)
1002{
1003 if (mins)
1004 {
1005 if (fc1.x < mins->x)
1006 mins->x = fc1.x;
1007 if (fc1.y < mins->y)
1008 mins->y = fc1.y;
1009 if (fc1.z < mins->z)
1010 mins->z = fc1.z;
1011 if (fc2.x < mins->x)
1012 mins->x = fc2.x;
1013 if (fc2.y < mins->y)
1014 mins->y = fc2.y;
1015 if (fc2.z < mins->z)
1016 mins->z = fc2.z;
1017 if (fc1.x > maxs->x)
1018 maxs->x = fc1.x;
1019 if (fc1.y > maxs->y)
1020 maxs->y = fc1.y;
1021 if (fc1.z > maxs->z)
1022 maxs->z = fc1.z;
1023 if (fc2.x > maxs->x)
1024 maxs->x = fc2.x;
1025 if (fc2.y > maxs->y)
1026 maxs->y = fc2.y;
1027 if (fc2.z > maxs->z)
1028 maxs->z = fc2.z;
1029 }
1030 /* DeltaE is meaningfully defined only in the range of xyz. */
1031 bool cl = false;
1032 if (fc1.x < 0)
1033 fc1.x = 0, cl = true;
1034 if (fc1.x > 1)
1035 fc1.x = 1, cl = true;
1036 if (fc1.y < 0)
1037 fc1.y = 0, cl = true;
1038 if (fc1.y > 1)
1039 fc1.y = 1, cl = true;
1040 if (fc1.z < 0)
1041 fc1.z = 0, cl = true;
1042 if (fc1.z > 1)
1043 fc1.z = 1, cl = true;
1044 if (fc2.x < 0)
1045 fc2.x = 0, cl = true;
1046 if (fc2.x > 1)
1047 fc2.x = 1, cl = true;
1048 if (fc2.y < 0)
1049 fc2.y = 0, cl = true;
1050 if (fc2.y > 1)
1051 fc2.y = 1, cl = true;
1052 if (fc2.z < 0)
1053 fc2.z = 0, cl = true;
1054 if (fc2.z > 1)
1055 fc2.z = 1, cl = true;
1056 if (cl && cln)
1057 {
1058 (*cln)++;
1059 }
1060 cie_lab cc1 (fc1, srgb_white);
1061 cie_lab cc2 (fc2, srgb_white);
1062 luminosity_t delta = deltaE2000 (cc1, cc2);
1063 if (!(delta >= 0))
1064 abort ();
1065 return delta;
1066}
