24 /* Return multiplicative factor of point p with bilinear interpolation. */
25 __attribute__ ((always_inline)) inline pure_attr rgbdata
26 interpolated_mult (point_t p) const
27 {
28 int sx, sy;
29 coord_t rx = my_modf (p.x * size, &sx);
30 coord_t ry = my_modf (p.y * size, &sy);
31 int sx1 = ((unsigned)sx + 1u) & (unsigned)(size - 1);
32 int sy1 = ((unsigned)sy + 1u) & (unsigned)(size - 1);
33 sx = (unsigned)sx & (unsigned)(size - 1);
34 sy = (unsigned)sy & (unsigned)(size - 1);
35 rgbdata d1 = {mult[sy][sx][0], mult[sy][sx][1], mult[sy][sx][2]};
36 rgbdata d2 = {mult[sy][sx1][0], mult[sy][sx1][1], mult[sy][sx1][2]};
37 rgbdata i1 = d1 * (1 - rx) + d2 * rx;
38 rgbdata dd1 = {mult[sy1][sx][0], mult[sy1][sx][1], mult[sy1][sx][2]};
39 rgbdata dd2 = {mult[sy1][sx1][0], mult[sy1][sx1][1], mult[sy1][sx1][2]};
40 rgbdata i2 = dd1 * (1 - rx) + dd2 * rx;
41 return i1 * (1 - ry) + i2 * ry;
42 }
43 /* Return multiplicative factor of point p with bilinear interpolation. */
44 inline pure_attr rgbdata
45 noninterpolated_mult (point_t p) const
46 {
47 int ix = (uint64_t) nearest_int ((p.x) * size) & (unsigned)(size - 1);
48 int iy = (uint64_t) nearest_int ((p.y) * size) & (unsigned)(size - 1);
49 return {mult[iy][ix][0], mult[iy][ix][1], mult[iy][ix][2]};
50 }
