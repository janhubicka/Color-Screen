18/* Square matrix template.
19 DIM is the dimension of the matrix. */
20template <typename T, size_t dim>
21class matrix
22{
23public:
24 static constexpr size_t m_dim = dim;
25
26 /* Default constructor: build identity matrix. */
27 constexpr matrix () : m_elements{}
28 {
29 for (size_t i = 0; i < m_dim; i++)
30 m_elements[i][i] = (T) 1;
31 }
32
33 /* Variadic constructor to initialize all elements in row-major order.
34 ARGS are the elements of the matrix. */
35 template <typename... Args, typename = std::enable_if_t<sizeof... (Args) == dim * dim>>
36 constexpr matrix (Args... args) : m_elements{}
37 {
38 T values[] = { static_cast<T> (args)... };
39 for (size_t i = 0; i < dim; i++)
40 for (size_t j = 0; j < dim; j++)
41 m_elements[i][j] = values[i * dim + j];
42 }
43
44 /* Support for Row, Column indexing. Storage is row-major.
45 ROW is the row index, COL is the column index. */
46 always_inline_attr inline T &
47 operator() (size_t row, size_t col) noexcept
48 {
49 return m_elements[row][col];
50 }
51
52 /* Support for Row, Column indexing. Storage is row-major.
53 ROW is the row index, COL is the column index. */
54 always_inline_attr inline const T &
55 operator() (size_t row, size_t col) const noexcept
56 {
57 return m_elements[row][col];
58 }
59
60 /* Make matrix random (for unit testing). */
61 void
62 randomize ()
63 {
64 for (size_t i = 0; i < m_dim; i++)
65 for (size_t j = 0; j < m_dim; j++)
66 m_elements[i][j] = (T) (rand () % 100);
67 }
68
69 /* Copy constructor. M is the matrix to copy from. */
70 constexpr matrix (const matrix &m) : m_elements{}
71 {
72 for (size_t i = 0; i < m_dim; i++)
73 for (size_t j = 0; j < m_dim; j++)
74 m_elements[i][j] = m.m_elements[i][j];
75 }
76
77 /* Add two matrices. RHS is the matrix to add. */
78 pure_attr inline matrix<T, dim>
79 operator+ (const matrix<T, dim> &rhs) const noexcept
80 {
81 matrix<T, dim> ret;
82 for (size_t i = 0; i < m_dim; i++)
83 for (size_t j = 0; j < m_dim; j++)
84 ret (i, j) = (*this) (i, j) + rhs (i, j);
85 return ret;
86 }
87
88 /* Multiply two matrices. RHS is the matrix to multiply with. */
89 pure_attr inline matrix<T, dim>
90 operator* (const matrix<T, dim> &rhs) const noexcept
91 {
92 matrix<T, dim> ret;
93 for (size_t i = 0; i < m_dim; i++)
94 for (size_t j = 0; j < m_dim; j++)
95 {
96 T a = 0;
97 for (size_t k = 0; k < m_dim; k++)
98 a += (*this) (i, k) * rhs (k, j);
99 ret (i, j) = a;
100 }
101 return ret;
102 }
103
104 /* Multiply matrix by a scalar. RHS is the scalar to multiply with. */
105 pure_attr inline matrix<T, dim>
106 operator* (const T rhs) const noexcept
107 {
108 matrix<T, dim> ret;
109 for (size_t i = 0; i < m_dim; i++)
110 for (size_t j = 0; j < m_dim; j++)
111 ret (i, j) = (*this) (i, j) * rhs;
112 return ret;
113 }
114
115 /* Transpose the matrix in-place. */
116 inline void
117 transpose () noexcept
118 {
119 for (size_t i = 0; i < m_dim; i++)
120 for (size_t j = 0; j < i; j++)
121 std::swap (m_elements[i][j], m_elements[j][i]);
122 }
123
124 /* Print the matrix to a file. F is the file pointer. */
125 void
126 print (FILE *f) const
127 {
128 for (size_t row = 0; row < m_dim; row++)
129 {
130 for (size_t col = 0; col < m_dim; col++)
131 fprintf (f, " %f", (double) (*this) (row, col));
132 fprintf (f, "\n");
133 }
134 }
135
136protected:
137 T m_elements[dim][dim];
138};

203 {
204 T a = (*this) (0, 0);
205 T b = (*this) (0, 1);
206 T c = (*this) (1, 0);
207 T d = (*this) (1, 1);
208 T det_rec = (T) 1 / (a * d - b * c);
209 return matrix2x2 (d * det_rec, -b * det_rec, -c * det_rec, a * det_rec);

274 /* Compute inversion. */
275 pure_attr inline matrix3x3
276 invert () const noexcept
277 {
278 T a = (*this) (0, 0), b = (*this) (0, 1), c = (*this) (0, 2);
279 T d = (*this) (1, 0), e = (*this) (1, 1), f = (*this) (1, 2);
280 T g = (*this) (2, 0), h = (*this) (2, 1), i = (*this) (2, 2);
281
282 T det = a * (e * i - f * h) - b * (d * i - f * g) + c * (d * h - e * g);
283 T idet = (T) 1 / det;
284
285 return matrix3x3 ((e * i - f * h) * idet, (c * h - b * i) * idet, (b * f - c * e) * idet,
286 (f * g - d * i) * idet, (a * i - c * g) * idet, (c * d - a * f) * idet,
287 (d * h - e * g) * idet, (b * g - a * h) * idet, (a * e - b * d) * idet);
288 }
