14/* Convert sRGB value C to linear space.
15 sRGB values are undefined outside range 0..1 but give them reasonable meanings. */
16inline luminosity_t
17srgb_to_linear (luminosity_t c)
18{
19 /* sRGB values are undefined outside range 0..1 but give them reasonable meanings. */
20 if (c < 0)
21 return -my_pow (-c, (luminosity_t)2.2);
22 else if (c > 1)
23 return my_pow (c, (luminosity_t)2.2);
24 if (c < (luminosity_t)0.04045)
25 return c / (luminosity_t)12.92;
26 return my_pow ((c + (luminosity_t)0.055) / (luminosity_t)1.055, (luminosity_t)2.4);
27}

29/* Apply gamma G to value VAL.
30 We handle special gama of -1 for sRGB color space curves. */
31inline luminosity_t
32apply_gamma (luminosity_t val, luminosity_t gamma)
33{
34 if (gamma == 1)
35 return val;
36 if (gamma == -1)
37 return srgb_to_linear (val);
38 if (val >= 0)
39 return my_pow (val, gamma);
40 else
41 return -my_pow (-val, gamma);
42}

71/* Convert linear luminescence C to sRGB space.
72 sRGB values are undefined outside range 0..1 but give them reasonable meanings. */
73inline luminosity_t
74linear_to_srgb (luminosity_t c)
75{
76 /* sRGB values are undefined outside range 0..1 but give them reasonable meanings. */
77 if (c < 0)
78 return -my_pow (-c, 1/(luminosity_t)2.2);
79 else if (c > 1)
80 return my_pow (c, 1/(luminosity_t)2.2);
81 if (c<(luminosity_t)0.0031308)
82 return (luminosity_t)12.92 * c;
83 return ((luminosity_t)1.055)*my_pow (c, 1/(luminosity_t)2.4)-(luminosity_t)0.055;
84}
