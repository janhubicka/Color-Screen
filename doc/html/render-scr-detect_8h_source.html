210 static rgbdata
211 normalize_color (rgbdata c)
212 {
213 c.red = std::max (c.red, (luminosity_t)0);
214 c.green = std::max (c.green, (luminosity_t)0);
215 c.blue = std::max (c.blue, (luminosity_t)0);
216 //luminosity_t min = /*std::max (*/std::min (std::min (c.red, c.green), c.blue)/*, (luminosity_t) 0*/);
217 //c.red -= min;
218 //c.green -= min;
219 //c.blue -= min;
220 luminosity_t sum = c.red + c.green + c.blue;
221 if (!(sum > 0))
222 return {0, 0, 0};
223 luminosity_t adj = 1 / sum;
224#if 1
225 c.red *= adj;
226 c.green *= adj;
227 c.blue *= adj;
228#else
229 c.red = (2 * c.red - sum) * adj;
230 c.green = (2 * c.green - sum) * adj;
231 c.blue = (2 * c.blue - sum) * adj;
232#endif
233 return c;
234 }

611 {
612 /* Search for nearest pixels of each known color. */
613 const coord_t inf = distance_list::max_distance + 1;
614 int rx[3], ry[3];
615 coord_t cdist[3] = {inf, inf, inf};
616 coord_t biggest = inf;
617 for (int i = 0; i < distance_list.num && distance_list.list[i].fdist < biggest + 2; i++)
618 {
619 int xx = (int)p.x + distance_list.list[i].x;
620 int yy = (int)p.y + distance_list.list[i].y;
621 if (xx < 0 || yy < 0 || xx >= m_img.width || yy >= m_img.height)
622 continue;
623 scr_detect::color_class t = m_color_class_map->get_class (xx, yy);
624 if (t == scr_detect::unknown)
625 continue;
626 //assert (t>=0 && t < 3);
627 if (distance_list.list[i].fdist > cdist[(int)t] + 2)
628 continue;
629 double dist = my_sqrt ((xx + (coord_t)0.5 - p.x) * (xx + (coord_t)0.5 - p.x) + (yy + (coord_t)0.5 - p.y) * (yy + (coord_t)0.5 - p.y));
630 if (dist < cdist[(int)t])
631 {
632 cdist[(int)t] = dist;
633 rx[(int)t] = xx;
634 ry[(int)t] = yy;
635 biggest = std::max (std::max (cdist[0], cdist[1]), cdist[2]);
636 }
637 }
638 if (biggest == inf)
639 return {0, 0, 0};
640 else
641 return {get_patch_density (rx[0], ry[0], scr_detect::red), get_patch_density (rx[1], ry[1], scr_detect::green), get_patch_density (rx[2], ry[2], scr_detect::blue)};
