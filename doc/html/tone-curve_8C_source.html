292tone_curve::tone_curve (enum tone_curves type, const std::vector<point_t> &control_points)
293: precomputed_function (0, 1)

295 m_linear = false;
296 switch (type)

299 m_linear = true;
300 return;
301 case tone_curve_dng:
302 init_by_y_values (dng_curve, sizeof (dng_curve) / sizeof (luminosity_t));
303 return;
304 case tone_curve_dng_contrast:
305 init_by_function (sizeof (dng_curve) / sizeof (luminosity_t),
306 [] (luminosity_t x) {
307 /* DNG curve is defined for 0..1 range. */
308 static precomputed_function<luminosity_t>
309 dng (0, 1, dng_curve, sizeof (dng_curve) / sizeof (luminosity_t));
310 return apply_gamma (dng.apply (x), 2.2);
311 });
312 return;
313 case tone_curve_safe:
314 init_by_sensitivity (spectrum_dyes_to_xyz::safe_reversal_output_curve);
315 break;
316 case tone_curve_kodachrome25:
317 init_by_sensitivity (spectrum_dyes_to_xyz::kodachrome25_curve);

324 break;

327 break;
328 case tone_curve_custom:
329 init_by_control_points (control_points);

368{
369 m_linear = false;
370 std::vector<point_t> cp = control_points;
371 if (cp.empty ())
372 cp = default_control_points ();
373 std::sort (cp.begin (), cp.end (), [](const point_t &a, const point_t &b) {
374 return a.x < b.x;
375 });
376 if (cp[0].x > 0)
377 cp.insert (cp.begin (), {0.0, 0.0});
378 if (cp.back ().x < 1.0)
379 cp.push_back ({1.0, 1.0});
380
381 int n = cp.size ();
382 double *x = new double[n];
383 double *y = new double[n];
384 for (int i = 0; i < n; i++)
385 {
386 x[i] = cp[i].x;
387 y[i] = cp[i].y;
388 }
389 spline<double> s (x, y, n);
390 init_by_function (1024, [&] (double x) {
391 double val = s.apply (x);
392 return std::max (0.0, std::min (1.0, val));
393 });
394 delete[] x;
395 delete[] y;
396}
