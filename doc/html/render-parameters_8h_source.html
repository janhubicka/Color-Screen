54/* Parameters of sharpening. */
55struct sharpen_parameters
56{
57 /* Supported sharpening modes. */
58 enum sharpen_mode
59 {
60 none,
61 unsharp_mask,
62 wiener_deconvolution,
63 richardson_lucy_deconvolution,
64 blur_deconvolution,
65 sharpen_mode_max
66 };
67 DLL_PUBLIC static const property_t sharpen_mode_names[(int)sharpen_mode_max];
68
69 /* Sharpening mode. */
70 enum sharpen_mode mode = none;
71
72 /* Radius (in pixels) and amount for unsharp-mask filter. */
73 luminosity_t usm_radius = 0, usm_amount = 0;
74
75 /* MTF curve of scanner. */
76 mtf_parameters scanner_mtf;
77
78 /* Signal to noise ratio of the scanner. */
79 luminosity_t scanner_snr = 2000;
80
81 /* Scale of scanner mtf. 0 disables deconvolution sharpening. */
82 luminosity_t scanner_mtf_scale = 1;
83
84 /* Number of iterations of Richardson-Lucy deconvolution sharpening.
85 If 0, much faster Wiener filter will be used. */
86 int richardson_lucy_iterations = 0;
87
88 /* Dampening parameter sigma. */
89 luminosity_t richardson_lucy_sigma = 0;
90
91 /* Supersampling for deconvolution sharpening. */
92 int supersample = 2;
93
94 /* Return effective sharpening mode. */
95 pure_attr enum sharpen_mode get_mode () const
96 {
97 switch (mode)
98 {
99 case none:
100 return none;
101 case unsharp_mask:
102 return usm_radius > 0 && usm_amount > 0 ? unsharp_mask : none;
103 case wiener_deconvolution:
104 return scanner_mtf_scale > 0 ? wiener_deconvolution : none;
105 case richardson_lucy_deconvolution:
106 return scanner_mtf_scale > 0 && richardson_lucy_iterations
107 ? richardson_lucy_deconvolution : none;
108 case blur_deconvolution:
109 return scanner_mtf_scale > 0
110 ? blur_deconvolution : none;
111 default:
112 abort ();
113 }
114 };
115
116 /* Return true if sharpening mode is a deconvolution. */
117 pure_attr bool deconvolution_p () const
118 {
119 enum sharpen_mode mode = get_mode ();
120 return mode == wiener_deconvolution || mode == richardson_lucy_deconvolution
121 || mode == blur_deconvolution;
122 }
123
124 /* Return true if THIS and O will produce same image.
125 Used for caching.
126
127 Allow small differences in scale since screen may change during editing. */
128 pure_attr bool
129 operator== (const sharpen_parameters &o) const
130 {
131 enum sharpen_mode mode = get_mode ();
132 if (o.get_mode () != mode)
133 return false;
134 switch (mode)
135 {
136 case none:
137 return true;
138 case unsharp_mask:
139 return fabs (usm_radius - o.usm_radius) < 0.001
140 && usm_amount == o.usm_amount;
141 case wiener_deconvolution:
142 return scanner_mtf == o.scanner_mtf
143 && fabs (scanner_mtf_scale - o.scanner_mtf_scale) < 0.001
144 && scanner_snr == o.scanner_snr
145 && supersample == o.supersample;
146 case richardson_lucy_deconvolution:
147 return scanner_mtf == o.scanner_mtf
148 && fabs (scanner_mtf_scale - o.scanner_mtf_scale) < 0.001
149 && richardson_lucy_iterations == o.richardson_lucy_iterations
150 && richardson_lucy_sigma == o.richardson_lucy_sigma
151 && supersample == o.supersample;
152 case blur_deconvolution:
153 return scanner_mtf == o.scanner_mtf
154 && fabs (scanner_mtf_scale - o.scanner_mtf_scale) < 0.001;
155 default:
156 abort ();
157 }
158 abort ();
159 }
160
161 /* Return true if THIS and O have same data. */
162 pure_attr bool equal_p (const sharpen_parameters &o) const
163 {
164 return mode == o.mode
165 && usm_radius == o.usm_radius
166 && usm_amount == o.usm_amount
167 && scanner_mtf.equal_p (o.scanner_mtf)
168 && scanner_snr == o.scanner_snr
169 && scanner_mtf_scale == o.scanner_mtf_scale
170 && richardson_lucy_iterations == o.richardson_lucy_iterations
171 && richardson_lucy_sigma == o.richardson_lucy_sigma
172 && supersample == o.supersample;
173 }
174
175 /* Default constructor. */
176 sharpen_parameters () = default;
177};

297 /* Apply tile adjustment to render parameters P. */
298 void
299 apply (render_parameters *p) const
300 {
301 p->dark_point = dark_point + exposure * p->dark_point;
302 p->scan_exposure *= exposure;
303 if (scanner_blur_correction)
304 p->scanner_blur_correction = scanner_blur_correction;
305 }
