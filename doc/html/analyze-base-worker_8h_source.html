23 scr_to_img *scr_to_img, render_to_scr *render, const screen *screen,
24 const simulated_screen *simulated_screen,
25 luminosity_t collection_threshold, luminosity_t *w_red,
26 luminosity_t *w_green, luminosity_t *w_blue, int_image_area area,
27 progress_info *progress)

30 int size2 = (openmp_min_size + m_area.width - 1) / m_area.width;
31#pragma omp parallel shared( \
32 progress, render, scr_to_img, screen, collection_threshold, w_blue, \
33 w_red, w_green, area, simulated_screen) default(none) \
34 if (area.height > size || this->m_area.height > size2)
35 {
36#pragma omp for
37 for (int y = area.y; y < area.y + area.height; y++)
38 {
39 if (!progress || !progress->cancel_requested ())
40 for (int x = area.x; x < area.x + area.width; x++)
41 {
42 point_t scr = scr_to_img->to_scr (
43 { x + (coord_t) 0.5, y + (coord_t) 0.5 });
44 scr += { (coord_t)m_area.xshift (), (coord_t)m_area.yshift () };
45 /* Dufay analyzer shifts red strip and some pixels gets accounted
46 to neighbouring screen tile; add extra bffer of 1 screen tile
47 to be sure we do not access uninitialized memory. */
48 if (!GEOMETRY::check_range
49 && (scr.x <= (coord_t) 0
50 || scr.x >= (coord_t) m_area.width - (coord_t) 1
51 || scr.y <= (coord_t) 0
52 || scr.y >= (coord_t) m_area.height - (coord_t) 1))
53 continue;
54
55 luminosity_t l = render->get_unadjusted_data ({ x, y });
56 rgbdata screen_color;
57 if (!simulated_screen)
58 screen_color = screen->noninterpolated_mult (scr);
59 else
60 screen_color = simulated_screen->get_pixel (x, y);
61 if (screen_color.red > collection_threshold)
62 {
63 data_entry e = GEOMETRY::red_scr_to_entry (scr);
64 if (!GEOMETRY::check_range
65 || (e.x >= 0
66 && e.x < m_area.width * GEOMETRY::red_width_scale
67 && e.y >= 0
68 && e.y < m_area.height
69 * GEOMETRY::red_height_scale))
70 {
71 if constexpr (debug)
72 assert (e.x >= 0 && e.y >= 0
73 && e.x < m_area.width
74 * GEOMETRY::red_width_scale
75 && e.y < m_area.height
76 * GEOMETRY::red_height_scale);
77 luminosity_t val
78 = (screen_color.red - collection_threshold);
79 int idx = e.y * m_area.width * GEOMETRY::red_width_scale
80 + e.x;
81 luminosity_t &c = m_red[idx];
82 luminosity_t vall = val * l;
83#pragma omp atomic
84 c += vall;
85 luminosity_t &w = w_red[idx];
86#pragma omp atomic
87 w += val;
88 }
89 }
90 if (screen_color.green > collection_threshold)
91 {
92 data_entry e = GEOMETRY::green_scr_to_entry (scr);
93 if (!GEOMETRY::check_range
94 || (e.x >= 0
95 && e.x < m_area.width * GEOMETRY::green_width_scale
96 && e.y >= 0
97 && e.y < m_area.height
98 * GEOMETRY::green_height_scale))
99 {
100 if constexpr (debug)
101 assert (e.x >= 0 && e.y >= 0
102 && e.x < m_area.width
103 * GEOMETRY::green_width_scale
104 && e.y < m_area.height
105 * GEOMETRY::green_height_scale);
106 luminosity_t val
107 = (screen_color.green - collection_threshold);
108 int idx
109 = e.y * m_area.width * GEOMETRY::green_width_scale
110 + e.x;
111 luminosity_t vall = val * l;
112 luminosity_t &c = m_green[idx];
113#pragma omp atomic
114 c += vall;
115 luminosity_t &w = w_green[idx];
116#pragma omp atomic
117 w += val;
118 }
119 }
120 if (screen_color.blue > collection_threshold)
121 {
122 data_entry e = GEOMETRY::blue_scr_to_entry (scr);
123 if (!GEOMETRY::check_range
124 || (e.x >= 0
125 && e.x < m_area.width * GEOMETRY::blue_width_scale
126 && e.y >= 0
127 && e.y < m_area.height
128 * GEOMETRY::blue_height_scale))
129 {
130 if constexpr (debug)
131 assert (e.x >= 0 && e.y >= 0
132 && e.x < m_area.width
133 * GEOMETRY::blue_width_scale
134 && e.y < m_area.height
135 * GEOMETRY::blue_height_scale);
136 luminosity_t val
137 = (screen_color.blue - collection_threshold);
138 int idx
139 = e.y * m_area.width * GEOMETRY::blue_width_scale
140 + e.x;
141 luminosity_t vall = val * l;
142 luminosity_t &c = m_blue[idx];
143#pragma omp atomic
144 c += vall;
145 luminosity_t &w = w_blue[idx];
146#pragma omp atomic
147 w += val;
148 }
149 }
150 }
151 if (progress)
152 progress->inc_progress ();
153 }
154 if (!progress || !progress->cancel_requested ())
155 {
156#pragma omp for nowait
157 for (int y = 0; y < m_area.height; y++)
158 {
159 if (!progress || !progress->cancel_requested ())
160 for (int yy = 0; yy < GEOMETRY::red_height_scale; yy++)
161 for (int x = 0; x < m_area.width * GEOMETRY::red_width_scale;
162 x++)
163 {
164 data_entry e = { x, y * GEOMETRY::red_height_scale + yy };
165 int idx = e.y * m_area.width * GEOMETRY::red_width_scale
166 + e.x;
167 if (w_red[idx] != (luminosity_t) 0)
168 m_red[idx] /= w_red[idx];
169 else
170 {
171 point_t scr = GEOMETRY::red_entry_to_scr (e);
172 m_red[idx]
173 = render->get_unadjusted_img_pixel_scr (
174 { scr.x - (coord_t)m_area.xshift (),
175 scr.y - (coord_t)m_area.yshift () });
176 }
177 }
178 if (progress)
179 progress->inc_progress ();
180 }
181#pragma omp for nowait
182 for (int y = 0; y < m_area.height; y++)
183 {
184 if (!progress || !progress->cancel_requested ())
185 for (int yy = 0; yy < GEOMETRY::green_height_scale; yy++)
186 for (int x = 0;
187 x < m_area.width * GEOMETRY::green_width_scale; x++)
188 {
189 data_entry e
190 = { x, y * GEOMETRY::green_height_scale + yy };
191 int idx = e.y * m_area.width * GEOMETRY::green_width_scale
192 + e.x;
193 if (w_green[idx] != (luminosity_t) 0)
194 m_green[idx] /= w_green[idx];
195 else
196 {
197 point_t scr = GEOMETRY::green_entry_to_scr (e);
198 m_green[idx]
199 = render->get_unadjusted_img_pixel_scr (
200 { scr.x - (coord_t)m_area.xshift (),
201 scr.y - (coord_t)m_area.yshift () });
202 }
203 }
204 if (progress)
205 progress->inc_progress ();
206 }
207#pragma omp for nowait
208 for (int y = 0; y < m_area.height; y++)
209 {
210 if (!progress || !progress->cancel_requested ())
211 for (int yy = 0; yy < GEOMETRY::blue_height_scale; yy++)
212 for (int x = 0; x < m_area.width * GEOMETRY::blue_width_scale;
213 x++)
214 {
215 data_entry e = { x, y * GEOMETRY::blue_height_scale + yy };
216 int idx = e.y * m_area.width * GEOMETRY::blue_width_scale
217 + e.x;
218 if (w_blue[idx] != (luminosity_t) 0)
219 m_blue[idx] /= w_blue[idx];
220 else
221 {
222 point_t scr = GEOMETRY::blue_entry_to_scr (e);

224 = render->get_unadjusted_img_pixel_scr (

228 }
229 if (progress)

231 }
232 }
233 }
234 return !progress || !progress->cancelled ();

472/* Collect data for deinterlaced rendering in original or profiled colors.
473 TODO: This does not make much sense. We should apply profile first or
474 collect in RGB colors.
475
476 SCR_TO_IMG is the map from image to screen.
477 RENDER is the renderer.
478 W_RED, W_GREEN, W_BLUE are the weights for the channels.
479 AREA is the region to analyze.
480 PROGRESS is the progress info object. */
481template <typename GEOMETRY>
482bool
483analyze_base_worker<GEOMETRY>::analyze_color (
484 scr_to_img *scr_to_img, render_to_scr *render, luminosity_t *w_red,
485 luminosity_t *w_green, luminosity_t *w_blue, int_image_area area,
486 progress_info *progress)
487{
488 luminosity_t weights[256];
489 luminosity_t half_weights[256];
490 int size = (openmp_min_size + area.width - 1) / area.width;
491 int size2 = (openmp_min_size + m_area.width - 1) / m_area.width;
492
493 /* FIXME: technically not right for paget where diagonal coordinates are not
494 of same size as normal ones. */
495 coord_t pixel_size = render->pixel_size ();
496 coord_t left = (coord_t) 128 - (pixel_size / (coord_t) 2) * (coord_t) 256;
497 coord_t right = (coord_t) 128 + (pixel_size / (coord_t) 2) * (coord_t) 256;
498 coord_t half_left = (coord_t) 128 - (pixel_size / (coord_t) 4) * (coord_t) 256;
499 coord_t half_right = (coord_t) 128 + (pixel_size / (coord_t) 4) * (coord_t) 256;
500
501 for (int i = 0; i < 256; i++)
502 {
503 if ((coord_t)i <= left)
504 weights[i] = (luminosity_t) 0;
505 else if ((coord_t)i >= right)
506 weights[i] = (luminosity_t)1;
507 else
508 weights[i]
509 = (luminosity_t)((coord_t)i - left) / (luminosity_t)(right - left);
510
511 if ((coord_t)i <= half_left)
512 half_weights[i] = (luminosity_t) 0;
513 else if ((coord_t)i >= half_right)
514 half_weights[i] = (luminosity_t)1;
515 else
516 half_weights[i] = (luminosity_t)((coord_t)i - half_left)
517 / (luminosity_t)(half_right - half_left);
518 }
519#pragma omp parallel shared( \
520 progress, render, scr_to_img, w_blue, w_red, w_green, area, \
521 half_weights, \
522 weights) default(none) if (area.height > size \
523 || this->m_area.height > size2)
524 {
525#pragma omp for
526 for (int y = area.y; y < area.y + area.height; y++)
527 {
528 int64_t red_minx = -2, red_miny = -2, green_minx = -2, green_miny = -2,
529 blue_minx = -2, blue_miny = -2;
530 int64_t red_maxx = 2, red_maxy = 2, green_maxx = 2, green_maxy = 2,
531 blue_maxx = 2, blue_maxy = 2;
532 if (!progress || !progress->cancel_requested ())
533 for (int x = area.x; x < area.x + area.width; x++)
534 {
535 point_t scr = scr_to_img->to_scr (
536 { x + (coord_t) 0.5, y + (coord_t) 0.5 });
537 scr += { (coord_t)m_area.xshift (), (coord_t)m_area.yshift () };
538 if (!GEOMETRY::check_range
539 && (scr.x < (coord_t) 0 || scr.x > (coord_t)m_area.width - 1
540 || scr.y < (coord_t) 0
541 || scr.y > (coord_t)m_area.height - 1))
542 continue;
543 rgbdata d = render->get_unadjusted_rgb_pixel ({ x, y });
544
545 point_t off;
546 analyze_base::data_entry e
547 = GEOMETRY::red_scr_to_entry (scr, &off);
548 if (e.x + red_minx >= 0
549 && e.x + red_maxx < m_area.width * GEOMETRY::red_width_scale
550 && e.y + red_miny >= 0
551 && e.y + red_maxy
552 < m_area.height * GEOMETRY::red_height_scale)
553 {
554 off.x = half_weights[(int)(off.x * (coord_t)255.5)];
555 off.y = weights[(int)(off.y * (coord_t)255.5)];
556 luminosity_t &l1
557 = w_red[(e.y) * m_area.width * GEOMETRY::red_width_scale
558 + e.x];
559 luminosity_t &v1
560 = m_red[(e.y) * m_area.width
561 * GEOMETRY::red_width_scale
562 + e.x];
563 luminosity_t val1
564 = ((luminosity_t)1 - (luminosity_t)off.x)
565 * ((luminosity_t)1 - (luminosity_t)off.y);
566#pragma omp atomic
567 l1 += val1;
568#pragma omp atomic
569 v1 += d.red * val1;
570 data_entry o
571 = GEOMETRY::offset_for_interpolation_red (e, { 1, 0 });
572 luminosity_t &l2
573 = w_red[o.y * m_area.width * GEOMETRY::red_width_scale
574 + o.x];
575 luminosity_t &v2
576 = m_red[o.y * m_area.width
577 * GEOMETRY::red_width_scale
578 + o.x];
579 luminosity_t val2
580 = ((luminosity_t)off.x)
581 * ((luminosity_t)1 - (luminosity_t)off.y);
582#pragma omp atomic
583 l2 += val2;
584#pragma omp atomic
585 v2 += d.red * val2;
586 o = GEOMETRY::offset_for_interpolation_red (e, { 0, 1 });
587 luminosity_t &l3
588 = w_red[o.y * m_area.width * GEOMETRY::red_width_scale
589 + o.x];
590 luminosity_t &v3
591 = m_red[o.y * m_area.width
592 * GEOMETRY::red_width_scale
593 + o.x];
594 luminosity_t val3 = ((luminosity_t)1 - (luminosity_t)off.x)
595 * ((luminosity_t)off.y);
596#pragma omp atomic
597 l3 += val3;
598#pragma omp atomic
599 v3 += d.red * val3;
600 o = GEOMETRY::offset_for_interpolation_red (e, { 1, 1 });
601 luminosity_t &l4
