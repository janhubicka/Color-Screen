66/* MTF can be either based on real measured data (then size() != 0)
67 or computed by diffraction limit or just a blur disk simulation.
68 In each case one can adjust sigma. */
69struct mtf_parameters
70{
71 /* Sigma (in pixels) used to estimate gaussian blur. */
72 double sigma = 0;
73
74 /* Size of blur diameter (in pixels) used to estimate defocus.
75 This parameter is used only if pixel_pitch/f_stop/wavelength
76 is not defined. */
77 double blur_diameter = 0;
78
79 /* Defocus (in millimeters) */
80 double defocus = 0;
81 /* F-stop. */
82 double f_stop = 0;
83 /* Wavelength of light in nm. */
84 double wavelength = 0;
85 /* Per-channel Wavelength of light in nm. */
86 std::array<double, 4> wavelengths = {0, 0, 0, 0};
87 /* Sensor pixel pitch (size of a pixel) in micrometers. */
88 double pixel_pitch = 0;
89 /* The ratio of the active area to the total pixel area of the sensor.
90 Usually in range 0..1 but for scanner and linear sensor it may be greater
91 than that. For Nikon Coolscan it seems to be 4.
92
93 0 disables accounting sensor MTF. */
94 double sensor_fill_factor = 1;
95
96 /* DPI of the scan; necessary to calculate magnification. */
97 double scan_dpi = 0;
98
99 /* Measurement to use. */
100 int measured_mtf_idx = -1;

125 bool
126 operator== (const mtf_parameters &o) const
127 {
128 if (use_measured_mtf ())
129 {
130 if (!o.use_measured_mtf ())
131 return false;
132 return measurements[measured_mtf_idx] == o.measurements[o.measured_mtf_idx];
133 }
134 else if (o.use_measured_mtf ())
135 return false;
136 if (sigma != o.sigma || sensor_fill_factor != o.sensor_fill_factor)
137 return false;
138 if (simulate_diffraction_p ())
139 return o.simulate_diffraction_p ()
140 && defocus == o.defocus
141 && blur_diameter == o.blur_diameter
142 && f_stop == o.f_stop
143 && wavelength == o.wavelength
144 && pixel_pitch == o.pixel_pitch
145 && scan_dpi == o.scan_dpi;
146 else if (o.simulate_diffraction_p ())
147 return false;
148 if (blur_diameter != o.blur_diameter)
149 return false;
150 return true;
151 }
152 bool equal_p (const mtf_parameters &o) const
153 {

163 && measurements == o.measurements
