48/* Write INPUT string to F with escaped characters. */
49static bool
50write_escaped_string (FILE *f, const char *input)
51{
52 if (fputc ('"', f) == EOF)
53 return false;
54
55 for (; *input; input++)
56 {
57 if (*input == '"' || *input == '\\')
58 {
59 if (fputc ('\\', f) == EOF)
60 return false;
61 }
62 if (fputc (*input, f) == EOF)
63 return false;
64 }
65
66 if (fputc ('"', f) == EOF)
67 return false;
68 return true;
69}

446static bool
447skipwhitespace (FILE *f)
448{
449 while (!feof (f))
450 {
451 int c = getc (f);
452 if (c == EOF)
453 return true;
454 if (c != '\n' && c != '\r' && !isspace (c))
455 {
456 ungetc (c, f);
457 return true;
458 }
459 }
460 return true;
461}

463std::string
464read_escaped_string (FILE *f)
465{
466 std::string result;
467 int c;
468
469 skipwhitespace (f);
470
471 c = getc (f);
472 if (c != '"')
473 {
474 if (c != EOF)
475 std::ungetc (c, stdin);
476 return "";
477 }
478
479 while ((c = getc (f)) != EOF)
480 {
481 if (c == '"')
482 break;
483 else if (c == '\\')
484 {
485 int next_c = getc (f);
486 if (next_c == EOF)
487 break;
488 result += static_cast<char> (next_c);
489 }
490 else
491 result += static_cast<char> (c);
492 }
493
494 return result;
495}
