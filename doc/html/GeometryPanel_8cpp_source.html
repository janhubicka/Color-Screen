273void GeometryPanel::updateRegistrationPointInfo(const ParameterState &state) {
274 int numPoints = state.solver.points.size();
275 colorscreen::scr_type type = state.scrToImg.type;
276
277 if (m_showRegistrationPointsBox) {
278 m_showRegistrationPointsBox->setText(QString("Show registration points (%1 points)").arg(numPoints));
279 }
280
281 auto updateMsg = [numPoints](QLabel *label, int threshold, const QString &task) {
282 if (!label) return;
283 if (numPoints < threshold) {
284 label->setText(QString("%1 additional registration points needed for %2").arg(threshold - numPoints).arg(task));
285 label->show();
286 } else {
287 label->hide();
288 }
289 };
290
291 updateMsg(m_optimizationMessageLabel, colorscreen::solver_parameters::min_points(type), "basic optimization");
292 updateMsg(m_lensMessageLabel, colorscreen::solver_parameters::min_lens_points(type), "lens correction");
293 updateMsg(m_tiltMessageLabel, colorscreen::solver_parameters::min_perspective_points(type), "perspective correction");
294 updateMsg(m_nonlinearMessageLabel, colorscreen::solver_parameters::min_mesh_points(type), "nonlinear correction");
295
296 bool canOptimize = numPoints >= colorscreen::solver_parameters::min_points(type);
297 if (m_optimizeButton) m_optimizeButton->setEnabled(canOptimize);
298
299 if (m_gridWidthSpin && m_gridHeightSpin) {
300 auto scan = m_imageGetter();
301 if (scan && scan->width > 0 && scan->height > 0) {
302 colorscreen::int_image_area crop = state.rparams.get_scan_crop(scan->width, scan->height);
303 int defW, defH;
304 m_finetuneAreaParams.get_grid_dimensions(crop, state.scrToImg, &defW, &defH);
305 m_gridWidthSpin->setSpecialValueText(QString("Default (%1)").arg(defW));
306 m_gridHeightSpin->setSpecialValueText(QString("Default (%1)").arg(defH));
307 } else {
308 m_gridWidthSpin->setSpecialValueText("Default");
309 m_gridHeightSpin->setSpecialValueText("Default");
310 }
311 }

320void GeometryPanel::updateDeformationChart() {
321 if (!m_deformationChart)
322 return;
323
324 // Get current state
325 ParameterState state = m_stateGetter();
326
327 // Sync tolerance from slider
328 double tol = 0.5;
329 if (m_heatmapToleranceSlider) {
330 tol = m_heatmapToleranceSlider->value() / 1000.0;
331 }
332 if (m_deformationChart) m_deformationChart->setHeatmapTolerance(tol);
333 if (m_lensChart) m_lensChart->setHeatmapTolerance(tol);
334 if (m_perspectiveChart) m_perspectiveChart->setHeatmapTolerance(tol);
335 if (m_nonlinearChart) m_nonlinearChart->setHeatmapTolerance(tol);
336
337 // Get scan image
338 auto scan = m_imageGetter();
339 bool hasScan = (scan && scan->width > 0 && scan->height > 0);
340
341 // Update visibility based on content
342 bool showLens = hasScan && !state.scrToImg.lens_correction.is_noop();
343 bool showPerspective = hasScan && (std::abs(state.scrToImg.tilt_x) > 1e-6 || std::abs(state.scrToImg.tilt_y) > 1e-6);
344 bool showNonlinear = hasScan && (state.scrToImg.mesh_trans != nullptr);
345 bool showFinal = hasScan && (state.scrToImg.type != colorscreen::Random);
346
347 // Using parentWidget() of the layout to get the container widget added to the form
348 if (m_lensChartContainer && m_lensChartContainer->parentWidget())
349 m_lensChartContainer->parentWidget()->setVisible(showLens);
350
351 if (m_perspectiveChartContainer && m_perspectiveChartContainer->parentWidget())
352 m_perspectiveChartContainer->parentWidget()->setVisible(showPerspective);
353
354 if (m_nonlinearChartContainer && m_nonlinearChartContainer->parentWidget())
355 m_nonlinearChartContainer->parentWidget()->setVisible(showNonlinear);
356
357 if (m_chartContainer && m_chartContainer->parentWidget())
358 m_chartContainer->parentWidget()->setVisible(showFinal);
359
360 if (!hasScan) {
361 if(m_deformationChart) m_deformationChart->clear();
362 if(m_lensChart) m_lensChart->clear();
363 if(m_perspectiveChart) m_perspectiveChart->clear();
364 if(m_nonlinearChart) m_nonlinearChart->clear();
365 return;
366 }
367 colorscreen::int_image_area crop = state.rparams.get_scan_crop(scan->width, scan->height);
368 int w = crop.width;
369 int h = crop.height;
370 int ox = crop.x;
371 int oy = crop.y;
372 int fw = scan->width;
373 int fh = scan->height;
374
375 // Create undeformed parameters by copying center and coordinates from current params
376 colorscreen::scr_to_img_parameters p0;
377 p0.center = state.scrToImg.center;
378 p0.coordinate1 = state.scrToImg.coordinate1;
379 p0.coordinate2 = state.scrToImg.coordinate2;
380 p0.scanner_type = state.scrToImg.scanner_type;
381
382 bool mirror = state.rparams.scan_mirror;
383 int rotation = state.rparams.scan_rotation;
384
385 // 1. Lens Chart: Undeformed vs Lens
386 colorscreen::scr_to_img_parameters p1 = p0;
387 p1.lens_correction = state.scrToImg.lens_correction;
388 if (m_lensChart) m_lensChart->setDeformationData(p1, p0, w, h, mirror, rotation, ox, oy, fw, fh);
389
390 // 2. Perspective Chart: Lens vs Perspective
391 colorscreen::scr_to_img_parameters p2 = p1;
392 p2.tilt_x = state.scrToImg.tilt_x;
393 p2.tilt_y = state.scrToImg.tilt_y;
394 p2.projection_distance = state.scrToImg.projection_distance;
395 if (m_perspectiveChart) m_perspectiveChart->setDeformationData(p2, p1, w, h, mirror, rotation, ox, oy, fw, fh);
396
397 // 3. Nonlinear Chart: Perspective vs Nonlinear
398 colorscreen::scr_to_img_parameters p3 = p2;
399 p3.mesh_trans = state.scrToImg.mesh_trans;
400 p3.mesh_trans_is_scr_to_img = state.scrToImg.mesh_trans_is_scr_to_img;
401 if (m_nonlinearChart) m_nonlinearChart->setDeformationData(p3, p2, w, h, mirror, rotation, ox, oy, fw, fh);
402
403 // 4. Final: Undeformed vs Current
404 if (m_deformationChart)
405 m_deformationChart->setDeformationData(state.scrToImg, p0, w, h, mirror, rotation, ox, oy, fw, fh);
406}
