1275sample_image_area (image_data *img, render *render, coord_t fx, coord_t fy,

1277{

1280 coord_t ry = my_modf (fy, &y);
1281 int xmin = x - range;

1315 sum += render->get_linearized_rgb_pixel ({x, y}) * yweight;
1316 sumweight += yweight;
