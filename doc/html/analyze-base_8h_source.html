413 /* Return the average color of a screen tile at (X, Y). */
414 inline rgbdata
415 screen_tile_color (int x, int y) const
416 {
417 rgbdata ret = {0,0,0};
418 for (int yy = 0; yy < GEOMETRY::red_height_scale; yy++)
419 for (int xx = 0; xx < GEOMETRY::red_width_scale; xx++)
420 ret.red += fast_red (x * GEOMETRY::red_width_scale + xx, y * GEOMETRY::red_height_scale + yy);
421 ret.red *= ((luminosity_t) 1.0 / (GEOMETRY::red_height_scale * GEOMETRY::red_width_scale));
422 for (int yy = 0; yy < GEOMETRY::green_height_scale; yy++)
423 for (int xx = 0; xx < GEOMETRY::green_width_scale; xx++)
424 ret.green += fast_green (x * GEOMETRY::green_width_scale + xx, y * GEOMETRY::green_height_scale + yy);
425 ret.green *= ((luminosity_t) 1.0 / (GEOMETRY::green_height_scale * GEOMETRY::green_width_scale));
426 for (int yy = 0; yy < GEOMETRY::blue_height_scale; yy++)
427 for (int xx = 0; xx < GEOMETRY::blue_width_scale; xx++)
428 ret.blue += fast_blue (x * GEOMETRY::blue_width_scale + xx, y * GEOMETRY::blue_height_scale + yy);
429 ret.blue *= ((luminosity_t) 1.0 / (GEOMETRY::blue_height_scale * GEOMETRY::blue_width_scale));
430 return ret;
431 }

478/* Linear interpolation between A and B with offset OFF. */
479inline luminosity_t linear_interpolate (luminosity_t a, luminosity_t b, luminosity_t off)
480{
481 return a * ((luminosity_t) 1 - off) + b * off;
482}
483
484/* 2D linear interpolation between V1, V2, V3, V4 with offset OFF. */
485inline flatten_attr pure_attr luminosity_t always_inline_attr
486do_linear_interpolate (luminosity_t v1, luminosity_t v2, luminosity_t v3, luminosity_t v4, point_t off)
487{
488 luminosity_t a = linear_interpolate (v1,v2, (luminosity_t)off.x);
489 luminosity_t b = linear_interpolate (v3,v4, (luminosity_t)off.x);
490 return linear_interpolate (a,b, (luminosity_t)off.y);
491}
