31/* Helper for downscaling template for grayscale rendering.
32 Account VAL * SCALE to DATA. */
33template <bool UseAtomic = false>
34inline void
35account_pixel (luminosity_t *data, luminosity_t val, luminosity_t scale)
36{
37 if constexpr (UseAtomic)
38 {
39#pragma omp atomic
40 *data += val * scale;
41 }
42 else
43 *data += val * scale;
44}
45
46/* Helper for downscaling template for color rendering.
47 Account VAL * SCALE to DATA. */
48template <bool UseAtomic = false>
49inline void
50account_rgb_pixel (rgbdata *data, rgbdata val, luminosity_t scale)
51{
52 if constexpr (UseAtomic)
53 {
54#pragma omp atomic
55 data->red += val.red * scale;
56#pragma omp atomic
57 data->green += val.green * scale;
58#pragma omp atomic
59 data->blue += val.blue * scale;
60 }
61 else
62 {
63 data->red += val.red * scale;
64 data->green += val.green * scale;
65 data->blue += val.blue * scale;
66 }
67}
68
69/* Unified accounting dispatcher. Account VAL * SCALE to DATA. */
70template <typename T, bool UseAtomic = false>
71inline void
72do_account (T *data, T val, luminosity_t scale)
73{
74 if constexpr (std::is_same_v<T, rgbdata>)
75 account_rgb_pixel<UseAtomic> (data, val, scale);
76 else
77 account_pixel<UseAtomic> (data, val, scale);
78}

424 interpolation without adjustments. */
425pure_attr inline luminosity_t
426render::get_unadjusted_img_pixel (point_t p) const noexcept
427{

432 coord_t rx = my_modf (p.x, &sx);

435 if (sx >= 1 && sx < m_img.width - 2 && sy >= 1 && sy < m_img.height - 2)

438 get_unadjusted_data ({ sx, sy - 1 }),

442 get_unadjusted_data ({ sx, sy - 0 }),
443 get_unadjusted_data ({ sx + 1, sy - 0 }),

557/* Inner loop for image downscaling processing single pixel PX, PY in result
558 DATA from source pixel VAL. */
559template <typename T, bool UseAtomic, auto account_p>
560inline void
561render::process_pixel (T *data, int width, int height, int px, int py, bool x0,
562 bool x1, bool y0, bool y1, T pixel, luminosity_t scale,
563 luminosity_t xweight, luminosity_t yweight) noexcept
564{
565 if (colorscreen_checking)
566 {
567 assert (px >= (x0 ? 0 : -1) && px < (x1 ? width - 1 : width));
568 assert (py >= (y0 ? 0 : -1) && py < (y1 ? height - 1 : height));
569 }
570 if (x0)
571 {
572 if (y0)
573 do_account<T, UseAtomic> (data + px + py * width, pixel,
574 scale * (1 - yweight) * (1 - xweight));
575 if (y1)
576 do_account<T, UseAtomic> (data + px + (py + 1) * width, pixel,
577 scale * yweight * (1 - xweight));
578 }
579 if (x1)
580 {
581 if (y0)
582 do_account<T, UseAtomic> (data + px + (py * width) + 1, pixel,
583 scale * (1 - yweight) * xweight);
584 if (y1)
585 do_account<T, UseAtomic> (data + px + (py + 1) * width + 1, pixel,
586 scale * yweight * xweight);
587 }
588}

590/* Inner loop for image downscaling processing single line. */
591template <typename T, typename D, T (D::*get_pixel) (int_point_t p) const,
592 bool UseAtomic, auto account_p>
593inline void
594render::process_line (T *data, int *pixelpos, luminosity_t *weights,
595 int xstart, int xend, int width, int height, int py,
596 int yy, bool y0, bool y1, luminosity_t scale,
597 luminosity_t yweight) noexcept
598{
599 int px = xstart;
600 int xx = pixelpos[px];
601 int stop;
602 if (yy < 0 || yy >= m_img.height || xx >= m_img.width)
603 return;
604 if (px >= 0 && xx >= 0)
605 {
606 T pixel = (((D *)this)->*get_pixel) ({ xx, yy });
607 process_pixel<T, UseAtomic, account_p> (data, width, height, px - 1, py,
608 false, true, y0, y1, pixel,
609 scale, weights[px], yweight);
610 }
611 xx++;
612 if (xx < 0)
613 xx = 0;
614 stop = pixelpos[px + 1];
615 for (; xx < stop; xx++)
616 {
617 T pixel = (((D *)this)->*get_pixel) ({ xx, yy });
618 process_pixel<T, UseAtomic, account_p> (data, width, height, px, py,
619 true, false, y0, y1, pixel,
620 scale, 0, yweight);
621 }
622 px++;
623 while (px <= xend)
624 {
625 T pixel = (((D *)this)->*get_pixel) ({ xx, yy });
626 process_pixel<T, UseAtomic, account_p> (data, width, height, px - 1, py,
627 true, true, y0, y1, pixel, scale,
628 weights[px], yweight);
629 stop = pixelpos[px + 1];
630 xx++;
631 for (; xx < stop; xx++)
632 {
633 T pixel = (((D *)this)->*get_pixel) ({ xx, yy });
634 process_pixel<T, UseAtomic, account_p> (data, width, height, px, py,
635 true, false, y0, y1, pixel,
636 scale, 0, yweight);
637 }
638 px++;
639 }
640 if (xx < m_img.width)
641 {
642 T pixel = (((D *)this)->*get_pixel) ({ xx, yy });
643 process_pixel<T, UseAtomic, account_p> (data, width, height, px - 1, py,
644 true, false, y0, y1, pixel,
645 scale, weights[px], yweight);
646 }
647}

649template <typename D, typename T, T (D::*get_pixel) (int_point_t p) const,
650 auto account_p>
651bool
652render::downscale (T *data, point_t p, int width, int height,
653 coord_t pixelsize, progress_info *progress) noexcept
654{
655 coord_t x = p.x;
656 coord_t y = p.y;
657 int pxstart = std::max (0, (int)(-x / pixelsize));
658 int pxend = std::min (width - 1, (int)((m_img.width - x) / pixelsize));
659
660 std::fill (data, data + (size_t)width * height, T{});
661
662 if (pxstart > pxend)
663 return true;
664
665 if (progress)
666 {
667 int pystart = std::max (0, (int)(-y / pixelsize));
668 int pyend = std::min (height - 1, (int)((m_img.height - y) / pixelsize));
669 progress->set_task ("downscaling", pyend - pystart + 1);
670 }
671
672 /* Precompute scaling weights for X coordinate. */
673 std::vector<int> pixelpos (width + 2);
674 std::vector<luminosity_t> weights (width + 2);
675
676 for (int px = pxstart; px <= pxend + 1; px++)
677 {
678 coord_t ix = x + pixelsize * px;
679 int xx = my_floor (ix);
680 pixelpos[px] = std::min (xx, m_img.width);
681 weights[px] = 1 - (ix - xx);
682 }
683
684#define ypixelpos(p) ((int)my_floor (y + pixelsize * (p)))
685#define weight(p) (1 - (y + pixelsize * (p) - ypixelpos (p)))
686
687#pragma omp parallel shared(progress, data, pixelsize, width, height, \
688 pixelpos, x, y, pxstart, pxend, \
689 weights) default(none)
690 {
691 luminosity_t scale = (luminosity_t)1.0 / (pixelsize * pixelsize);
692 int pystart = std::max (0, (int)(-y / pixelsize));
693 int pyend = std::min (height - 1, (int)((m_img.height - y) / pixelsize));
694
695#ifdef _OPENMP
696 int tn = omp_get_thread_num ();
697 int threads = omp_get_num_threads ();
698#else
699 int tn = 0;
700 int threads = 1;
701#endif
702
703 int ystart = pystart + (pyend + 1 - pystart) * tn / threads;
704 int yend = pystart + (pyend + 1 - pystart) * (tn + 1) / threads - 1;
705
706 int py = ystart;
707 int yy = ypixelpos (py);
708 int stop;
709
710 if (ystart <= yend)
711 {
712 /* The first line of each thread's range may overlap with the previous
713 thread's last line because bilinear interpolation spans two output
714 rows. We use UseAtomic=true for boundary rows. */
715 if (py >= 0 && yy >= 0)
716 process_line<T, D, get_pixel, true, account_p> (
717 data, pixelpos.data (), weights.data (), pxstart, pxend, width,
718 height, py - 1, yy, false, true, scale, weight (py));
719 yy++;
720 stop = std::min (ypixelpos (py + 1), m_img.height);
721 for (; yy < stop; yy++)
722 {
723 /* If we have only one output row in our chunk, use atomics. */
724 if (py == yend)
725 process_line<T, D, get_pixel, true, account_p> (
726 data, pixelpos.data (), weights.data (), pxstart, pxend,
727 width, height, py, yy, true, false, scale, 0);
728 else
729 process_line<T, D, get_pixel, false, account_p> (
730 data, pixelpos.data (), weights.data (), pxstart, pxend,
731 width, height, py, yy, true, false, scale, 0);
732 }
733 py++;
734
735 while (py <= yend && (!progress || !progress->cancel_requested ()))
736 {
737 /* Only use atomics if this is the last row of our range, which
738 might be shared with the next thread. */
739 if (py == yend)
740 process_line<T, D, get_pixel, true, account_p> (
741 data, pixelpos.data (), weights.data (), pxstart, pxend,
742 width, height, py - 1, yy, true, true, scale, weight (py));
743 else
744 process_line<T, D, get_pixel, false, account_p> (
745 data, pixelpos.data (), weights.data (), pxstart, pxend,
746 width, height, py - 1, yy, true, true, scale, weight (py));
747
748 stop = std::min (ypixelpos (py + 1), m_img.height);
749 yy++;
750 for (; yy < stop; yy++)
751 {
752 if (py == yend)
753 process_line<T, D, get_pixel, true, account_p> (
754 data, pixelpos.data (), weights.data (), pxstart, pxend,
755 width, height, py, yy, true, false, scale, 0);
756 else
757 process_line<T, D, get_pixel, false, account_p> (
758 data, pixelpos.data (), weights.data (), pxstart, pxend,
759 width, height, py, yy, true, false, scale, 0);
760 }
761 py++;
762 if (progress)
763 progress->inc_progress ();
764 }
765 /* Final line of input image might also contribute to the last row of
766 * our range. */
767 if (yy < m_img.height)
768 process_line<T, D, get_pixel, true, account_p> (
769 data, pixelpos.data (), weights.data (), pxstart, pxend, width,
770 height, py - 1, yy, true, false, scale, weight (py));
771 }
772 }
773
774#undef ypixelpos
775#undef weight
776 return !progress || !progress->cancelled ();
777}
