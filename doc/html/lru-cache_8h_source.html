36/* LRU cache used keep various data between invocations of renderers.
37 P represents parameters which are used to produce T.
38 get_new is a function computing T based on P. It is expected to
39 allocate memory via std::unique_ptr and the cache will manage it.
40
41 Template Architecture:
42 The implementation uses a template-based design to support different
43 types of keys and values while sharing common logic via abstract_lru_cache.
44 - P: Parameter type (key)
45 - T: Result type (value)
46 - Entry: The struct type representing a cache entry
47 - Derived: The final cache class (CRTP) used to fetch base configuration
48
49 Synchronization Model:
50 The cache uses a combination of a global lock (std::shared_timed_mutex) and
51 per-entry state ("computing" flag) to ensure thread-safety while maximizing
52 concurrency.
53
54 1. Lock Granularity: The global "lock" protects the integrity of the linked list
55 ("entries") and metadata (last_used, id). Using shared_timed_mutex allows
56 concurrent lookup attempts while waiting for long-running computations.
57
58 2. Non-blocking Computation: To prevent the cache from stalling the entire
59 application during a long "get_new" call, the implementation:
60 a) Identifies/allocates the target hit/miss while locked.
61 b) Sets "computing = true" on the entry.
62 c) Unlocks the global lock.
63 d) Executes "get_new" concurrently.
64 e) Re-locks to store the result and unset "computing".
65 Other threads can freely access, add, or release independent entries
66 while "get_new" is running.
67
68 3. Race Condition Prevention:
69 - Redundant Computation: If a second thread requests the same parameters
70 while an entry is already marked "computing", it waits on a
71 condition variable ("cond") until the first thread finishes.
72 It does NOT start a second "get_new" call.
73 - Premature Reuse: The "computing" flag ensures that "prune()" or lookup
74 logic for "longest_unused" will skip entries currently being generated
75 or updated out-of-lock.
76 - Task Cancellation: All wait loops (both for the global lock and the
77 condition variable) periodically wake up (333ms) to check if the
78 caller has requested task cancellation via the "progress" object. */
79

159 /* Internal lookup and management logic shared by all cache types.
160 P is the parameter set.
161 PROGRESS is the progress info for task cancellation.
162 ID_OUT will receive the unique ID of the entry.
163 MATCH_FUNC is a predicate checking if an entry matches P.
164 INIT_FUNC initializes a new entry for P.
165 FETCH_FUNC produces the value T for P. */
166 template <typename Matcher, typename Init, typename Fetcher>
167 std::shared_ptr<T>
168 get_internal (P &p, progress_info *progress, uint64_t *id_out, Matcher &&match_func, Init &&init_func, Fetcher &&fetch_func)
169 {
170 uint64_t time = lru_caches::get ();
171 Entry *longest_unused = NULL, *e;
172 int size = 0;
173
174 if (progress)
175 progress->wait ("unlocking cache");
176
177 std::unique_lock<std::shared_timed_mutex> guard (lock, std::defer_lock);
178 while (!guard.try_lock_for (std::chrono::milliseconds (333)))
179 {
180 if (progress && progress->cancel_requested ())
181 return NULL;
182 }
183 time++;
184
185 restart:
186 size = 0;
187 longest_unused = NULL;
188 for (e = entries; e; e = e->next)
189 {
190 if (match_func (e))
191 {
192 while (e->computing)
193 {
194 uint64_t id_val = e->id;
195 if (progress)
196 progress->wait ("waiting for other thread to finish computation");
197 if (cond.wait_for (guard, std::chrono::milliseconds (333))
198 == std::cv_status::timeout)
199 {
200 if (progress && progress->cancel_requested ())
201 return NULL;
202 }
203 bool found = false;
204 for (Entry *e2 = entries; e2; e2 = e2->next)
205 if (e2 == e && e2->id == id_val)
206 {
207 found = true;
208 break;
209 }
210 if (!found)
211 goto restart;
212 }
213 e->last_used = time;
214 if (verbose)
215 fprintf (stderr, "Cache %s: hit id %i\n", name, (int)e->id);
216 std::shared_ptr<T> ret = e->val;
217 if (id_out)
218 *id_out = e->id;
219 return ret;
220 }
221 if (e->val.use_count () <= 1 && !e->computing
222 && (!longest_unused || longest_unused->last_used < e->last_used))
223 longest_unused = e;
224 size++;
225 }
226
227 if (size >= cache_size && longest_unused)
228 {
229 e = longest_unused;
230 if (verbose)
231 fprintf (stderr, "Cache %s: deleting id %i\n", name, (int)e->id);
232 e->val = nullptr;
233 }
234 else
235 {
236 e = new Entry;
237 e->next = entries;
238 entries = e;
239 }
240
241 e->params = p;
242 init_func (e);
243 e->computing = true;
244 e->id = time;
245 e->last_used = time;
246
247 guard.unlock ();
248 std::shared_ptr<T> ret_val;
249 {
250 computing_guard<std::shared_timed_mutex> cguard (e->computing, lock, cond);
251 std::unique_ptr<T> val = fetch_func (e);
252 guard.lock ();
253 e->val = std::move (val);
254 e->computing = false;
255 ret_val = e->val;
256 cguard.finished ();
257 }
258 cond.notify_all ();
259
260 if (id_out)
261 *id_out = e->id;
262 if (!ret_val)
263 {
264 for (Entry **e2 = &entries;; e2 = &(*e2)->next)
265 if (*e2 == e)
266 {
267 *e2 = e->next;
268 delete e;
269 break;
270 }
271 }
272 if (verbose && ret_val)
273 fprintf (stderr, "Cache %s: added id %i size %i\n", name, (int)e->id, (int)size);
274 return ret_val;
275 }

296 }
