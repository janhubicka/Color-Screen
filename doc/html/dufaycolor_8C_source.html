160static void
161initialize_spec (spectrum_dyes_to_xyz &spec, int mode, bool normalized_dyes)
162{
163 if (!mode)
164 spec.set_dyes (spectrum_dyes_to_xyz::dufaycolor_color_cinematography);
165 else if (mode == 1)
166 spec.set_dyes (spectrum_dyes_to_xyz::dufaycolor_harrison_horner);
167 else if (mode == 2)
168 spec.set_dyes (spectrum_dyes_to_xyz::dufaycolor_photography_its_materials_and_processes);
169 else if (mode == 3)
170 spec.set_dyes (spectrum_dyes_to_xyz::dufaycolor_collins_giles);
171 else
172 abort ();
173
174 /* Set scales so we simulate look of the overall screen. */
175 if (normalized_dyes)
176 {
177 spec.rscale = dufaycolor::red_portion;
178 spec.gscale = dufaycolor::green_portion;
179 spec.bscale = dufaycolor::blue_portion;
180 }
181 else
182 spec.rscale = spec.gscale = spec.bscale = 1;
183
184 initialize_spec_response (spec);
185}
