206void ThamesAnimation::checkBallCollisions(Ball &ball1, Ball &ball2) {
207 QPointF delta = ball2.pos - ball1.pos;
208 double dist = std::sqrt(delta.x() * delta.x() + delta.y() * delta.y());
209 double minDist = ball1.radius + ball2.radius;
210
211 if (dist < minDist && dist > 0.001) {
212 // Collision detected - elastic collision
213 QPointF normal = delta / dist;
214
215 // Separate balls
216 double overlap = minDist - dist;
217 ball1.pos -= normal * (overlap / 2.0);
218 ball2.pos += normal * (overlap / 2.0);
219
220 // Calculate relative velocity
221 QPointF relVel = ball2.velocity - ball1.velocity;
222 double velAlongNormal = relVel.x() * normal.x() + relVel.y() * normal.y();
223
224 // Only resolve if balls are moving toward each other
225 if (velAlongNormal < 0) {
226 return;
227 }
228
229 // Apply impulse (assuming equal mass)
230 QPointF impulse = normal * velAlongNormal;
231 ball1.velocity += impulse * BALL_BOUNCE_DAMPING;
232 ball2.velocity -= impulse * BALL_BOUNCE_DAMPING;
233 }
