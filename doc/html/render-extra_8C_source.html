41bool
42render_extra::precompute (coord_t xmin, coord_t ymin, coord_t xmax, coord_t ymax, progress_info *progress)
43{
44 uint64_t screen_id = 0;
45 /* When doing profiled matrix, we need to pre-scale the profile so black point corretion goes right.
46 Without doing so, for exmaple black from red pixels would be subtracted too agressively, since
47 we account for every pixel in image, not only red patch portion. */
48 if (!render_to_scr::precompute (true, true, xmin, ymin, xmax, ymax, progress))
49 return false;
50 if (m_params.precise)
51 {
52 coord_t radius = m_params.screen_blur_radius * pixel_size ();
53 sharpen_parameters sp = m_params.sharpen;
54 sp.usm_radius = radius;
55 m_screen = get_screen (m_scr_to_img.get_type (), false,
56 sp.deconvolution_p (), sp,
57 m_params.red_strip_width,
58 m_params.green_strip_width, progress, &screen_id);
59 if (!m_screen)
60 return false;
61 }
62 int xshift = -xmin;
63 int yshift = -ymin;
64 int width = xmax - xmin;
65 int height = ymax - ymin;
66 int xshift2, yshift2, width2, height2;
67 m_scr_to_img.get_range (0, 0, m_img.width, m_img.height, &xshift2, &yshift2, &width2, &height2);
68 if (xshift > xshift2)
69 width -= xshift-xshift2, xshift = xshift2;
70 if (yshift > yshift2)
71 height -= yshift-yshift2, yshift = yshift2;
72 if (width - xshift > width2 - xshift2)
73 width = width2 - xshift2 + xshift;
74 if (height - yshift > height2 - yshift2)
75 height = height2 - yshift2 + yshift;
76 /* For UI response,
77 it is better to compute whole image then significant portion of it. */
78 if (width * height > width2 * height2 / 2)
79 {
80 xshift = xshift2;
81 yshift = yshift2;
82 width = width2;
83 height = height2;
84 }
85 /* We need to compute bit more to get interpolation right.
86 TODO: figure out how much. */
87 xshift += 4;
88 yshift += 4;
89 width += 8;
90 height += 8;
91 struct analyzer_params p
92 {
93 m_img.id,
94 m_gray_data_id,
95 screen_id,
96 !m_params.precise ? analyze_base::fast : analyze_base::precise,
97 m_params.collection_threshold,
98 m_scr_to_img.get_param ().mesh_trans ? m_scr_to_img.get_param ().mesh_trans->id : 0,
99 m_scr_to_img.get_param (),
100 &m_img,
101 m_screen.get (),
102 this,
103 &m_scr_to_img,
104 };
105 if (m_scr_to_img.get_type () != Dufay)
106 {
107 m_paget = paget_analyzer_cache.get (p, xshift, yshift, width, height, progress);
108 if (!m_paget)
109 return false;
110 }
111 else
112 {
113 m_dufay = dufay_analyzer_cache.get (p, xshift, yshift, width, height, progress);
114 if (!m_dufay)
115 return false;
116 }
117 return !progress || !progress->cancelled ();
118}
