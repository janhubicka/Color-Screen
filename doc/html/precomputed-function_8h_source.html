16
17 This class stores a precomputed table of a 1D function f(x) over a specified

113 /* Evaluate the function at X using linear interpolation. */
114 T pure_attr
115 apply (T x) const noexcept
116 {
117 if (m_table.empty ())
118 return x;
119 int index = my_floor ((x - m_min_x) * m_step_inv);
120 index = std::max (std::min (index, (int)m_table.size () - 1), 0);
121 return m_table[index].add + m_table[index].slope * x;
122 }
123
124 /* Determine the inverse of the function for value Y. Works only for
125 monotone functions. */
126 T pure_attr
127 invert (T y) const noexcept
128 {
129 unsigned int min = 0;
130 unsigned int max = m_table.size ();
131 while (max != min)
132 {
133 unsigned int ix = (min + max) / 2;
134 T xx = m_min_x + ix * m_step;
135 T val = m_table[ix].add + m_table[ix].slope * xx;
136 T val2 = val + m_table[ix].slope * m_step;
137 if (!increasing)
138 std::swap (val, val2);
139 if (val <= y && y <= val2)
140 min = max = ix;
141 else if ((val < y) ^ increasing)
142 max = ix;
143 else if (min != ix)
144 min = ix;
145 else
146 break;
147 }
148 T ret = m_table[min].slope
149 ? (y - m_table[min].add) / m_table[min].slope
150 : m_min_x + min * m_step;
151
152 /* Verify that inverse works. This also crashes if the function is not
153 monotone. */
154 if (debug && std::abs (apply (ret) - y) > epsilon)
155 {
156 printf ("%i %i:%f...%i:%f min:%i (%f..%f) ret:%f %f should be %f\n",
157 increasing, 0, m_min_x, (int)m_table.size () - 1, m_max_x, min,
158 m_min_x + min * m_step, m_min_x + (min + 1) * m_step, (double)ret,
159 (double)apply (ret), (double)y);
160 abort ();
161 }
162 return ret;
163 }

165 /* Initialize the table using Y values at equidistant X points.
166 LEN specifies the number of Y values. */
167 void
168 init_by_y_values (const T *y, int len)
169 {
170 if (len < 2)
171 abort ();
172 int n_entries = len - 1;
173 m_table.resize (n_entries);
174 m_step = (m_max_x - m_min_x) / (T)n_entries;
175 m_step_inv = 1 / m_step;
176 for (int i = 0; i < n_entries; i++)
177 {
178 T xleft = m_min_x + i * m_step;
179 m_table[i].slope = (y[i + 1] - y[i]) * m_step_inv;
180 m_table[i].add = y[i] - xleft * m_table[i].slope;
181 }
182 increasing = y[0] < y[len - 1];
183 }

185 /* Initialize the table using an evaluation function F. LEN specifies
186 the number of entries in the lookup table. If PARALLEL is true,
187 initialization is performed in parallel using OpenMP. */
188 template <typename Func>
189 void
190 init_by_function (int len, Func f, bool parallel = false)
191 {
192 if (len < 2)
193 abort ();
194 std::vector<T> y (len);
195 T step = (m_max_x - m_min_x) / (T)(len - 1);
196
197 if (parallel)
198 {
199#pragma omp parallel for
200 for (int i = 0; i < len; i++)
201 y[i] = f (m_min_x + i * step);
202 }
203 else
204 {
205 for (int i = 0; i < len; i++)
206 y[i] = f (m_min_x + i * step);
207 }
208 init_by_y_values (y.data (), len);
209 }

211 /* Initialize using linear interpolation between control points X and Y.
212 NPOINTS specifies the number of control points. LEN is the number
213 of lookup table entries. */
214 void
215 init_by_x_y_values (const T *x, const T *y, int npoints, int len)
216 {
217 /* If there are only 2 npoints we represent linear function. */
218 if (npoints <= 2)
219 len = 2;
220
221 init_by_function (len, [&](T xx) {
222 if (!npoints)
223 return xx;
224 if (npoints == 1)
225 return xx + y[0] - x[0];
226
227 int p = 0;
228 while (p < npoints - 1 && x[p + 1] < xx)
229 p++;
230 return y[p] + (y[p + 1] - y[p]) * (xx - x[p]) / (x[p + 1] - x[p]);
231 });
232 }

236 of lookup table entries. */

240 if (npoints <= 2)
241 len = 2;
242
243 init_by_function (len, [&](T xx) {
244 if (!npoints)
245 return xx;
246 if (npoints == 1)
247 return xx + table[0][1] - table[0][0];
248
249 int p = 0;
250 while (p < npoints - 1 && table[p + 1][0] < xx)
251 p++;
252 return table[p][1]
253 + (table[p + 1][1] - table[p][1]) * (xx - table[p][0])
254 / (table[p + 1][0] - table[p][0]);
