31/* Determine homography matrix matching points specified by POINTS.
32 Update PARAM for desired transformations.
33 W_CENTER specifies point where to optimize for if solve_screen_weights
34 or solve_image_weights are set in FLAGS.
35 If FINAL_RUN is true output info on results. */
36
37nodiscard_attr coord_t
38solver (scr_to_img_parameters *param, const image_data &img_data,
39 const std::vector<solver_parameters::solver_point_t> &points,
40 const point_t w_center, int flags, bool final_run = false)
41{
42 if (debug_output && final_run)
43 {
44 printf ("Old Translation %f %f\n", param->center.x, param->center.y);
45 printf ("Old coordinate1 %f %f\n", param->coordinate1.x,
46 param->coordinate1.y);
47 printf ("Old coordinate2 %f %f\n", param->coordinate2.x,
48 param->coordinate2.y);
49 }
50 /* Clear previous map. */
51 param->center = { (coord_t)0, (coord_t)0 };
52 param->coordinate1 = { (coord_t)1, (coord_t)0 };
53 param->coordinate2 = { (coord_t)0, (coord_t)1 };
54 if (flags & homography::solve_rotation)
55 {
56 param->tilt_x = 0;
57 param->tilt_y = 0;
58 }
59 /* This map applies only non-linear part of corrections (that are not
60 optimized). */
61 scr_to_img map;
62 if (!map.set_parameters (*param, img_data))
63 return 1e30;
64
65 double chisq;
66 if (screen_with_vertical_strips_p (param->type))
67 flags |= homography::solve_vertical_strips;
68 bool do_ransac = !(flags
69 & (homography::solve_image_weights
70 | homography::solve_screen_weights));
71
72 trans_4d_matrix h;
73 if (do_ransac)
74 h = homography::get_matrix_ransac (points, flags, param->scanner_type,
75 &map, w_center, &chisq, final_run);
76 else
77 h = homography::get_matrix (points, flags, param->scanner_type, &map,
78 w_center, &chisq);
79 /* Determine center and coordinate vectors. */
80 point_t center = h.perspective_transform ({ (coord_t) 0, (coord_t) 0 });
81 point_t coordinate1 = h.perspective_transform ({ (coord_t) 1, (coord_t) 0 });
82 point_t coordinate2 = h.perspective_transform ({ (coord_t) 0, (coord_t) 1 });
83
84 param->center = map.inverse_early_correction (center);
85 param->coordinate1 = map.inverse_early_correction (coordinate1) - param->center;
86 param->coordinate2 = map.inverse_early_correction (coordinate2) - param->center;
87 /* TODO: Can we decompose matrix in the way scr_to_img expects the
88 parameters? */
89 coord_t minsq = INT_MAX;
90 coord_t best_tilt_x = 1, best_tilt_y = 1;
91 if (flags & homography::solve_rotation)
92 {
93 coord_t tilt_x_min = -0.003, tilt_x_max = 0.003;
94 int tilt_x_steps = 21;
95 coord_t tilt_y_min = -0.003, tilt_y_max = 0.003;
96 int tilt_y_steps = 21;
97 scr_to_img map2;
98 if (!map2.set_parameters (*param, img_data))
99 return 1e30;
100 for (int i = 0; i < 10; i++)
101 {
102 coord_t txstep = (tilt_x_max - tilt_x_min) / (tilt_x_steps - 1);
103 coord_t tystep = (tilt_y_max - tilt_y_min) / (tilt_y_steps - 1);
104 for (int tx = 0; tx < tilt_x_steps; tx++)
105 for (int ty = 0; ty < tilt_y_steps; ty++)
106 {
107 param->tilt_x = tilt_x_min + txstep * tx;
108 param->tilt_y = tilt_y_min + tystep * ty;
109 coord_t sq = 0;
110 map2.update_linear_parameters (*param);
111 for (int sy = -100; sy <= 100; sy += 100)
112 for (int sx = -100; sx <= 100; sx += 100)
113 {
114 point_t t = map2.to_img ({ (coord_t)sx, (coord_t)sy });
115 point_t p;
116 p = h.perspective_transform ({ (coord_t)sx, (coord_t)sy });
117 p = map.inverse_early_correction (p);
118 sq += p.dist_sq2_from (t);
119 }
120
121 if (sq < minsq)
122 {
123 minsq = sq;
124 best_tilt_x = param->tilt_x;
125 best_tilt_y = param->tilt_y;
126 //printf ("Tilts %f %f %f %i\n", best_tilt_x, best_tilt_y, minsq, i);
127 }
128 }
129 param->tilt_x = best_tilt_x;
130 param->tilt_y = best_tilt_y;
131 tilt_x_min = best_tilt_x - txstep;
132 tilt_x_max = best_tilt_x + txstep;
133 tilt_y_min = best_tilt_y - tystep;
134 tilt_y_max = best_tilt_y + tystep;
135 }
136 param->tilt_x = best_tilt_x;
137 param->tilt_y = best_tilt_y;
138 }
139 if (debug_output && final_run)
140 {
141 printf ("New Translation %f %f\n", param->center.x, param->center.y);
142 printf ("New coordinate1 %f %f\n", param->coordinate1.x,
143 param->coordinate1.y);
144 printf ("New coordinate2 %f %f\n", param->coordinate2.x,
145 param->coordinate2.y);
146 }
147 /* This tests that we found corect rotations and homography matrix determined
148 corresponds to what is in the scr_to_img parameters. */
149 if (final_run && ((debug || debug_output)))
150 {
151 scr_to_img map2;
152 if (!map2.set_parameters (*param, img_data))
153 return 1e30;
154 // map2.m_matrix.print (stdout);
155 bool found = false;
156 /* For vertical strips we need to compare screen coordinates only,
157 since solver points have no meaningful y coordinate. */
158 if (!(flags & homography::solve_vertical_strips))
159 for (auto point : points)
160 {
161 point_t img = point.img;
162 point_t scr = point.scr;
163 point_t t = map2.to_img (scr);
164 point_t p;
165 p = h.perspective_transform (scr);
166 p = map.inverse_early_correction (p);
167 if (!p.almost_eq (t, 1))
168 {
169 printf ("Solver model mismatch %f %f should be %f %f (ideally "
170 "%f %f)\n",
171 t.x, t.y, p.x, p.y, img.x, img.y);
172 found = true;
173 }
174 }
175 else
176 for (auto point : points)
177 {
178 point_t img = point.img;
179 point_t t = map2.to_scr (img);
180 point_t p = map.apply_early_correction (img);
181 p = h.inverse_perspective_transform (p);
182 if (!p.almost_eq (t, 0.03))
183 {
184 printf ("Solver model mismatch %f %f should be %f %f (ideally "
185 "first coord %f)\n",
186 t.x, t.y, p.x, p.y, point.scr.x);
187 found = true;
188 }
189 }
190 if (found)
191 {
192 h.print (stdout);
193 printf ("Tilts %f %f %f\n", best_tilt_x, best_tilt_y, minsq);
194 }
195 }
196 return (coord_t)chisq;
197}

199/* Pick N nearest points from POINTS to P. If SCREEN is true use screen coordinates,
200 otherwise use image coordinates. Store results in OUT. */
201
202void
203pick_nearest_points (std::vector<solver_parameters::solver_point_t> &out,
204 const std::vector<solver_parameters::solver_point_t> &points,
205 point_t p, int n, bool screen)
206{
207 if ((int)points.size () <= n)
208 {
209 out = points;
210 return;
211 }
212 struct entry
213 {
214 int index;
215 coord_t dist_sq;
216 bool operator< (const entry &other) const
217 {
218 return dist_sq < other.dist_sq;
219 }
220 };
221 std::vector<entry> heap;
222 heap.reserve (n);
223
224 for (int i = 0; i < (int)points.size (); i++)
225 {
226 coord_t d = screen ? points[i].scr.dist_sq2_from (p)
227 : points[i].img.dist_sq2_from (p);
228 if ((int)heap.size () < n)
229 {
230 heap.push_back ({ i, d });
231 if ((int)heap.size () == n)
232 std::make_heap (heap.begin (), heap.end ());
233 }
234 else if (d < heap.front ().dist_sq)
235 {
236 std::pop_heap (heap.begin (), heap.end ());
237 heap.back () = { i, d };
238 std::push_heap (heap.begin (), heap.end ());
239 }
240 }
241
242 out.clear ();
243 out.reserve (n);
244 for (const auto &e : heap)
245 out.push_back (points[e.index]);
246}

248/* Nonlinear optimizer for determining radial lens warp parameters. */
249class lens_solver
250{
251public:
252 lens_solver (scr_to_img_parameters &param, const image_data &img_data,
253 const solver_parameters &sparam, progress_info *progress)
254 : m_param (param), m_img_data (img_data), m_sparam (sparam),
255 m_progress (progress), m_start{ (coord_t)0.5, (coord_t)0.5, (coord_t)0, (coord_t)0, (coord_t)0 }
256 {
257 if (num_coordinates () == 1)
258 m_start[1] = 0;
259 }
260 scr_to_img_parameters &m_param;
261 const image_data &m_img_data;
262 const solver_parameters &m_sparam;
263 progress_info *m_progress;
264 static constexpr coord_t scale_kr = 128;
265
266 /* Return number of lens center coordinates. */
267 int
268 num_coordinates () const
269 {
270 return is_fixed_lens (m_param.scanner_type) ? 2 : 1;
271 }
272
273 /* Return total number of parameters to optimize. */
274 int
275 num_values () const
276 {
277 return num_coordinates () + 3;
278 }
279 std::array<coord_t, 5> m_start;
280 coord_t *start = m_start.data ();
281
282 /* Return epsilon for solver convergence. */
283 coord_t
284 epsilon () const
285 {
286 return (coord_t)0.00000001;
287 }
288
289 /* Return perturbation for finite differences derivatives. */
290 coord_t
291 derivative_perturbation () const
292 {
293 return (coord_t)0.0001;
294 }
295
296 /* Return initial scale for simplex solver. */
297 coord_t
298 scale () const
299 {
300 return (coord_t)0.3;
301 }
302
303 /* Return true if solver should be verbose. */
304 bool
305 verbose () const
306 {
307 return false;
308 }
309
310 /* Constrain parameters to valid range.
311 VALS are parameters to be constrained. */
312 void
313 constrain (coord_t *vals) const
314 {
315 int n = num_coordinates ();
316 /* Also consider the case that lens center is outside of the scan. */
317 if (vals[0] < (coord_t)-10)
318 vals[0] = (coord_t)-10;
319 if (vals[0] > (coord_t)10)
320 vals[0] = (coord_t)10;
321 if (n == 2)
322 {
323 if (vals[1] < (coord_t)-10)
324 vals[1] = (coord_t)-10;
325 if (vals[1] > (coord_t)10)
326 vals[1] = (coord_t)10;
327 }
328 /*
329 Coefficient Typical Range Extreme Range
330 k1 (3rd Order) ±0.001-±0.05 ±0.15
331 k2 (5th Order) ±0.0001-±0.01 ±0.05
332 k3 (7th Order) ≈0 (often unused) ±0.01 */
333 constexpr coord_t range [3] = {0.15, 0.05, 0.01};
334 for (int i = n, j = 0; i < n + 3; i++, j++)
335 {
336 if (vals[i] < (coord_t)-range[j] * scale_kr)
337 vals[i] = (coord_t)-range[j] * scale_kr;
338 if (vals[i] > (coord_t)range[j] * scale_kr)
339 vals[i] = (coord_t)range[j] * scale_kr;
340 }
341 }
342
343 /* Solve for lens parameters.
344 VALS are current lens parameters.
345 CHISQ is updated with resulting chi square.
346 TRANSFORMED is updated with transformed points. */
347 bool
348 solve (const coord_t *vals, coord_t *chisq, std::vector <point_t> *transformed) const
349 {
350 static constexpr coord_t bad_val = 100000000;
351 m_param.center = { (coord_t)0, (coord_t)0 };
352 m_param.coordinate1 = { (coord_t)1, (coord_t)0 };
353 m_param.coordinate2 = { (coord_t)0, (coord_t)1 };
354 int n = num_coordinates ();
355 if (is_fixed_lens (m_param.scanner_type))
356 m_param.lens_correction.center = { vals[0], vals[1] };
357 else if (m_param.scanner_type == lens_move_horizontally)
358 m_param.lens_correction.center = { 0, vals[0] };
359 else if (m_param.scanner_type == lens_move_vertically)
360 m_param.lens_correction.center = { vals[0], 0 };
361 else
362 abort ();
363 m_param.lens_correction.kr[0] = 1;
364 m_param.lens_correction.kr[1] = vals[n] * (1 / scale_kr);
365 m_param.lens_correction.kr[2] = vals[n + 1] * (1 / scale_kr);
366 m_param.lens_correction.kr[3] = vals[n + 2] * (1 / scale_kr);
367 if (!m_param.lens_correction.is_monotone ())
368 {
369 if (colorscreen_checking)
370 printf ("Non monotone lens correction %f %f: %f %f %f %f\n",
371 m_param.lens_correction.center.x,
372 m_param.lens_correction.center.y,
373 m_param.lens_correction.kr[0], m_param.lens_correction.kr[1],
374 m_param.lens_correction.kr[2], m_param.lens_correction.kr[3]);
375 if (chisq)
376 *chisq = bad_val;
377 if (transformed)
378 for (size_t i = 0; i < m_sparam.points.size (); i++)
379 (*transformed)[i] = m_sparam.points[i].scr;
380 return false;
381 }
382 if (!m_param.lens_correction.normalize ())
383 {
384 if (chisq)
385 *chisq = bad_val;
386 if (transformed)
387 for (size_t i = 0; i < m_sparam.points.size (); i++)
388 (*transformed)[i] = m_sparam.points[i].scr;
389 return false;
390 }
391 scr_to_img map;
392 /* We may save some inversions if points are relatively few. */
393 bool ok;
394 if (m_sparam.n_points () * 2 < lens_warp_correction::size)
395 ok = map.set_parameters_for_early_correction (m_param, m_img_data.width, m_img_data.height);
396 else
397 ok = map.set_parameters (m_param, m_img_data);
398 if (!ok)
399 return false;
400 coord_t chi = -5;
401 /* Do not use ransac here, since it is not smooth and will confuse solver. */
402 homography::get_matrix (m_sparam.points,
403 (screen_with_vertical_strips_p (m_param.type) ? homography::solve_vertical_strips : 0)
404 | homography::solve_rotation, m_param.scanner_type, &map, m_sparam.center, &chi, transformed);
405 if (chisq)
406 {
407 if (!(chi >= 0 && chi < bad_val))
408 {
409 printf ("Bad chi %f\n", chi);
410 *chisq = bad_val;
411 }
412 else
413 *chisq = chi;
414 }
415 return (chi >= 0 && chi < bad_val);
416 }
417 /* Return number of observations. */
418 int
419 num_observations ()
420 {
421 return m_sparam.points.size () * (screen_with_vertical_strips_p (m_param.type) ? 1 : 2);
422 }
423
424 /* Compute residuals. VALS is current set of parameters, F_VEC is the
425 output vector of residuals. */
426 int
427 residuals(const coord_t *vals, coord_t *f_vec)
428 {
429 std::vector<point_t> transformed (m_sparam.points.size ());
430 coord_t chisq;
431 if (!solve (vals, &chisq, &transformed))
432 {
433 for (int i = 0; i < (int)num_observations (); i++)
434 f_vec[i] = 1000;
435 return GSL_EDOM;
436 }
437 if (!screen_with_vertical_strips_p (m_param.type))
438 for (size_t i = 0; i < m_sparam.points.size (); i++)
439 {
440 f_vec[2 * i] = transformed[i].x - m_sparam.points[i].scr.x;
441 f_vec[2 * i + 1] = transformed[i].y - m_sparam.points[i].scr.y;
442 //printf ("%f %f %f %f\n",transformed[i].x, transformed[i].y, m_sparam.points[i].scr.x, m_sparam.points[i].scr.y);
443 }
444 else
445 for (size_t i = 0; i < m_sparam.points.size (); i++)
446 {
447 f_vec[i] = transformed[i].x - m_sparam.points[i].scr.x;
448 }
449 //printf ("%g %g %g %g %g error sq %.10g\n",vals[0], vals[1], vals[2], vals[3], vals[4], chisq);
450 return GSL_SUCCESS;
451 }
452
453 /* Objective function for simplex solver. */
454 coord_t
455 objfunc (coord_t *vals)
456 {
457 coord_t chisq;
458 solve (vals, &chisq, nullptr);
459 return chisq;
460 }
461};
462}

481/* Determine geometry and lens warp. PARAM is updated with results.
482 IMG_DATA is the source image. SPARAM contains solver points.
483 PROGRESS is used for progress reporting. */
484
485coord_t
486solver (scr_to_img_parameters *param,const image_data &img_data,
487 const solver_parameters &sparam, progress_info *progress)
488{
489 /* 3 points may be enough for strips; we only solve homography on 1d. */
490 if ((int)sparam.n_points () < solver_parameters::min_points (param->type))
491 return 1e30;
492
493 param->mesh_trans = nullptr;
494
495 /* Require more points for strips; we only can verify 1d info. */
496 bool optimize_lens = sparam.optimize_lens && ((int)sparam.n_points () > solver_parameters::min_lens_points (param->type));
497 bool optimize_rotation = sparam.optimize_tilt && ((int)sparam.n_points () > solver_parameters::min_perspective_points (param->type));
498
499 if (optimize_lens)
500 {
501 lens_solver s (*param, img_data, sparam, progress);
502 bool use_early_multifit = false;
503 bool use_simplex = true;
504 bool use_gsl_simplex = false;
505 bool use_multifit = false;
506 if (use_early_multifit)
507 gsl_multifit<coord_t, lens_solver> (s, "optimizing lens correction (multifit)",
508 progress);
509 if (use_simplex)
510 simplex<coord_t, lens_solver> (s, "optimizing lens correction (simplex)",
511 progress);
512 if (use_gsl_simplex)
513 gsl_simplex<coord_t, lens_solver> (s, "optimizing lens correction (GSL simplex)",
514 progress);
515 if (use_multifit)
516 gsl_multifit<coord_t, lens_solver> (s, "optimizing lens correction (multifit)",
517 progress);
518 int n = s.num_coordinates ();
519 if (is_fixed_lens (param->scanner_type))
520 param->lens_correction.center = { s.start[0], s.start[1] };
521 else if (param->scanner_type == lens_move_horizontally)
522 param->lens_correction.center = { 0, s.start[0] };
523 else if (param->scanner_type == lens_move_vertically)
524 param->lens_correction.center = { s.start[0], 0 };
525 param->lens_correction.kr[0] = 1;
526 param->lens_correction.kr[1] = s.start[n] * (1 / lens_solver::scale_kr);
527 param->lens_correction.kr[2]
528 = s.start[n + 1] * (1 / lens_solver::scale_kr);
529 param->lens_correction.kr[3]
530 = s.start[n + 2] * (1 / lens_solver::scale_kr);
531 if (!param->lens_correction.normalize ())
532 return 1e30;
533 }
534 if (progress)
535 progress->set_task ("optimizing perspective correction", 1);
536 return solver (param, img_data, sparam.points.read (), sparam.center,
537
538 (sparam.weighted ? homography::solve_image_weights : 0)
539 | (optimize_rotation ? homography::solve_rotation : 0),
540 true);
541}

596/* Determine mesh transformation for PARAM. IMG_DATA is the source image.
597 SPARAM contains solver points. PROGRESS is used for progress reporting. */
598
599std::unique_ptr <mesh>
600solver_mesh (scr_to_img_parameters *param, image_data &img_data,
601 solver_parameters &sparam, progress_info *progress)
602{
603 if ((int)sparam.n_points () < solver_parameters::min_mesh_points (param->type))
604 return nullptr;
605 int step = 10;
606 if (param->mesh_trans)
607 abort ();
608 scr_to_img map;
609 if (!map.set_parameters (*param, img_data))
610 return nullptr;
611 int_image_area r1 (map.get_range (img_data.width, img_data.height));
612 int width = (r1.width + step - 1) / step, height = (r1.height + step - 1) / step;
613 if (progress)
614 progress->set_task ("computing mesh", width * height);
615 std::unique_ptr <mesh> mesh_trans = std::make_unique<mesh> (r1, step, step);
616#pragma omp parallel for default(none) schedule(dynamic) collapse(2) \
617 shared(progress, r1, step, width, height, sparam, img_data, \
618 mesh_trans, param)
619 for (int y = 0; y < height; y++)

725/* Determine mesh transformation for PARAM. IMG_DATA is the source image.
726 SPARAM contains solver points. PROGRESS is used for progress reporting. */
727
728std::unique_ptr <mesh>
729solver_mesh (const scr_to_img_parameters *param, const image_data &img_data,
730 const solver_parameters &sparam, progress_info *progress)
731{
732 if ((int)sparam.n_points () < solver_parameters::min_mesh_points (param->type))
733 return nullptr;
734 if (param->mesh_trans)
735 abort ();
736 scr_to_img map;
737 if (!map.set_parameters (*param, img_data))
738 return NULL;
739 int_image_area r1 = {0, 0, img_data.width, img_data.height};
740 const int steps = 200;
741 int step = std::max (r1.width / steps, r1.height / steps);
742 if (!step)
743 return NULL;
744 int width = (r1.width + step - 1) / step, height = (r1.height + step - 1) / step;
745 if (progress)
746 progress->set_task ("computing mesh", width * height);
747
748 /* Expand the range so inversion is not using out of range points. */
749 int_image_area r2 = {-2*step, -2*step, img_data.width + 4*step, img_data.height + 4*step};
750 std::unique_ptr <mesh> mesh_trans = std::make_unique<mesh> (r2, step, step);
751 width = mesh_trans->get_width ();
752 height = mesh_trans->get_height ();
753#pragma omp parallel for default(none) schedule(dynamic) collapse(2) \
754 shared(progress, r1, step, width, height, sparam, img_data, \
755 mesh_trans, param)
756 for (int y = 0; y < height; y++)
757 for (int x = 0; x < width; x++)
758 if (!progress || !progress->cancel_requested ())
759 {
760 compute_img_to_scr_mesh_point (sparam, param->scanner_type, mesh_trans.get (), {x, y});
761 if (progress)
762 progress->inc_progress ();
763 }
764 if (progress && progress->cancel_requested ())
765 return nullptr;
766 // mesh_trans->print (stdout);
767 if (progress)
768 progress->set_task ("inverting mesh", 1);
769 mesh_trans->precompute_inverse ();
770 return mesh_trans;

773/* Determine single mesh point coordinates for SMAP.
774 SPARAM is updated with nearby solver points.
775 LPARAM determines the scanner geometry. */
776
777static void
778compute_mesh_point (const screen_map &smap, solver_parameters &sparam,
779 const scr_to_img_parameters &lparam, const image_data &img_data,
780 mesh *mesh_trans, int x, int y)
781{
782 int_point_t e = { x, y };
783 point_t scrp = mesh_trans->get_screen_point (e);
784 smap.get_solver_points_nearby (scrp.x, scrp.y, 100, sparam);
785 trans_4d_matrix h = homography::get_matrix (
786 sparam.points.read (), homography::solve_screen_weights, lparam.scanner_type,
787
788 nullptr, scrp, nullptr);
789 point_t imgp;
790 imgp = h.perspective_transform (scrp);
791 /* We need to set weight assymetrically based on image distance. Problem is
792 that without knowing the image distance we can not set one, so iteratively
793 find right one. */
794 if (lparam.scanner_type != fixed_lens)
795 {
796 int i;
797 for (i = 0; i < 100; i++)
798 {
799 point_t last_imgp = imgp;
800 trans_4d_matrix h = homography::get_matrix (
801 sparam.points.read (), homography::solve_image_weights,
802
803 lparam.scanner_type, nullptr, imgp, nullptr);
804 imgp = h.perspective_transform (scrp);
805 if (last_imgp.almost_eq (imgp, 0.5))
806 break;
807 }
808 if (i == 100)
809 printf ("Osclation instability\n");
810 }
811 mesh_trans->set_point (e, imgp);
812}

1015/* Fit points to a line and return distance threshold for 90% of points. */
1016double
1017solver_parameters::fit_line (point_t &origin, point_t &dir) const
1018{
1019 if (points.empty ())
1020 return 0;
1021 const auto &pointsVec = points.read ();
1022 if (pointsVec.size () == 1)
1023 {
1024 origin = pointsVec[0].img;
1025 dir = { 1, 0 };
1026 return 0;
1027 }
1028
1029 double mean_x = 0, mean_y = 0;
1030 for (const auto &p : pointsVec)
1031 {
1032 mean_x += p.img.x;
1033 mean_y += p.img.y;
1034 }
1035 mean_x /= pointsVec.size ();
1036 mean_y /= pointsVec.size ();
1037 origin = { (coord_t)mean_x, (coord_t)mean_y };
1038
1039 double mxx = 0, myy = 0, mxy = 0;
1040 for (const auto &p : pointsVec)
1041 {
1042 double dx = p.img.x - mean_x;
1043 double dy = p.img.y - mean_y;
1044 mxx += dx * dx;
1045 myy += dy * dy;
1046 mxy += dx * dy;
1047 }
1048
1049 gsl_matrix *m = gsl_matrix_alloc (2, 2);
1050 gsl_matrix_set (m, 0, 0, mxx);
1051 gsl_matrix_set (m, 0, 1, mxy);
1052 gsl_matrix_set (m, 1, 0, mxy);
1053 gsl_matrix_set (m, 1, 1, myy);
1054
1055 gsl_vector *eval = gsl_vector_alloc (2);
1056 gsl_matrix *evec = gsl_matrix_alloc (2, 2);
1057 gsl_eigen_symmv_workspace *w = gsl_eigen_symmv_alloc (2);
1058 gsl_eigen_symmv (m, eval, evec, w);
1059 gsl_eigen_symmv_sort (eval, evec, GSL_EIGEN_SORT_VAL_DESC);
1060
1061 dir.x = gsl_matrix_get (evec, 0, 0);
1062 dir.y = gsl_matrix_get (evec, 1, 0);
1063
1064 point_t normal = { (coord_t)gsl_matrix_get (evec, 0, 1),
1065 (coord_t)gsl_matrix_get (evec, 1, 1) };
1066
1067 gsl_eigen_symmv_free (w);
1068 gsl_matrix_free (m);
1069 gsl_vector_free (eval);
1070 gsl_matrix_free (evec);
1071
1072 std::vector<double> dists;
1073 dists.reserve (pointsVec.size ());
1074 for (const auto &p : pointsVec)
1075 {
1076 double dx = p.img.x - mean_x;
1077 double dy = p.img.y - mean_y;
1078 dists.push_back (fabs (dx * normal.x + dy * normal.y));
1079 }
1080 std::sort (dists.begin (), dists.end ());
1081 return dists[(size_t)(dists.size () * 0.9)];
1082}
