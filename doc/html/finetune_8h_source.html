129 /* Determine gris WIDTH and HEIGHT for given CROP and PARAM. */
130 void
131 get_grid_dimensions (const int_image_area &crop, const scr_to_img_parameters &param, int *width, int *height) const
132 {
133 int grid_w = grid_width;
134 int grid_h = grid_height;
135 int scalex = 1, scaley = 1;
136 if (param.scanner_type == lens_move_horizontally
137 || param.scanner_type == fixed_lens_sensor_move_horizontally)
138 scalex *= 3;
139 else if (param.scanner_type == lens_move_vertically
140 || param.scanner_type == fixed_lens_sensor_move_vertically)
141 scaley *= 3;
142 if (!grid_w && !grid_h)
143 {
144 int n = 100;
145 /* Dufaycolor has flexible base that is prone to defomrations. */
146 if (param.type == Dufay)
147 n = 300;
148 if (crop.width > crop.height)
149 grid_w = n * scalex;
150 else
151 grid_h = n * scaley;
152 }
153 if (!grid_w)
154 grid_w = nearest_int (grid_h * (luminosity_t)crop.width / crop.height * scalex / scaley);
155 if (!grid_h)
156 grid_h = nearest_int (grid_w * (luminosity_t)crop.height / crop.width * scaley / scalex);
157 *width = grid_w;
158 *height = grid_h;
159 }
