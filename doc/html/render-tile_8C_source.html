147/* Sanitize rendering parameters RTPARAM according to screen-to-image mapping
148 parameters PARAM and image data IMG. This ensures that requested rendering
149 type is compatible with available data. */
150static void
151sanitize_render_parameters (render_type_parameters &rtparam,
152 scr_to_img_parameters &param, image_data &img)
153{
154 if (rtparam.color && !img.has_rgb ())
155 rtparam.color = false;
156
157 if (!img.has_rgb ()
158 && (render_type_properties[(int)rtparam.type].flags
159 & render_type_property::NEEDS_RGB))
160 rtparam.type = render_type_original;
161
162 if (param.type == Random
163 && (render_type_properties[(int)rtparam.type].flags
164 & render_type_property::NEEDS_SCR_TO_IMG))
165 rtparam.type = render_type_original;
166
167 if (rtparam.type == render_type_fast
168 || rtparam.type == render_type_interpolated_original
169 || rtparam.type == render_type_interpolated_profiled_original
170 || rtparam.type == render_type_interpolated)
171 rtparam.antialias = false;
172 if (rtparam.type == render_type_original && img.has_rgb ())
173 rtparam.color = true;
174 if (rtparam.type == render_type_image_layer)
175 rtparam.color = false;
176 if (rtparam.type == render_type_profiled_original)
177 rtparam.color = true;
178#if 0
179 if (rtparam.type == render_type_realistic)
180 rtparam.color = false;
181#endif
182}

184/* Render tile of image to screen buffer PIXELS. RTPARAM specifies rendering
185 type, PARAM is the screen-to-image mapping parameters, IMG is the image
186 data, RPARAM is the rendering parameters, PIXELBYTES is the bytes per pixel,
187 ROWSTRIDE is the row stride, WIDTH and HEIGHT are tile dimensions,
188 XOFFSET and YOFFSET are coordinates in the output image, STEP is the
189 sampling step, and PROGRESS is used for progress reporting. */
190bool
191render_to_scr::render_tile (render_type_parameters rtparam,
192 scr_to_img_parameters &param, image_data &img,
193 render_parameters &rparam, unsigned char *pixels,
194 int pixelbytes, int rowstride, int width,
195 int height, double xoffset, double yoffset,
196 double step, progress_info *progress)
197{
198 if (width <= 0 || height <= 0)
199 return true;
200
201 if (param.type == Random && rtparam.type != render_type_original
202 && rtparam.type != render_type_profiled_original)
203 rtparam.type = render_type_original;
204
205 /* Avoid rendering outside of image area. This saves some time and prevents
206 * ugly artifacts. */
207 if ((int)xoffset < 0)
208 {
209 int border = -xoffset;
210 if (border > width)
211 border = width;
212 for (int y = 0; y < height; y++)
213 for (int x = 0; x < border; x++)
214 putpixel (pixels, pixelbytes, rowstride, x, y, 128, 128, 128);
215 pixels += border * pixelbytes;
216 xoffset += border;
217 width -= border;
218 if (!width)
219 return true;
220 }
221 if ((int)yoffset < 0)
222 {
223 int border = -yoffset;
224 if (border > height)
225 border = height;
226 for (int y = 0; y < border; y++)
227 for (int x = 0; x < width; x++)
228 putpixel (pixels, pixelbytes, rowstride, x, y, 128, 128, 128);
229 pixels += border * rowstride;
230 yoffset += border;
231 height -= border;
232 if (!height)
233 return true;
234 }
235 if ((int)((xoffset + width) - img.width / step) > 0)
236 {
237 int border = (int)((xoffset + width) - img.width / step);
238 if (border > width)
239 border = width;
240 for (int y = 0; y < height; y++)
241 for (int x = width - border; x < width; x++)
242 putpixel (pixels, pixelbytes, rowstride, x, y, 128, 128, 128);
243 width -= border;
244 if (!width)
245 return true;
246 }
247 if ((int)((yoffset + height) - img.height / step) > 0)
248 {
249 int border = (int)((yoffset + height) - img.height / step);
250 if (border > height)
251 border = height;
252 for (int y = height - border; y < height; y++)
253 for (int x = 0; x < width; x++)
254 putpixel (pixels, pixelbytes, rowstride, x, y, 128, 128, 128);
255 height -= border;
256 if (!height)
257 return true;
258 }
259
260 /* Do not render out of scan area; it is slow. */
261 if (stats == -1)
262 stats = getenv ("CSSTATS") != NULL;
263 struct timeval start_time;
264 if (stats)
265 gettimeofday (&start_time, NULL);
266 bool ok = true;
267 const bool lock_p = false;
268 if (lock_p)
269 global_rendering_lock.lock ();
270
271 sanitize_render_parameters (rtparam, param, img);
272
273 render_parameters my_rparam;
274 my_rparam.adjust_for (rtparam, rparam);
275
276 if (progress)
277 progress->set_task ("precomputing", 1);
278 switch (rtparam.type)
279 {
280 case render_type_original:
281 case render_type_profiled_original:
282 case render_type_image_layer:
283 ok = do_render_tile_with_gray<render_img> (
284 rtparam, param, img, my_rparam, pixels, pixelbytes, rowstride, width,
285 height, xoffset, yoffset, step, progress);
286 break;
287 case render_type_preview_grid:
288 case render_type_realistic:
289 ok = do_render_tile<render_superpose_img> (
290 rtparam, param, img, my_rparam, pixels, pixelbytes, rowstride, width,
291 height, xoffset, yoffset, step, progress);
292 break;
293 case render_type_screen:
294 my_rparam.brightness = 1;
295 ok = do_render_tile<render_screen> (
296 rtparam, param, img, my_rparam, pixels, pixelbytes, rowstride, width,
297 height, xoffset, yoffset, step, progress);
298 break;
299 case render_type_simulate_process:
300 ok = do_render_tile<render_simulate_process> (
301 rtparam, param, img, my_rparam, pixels, pixelbytes, rowstride, width,
302 height, xoffset, yoffset, step, progress);
303 break;
304 case render_type_interpolated_original:
305 case render_type_interpolated_profiled_original:
306 case render_type_interpolated:
307 case render_type_combined:
308 case render_type_predictive:
309 ok = do_render_tile<render_interpolate> (
310 rtparam, param, img, my_rparam, pixels, pixelbytes, rowstride, width,
311 height, xoffset, yoffset, step, progress);
312 break;
313 case render_type_interpolated_diff:
314 ok = do_render_tile<render_diff> (rtparam, param, img, my_rparam, pixels,
315 pixelbytes, rowstride, width, height,
316 xoffset, yoffset, step, progress);
317 break;
318 case render_type_extra:
319#ifdef RENDER_EXTRA
320 ok = do_render_tile<render_extra> (
321 rtparam, param, img, my_rparam, pixels, pixelbytes, rowstride, width,
322 height, xoffset, yoffset, step, progress);
323 break;
324#endif
325 case render_type_fast:
326 ok = do_render_tile<render_fast> (rtparam, param, img, my_rparam, pixels,
327 pixelbytes, rowstride, width, height,
328 xoffset, yoffset, step, progress);
329 break;
330 default:
331 abort ();
332 }
333 if (stats)
334 {
335 struct timeval end_time;
336 static struct timeval prev_time;
337 static bool prev_time_set = true;
338
339 gettimeofday (&end_time, NULL);
340 double time = end_time.tv_sec + end_time.tv_usec / 1000000.0
341 - start_time.tv_sec - start_time.tv_usec / 1000000.0;
342 printf ("\nRender type:%i resolution:%ix%i time:%.3fs fps:%.3f",
343 rtparam.type, width, height, time, 1 / time);
344 if (prev_time_set)
345 {
346 double time2 = end_time.tv_sec + end_time.tv_usec / 1000000.0
347 - prev_time.tv_sec - prev_time.tv_usec / 1000000.0;
348 printf (" last run %.3f s ago overall FPS:%.3f\n", time2,
349 1 / (time2 + time));
350 }
351 else
352 printf ("\n");
353 prev_time = end_time;
354 prev_time_set = true;
355 fflush (stdout);
356 }
357 if (lock_p)
358 global_rendering_lock.unlock ();
359 return (!progress || !progress->cancelled ()) && ok;
360}

362/* Render whole image to a file. RFPARAMS specifies file output parameters,
363 RTPARAM specifies rendering type, PARAM is the screen-to-image mapping
364 parameters, RPARAM is the rendering parameters, IMG is the image data,
365 BLACK specifies if black border should be rendered, and PROGRESS is used
366 for progress reporting. */
367const char *
368render_to_scr::render_to_file (render_to_file_params &rfparams,
369 render_type_parameters rtparam,
370 scr_to_img_parameters param,
371 render_parameters rparam, image_data &img,
372 int black, progress_info *progress)
373{
374 switch (rtparam.type)
375 {
376 case render_type_original:
377 case render_type_profiled_original:
378 case render_type_image_layer:
379 return produce_file<render_img, supports_img> (
380 rfparams, rtparam, param, param, rparam, img, black, progress);
381 break;
382 case render_type_preview_grid:
383 case render_type_realistic:
384 return produce_file<render_superpose_img, supports_img> (
385 rfparams, rtparam, param, param, rparam, img, black, progress);
386 break;
387 case render_type_screen:
388 rparam.brightness = 1;
389 return produce_file<render_screen, supports_scr> (
390 rfparams, rtparam, param, param, rparam, img, black, progress);
391 break;
392 case render_type_simulate_process:
393 return produce_file<render_simulate_process, supports_img> (
394 rfparams, rtparam, param, param, rparam, img, black, progress);
395 break;
396 case render_type_interpolated_original:
397 case render_type_interpolated_profiled_original:
398 case render_type_interpolated:
399 case render_type_combined:
400 case render_type_predictive:
401 return produce_file<render_interpolate, supports_final> (
402 rfparams, rtparam, param, param, rparam, img, black, progress);
403 break;
404 case render_type_interpolated_diff:
405 return produce_file<render_diff, supports_scr> (
406 rfparams, rtparam, param, param, rparam, img, black, progress);
407 break;
408 case render_type_extra:
409#ifdef RENDER_EXTRA
410 return produce_file<render_extra, supports_final> (
411 rfparams, rtparam, param, param, rparam, img, black, progress);
412 break;
413#endif
414 case render_type_fast:
415 return produce_file<render_fast, supports_scr> (
416 rfparams, rtparam, param, param, rparam, img, black, progress);
417 break;
418 default:
419 abort ();
420 }
421}

423/* Global entry point for rendering a tile. SCAN is the scanned image data,
424 PARAM is the screen-to-image mapping parameters, DPARAM is the screen
425 detection parameters, RPARAM is the rendering parameters, RTPARAM specifies
426 rendering type, TILE is the tile parameters, and PROGRESS is used for
427 progress reporting. */
428DLL_PUBLIC bool
429render_tile (image_data &scan, scr_to_img_parameters &param,
430 scr_detect_parameters &dparam, render_parameters &rparam,
431 render_type_parameters &rtparam, tile_parameters &tile,
432 progress_info *progress)
433{
434 if ((int)rtparam.type < (int)render_type_first_scr_detect
435 && rtparam.type != render_type_interpolated_diff)
436 return render_to_scr::render_tile (
437 rtparam, param, scan, rparam, tile.pixels, tile.pixelbytes,
438 tile.rowstride, tile.width, tile.height, tile.pos.x, tile.pos.y,
439 tile.step, progress);
440 else
441 return render_scr_detect::render_tile (
442 rtparam, dparam, scan, rparam, tile.pixels, tile.pixelbytes,
443 tile.rowstride, tile.width, tile.height, tile.pos.x, tile.pos.y,
444 tile.step, progress);
445}
