18inline uint16_t
19float_to_half (float f)
20{
21 union
22 {
23 float f;
24 uint32_t i;
25 } tmp;
26
27 tmp.f = f;
28 const int32_t lsign = (tmp.i >> 16) & 0x00008000;
29 int32_t exponent = ((tmp.i >> 23) & 0x000000ff) - (127 - 15);
30 int32_t mantissa = tmp.i & 0x007fffff;
31 if (exponent <= 0)
32 {
33 if (exponent < -10)
34 return (uint16_t)lsign;
35 mantissa = (mantissa | 0x00800000) >> (1 - exponent);
36 if (mantissa & 0x00001000)
37 mantissa += 0x00002000;
38 return (uint16_t)(lsign | (mantissa >> 13));
39 }
40 else if (exponent == 0xff - (127 - 15))
41 {
42 if (mantissa == 0)
43 return (uint16_t)(lsign | 0x7c00);
44 else
45 return (uint16_t)(lsign | 0x7c00 | (mantissa >> 13));
46 }
47 if (mantissa & 0x00001000)
48 {
49 mantissa += 0x00002000;
50 if (mantissa & 0x00800000)
51 {
52 mantissa = 0; // overflow in significand,
53 exponent += 1; // adjust exponent
54 }
55 }
56 if (exponent > 30)
57 return (uint16_t)(lsign | 0x7c00); // infinity with the same sign as f.
58 return (uint16_t)(lsign | (exponent << 10) | (mantissa >> 13));
59}
