52 /* Verify that the correction is monotone on a interval [0, 1].
53 Monotonicity is required for the inverse mapping to be well-defined. */
54 pure_attr bool
55 is_monotone () const
56 {
57 /* The source radius is r_src = r * (k0 + k1*r^2 + k2*r^4 + k3*r^6).
58 For monotonicity we need d(r_src)/dr > 0 for r in [0, 1].
59 d(r_src)/dr = k0 + 3*k1*r^2 + 5*k2*r^4 + 7*k3*r^6.
60 Let x = r^2. We need f(x) = k0 + 3*k1*x + 5*k2*x^2 + 7*k3*x^3 > 0
61 for x in [0, 1]. */
62 auto f = [this] (coord_t x) {
63 return kr[0] + x * (3 * kr[1] + x * (5 * kr[2] + x * 7 * kr[3]));
64 };
65
66 /* Check boundaries. */
67 if (f (0) <= 0 || f (1) <= 0)
68 return false;
69
70 /* Check critical points of f(x) where f'(x) = 0.
71 f'(x) = 3*k1 + 10*k2*x + 21*k3*x^2. */
72 coord_t a = 21 * kr[3];
73 coord_t b = 10 * kr[2];
74 coord_t c = 3 * kr[1];
75
76 if (my_fabs (a) < 1e-12)
77 {
78 /* Linear or constant derivative. */
79 if (my_fabs (b) > 1e-12)
80 {
81 coord_t x = -c / b;
82 if (x > 0 && x < 1 && f (x) <= 0)
83 return false;
84 }
85 }
86 else
87 {
88 /* Quadratic derivative. */
89 coord_t disc = b * b - 4 * a * c;
90 if (disc >= 0)
91 {
92 coord_t sq = my_sqrt (disc);
93 coord_t x1 = (-b + sq) / (2 * a);
94 coord_t x2 = (-b - sq) / (2 * a);
95 if (x1 > 0 && x1 < 1 && f (x1) <= 0)
96 return false;
97 if (x2 > 0 && x2 < 1 && f (x2) <= 0)
98 return false;
99 }
100 }

102 }
