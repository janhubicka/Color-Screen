273static inline double
274hd_linear_to_axis_x (double linear, hd_axis_type axis, luminosity_t preflash, luminosity_t exposure)
275{
276 double val = (linear + (preflash / 100)) * exposure;
277 if (axis == hd_axis_hd)
278 return std::log10 (std::max (0.0000001, val));
279 if (axis == hd_axis_gamma22)
280 return std::pow (std::max (0.0000001, val), 1.0 / 2.2);
281 return val;
282}

284static inline double
285hd_log_exposure_to_axis_x (double logE, hd_axis_type axis)
286{
287 if (axis == hd_axis_hd)
288 return logE;
289 if (axis == hd_axis_gamma22)
290 return std::pow (std::pow (10.0, logE), 1.0 / 2.2);
291 return std::pow (10.0, logE);
292}

294static inline double
295hd_axis_x_to_log_exposure (double axisX, hd_axis_type axis)
296{
297 if (axis == hd_axis_hd)
298 return axisX;
299 if (axis == hd_axis_gamma22)
300 return std::log10 (std::max (0.0000001, std::pow (axisX, 2.2)));
301 return std::log10 (std::max (0.0000001, axisX));
302}

304static inline double
305hd_density_to_axis_y (double density, double boost, hd_axis_type axis)
306{
307 if (axis == hd_axis_hd)
308 return density;
309 double transmittance = std::pow (10.0, -density * boost);
310 if (axis == hd_axis_gamma10)
311 return transmittance;
312 return std::pow (transmittance, 1.0 / 2.2);
313}
314
315static inline double
316hd_axis_y_to_density (double axisY, double boost, hd_axis_type axis)
317{
318 if (axis == hd_axis_hd)
319 return axisY;
320 double transmittance;
321 if (axis == hd_axis_gamma10)
322 transmittance = axisY;
323 else
324 transmittance = std::pow (std::max (0.0000001, axisY), 2.2);
325 return -std::log10 (std::max (0.0000001, transmittance)) / std::max (0.01, boost);
326}
