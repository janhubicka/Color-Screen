118/* This implements to translate image coordinates to coordinates of the viewing
119 screen. In the viewing screen the coordinates (0,0) describe a green dot and
120 the screen is periodic with period 1: that is all integer coordinates
121 describes green dots again.
122
123 In order to turn scan coordinates to screen the following transformations
124 are performed
125 1) translation to move lens_center to (0,0)
126 2) lens correction
127 3) perspective correction (with tilt applied)
128 4) translation to move center to (0,0)
129 5) change of basis so center+coordinate1 becomes (1,0) and
130 center+coordinate2 becomes (0,1)
131*/
132
133struct scr_to_img_parameters
134{
135 /* Coordinates (in the image) of the center of the screen (a green dot). */
136 point_t center = { (coord_t)0, (coord_t)0 };
137 /* First coordinate vector:
138 image's (center.x+coordinate1.x, center.y+coordinate1.y) should describe
139 a green dot just on the right side of (center_x, center_y). */
140 point_t coordinate1 = { (coord_t)1, (coord_t)0 };
141 /* Second coordinate vector:
142 image's (center.x+coordinate2.x, center.y+coordinate2.y) should describe
143 a green dot just below (center.x, center.y). */
144 point_t coordinate2 = { (coord_t)0, (coord_t)1 };
145
146 /* Distance of the perspective plane from the camera coordinate. */
147 coord_t projection_distance = 1;
148 /* Perspective tilt in x and y coordinate in degrees. */
149 coord_t tilt_x = 0, tilt_y = 0;
150
151 /* Rotation from screen coordinates to final coordinates. */
152 coord_t final_rotation = 0;
153 /* Angle of the screen X and Y axis in the final coordinates. */
154 coord_t final_angle = 90;
155 /* Ratio of the X and Y axis in the final coordinates. */
156 coord_t final_ratio = 1;
157
158 std::shared_ptr<mesh> mesh_trans = nullptr;
159 bool mesh_trans_is_scr_to_img = false;
160
161 enum scr_type type = Random;
162 enum scanner_type scanner_type = fixed_lens;
163
164 lens_warp_correction_parameters lens_correction;
165
166 bool
167 operator== (const scr_to_img_parameters &other) const
168 {
169 return center == other.center && coordinate1 == other.coordinate1
170 && coordinate2 == other.coordinate2
171 && projection_distance == other.projection_distance
172 && final_rotation == other.final_rotation
173 && final_angle == other.final_angle
174 && final_ratio == other.final_ratio && tilt_x == other.tilt_x
175 && tilt_y == other.tilt_y && type == other.type
176 && scanner_type == other.scanner_type
177 && lens_correction == other.lens_correction
178 && mesh_trans == other.mesh_trans
179 && mesh_trans_is_scr_to_img == other.mesh_trans_is_scr_to_img;
180 }
181 /* Merge solution from solver into the parameters.
182 OTHER is the solution from the solver. */
183 void
184 merge_solver_solution (const scr_to_img_parameters &other)
185 {
186 center = other.center;
187 coordinate1 = other.coordinate1;
188 coordinate2 = other.coordinate2;
189 tilt_x = other.tilt_x;
190 tilt_y = other.tilt_y;
191 lens_correction = other.lens_correction;
192 projection_distance = other.projection_distance;
193 mesh_trans = other.mesh_trans;
194 mesh_trans_is_scr_to_img = other.mesh_trans_is_scr_to_img;
195 }
196
197 /* Return true if OTHER is not equal to this. */
198 bool
199 operator!= (const scr_to_img_parameters &other) const
200 {
201 return !(*this == other);
202 }
203
204 /* Return estimated screen per inch for the current screen type. */
205 coord_t
206 screen_per_inch () const
207 {
208 if (type == Dufay)
209 /* Dufaycolor manual promises 500 lines per inch. Screen has 2 lines */
210 return 250;
211 else
212 /* 2 squares per screen. */
213 return 25.4 / 2;
214 }
215
216 /* Return the length of the first coordinate vector. */
217 pure_attr coord_t
218 get_x_len () const
219 {
220 return coordinate1.length ();
221 }
222
223 /* Return the length of the second coordinate vector. */
224 pure_attr coord_t
225 get_y_len () const
226 {
227 return coordinate2.length ();
228 }
229
230 /* Return the angle between the two coordinate vectors in degrees. */
231 pure_attr coord_t
232 get_angle () const
233 {
234 coord_t dot
235 = coordinate1.x * coordinate2.x + coordinate1.y * coordinate2.y;
236 // coord_t det = coordinate1.x*coordinate2.y - coordinate1.y*coordinate2.x;
237 // return atan2(det, dot) * 180 / M_PI;
238 return std::acos (dot / (get_x_len () * get_y_len ())) * (180 / M_PI);
239 }
