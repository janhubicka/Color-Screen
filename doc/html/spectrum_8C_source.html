29void
30compute_spectrum (spectrum s, luminosity_t start, luminosity_t end, int size,
31 const luminosity_t data[], bool absorbance, luminosity_t norm, bool limit_range, bool clamp)
32{
33 luminosity_t step = (end - start) / (luminosity_t) (size - 1);
34 luminosity_t repnorm = 1 / norm;
35 //fprintf (stderr, "start %f %f %f %i\n",start,end,step,size);
36 //printf ("start %f end %f step %f size %i\n",start,end,step,size);
37 for (int i = 0; i < SPECTRUM_SIZE; i++)
38 {
39 int p = SPECTRUM_START + i * SPECTRUM_STEP;
40 float sp = (p - start) / (float) step;
41 int ri;
42 float rp = my_modf (sp, &ri);
43
44
45 /* If out of range continue interpolating linearly.
46 * ??? maybe this gets to too extreme values towards the end. */
47 if (ri < 1)
48 {
49 rp += ri - 1;
50 ri = 1;
51 s[i] = data[1] + rp * (data[1] - data[0]);
52 /* Just clamp when data are missing. */
53 if (clamp && p < start)
54 {
55 s[i] = CLAMP;
56 continue;
57 }
58 }
59 else if (ri >= size - 2)
60 {
61 rp += ri - size + 2;
62 ri = size - 2;
63 s[i] = data[size - 2] + rp * (data[size - 1] - data[size - 2]);
64 if (clamp && p > end)
65 {
66 s[i] = CLAMP;
67 continue;
68 }
69 }
70 else
71 s[i] =
72 cubic_interpolate (data[ri - 1], data[ri], data[ri + 1],
73 data[ri + 2], rp);
74 if (absorbance)
75 s[i] = absorbance_to_transmitance (s[i] * repnorm);
76 else
77 s[i] *= repnorm;
78 if (limit_range && s[i] < 0)
79 s[i] = 0;
80 if (limit_range && s[i] > 1)
81 s[i] = 1;
