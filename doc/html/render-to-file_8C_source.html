38bool
39complete_rendered_file_parameters (render_type_parameters *rtparams, scr_to_img_parameters * param, image_data *scan, stitch_project *stitch, render_to_file_params *p)
40{
41 render_parameters rparam;
42 if (scan && scan->stitch)
43 {
44 stitch = scan->stitch;
45 scan = nullptr;
46 }
47 const render_type_property &prop = render_type_properties [(int)rtparams->type];
48
49 if (p->geometry == render_to_file_params::default_geometry)
50 {
51 if (prop.flags & render_type_property::NEEDS_SCR_TO_IMG)
52 p->geometry = render_to_file_params::screen_geometry;
53 }
54 /* Stitching always needs geometry correction. */
55 if (stitch)
56 p->geometry = render_to_file_params::screen_geometry;
57
58 /* Do we render using scr_to_img map? */
59 if (p->geometry == render_to_file_params::screen_geometry)
60 {
61 coord_t render_width, render_height;
62 if (!stitch)
63 {
64 render_img render (*param, *scan, rparam, 65535);
65 render.compute_final_range ();
66 render_width = render.get_final_width ();
67 render_height = render.get_final_height ();
68 if (!p->pixel_size)
69 {
70 p->pixel_size = render.pixel_size ();
71 if (prop.flags & render_type_property::PATCH_RESOLUTION)
72 p->pixel_size = param->type == Dufay ? 0.5 : 1.0/3;
73 else if (prop.flags & render_type_property::SCREEN_RESOLUTION)
74 p->pixel_size = 1;
75 }
76 }
77 else
78 {
79 int xmin, xmax, ymin, ymax;
80 stitch->determine_viewport (xmin, xmax, ymin, ymax);
81 int_image_area viewport (xmin, ymin, xmax - xmin, ymax - ymin);
82 render_width = viewport.width;
83 render_height = viewport.height;
84 p->xpos = viewport.x;
85 p->ypos = viewport.y;
86 if (!p->pixel_size)
87 {
88 p->pixel_size = stitch->pixel_size;
89 if (prop.flags & render_type_property::PATCH_RESOLUTION)
90 p->pixel_size = stitch->param.type == Dufay ? 0.5 : 1.0/3;
91 else if (prop.flags & render_type_property::SCREEN_RESOLUTION)
92 p->pixel_size = 1;
93 }
94 }
95 if (!p->xstep)
96 {
97 if (!p->screen_scale)
98 {
99 p->xstep = p->pixel_size / p->scale;
100 p->ystep = p->pixel_size / p->scale;
101 }
102 else
103 {
104 p->xstep = 1 / p->screen_scale;
105 p->ystep = 1 / p->screen_scale;
106 }
107 }
108 if (!p->antialias)
109 {
110 if (prop.flags & render_type_property::ANTIALIAS)
111 p->antialias = round (4 * p->xstep / p->pixel_size);
112 else
113 p->antialias = round (1 * p->xstep / p->pixel_size);
114 /* Antialias should be odd, so middle of tile is sampled. */
115 if (!(p->antialias & 1))
116 p->antialias++;
117 }
118 if (!p->width && !p->height)
119 {
120 p->width = render_width / p->xstep;
121 p->height = render_height / p->ystep;
122 }
123 else if (!p->width && p->height)
124 {
125 /* height specified: derive step and width to preserve aspect ratio. */
126 double new_step = render_height / (double)p->height;
127 p->xstep = p->ystep = new_step;
128 p->width = (int)(render_width / new_step + 0.5);
129 }
130 else if (p->width && !p->height)
131 {
132 /* width specified: derive step and height to preserve aspect ratio. */
133 double new_step = render_width / (double)p->width;
134 p->xstep = p->ystep = new_step;
135 p->height = (int)(render_height / new_step + 0.5);
136 }
137 /* TODO: Make this work with stitched projects. */
138 if ((!p->xdpi || !p->ydpi) && scan && scan->xdpi && scan->ydpi)
139 {
140 scr_to_img map;
141 if (!map.set_parameters (*param, *scan))
142 return false;
143 point_t c = map.img_to_final ({(coord_t)(scan->width / 2), (coord_t)(scan->height / 2)});
144 point_t z = map.final_to_img (c);
145 point_t imgp = map.final_to_img ({c.x + p->xstep, c.y});
146 imgp.x = (imgp.x - z.x) / scan->xdpi;
147 imgp.y = (imgp.y - z.y) / scan->ydpi;
148 coord_t len = my_sqrt (imgp.x * imgp.x + imgp.y * imgp.y);
149 /* This is approximate for deformed screens, so take average of X and Y resolution. */
150 if (len && !p->xdpi)
151 {
152 coord_t xdpi = 1 / len;
153
154 imgp = map.final_to_img ({c.x, c.y + p->ystep});
155 imgp.x = (imgp.x - z.x) / scan->xdpi;
156 imgp.y = (imgp.y - z.y) / scan->ydpi;
157 coord_t len2 = my_sqrt (imgp.x * imgp.x + imgp.y * imgp.y);
158 if (len2 && !p->ydpi)
159 {
160 p->xdpi = p->ydpi = (xdpi + 1 / len2) / 2;
161 }
162 }
163 }
164 }
165 else
166 {
167 if (!p->xstep)
168 p->xstep = p->ystep = 1 / p->scale;
169 if (!p->width && !p->height)
170 {
171 p->width = scan->width / p->xstep;
172 p->height = scan->height / p->ystep;
173 }
174 else if (!p->width && p->height)
175 {
176 double new_step = (double)scan->height / p->height;
177 p->xstep = p->ystep = new_step;
178 p->width = (int)(scan->width / new_step + 0.5);
179 }
180 else if (p->width && !p->height)
181 {
182 double new_step = (double)scan->width / p->width;
183 p->xstep = p->ystep = new_step;
184 p->height = (int)(scan->height / new_step + 0.5);
185 }
186 if (!p->antialias)
187 {
188 if (prop.flags & render_type_property::ANTIALIAS)
189 p->antialias = round (4 * p->xstep);
190 else
191 p->antialias = round (1 * p->ystep);
192 /* Antialias should be odd, so middle of tile is sampled. */
193 if (!(p->antialias & 1))
194 p->antialias++;
195 }
196 if (!p->xdpi)
197 p->xdpi = scan->xdpi / p->xstep;
198 if (!p->ydpi)
199 p->ydpi = scan->ydpi / p->ystep;
200 }
201 return true;
202}

221bool
222render_to_file (image_data & scan, scr_to_img_parameters & param,
223 scr_detect_parameters & dparam, render_parameters &in_rparam,
224 struct render_to_file_params rfparams /* modified */,
225 render_type_parameters &rtparam,
226 progress_info * progress, const char **error)
227{
228 bool free_profile = false;
229 int black = 0;
230 if (scan.stitch)
231 return scan.stitch->write_tiles (in_rparam, &rfparams, rtparam, 1, progress, error);
232 render_parameters rparam;
233 rparam.adjust_for (rtparam, in_rparam);
234 if (rfparams.dng)
235 {
236 rparam.output_gamma = 1;
237 black = rparam.dark_point * 65536;
238 rparam.dark_point = 0;
239 rparam.scan_exposure = rparam.brightness = 1;
240 rparam.white_balance.red = rparam.white_balance.green = rparam.white_balance.blue = 1;
241 }
242 if (rfparams.verbose)
243 {
244 if (progress)
245 progress->pause_stdout ();
246 printf ("Precomputing\n");
247 fflush (stdout);
248 if (progress)
249 progress->resume_stdout ();
250 }
251 void *icc_profile = nullptr;
252 size_t icc_profile_len = 0;
253
254 const render_type_property &prop = render_type_properties [(int)rtparam.type];
255 if ((prop.flags & render_type_property::NEEDS_RGB) && !scan.has_rgb ())
256 {
257 *error = "Selected rendering algorithm is impossible on monochromatic scan";
258 return false;
259 }
260 if (rfparams.hdr || rfparams.dng)
261 rparam.output_profile = render_parameters::output_profile_original;
262
263 if (rparam.output_profile == render_parameters::output_profile_original)
264 {
265 if (prop.flags & render_type_property::OUTPUTS_SCAN_PROFILE)
266 {
267 if (scan.icc_profile
268 && rparam.gamma == rparam.output_gamma)
269 {
270 rfparams.icc_profile = scan.icc_profile;
271 rfparams.icc_profile_len = scan.icc_profile_size;
272 }
273 else
274 {
275 xyz red = xyY_to_xyz (scan.primary_red.x, scan.primary_red.y, scan.primary_red.Y);
276 xyz green = xyY_to_xyz (scan.primary_green.x, scan.primary_green.y, scan.primary_green.Y);
277 xyz blue = xyY_to_xyz (scan.primary_blue.x, scan.primary_blue.y, scan.primary_blue.Y);
278 icc_profile_len = create_profile ("ColorScreen produced profile based on original scan", red, green, blue, red+green+blue, rparam.output_gamma, &icc_profile);
279 free_profile = true;
280 }
281 }
282 if (prop.flags & render_type_property::OUTPUTS_SRGB_PROFILE)
283 {
284 rfparams.icc_profile = sRGB_icc;
285 rfparams.icc_profile_len = sRGB_icc_len;
286 rparam.output_gamma = -1;
287 }
288 else if (!free_profile)
289 {
290 icc_profile_len = rparam.get_icc_profile (&icc_profile, &scan, false /*TODO*/);
291 free_profile = true;
292 }
293 }
294
295 /* Initialize rendering engine. */
296 if (!complete_rendered_file_parameters (rtparam, param, scan, &rfparams))
297 {
298 *error = "Precomputation failed (out of memory)";
299 if (free_profile)
300 free (icc_profile);
301 return false;
302 }
303 bool icc_profile_set = rfparams.icc_profile;
304 if (!icc_profile_set)
305 {
306 rfparams.icc_profile = icc_profile;
307 rfparams.icc_profile_len = icc_profile_len;
308 }
309 if (progress)
310 progress->set_task ("precomputing", 1);
311 if ((int)rtparam.type < (int)render_type_adjusted_color || rtparam.type >= (int)render_type_profiled_original)
312 render_to_scr::render_to_file (rfparams, rtparam, param, rparam, scan, black, progress);
313 else
314 render_scr_detect::render_to_file (rfparams, rtparam, param, dparam, rparam, scan, black, progress);
315 if (free_profile)
316 free (icc_profile);
317 return true;
318}
