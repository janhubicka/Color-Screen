21/* Lanczos resample LINE of length IN_LEN and with given SUPERSAMPLE factor
22 into OUTPUT of length OUT_LEN with OUT_STRIDE. KERNELS are precomputed
23 Lanczos kernels of size A. */
24
25template <typename T>
26inline __attribute__ ((always_inline))
27void
28resample_line (T *output,
29 const T *input, int out_len,
30 int out_stride,
31 int in_len,
32 int supersample,
33 const std::vector<T, fft_allocator<T>>& kernels,
34 int a = 3)
35{
36 /* Let compiler know that supersample is small integer. */
37 if (supersample <= 1 || supersample > 1024)
38 __builtin_unreachable ();
39 /* We do mirroring on the edges of image; special case first few iterations. */
40#pragma omp simd
41 for (int i = 0; i < std::min (a * supersample, out_len); ++i)
42 {
43 /* Map output pixel to input space center. */
44 int center = (2 * i + 1) - supersample;
45 int start = center / (2 * supersample) - a + 1;
46 int io = (center / 2) % supersample;
47
48 T sum = 0.0;
49 const T *k = kernels.data () + io * a * 2;
50
51#pragma omp simd
52 for (int jj = 0; jj < 2 * a; ++jj)
53 {
54 int j = start + jj;
55 /* Mirror padding for boundaries. */
56 int index = j;
57 if (index < 0)
58 index = -index;
59 if (index >= in_len)
60 index = 2 * in_len - index - 1;
61
62 sum += input[index] * k[jj];
63 }
64
65 output[i * out_stride] = sum;
66 }
67#pragma omp simd
68 for (int i = a * supersample; i < std::min (out_len - a * supersample, out_len); ++i)
69 {
70 /* Map output pixel to input space center. */
71 int center = (2 * i + 1) - supersample;
72 int start = center / (2 * supersample) - a + 1;
73 int io = (center / 2) % supersample;
74
75 T sum = 0.0;
76 const T *k = kernels.data () + io * a * 2;
77
78#pragma omp simd
79 for (int jj = 0; jj < 2 * a; ++jj)
80 {
81 int j = start + jj;
82 sum += input[j] * k[jj];
83 }
84
85 output[i * out_stride] = sum;
86 }
87#pragma omp simd
88 for (int i = std::max (out_len - a * supersample, a * supersample); i < out_len; ++i)
89 {
90 /* Map output pixel to input space center. */
91 int center = (2 * i + 1) - supersample;
92 int start = center / (2 * supersample) - a + 1;
93 int io = (center / 2) % supersample;
94
95 T sum = 0.0;
96 const T *k = kernels.data () + io * a * 2;
97
98#pragma omp simd
99 for (int jj = 0; jj < 2 * a; ++jj)
100 {
101 int j = start + jj;
102 /* Mirror padding for boundaries. */
103 int index = j;
104 if (index >= in_len)
105 index = 2 * in_len - index - 1;
106
107 sum += input[index] * k[jj];
108 }
109 output[i * out_stride] = sum;
110 }
111}
112}

114/* Set up deconvolution for given MTF and MTF_SCALE. SNR specifies signal to
115 noise ratio for Weiner filter. SIGMA specifies dampening parameter for
116 Richardson-Lucy. MAX_THREADS specifies number of threads. MODE is
117 deconvolution mode. ITERATIONS specifies number of iterations for
118 Richardson-Lucy. SUPERSAMPLE specifies supersampling factor. */
119template <typename T>
120deconvolution<T>::deconvolution (mtf *mtf, luminosity_t mtf_scale,
121 luminosity_t snr, luminosity_t sigma,
122 int max_threads, enum mode mode,
123 int iterations, int supersample)
124 : m_supersample (supersample), m_sigma ((T)sigma),
125 m_iterations (iterations)
126{
127 (void)mtf->precompute ();
128 T k_const = (T)1.0 / (T)snr;
129 m_border_size = mtf->psf_radius (mtf_scale);
130 if (m_supersample && m_border_size == 0)
131 m_border_size = 1;
132
133 if (taper_edges)
134 {
135 m_taper_size = m_border_size * m_supersample;
136 m_border_size *= 2;
137 }
138
139 while (m_enlarged_tile_size < m_border_size * 4 * m_supersample)
140 m_enlarged_tile_size *= 2;
141 m_tile_size = (m_enlarged_tile_size + m_supersample - 1) / m_supersample;
142
143 m_data.resize (max_threads);
144 /* Result of real fft is symmetric. We need only N / 2 + 1 complex values.
145 Moreover point spread functions we compute are symmetric real functions so
146 the FFT result is again a real function (all complex values should be 0 up
147 to roundoff errors). */
148 m_fft_size = m_enlarged_tile_size / 2 + 1;
149 m_blur_kernel = fft_alloc_complex<T> (m_enlarged_tile_size * m_fft_size);
150 T scale = (T)1.0 / (T)(m_enlarged_tile_size * m_enlarged_tile_size);
151 T rev_tile_size = (T)m_supersample * (T)mtf_scale / (T)m_enlarged_tile_size;
152#pragma omp parallel for shared(scale,mtf,rev_tile_size,mode,k_const) default(none) collapse(2)
153 for (int y = 0; y < m_fft_size; y++)
154 for (int x = 0; x < m_fft_size; x++)
155 {
156 std::complex<T> ker (
157 std::clamp (
158 (T)mtf->get_mtf (x, y, rev_tile_size),
159 (T)0, (T)1),
160 (T)0);
161
162 /* If sharpening, apply Wiener Filter.
163 Result = Img * conj(Ker) / (|Ker|^2 + 1/SNR). */
164 if (mode == sharpen)
165 ker = std::conj (ker) / (std::norm (ker) + k_const);
166 ker = ker * scale;
167 m_blur_kernel[y * m_fft_size + x][0] = ker.real ();
168 m_blur_kernel[y * m_fft_size + x][1] = ker.imag ();
169 if (y)
170 {
171 m_blur_kernel[(m_enlarged_tile_size - y) * m_fft_size + x][0]
172 = ker.real ();
173 m_blur_kernel[(m_enlarged_tile_size - y) * m_fft_size + x][1]
174 = ker.imag ();
175 }
176 }
177 T sc = scale / m_blur_kernel[0][0];
178 if (sc != (T)1)
179 for (int x = 0; x < m_fft_size * m_enlarged_tile_size; x++)
180 {
181 m_blur_kernel[x][0] *= sc;
182 m_blur_kernel[x][1] *= sc;
183 }
184 m_richardson_lucy = (mode == richardson_lucy_sharpen);
185 if (taper_edges)
186 {
187 m_weights.resize (m_taper_size);
188 for (int i = 0; i < m_taper_size; i++)
189 /* Cosine bell curve: 0.0 at edge, 1.0 at taper_width. */
190 m_weights[i] = (T)0.5 * ((T)1.0 - std::cos ((T)M_PI * (T)i / (T)m_taper_size));
191 }
192 if (m_supersample > 1)
193 {
194 m_lanczos_kernels.resize (lanczos_a * 2 * m_supersample);
195 for (int off = 0; off < m_supersample; off++)
196 {
197 T sum = 0;
198 for (int i = 0 ; i < lanczos_a * 2; i++)
199 sum += m_lanczos_kernels[off * lanczos_a * 2 + i]
200 = lanczos_kernel ((T)i - (T)lanczos_a + ((T)off + (T)0.5) / (T)m_supersample, lanczos_a);
201 for (int i = 0 ; i < lanczos_a * 2; i++)
202 m_lanczos_kernels[off * lanczos_a * 2 + i] *= ((T)1 / sum);
203 }
204 }
205}

207/* Allocate memory for tiles and initialize fftw plans for given THREAD_ID. */
208template <typename T>
209void
210deconvolution<T>::init (int thread_id)
211{
212 if (m_data[thread_id].initialized)
213 return;
214 m_data[thread_id].in = fft_alloc_complex<T> (m_enlarged_tile_size * m_fft_size);
215 m_data[thread_id].tile.resize (m_tile_size * m_tile_size);
216 if (m_supersample > 1)
217 {
218 m_data[thread_id].enlarged_tile_data.resize (m_enlarged_tile_size
219 * m_enlarged_tile_size);
220 m_data[thread_id].enlarged_tile = &m_data[thread_id].enlarged_tile_data;
221 }
222 else
223 m_data[thread_id].enlarged_tile = &m_data[thread_id].tile;
224 if (m_richardson_lucy)
225 {
226 m_data[thread_id].ratios.resize (m_enlarged_tile_size
227 * m_enlarged_tile_size);
228 m_data[thread_id].observed.resize (m_enlarged_tile_size
229 * m_enlarged_tile_size);
230 }
231 m_data[thread_id].initialized = true;
232 if (!m_plans_exists)
233 {
234 m_plan_2d_inv = fft_plan_c2r_2d<T> (m_enlarged_tile_size,
235 m_enlarged_tile_size,
236 m_data[thread_id].in.get (),
237 m_data[thread_id].enlarged_tile->data ());
238 m_plan_2d = fft_plan_r2c_2d<T> (m_enlarged_tile_size,
239 m_enlarged_tile_size,
240 m_data[thread_id].enlarged_tile->data (),
241 m_data[thread_id].in.get ());
242 m_plans_exists = true;
243 }

246/* Apply the deconvolution kernel for given THREAD_ID. */
247template <typename T>
248void
249deconvolution<T>::process_tile (int thread_id, progress_info *progress)
250{
251 if (progress && progress->cancelled ())
252 return;
253 if (m_supersample > 1)
254 {
255 for (int y = 0; y < m_tile_size; y++)
256 resample_line (m_data[thread_id].enlarged_tile->data ()
257 + y * m_enlarged_tile_size,
258 m_data[thread_id].tile.data () + y * m_tile_size,
259 m_enlarged_tile_size, 1, m_tile_size,
260 m_supersample, m_lanczos_kernels,
261 lanczos_a);
262 std::vector<T> line (m_tile_size);
263 for (int x = 0; x < m_enlarged_tile_size; x++)
264 {
265#pragma omp simd
266 for (int y = 0; y < m_tile_size; y++)
267 line[y] = get_enlarged_pixel (thread_id, x, y);
268 resample_line (m_data[thread_id].enlarged_tile->data () + x,
269 line.data (), m_enlarged_tile_size,
270 m_enlarged_tile_size, m_tile_size,
271 m_supersample, m_lanczos_kernels,
272 lanczos_a);
273 }
274 if (progress && progress->cancelled ())
275 return;
276 }
277 if (taper_edges)
278 {
279 double sum = 0;
280
281 /* Compute average pixel. */
282 for (int y = 0; y < m_taper_size; y++)
283 for (int x = 0; x < m_enlarged_tile_size; x++)
284 sum += get_enlarged_pixel (thread_id, x, y);
285 for (int y = 0; y < m_taper_size; y++)
286 for (int x = 0; x < m_enlarged_tile_size; x++)
287 sum += get_enlarged_pixel (thread_id, x,
288 y + m_enlarged_tile_size - m_taper_size);
289 for (int y = m_taper_size; y < m_enlarged_tile_size - m_taper_size; y++)
290 {
291 for (int x = 0; x < m_taper_size; x++)
292 sum += get_enlarged_pixel (thread_id, x, y);
293 for (int x = 0; x < m_taper_size; x++)
294 sum += get_enlarged_pixel (
295 thread_id, x + m_enlarged_tile_size - m_taper_size, y);
296 }
297 sum /= (double)(m_enlarged_tile_size * m_taper_size * 2
298 + (m_enlarged_tile_size - 2 * m_taper_size) * m_taper_size * 2);
299 /* Taper top edge. */
300 for (int y = 0; y < m_taper_size; y++)
301 {
302 T weight = m_weights[y];
303#pragma omp simd
304 for (int x = 0; x < y; x++)
305 put_enlarged_pixel (
306 thread_id, x, y,
307 sum
308 + (get_enlarged_pixel (thread_id, x, y) - sum)
309 * m_weights[x]);
310#pragma omp simd
311 for (int x = y; x < m_enlarged_tile_size - y; x++)
312 put_enlarged_pixel (
313 thread_id, x, y,
314 sum + (get_enlarged_pixel (thread_id, x, y) - sum) * weight);
315#pragma omp simd
316 for (int x = m_enlarged_tile_size - y; x < m_enlarged_tile_size; x++)
317 put_enlarged_pixel (
318 thread_id, x, y,
319 sum
320 + (get_enlarged_pixel (thread_id, x, y) - sum)
321 * m_weights[m_enlarged_tile_size - 1 - x]);
322 }
323 /* Taper left and right edge. */
324 for (int y = m_taper_size; y < m_enlarged_tile_size - m_taper_size; y++)
325 {
326#pragma omp simd
327 for (int x = 0; x < m_taper_size; x++)
328 put_enlarged_pixel (
329 thread_id, x, y,
330 sum
331 + (get_enlarged_pixel (thread_id, x, y) - sum)
332 * m_weights[x]);
333#pragma omp simd
334 for (int x = m_enlarged_tile_size - m_taper_size;
335 x < m_enlarged_tile_size; x++)
336 put_enlarged_pixel (
337 thread_id, x, y,
338 sum
339 + (get_enlarged_pixel (thread_id, x, y) - sum)
340 * m_weights[m_enlarged_tile_size - 1 - x]);
341 }
342 /* Taper bottom edge. */
343 for (int y = m_enlarged_tile_size - m_taper_size;
344 y < m_enlarged_tile_size; y++)
345 {
346 int d = m_enlarged_tile_size - 1 - y;
347 T weight = m_weights[d];
348#pragma omp simd
349 for (int x = 0; x < d; x++)
350 put_enlarged_pixel (
351 thread_id, x, y,
352 sum
353 + (get_enlarged_pixel (thread_id, x, y) - sum)
354 * m_weights[x]);
355#pragma omp simd
356 for (int x = d; x < m_enlarged_tile_size - d; x++)
357 put_enlarged_pixel (
358 thread_id, x, y,
359 sum + (get_enlarged_pixel (thread_id, x, y) - sum) * weight);
360#pragma omp simd
361 for (int x = m_enlarged_tile_size - d; x < m_enlarged_tile_size; x++)
362 put_enlarged_pixel (
363 thread_id, x, y,
364 sum
365 + (get_enlarged_pixel (thread_id, x, y) - sum)
366 * m_weights[m_enlarged_tile_size - 1 - x]);
367 }
368 }
369 if (progress && progress->cancelled ())
370 return;
371
372 if (!m_richardson_lucy)
373 {
374 typename fft_complex_t<T>::type *in = m_data[thread_id].in.get ();
375 m_plan_2d.execute_r2c (m_data[thread_id].enlarged_tile->data (), in);
376#pragma omp simd
377 for (int i = 0; i < m_fft_size * m_enlarged_tile_size; i++)
378 {
379 T vr = in[i][0];
380 T vi = in[i][1];
381 T wr = m_blur_kernel[i][0];
382 T wi = m_blur_kernel[i][1];
383 in[i][0] = vr * wr - vi * wi;
384 in[i][1] = vr * wi + vi * wr;
385 }
386 m_plan_2d_inv.execute_c2r (in, m_data[thread_id].enlarged_tile->data ());
387 }
388 else
389 {
390 std::vector<T, fft_allocator<T>> &observed = m_data[thread_id].observed;
391 std::copy (m_data[thread_id].enlarged_tile->begin (),
392 m_data[thread_id].enlarged_tile->end (),
393 observed.begin ());
394 std::vector<T, fft_allocator<T>> &estimate
395 = *m_data[thread_id].enlarged_tile;
396 std::vector<T, fft_allocator<T>> &ratios = m_data[thread_id].ratios;
397 T scale = (T)1;
398 typename fft_complex_t<T>::type *in = m_data[thread_id].in.get ();
399 T sigma = m_sigma;
400 for (int iteration = 0; iteration < m_iterations; iteration++)
401 {
402 if (progress && progress->cancelled ())
403 return;
404 /* Step A: Re-blur the current estimate. */
405
406 /* Blur current estimate to IN. */
407 m_plan_2d.execute_r2c (estimate.data (), in);
408#pragma omp simd
409 for (int i = 0; i < m_fft_size * m_enlarged_tile_size; i++)
410 {
411 T vr = in[i][0];
412 T vi = in[i][1];
413 T wr = m_blur_kernel[i][0];
414 T wi = m_blur_kernel[i][1];
415 in[i][0] = vr * wr - vi * wi;
416 in[i][1] = vr * wi + vi * wr;
417 }
418 m_plan_2d_inv.execute_c2r (in, ratios.data ());
419
420 /* Step B: ratio = observed / (re-blurred + epsilon). */
421
422 T epsilon = (T)1e-12;
423
424 /* RATIOS is now blurred ESTIMATE; compute ratios. */
425 if (sigma > (T)0)
426#pragma omp simd
427 for (int i = 0; i < m_enlarged_tile_size * m_enlarged_tile_size;
428 i++)
429 {
430 T reblurred = ratios[i] * scale;
431 T diff = observed[i] - reblurred;
432 if (reblurred > epsilon && std::abs (diff) > (T)2 * sigma)
433 ratios[i] = (T)1.0
434 + (reblurred * diff)
435 / (reblurred * reblurred + sigma * sigma);
436 else
437 ratios[i] = (T)1.0;
438 }
439 else
440#pragma omp simd
441 for (int i = 0; i < m_enlarged_tile_size * m_enlarged_tile_size;
442 i++)
443 {
444 T reblurred = ratios[i] * scale;
445 if (reblurred > epsilon)
446 ratios[i] = observed[i] / reblurred;
447 else
448 ratios[i] = (T)1.0;
449 }
450
451 /* Step C: Update estimate
452 FFT(ratio) -> multiply by FFT(PSF_flipped) -> IFFT
453 estimate = estimate * result_of_Step_C. */
454
455 /* Do FFT of ratio. */
456 m_plan_2d.execute_r2c (ratios.data (), in);
457 /* Scale by complex conjugate of blur kernel. */
458#pragma omp simd
459 for (int i = 0; i < m_fft_size * m_enlarged_tile_size; i++)
460 {
461 T vr = in[i][0];
462 T vi = in[i][1];
463 T wr = m_blur_kernel[i][0];
464 T wi = -m_blur_kernel[i][1];
465 in[i][0] = vr * wr - vi * wi;
466 in[i][1] = vr * wi + vi * wr;
467 }
468 /* Now initialize ratios. */
469 m_plan_2d_inv.execute_c2r (in, ratios.data ());
470
471 /* estimate = estimate * result_of_Step_C. */
472#pragma omp simd
473 for (int i = 0; i < m_enlarged_tile_size * m_enlarged_tile_size; i++)
474 estimate[i] *= ratios[i] * scale;
475 }
476 }
477 /* Use bicubic interpolation for upscaling by 2. */
478 if (m_supersample == 2)
479 {
480 const T* enlarged_ptr = m_data[thread_id].enlarged_tile->data();
481 for (int y = m_border_size; y < m_tile_size - m_border_size; y++)
482#pragma omp simd
483 for (int x = m_border_size; x < m_tile_size - m_border_size; x++)
484 {
485 int sx = x * 2;
486 int sy = y * 2;
487 T val = cubic_interpolate (
488 cubic_interpolate (
489 enlarged_ptr[(sy - 1) * m_enlarged_tile_size + sx - 1],
490 enlarged_ptr[(sy) * m_enlarged_tile_size + sx - 1],
491 enlarged_ptr[(sy + 1) * m_enlarged_tile_size + sx - 1],
492 enlarged_ptr[(sy + 2) * m_enlarged_tile_size + sx - 1], (T)0.5),
493 cubic_interpolate (
494 enlarged_ptr[(sy - 1) * m_enlarged_tile_size + sx - 0],
495 enlarged_ptr[(sy) * m_enlarged_tile_size + sx - 0],
496 enlarged_ptr[(sy + 1) * m_enlarged_tile_size + sx - 0],
497 enlarged_ptr[(sy + 2) * m_enlarged_tile_size + sx - 0], (T)0.5),
498 cubic_interpolate (
499 enlarged_ptr[(sy - 1) * m_enlarged_tile_size + sx + 1],
500 enlarged_ptr[(sy) * m_enlarged_tile_size + sx + 1],
501 enlarged_ptr[(sy + 1) * m_enlarged_tile_size + sx + 1],
502 enlarged_ptr[(sy + 2) * m_enlarged_tile_size + sx + 1], (T)0.5),
503 cubic_interpolate (
504 enlarged_ptr[(sy - 1) * m_enlarged_tile_size + sx + 2],
505 enlarged_ptr[(sy) * m_enlarged_tile_size + sx + 2],
506 enlarged_ptr[(sy + 1) * m_enlarged_tile_size + sx + 2],
507 enlarged_ptr[(sy + 2) * m_enlarged_tile_size + sx + 2], (T)0.5),
508 (T)0.5);
509 put_pixel (thread_id, x, y, val);
510 }
511 }
512 /* Bigger upscaling is unlikely to be useful. */
513 else if (m_supersample > 1)
514 {
515 if (progress && progress->cancelled ())
516 return;
517 T scale = (T)1 / (T)(m_supersample * m_supersample);
518 for (int y = m_border_size; y < m_tile_size - m_border_size; y++)
519 for (int x = m_border_size; x < m_tile_size - m_border_size; x++)
520 {
521 T sum = 0;
522 for (int yy = 0; yy < m_supersample; yy++)
523 for (int xx = 0; xx < m_supersample; xx++)
524 sum += get_enlarged_pixel (thread_id, x * m_supersample + xx,
525 y * m_supersample + yy);
526 put_pixel (thread_id, x, y, sum * scale);
527 }
528 }
