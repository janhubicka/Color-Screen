312/* Initialize TIFF loader for file NAME.
313 On failure, set ERROR to the error message.
314 PROGRESS is used for progress reporting.
315 DEMOSAIC is the demosaicing algorithm to use (ignored for TIFF). */
316bool
317tiff_image_data_loader::init_loader (const char *name, const char **error,
318 progress_info *progress,
319 image_data::demosaicing_t demosaic)
320{
321 if (debug)
322 printf ("TIFFopen\n");
323 TIFFSetWarningHandler (warning_handler);
324 m_tif = TIFFOpen (name, "r");
325 if (!m_tif)
326 {
327 *error = "can not open file";
328 return false;
329 }
330 uint32_t w, h;
331 uint16_t photometric;
332 float dpi;
333 if (debug)
334 printf ("checking width/height\n");
335 TIFFGetField (m_tif, TIFFTAG_IMAGEWIDTH, &w);
336 TIFFGetField (m_tif, TIFFTAG_IMAGELENGTH, &h);
337 if (TIFFGetField (m_tif, TIFFTAG_XRESOLUTION, &dpi))
338 m_img->xdpi = dpi;
339 if (TIFFGetField (m_tif, TIFFTAG_YRESOLUTION, &dpi))
340 m_img->ydpi = dpi;
341 if (m_img->xdpi && !m_img->ydpi)
342 m_img->ydpi = m_img->xdpi;
343 else if (!m_img->xdpi && m_img->ydpi)
344 m_img->xdpi = m_img->ydpi;
345 if (debug)
346 printf ("checking bits per sample\n");
347 TIFFGetFieldDefaulted (m_tif, TIFFTAG_BITSPERSAMPLE, &m_bitspersample);
348 if (m_bitspersample != 8 && m_bitspersample != 16)
349 {
350 *error = "bit depth should be 8 or 16";
351 return false;
352 }
353 if (debug)
354 printf ("checking smaples per pixel\n");
355 TIFFGetFieldDefaulted (m_tif, TIFFTAG_SAMPLESPERPIXEL, &m_samples);
356 void *iccprof;
357 uint32_t size;
358 if (TIFFGetField (m_tif, TIFFTAG_ICCPROFILE, &size, &iccprof))
359 {
360 // m_img->icc_profile = iccprof;
361 // m_img->icc_profile_size = size;
362 m_img->icc_profile = malloc (size);
363 memcpy (m_img->icc_profile, iccprof, size);
364 m_img->icc_profile_size = size;
365 }
366 if (m_samples != 1 && m_samples != 3 && m_samples != 4)
367 {
368 if (debug)
369 printf ("Samples:%i\n", m_samples);
370 *error = "only 1 sample per pixel (grayscale), 3 samples per pixel "
371 "(RGB) or 4 samples per pixel (RGBa) are supported";
372 return false;
373 }
374 if (TIFFGetField (m_tif, TIFFTAG_PHOTOMETRIC, &photometric))
375 {
376 if (photometric == PHOTOMETRIC_MINISBLACK && m_samples == 1)
377 ;
378 else if (photometric == PHOTOMETRIC_RGB
379 && (m_samples == 4 || m_samples == 3))
380 ;
381 else
382 {
383 *error = "only RGB, RGBa or grayscale images are suppored";
384 return false;
385 }
386 }
387 if (debug)
388 printf ("Getting scanlinel\n");
389 m_buf = _TIFFmalloc (TIFFScanlineSize (m_tif));
390 if (!m_buf)
391 {
392 *error = "out of memory allocating tiff scanline";
393 return false;
394 }
395 m_img->width = w;
396 m_img->height = h;
397 grayscale = m_samples == 4 || m_samples == 1;
398 rgb = m_samples != 1;
399 if (m_bitspersample == 8)
400 m_img->maxval = 255;
401 else if (m_bitspersample == 16)
402 m_img->maxval = 65535;
403 else
404 {
405 *error = "Only bith depths 8 and 16 are supported";
406 return false;
407 }
408 m_row = 0;
409 m_img->load_exif (name);
410 return true;
411}

413/* Load part of the TIFF image.
414 Update PERMILLE with progress (0-1000).
415 On failure, set ERROR to the error message.
416 PROGRESS is used for progress reporting. */
417bool
418tiff_image_data_loader::load_part (int *permille, const char **error,
419 progress_info *progress)
420{
421 if ((int)m_row < m_img->height)
422 {
423 uint32_t row = m_row;
424 uint32_t w = m_img->width;
425 image_data::gray *data = m_img->get_row (row);
426 image_data::pixel *rgbdata = m_img->get_rgb_row (row);
427
428 if (debug)
429 printf ("Decoding scanline %i\n", row);
430 if (!TIFFReadScanline (m_tif, m_buf, row))
431 {
432 *error = "scanline decoding failed";
433 return false;
434 }
435 if (m_bitspersample == 8 && m_samples == 1)
436 {
437 uint8_t *buf2 = (uint8_t *)m_buf;
438 if (data)
439 for (uint32_t x = 0; x < w; x++)
440 data[x] = buf2[x];
441 }
442 else if (m_bitspersample == 8 && m_samples == 3)
443 {
444 uint8_t *buf2 = (uint8_t *)m_buf;
445 if (rgbdata)
446 for (uint32_t x = 0; x < w; x++)
447 {
448 rgbdata[x].r = buf2[3 * x + 0];
449 rgbdata[x].g = buf2[3 * x + 1];
450 rgbdata[x].b = buf2[3 * x + 2];
451 }
452 }
453 else if (m_bitspersample == 8 && m_samples == 4)
454 {
455 uint8_t *buf2 = (uint8_t *)m_buf;
456 for (uint32_t x = 0; x < w; x++)
457 {
458 if (rgbdata)
459 {
460 rgbdata[x].r = buf2[4 * x + 0];
461 rgbdata[x].g = buf2[4 * x + 1];
462 rgbdata[x].b = buf2[4 * x + 2];
463 }
464 if (data)
465 data[x] = buf2[4 * x + 3];
466 }
467 }
468 else if (m_bitspersample == 16 && m_samples == 1)
469 {
470 uint16_t *buf2 = (uint16_t *)m_buf;
471 if (data)
472 for (uint32_t x = 0; x < w; x++)
473 data[x] = buf2[x];
474 }
475 else if (m_bitspersample == 16 && m_samples == 3)
476 {
477 uint16_t *buf2 = (uint16_t *)m_buf;
478 if (rgbdata)
479 for (uint32_t x = 0; x < w; x++)
480 {
481 rgbdata[x].r = buf2[3 * x + 0];
482 rgbdata[x].g = buf2[3 * x + 1];
483 rgbdata[x].b = buf2[3 * x + 2];
484 }
485 }
486 else if (m_bitspersample == 16 && m_samples == 4)
487 {
488 uint16_t *buf2 = (uint16_t *)m_buf;
489 for (uint32_t x = 0; x < w; x++)
490 {
491 if (rgbdata)
492 {
493 rgbdata[x].r = buf2[4 * x + 0];
494 rgbdata[x].g = buf2[4 * x + 1];
495 rgbdata[x].b = buf2[4 * x + 2];
496 }
497 if (data)
498 data[x] = buf2[4 * x + 3];
499 }
500 }
501 else
502 {
503 /* We should have given up earlier. */
504 fprintf (stderr,
505 "Wrong combinations of bitspersample %i and samples %i\n",
506 m_bitspersample, m_samples);
507 abort ();
508 }
509 *permille = (999 * m_row + m_img->height / 2) / m_img->height;
510 m_row++;
511 }
512 else
513 {
514 if (debug)
515 printf ("done\n");
516 *permille = 1000;
517 }
518 return true;
519}

521/* Initialize JPEG loader for file NAME.
522 On failure, set ERROR to the error message.
523 PROGRESS is used for progress reporting.
524 DEMOSAIC is the demosaicing algorithm to use (ignored for JPEG). */
525bool
526jpg_image_data_loader::init_loader (const char *name, const char **error,
527 progress_info *progress, image_data::demosaicing_t demosaic)
528{
529 FILE *jpegFile;
530 if ((jpegFile = fopen (name, "rb")) == NULL)
531 {
532 *error = "can not open file";
533 return false;
534 }
535 size_t size;
536 if (fseek (jpegFile, 0, SEEK_END) < 0 || ((size = ftell (jpegFile)) < 0)
537 || fseek (jpegFile, 0, SEEK_SET) < 0)
538 {
539 *error = "can not determine input file size";
540 fclose (jpegFile);
541 return false;
542 }
543 if (size == 0)
544 {
545 *error = "input file is empty";
546 fclose (jpegFile);
547 return false;
548 }
549 unsigned long jpegSize = (unsigned long)size;
550 if ((m_jpeg_buf = (unsigned char *)tjAlloc (jpegSize)) == NULL)
551 {
552 *error = "input file is empty";
553 fclose (jpegFile);
554 return false;
555 }
556 if (fread (m_jpeg_buf, jpegSize, 1, jpegFile) < 1)
557 {
558 *error = "can not read file";
559 fclose (jpegFile);
560 return false;
561 }
562 fclose (jpegFile);
563 m_tj_instance = tjInitDecompress ();
564 if (!m_tj_instance)
565 {
566 *error = "can not initialize jpeg decompressor";
567 return false;
568 }
569 int inSubsamp, inColorspace;
570 /* TODO: use Exif to determine DPI. */
571 if (tjDecompressHeader3 (m_tj_instance, m_jpeg_buf, jpegSize, &m_img->width,
572 &m_img->height, &inSubsamp, &inColorspace)
573 < 0)
574 {
575 *error = "can not read header";
576 return false;
577 }
578 /* RGB is 1 and gray is 2. */
579 if (inColorspace != 1 && inColorspace != 2)
580 {
581 *error = "only grayscale and rgb jpeg files are supported";
582 return false;
583 }
584 rgb = inColorspace == 1;
585 int pixelFormat = rgb ? TJPF_RGB : TJPF_GRAY;
586 // m_img_buf = (unsigned char *)tjAlloc(m_img->width * (size_t) m_img->height
587 // * tjPixelSize[pixelFormat]);
588 m_img_buf = (unsigned char *)MapAlloc::Alloc (
589 m_img->width * (size_t)m_img->height * tjPixelSize[pixelFormat],
590 "Decompressed jpeg image");
591 if (!m_img_buf)
592 {
593 *error = "can not allocate decompressed image buffer";
594 return false;
595 }
596 if (tjDecompress2 (m_tj_instance, m_jpeg_buf, jpegSize, m_img_buf,
597 m_img->width, 0, m_img->height, pixelFormat,
598 TJFLAG_ACCURATEDCT)
599 < 0)
600 {
601 *error = "jpeg decompression failed";
602 return false;
603 }
604 free (m_jpeg_buf);
605 m_jpeg_buf = NULL;
606 tjDestroy (m_tj_instance);
607 m_tj_instance = NULL;
608 grayscale = !rgb;
609 m_img->maxval = 255;
610 m_img->load_exif (name);
611 return true;
612}

979 {

981
982 if (bayer_correction)
983 {
984 luminosity_t grsum = 0, rsum = 0, gbsum = 0, bsum = 0;
985 /* Pass 1: find approximate ratio */

987 for (int x = 0; x < m_img->width - 1; x++)
988 {
989 int i = y * m_img->width + x;

992 {
993 int r = m_processor->imgdata.image[i + 1][0];
994 if (r > 0 && r < 65535 - 256)
995 grsum += g, rsum += r;
996 int b = m_processor->imgdata.image[i + 1][2];
997 if (b > 0 && b < 65535 - 256)
998 gbsum += g, bsum += b;
999 }
1000 }
1001 if (!grsum || !gbsum)
1002 {
1003 *error = "image is too overexposed or completely dark in green channel";
1004 return false;
1005 }
1006 if (!bsum)
1007 {
1008 *error = "image has no data in blue channel";
1009 return false;
1010 }
1011 if (!rsum)
1012 {
1013 *error = "image has no data in red channel";

1018 //fprintf (stderr, "rratio %f bratio %f\n", rratio, bratio);

1020 bhistogram.set_range (1 - range, 1 + range, 65535 * 4);
1021 for (int y = 0; y < m_img->height; y++)
1022 for (int x = 0; x < m_img->width - 1; x++)
1023 {
1024 int i = y * m_img->width + x;
1025 int g = m_processor->imgdata.image[i][1];

1037 }

1040 {

1043 bhistogram.account (ratio);

1045 }

1054 }

1056 rscale = rhistogram.find_avg (0.2, 0.2) / rratio;
1057 //fprintf (stderr, "rscale %f bscale %f\n", rscale, bscale);
1058 }
1059#pragma omp parallel for default(none) \
1060 shared(m_img, m_processor, bscale, rscale)

1062 {

1064 if (row)
1065 for (int x = 0; x < m_img->width; x++)
1066 {
1067 int i = y * m_img->width + x;

1070 + m_processor->imgdata.image[i][1]

1078#pragma omp parallel for default(none) shared(m_img, m_processor)

1080 {

1082 if (row)
1083 for (int x = 0; x < m_img->width; x++)
1084 {
1085 int i = y * m_img->width + x;
1086 row[x].r = m_processor->imgdata.image[i][0];
1087 row[x].g = m_processor->imgdata.image[i][1];
1088 row[x].b = m_processor->imgdata.image[i][2];
1089 }
1090 }
1091 }
1092 else
1093 {
1094#pragma omp parallel for default(none) shared(m_img, m_processor)
1095 for (int y = 0; y < m_img->height; y++)
1096 {
1097 image_data::gray *row = m_img->get_row (y);

1099 {

1104 }

1533 }
