74class ChangeParametersCommand : public QUndoCommand {
75public:
76 ChangeParametersCommand(MainWindow *window, const ParameterState &oldState,
77 const ParameterState &newState,
78 const QString &description = QString())
79 : m_window(window), m_oldState(oldState), m_newState(newState) {
80 setText(description.isEmpty() ? "Change Parameters" : description);
81 m_timestamp = QDateTime::currentMSecsSinceEpoch();
82 }
83
84 int id() const override {
85 return 1; // All parameter changes have the same ID
86 }
87
88 bool mergeWith(const QUndoCommand *other) override {
89 if (other->id() != id())
90 return false;
91
92 const ChangeParametersCommand *cmd =
93 static_cast<const ChangeParametersCommand *>(other);
94
95 // Only merge if commands are within 500ms of each other (e.g., slider
96 // dragging)
97 qint64 timeDiff = cmd->m_timestamp - m_timestamp;
98 if (timeDiff > 500) {
99 return false; // Don't merge - create separate undo step
100 }
101
102 // Merge: update our newState to the newer command's newState
103 // This allows slider dragging to be one undo operation
104 m_newState = cmd->m_newState;
105 m_timestamp = cmd->m_timestamp; // Update timestamp for next merge check
106 return true;
107 }
108
109 void undo() override { m_window->applyState(m_oldState); }
110
111 void redo() override { m_window->applyState(m_newState); }
112
113private:
114 MainWindow *m_window;
115 ParameterState m_oldState;
116 ParameterState m_newState;
117 qint64 m_timestamp; // Timestamp for merge window
118};

577 "background: Palette(Window); }");

580 outerLayout->setContentsMargins(1, 1, 1, 1);
581 outerLayout->setSpacing(0);

584 QGridLayout *gLayout = new QGridLayout(container);
585 gLayout->setContentsMargins(0, 0, 0, 0);
586 gLayout->setSpacing(0);

591 gLayout->addWidget(grip, 0, 0, Qt::AlignRight | Qt::AlignBottom);
592
593 outerLayout->addWidget(container);

595 dock->setWidget(wrapper);
596 w->show(); // Ensure widget is visible
597 dock->setFloating(true);
598 dock->show();

610 if (dock->widget()) {

612 QWidget *wrapper = dock->widget();

614 // The structure is Wrapper -> QVBoxLayout -> Container ->
615 // QGridLayout -> Widget
616 if (wrapper->layout() && wrapper->layout()->count() > 0) {

618 if (containerItem && containerItem->widget() &&

623 if (child && !qobject_cast<QSizeGrip *>(child)) {

1066QIcon getSymbolicIcon(const QString &name) {
1067 // If it is a resource, use it directly (Qt handles SVG scaling properly)
1068 // We assume resources are already correct color (white)
1069 if (name.startsWith(":/")) {
1070 return QIcon(name);
1071 }
1072
1073 QString path;
1074#ifdef Q_OS_WIN
1075 // Fallback logic for Windows specific paths if needed,
1076 // but mostly we should use resources or standard theme.
1077 // Keeping existing logic for finding files if they are not resources.
1078 static QStringList subdirs = {"actions", "devices", "places",
1079 "status", "ui", "legacy",
1080 "categories", "apps", "mimetypes"};
1081 QString appDir = QCoreApplication::applicationDirPath();
1082 QStringList bases = {appDir + "/share/icons/Adwaita/symbolic",
1083 appDir + "/../share/icons/Adwaita/symbolic"};
1084
1085 for (const auto &base : bases) {
1086 for (const auto &subdir : subdirs) {
1087 QString tryPath = base + "/" + subdir + "/" + name + ".svg";
1088 if (QFile::exists(tryPath)) {
1089 path = tryPath;
1090 break;
1091 }
1092 tryPath = base + "/" + subdir + "/" + name + ".symbolic.svg";
1093 if (QFile::exists(tryPath)) {
1094 path = tryPath;
1095 break;
1096 }
1097 }
1098 if (!path.isEmpty())
1099 break;
1100 }
1101#endif
1102
1103 if (!path.isEmpty()) {
1104 QIcon icon;
1105 QSvgRenderer renderer(path);
1106 if (!renderer.isValid())
1107 return QIcon::fromTheme(name);
1108
1109 // Generate multiple sizes for DPI
1110 QList<int> sizes = {16, 24, 32, 48, 64, 96, 128};
1111 for (int size : sizes) {
1112 QPixmap pix(size, size);
1113 pix.fill(Qt::transparent);
1114
1115 QPainter p(&pix);
1116 renderer.render(&p);
1117
1118 // Recolor to white
1119 p.setCompositionMode(QPainter::CompositionMode_SourceIn);
1120 p.fillRect(pix.rect(), Qt::white);
1121 p.end();
1122
1123 icon.addPixmap(pix);
1124 }
1125 return icon;
1126 }
1127
1128 return QIcon::fromTheme(name); // Fallback to system theme
1129}

1420

1422 QSize baseSize = m_imageWidget->size();

1472 // We access the static array in anonymous namespace from
1473 // render-type-parameters.h Since we included the header, we can try accessing
1474 // it via the namespace. However, anonymous namespace members have internal
1475 // linkage. Usually headers shouldn't define static data in anonymous
1476 // namespaces unless used carefully. Assuming we can access
1477 // colorscreen::render_type_properties NOTE: In C++, anonymous namespace
1478 // members are accessible in the same TU. If render_type_properties is in a
1479 // header in anonymous namespace, each TU gets a copy. But we need to refer to
1480 // it. It's inside namespace colorscreen { namespace { ... } } or just
1481 // namespace { ... } inside colorscreen? The header has: namespace colorscreen

1487 updateColorCheckBoxState();
1488 for (int i = 0; i < render_type_max; ++i) {
1489 const render_type_property &prop = render_type_properties[i];
1490
1491 // Filter logic
1492 bool show = true;
1493
1494 if (prop.flags & render_type_property::HIDE_IN_GUI)
1495 show = false;
1496

1506 // m_scan->rgbdata is NULL.

1508 // Check m_scan
1509 if (!m_scan || !m_scan->has_rgb()) {
1510 show = false;
1511 }
1512 }
1513 if ((prop.flags & render_type_property::NEEDS_CORRECTION_PROFILE) &&
1514 !m_rparams.has_correction_profile())
1515 show = false;
1516
1517 if (show) {
1518 m_modeComboBox->addItem(prop.pretty_name, QVariant(i));
1519 if (prop.help)
1520 m_modeComboBox->setItemData(i, QString::fromUtf8(prop.help),
1521 Qt::ToolTipRole);
1522 }
1523 }
1524
1525 // Select current type if present
1526 int idx = m_modeComboBox->findData((int)m_renderTypeParams.type);
1527 if (idx != -1) {
1528 m_modeComboBox->setCurrentIndex(idx);
1529 } else if (m_modeComboBox->count() > 0) {
1530 // Fallback
1531 m_modeComboBox->setCurrentIndex(0);

1533 // Let's defer that to user interaction or explicit set.

1538 if (i < m_modeComboBox->count()) {
1539 m_modeActions[i]->setEnabled(true);

1544 help = m_modeComboBox->itemText(i);

1547 Qt::ToolTipRole);

1550 }
1551 }

1553 m_modeComboBox->blockSignals(false);

2082ProgressEntry *MainWindow::getLongestRunningTask() {
2083 ProgressEntry *oldestActive = nullptr;
2084 ProgressEntry *oldestAny = nullptr;
2085 qint64 maxActiveTime = -1;
2086 qint64 maxAnyTime = -1;
2087
2088 for (auto &entry : m_activeProgresses) {
2089 qint64 elapsed = entry.startTime.elapsed();
2090
2091 // Check if this task has non-zero progress
2092 const char *tName = "";
2093 float percent = 0;
2094 entry.info->get_status(&tName, &percent);
2095
2096 // Track oldest task with non-zero progress
2097 if (percent > 0 && elapsed > maxActiveTime) {
2098 maxActiveTime = elapsed;
2099 oldestActive = &entry;
2100 }
2101
2102 // Also track oldest task overall as fallback
2103 if (elapsed > maxAnyTime) {
2104 maxAnyTime = elapsed;
2105 oldestAny = &entry;
2106 }
2107 }
2108
2109 // Prefer task with non-zero progress, otherwise use oldest

2200
