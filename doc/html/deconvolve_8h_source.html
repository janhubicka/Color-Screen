163/* Deconvolution worker. Sharpen DATA to OUT which both has dimensions
164 WIDTH*HEIGHT. DATA are accessed using GETDATA function and PARAM can be used
165 to pass extra data around. SHARPEN specifies sharpening parameters.
166 PROGRESS is progress info. PARALLEL is true if parallel execution is
167 requested. O is output type name, MEM_O is memory output type, T is data
168 type name, P is extra bookkeeping parameter type. DT is a type to do
169 deconvolution in. */
170template <typename O, typename mem_O, typename T, typename P,
171 O (*getdata) (T data, int_point_t p, int width, P param), typename DT>
172nodiscard_attr bool
173deconvolve (mem_O *out, T data, P param, int width, int height,
174 const sharpen_parameters &sharpen, progress_info *progress,
175 bool parallel = true)
176{
177 int nthreads = parallel ? omp_get_max_threads () : 1;
178 typename deconvolution<DT>::mode mode;
179 if (progress)
180 progress->set_task ("initializing mtf based deconvolution", 1);
181 switch (sharpen.mode)
182 {
183 case sharpen_parameters::richardson_lucy_deconvolution:
184 mode = deconvolution<DT>::richardson_lucy_sharpen;
185 break;
186 case sharpen_parameters::wiener_deconvolution:
187 mode = deconvolution<DT>::sharpen;
188 break;
189 case sharpen_parameters::blur_deconvolution:
190 mode = deconvolution<DT>::blur;
191 break;
192 default:
193 abort ();
194 }
195 std::shared_ptr<mtf> scanner_mtf = mtf::get_mtf (sharpen.scanner_mtf, progress);
196 deconvolution<DT> d (scanner_mtf.get (), sharpen.scanner_mtf_scale,
197 sharpen.scanner_snr, sharpen.richardson_lucy_sigma,
198 nthreads, mode, sharpen.richardson_lucy_iterations,
199 sharpen.supersample);
200
201 int xtiles
202 = (width + d.get_basic_tile_size () - 1) / d.get_basic_tile_size ();
203 int ytiles
204 = (height + d.get_basic_tile_size () - 1) / d.get_basic_tile_size ();
205 if (progress)
206 {
207 if (mode == deconvolution<DT>::sharpen)
208 progress->set_task ("deconvolution sharpening (Weiner filter)",
209 xtiles * ytiles);
210 else if (mode == deconvolution<DT>::blur)
211 progress->set_task ("deconvolution blurring",
212 xtiles * ytiles);
213 else
214 progress->set_task ("deconvolution sharpening (Richardson-Lucy)",
215 xtiles * ytiles);
216 }
217#pragma omp parallel for default(none) schedule(dynamic) collapse(2) shared( \
218 width, height, d, progress, out, param, parallel, data) if (parallel)
219 for (int y = 0; y < height; y += d.get_basic_tile_size ())
220 for (int x = 0; x < width; x += d.get_basic_tile_size ())
221 {
222 if (progress && progress->cancel_requested ())
223 continue;
224 int id = parallel ? omp_get_thread_num () : 0;
225 d.init (id);
226 for (int yy = 0; yy < d.get_tile_size_with_borders (); yy++)
227 for (int xx = 0; xx < d.get_tile_size_with_borders (); xx++)
228 {
229 int px = x + xx - d.get_border_size ();
230 int py = y + yy - d.get_border_size ();
231
232 /* Do mirroring to avoid sharp edge at the border of image. */
233 if (px < 0)
234 px = -px;
235 if (py < 0)
236 py = -py;
237 if (px >= width)
238 px = width - (px - width) - 1;
239 if (py >= height)
240 py = height - (py - height) - 1;
241 px = std::clamp (px, 0, width - 1);
242 py = std::clamp (py, 0, height - 1);
243 d.put_pixel (id, xx, yy, getdata (data, {px, py}, width, param));
244 }
245 d.process_tile (id, progress);
246 for (int yy = 0; yy < d.get_basic_tile_size (); yy++)
247 for (int xx = 0; xx < d.get_basic_tile_size (); xx++)
248 if (y + yy < height && x + xx < width)
249 out[(y + yy) * width + x + xx] = d.get_pixel (
250 id, xx + d.get_border_size (), yy + d.get_border_size ());
251 if (progress)
252 progress->inc_progress ();
253 }
254 if (progress && progress->cancelled ())
255 return false;
256 return true;
257}

259/* Deconvolution worker for rgbdata and related types. Sharpen DATA to OUT.
260 DATA are accessed using GETDATA function and PARAM can be used to pass extra
261 data around. SHARPEN specifies sharpening parameters. PROGRESS is progress
262 info. PARALLEL is true if parallel execution is requested. O is output type

264 bookkeeping parameter type. DT is a type to do deconvolution in. */

267nodiscard_attr bool
268deconvolve_rgb (mem_O *out, T data, P param, int width, int height,
269 const sharpen_parameters &sharpen,
270 progress_info *progress,

272{

274 typename deconvolution<DT>::mode mode;
275 if (progress)
276 progress->set_task ("initializing mtf based deconvolution", 1);
277 switch (sharpen.mode)
278 {
279 case sharpen_parameters::richardson_lucy_deconvolution:
280 mode = deconvolution<DT>::richardson_lucy_sharpen;
281 break;
282 case sharpen_parameters::wiener_deconvolution:
283 mode = deconvolution<DT>::sharpen;
284 break;
285 case sharpen_parameters::blur_deconvolution:
286 mode = deconvolution<DT>::blur;
287 break;
288 default:
289 abort ();
290 }
291 std::shared_ptr<mtf> scanner_mtf = mtf::get_mtf (sharpen.scanner_mtf, progress);
292 deconvolution<DT> d (scanner_mtf.get (), sharpen.scanner_mtf_scale,
293 sharpen.scanner_snr, sharpen.richardson_lucy_sigma, nthreads * 3, mode,
294 sharpen.richardson_lucy_iterations,
295 sharpen.supersample);
296
297 int xtiles
298 = (width + d.get_basic_tile_size () - 1) / d.get_basic_tile_size ();
299 int ytiles
300 = (height + d.get_basic_tile_size () - 1) / d.get_basic_tile_size ();
301 if (progress)
302 {
303 if (mode == deconvolution<DT>::blur)
304 progress->set_task ("deconvolution blurring", xtiles * ytiles);
305 else if (mode == deconvolution<DT>::sharpen)
306 progress->set_task ("deconvolution sharpening (weiner filter)", xtiles * ytiles);
307 else
308 progress->set_task ("deconvolution sharpening (richardson-lucy)", xtiles * ytiles);
309 }
310#pragma omp parallel for default(none) schedule(dynamic) collapse(2) \
311 shared (width, height,d,progress,out,param,parallel,data) if (parallel)
312 for (int y = 0; y < height; y += d.get_basic_tile_size ())
313 for (int x = 0; x < width; x += d.get_basic_tile_size ())
314 {
315 if (progress && progress->cancelled ())
316 continue;
317 int id = parallel ? omp_get_thread_num () : 0;
318 d.init (3 * id);
319 d.init (3 * id + 1);
320 d.init (3 * id + 2);
321
322 for (int yy = 0; yy < d.get_tile_size_with_borders (); yy++)
323 for (int xx = 0; xx < d.get_tile_size_with_borders (); xx++)
324 {
325 int px = x + xx - d.get_border_size ();
326 int py = y + yy - d.get_border_size ();
327
328 /* Do mirroring to avoid sharp edge at the border of image. */
329 if (px < 0)
330 px = -px;
331 if (py < 0)
332 py = -py;
333 if (px >= width)
334 px = width - (px - width) - 1;
335 if (py >= height)
336 py = height - (py - height) - 1;
337 px = std::clamp (px, 0, width - 1);
338 py = std::clamp (py, 0, height - 1);
339 O pixel = getdata (data, {px, py}, width, param);
340 d.put_pixel (3 * id, xx, yy, pixel.red);
341 d.put_pixel (3 * id + 1, xx, yy, pixel.green);
342 d.put_pixel (3 * id + 2, xx, yy, pixel.blue);
343 }
344 d.process_tile (3 * id, progress);
345 d.process_tile (3 * id + 1, progress);
346 d.process_tile (3 * id + 2, progress);
347 for (int yy = 0; yy < d.get_basic_tile_size (); yy++)
348 for (int xx = 0; xx < d.get_basic_tile_size (); xx++)
349 if (y + yy < height && x + xx < width)
350 {
351 out[(y + yy) * width + x + xx].red = d.get_pixel (
352 3 * id, xx + d.get_border_size (), yy + d.get_border_size ());
353 out[(y + yy) * width + x + xx].green = d.get_pixel (
354 3 * id + 1, xx + d.get_border_size (), yy + d.get_border_size ());
355 out[(y + yy) * width + x + xx].blue = d.get_pixel (
356 3 * id + 2, xx + d.get_border_size (), yy + d.get_border_size ());
357 }
358
359 if (progress)
360 progress->inc_progress ();
361 }
362 if (progress && progress->cancelled ())
363 return false;
364 return true;
365}

367/* Auto-select the type for deconvolution.
368 O is output type, MEM_O is memory output type, T is data type, P is extra
369 bookkeeping parameter type. */
370template <typename O, typename mem_O, typename T, typename P,
371 O (*getdata) (T data, int_point_t p, int width, P param)>
372nodiscard_attr bool
373deconvolve (mem_O *out, T data, P param, int width, int height,
374 const sharpen_parameters &sharpen, progress_info *progress,
375 bool parallel = true)
376{
377 /* For many iterations use double; otherwise float is good and faster. */
378 if (sharpen.mode != sharpen_parameters::richardson_lucy_deconvolution
379 || sharpen.richardson_lucy_iterations < 300)
380 return deconvolve<O, mem_O, T, P, getdata, float>(out, data, param, width,
381 height, sharpen, progress,
382 parallel);
383 else
384 return deconvolve<O, mem_O, T, P, getdata, double>(out, data, param, width,
385 height, sharpen, progress,
386 parallel);
387}
