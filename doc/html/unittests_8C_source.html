608 /* Stage 1: Parallel range. */
609#pragma omp parallel for reduction(histogram_range : h)
610 for (int i = 0; i < n; i++)
611 h.pre_account ((luminosity_t)(i % 1000));
612
613 if (h.find_min (0) != 0 || h.find_max (0) != 999)
614 {
615 printf ("FAILED: Parallel range mismatch! Min: %f Max: %f\n",
616 (double)h.find_min (0), (double)h.find_max (0));
617 return false;
618 }
619
620 h.finalize_range (1000);
621
622 /* Stage 2: Parallel entries. */
623#pragma omp parallel for reduction(histogram_entries : h)
624 for (int i = 0; i < n; i++)
625 h.account ((luminosity_t)(i % 1000));
626
627 h.finalize ();
628
629 if (h.num_samples () != n)
630 {
631 printf ("FAILED: Total count mismatch! Expected %i, got %i\n", n,
632 h.num_samples ());
633 return false;
634 }
635
636 for (int i = 0; i < 1000; i++)
637 if (h.entry (i) != 1000)
638 {
639 printf ("FAILED: Entry %i count mismatch! Expected 1000, got %llu\n", i,
640 (unsigned long long)h.entry (i));
641 return false;
642 }
643
644 return true;
