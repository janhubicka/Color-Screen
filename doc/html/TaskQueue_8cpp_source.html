144void TaskQueue::runAsync (std::function<void (colorscreen::progress_info *)> worker,
145 std::function<void ()> done,
146 const QVariant &userData)
147{
148 /* Pre-allocate the request ID so we can capture it in the lambda BEFORE
149 calling requestRender(). requestRender() → startTask() → emit
150 triggerRender() happens synchronously, so the connection must already
151 be live when that emit fires.
152
153 Qt::QueuedConnection guarantees the lambda runs from the event loop
154 even when sender and receiver share a thread, so it always executes
155 after this function returns and the connection is fully established. */
156 int reqId = m_nextReqId; /* peek — requestRender() will assign the same ID */
157
158 auto connHolder = std::make_shared<QMetaObject::Connection> ();
159
160 *connHolder = connect (
161 this, &TaskQueue::triggerRender, this,
162 [this, reqId,
163 worker = std::move (worker),
164 done = std::move (done),
165 connHolder]
166 (int firedId, std::shared_ptr<colorscreen::progress_info> progress,
167 const QVariant &) mutable
168 {
169 if (firedId != reqId)
170 return;
171
172 /* One-shot: disconnect immediately so later tasks don't re-trigger. */
173 disconnect (*connHolder);
174 connHolder.reset ();
175
176 /* Launch worker on Qt thread-pool; progress is read-only in worker. */
177 auto *watcher = new QFutureWatcher<void> (this);
178 connect (watcher, &QFutureWatcher<void>::finished, this,
179 [this, reqId, watcher, done = std::move (done)] () mutable
180 {
181 watcher->deleteLater ();
182 done ();
183 reportFinished (reqId, true);
184 });
185 watcher->setFuture (
186 QtConcurrent::run (
187 [w = std::move (worker), progress] () mutable
188 { w (progress.get ()); }));
189 },
190 Qt::QueuedConnection);
191
192 /* Now request the render — this will emit triggerRender(reqId, ...) which
193 is queued and delivered on the next event-loop iteration. */
194 requestRender (userData);
195}
