20inline int
21fast_rand16 (unsigned int *g_seed)
22{
23 *g_seed = (214013 * *g_seed + 2531011);
24 return ((*g_seed) >> 16) & 0x7FFF;
25}

27/* Random number generator used by RANSAC. It is re-initialized every time
28 RANSAC is run so results are deterministic. */
29static inline int
30fast_rand32 (unsigned int *g_seed) noexcept
31{
32 return fast_rand16 (g_seed) | (fast_rand16 (g_seed) << 15);
33}

35class translation_scale_matrix : public trans_4d_matrix
36{
37public:
38 translation_scale_matrix (coord_t tx, coord_t ty, coord_t s)
39 {
40 (*this)(0, 0) = s; (*this)(0, 1) = 0; (*this)(0, 2) = s * tx; (*this)(0, 3) = 0;
41 (*this)(1, 0) = 0; (*this)(1, 1) = s; (*this)(1, 2) = s * ty; (*this)(1, 3) = 0;
42 (*this)(2, 0) = 0; (*this)(2, 1) = 0; (*this)(2, 2) = 1; (*this)(2, 3) = 0;
43 (*this)(3, 0) = 0; (*this)(3, 1) = 0; (*this)(3, 2) = 0; (*this)(3, 3) = 1;
44 }
45};

47/* Used to do two passes across set of points and produces a translation and
48 scale matrix which normalizes them so they have center in 0 and scale
49 sqrt(2). This improves numerical stability of homography computations. */
50
51class normalize_points
52{
53public:
54 normalize_points (int nn) : n (nn) {}
55
56 /* Account for point P in first pass. */
57 void
58 account1 (point_t p, enum scanner_type type) noexcept
59 {
60 if (is_fixed_lens (type))
61 {
62 avg_x += p.x;
63 avg_y += p.y;
64 }
65 }
66 /* Screens with vertical strips has only x coordinate. */
67 void
68 account1_xonly (point_t p, enum scanner_type type) noexcept
69 {
70 if (is_fixed_lens (type))
71 avg_x += p.x;
72 }
73 /* Finish first pass. */
74 void
75 finish1 () noexcept
76 {
77 avg_x /= n;
78 avg_y /= n;
79 }
80 /* Account for point P in second pass. */
81 void
82 account2 (point_t p) noexcept
83 {
84 dist_sum += my_sqrt ((p.x - avg_x) * (p.x - avg_x)
85 + (p.y - avg_y) * (p.y - avg_y));
86 }
87 /* Screens with vertical strips has only x coordinate. */
88 void
89 account2_xonly (point_t p) noexcept
90 {
91 dist_sum += my_fabs (p.x - avg_x);
92 }
93 /* Return normalization matrix. */
94 trans_4d_matrix
95 get_matrix () noexcept
96 {
97 return translation_scale_matrix (-avg_x, -avg_y, my_sqrt ((coord_t)2) * n / dist_sum);
98 }
99
100private:
101 coord_t avg_x = 0, avg_y = 0;
102 coord_t dist_sum = 0;
103 int n;
104};

129/* Produce rows of homography equation converting point S to D at index N.
130 A and V are GSL matrix and vector to initialize. INVERT is true if
131 we are solving for inverse homography. FLAGS and SCANNER_TYPE determine
132 the setup of the equations. TS and TD are normalization matrices for
133 source and destination coordinates. */
134
135static inline void
136init_equation (gsl_matrix *A, gsl_vector *v, int n, bool invert, int flags,
137 enum scanner_type scanner_type, point_t s, point_t d,
138 trans_4d_matrix ts, trans_4d_matrix td)
139{
140 s = ts.perspective_transform (s);
141 d = td.perspective_transform (d);
142 if (!(flags & homography::solve_vertical_strips))
143 {
144 if (invert)
145 std::swap (s, d);
146 gsl_vector_set (v, n * 2, d.x);
147 gsl_vector_set (v, n * 2 + 1, d.y);
148
149 gsl_matrix_set (A, n * 2, 0, s.x);
150 gsl_matrix_set (A, n * 2, 1, s.y);
151 gsl_matrix_set (A, n * 2, 2, 1);
152 gsl_matrix_set (A, n * 2, 3, 0);
153 gsl_matrix_set (A, n * 2, 4, 0);
154 gsl_matrix_set (A, n * 2, 5, 0);
155 if (flags & (homography::solve_rotation | homography::solve_free_rotation))
156 {
157 if ((flags & homography::solve_free_rotation)
158 || scanner_type != lens_move_horizontally)
159 {
160 gsl_matrix_set (A, n * 2, 6, -d.x * s.x);
161 gsl_matrix_set (A, n * 2, 7, -d.x * s.y);
162 }
163 else
164 {
165 gsl_matrix_set (A, n * 2, 6, 0);
166 gsl_matrix_set (A, n * 2, 7, 0);
167 }
168 if (flags & homography::solve_free_rotation)
169 {
170 gsl_matrix_set (A, n * 2, 8, 0);
171 gsl_matrix_set (A, n * 2, 9, 0);
172 }
173 }
174
175 gsl_matrix_set (A, n * 2 + 1, 0, 0);
176 gsl_matrix_set (A, n * 2 + 1, 1, 0);
177 gsl_matrix_set (A, n * 2 + 1, 2, 0);
178 gsl_matrix_set (A, n * 2 + 1, 3, s.x);
179 gsl_matrix_set (A, n * 2 + 1, 4, s.y);
180 gsl_matrix_set (A, n * 2 + 1, 5, 1);
181 if (flags & homography::solve_rotation)
182 {
183 if (scanner_type != lens_move_vertically)
184 {
185 gsl_matrix_set (A, n * 2 + 1, 6, -d.y * s.x);
186 gsl_matrix_set (A, n * 2 + 1, 7, -d.y * s.y);
187 }
188 else
189 {
190 gsl_matrix_set (A, n * 2 + 1, 6, 0);
191 gsl_matrix_set (A, n * 2 + 1, 7, 0);
192 }
193 }
194 else if (flags & homography::solve_free_rotation)
195 {
196 gsl_matrix_set (A, n * 2 + 1, 6, 0);
197 gsl_matrix_set (A, n * 2 + 1, 7, 0);
198 gsl_matrix_set (A, n * 2 + 1, 8, -d.y * s.x);
199 gsl_matrix_set (A, n * 2 + 1, 9, -d.y * s.y);
200 }
201 }
202 else
203 {
204 /* Normally we compute error in screen coordinates, so the solver does not
205 increase size of screen to reduce square error.
206 This is not possible when we do linear solution, so compute inverse
207 homography and invert it later. Consequently we can not support free
208 rotation since scanner homography is not invertible and we here
209 compute inverse homography.
210
211 We look for matrix 3x2 (missing row for s.y) as follows:
212
213 /s.x\ /v0 v1 v2 \ /d.x\
214 \ 1/ = \v3 v4 v5 / |d.y|
215 \1 /
216
217 s.x = v0*d.x + v1*d.y + v2 | equation (1)
218 1 = v3*d.x + v4*d.y + v5 | equation (2)
219 multiply (1) by s.x
220 s.x = v3*d.x*s.x + v4*d.y*s.x + v5*s.x | equation (3)
221 now (3)-(1)
222 0 = v0*d.x + v1*d.y + v2 - v3*d.x*s.x - v4*d.y*s.x - v5*s.x
223 Fix v5=1 and reorder
224 s.x = v0*d.x + v1*d.y + v2 - v3*d.x*s.x - v4*d.y*s.x
225 */
226
227 gsl_matrix_set (A, n, 0, d.x);
228 gsl_matrix_set (A, n, 1, d.y);
229 gsl_matrix_set (A, n, 2, 1);
230 if ((flags & homography::solve_rotation)
231 || (flags & homography::solve_free_rotation))
232 {
233 gsl_matrix_set (A, n, 3, -d.x * s.x);
234 gsl_matrix_set (A, n, 4, -d.y * s.x);
235 }
236 gsl_vector_set (v, n, s.x);
237 assert (!colorscreen_checking || !invert);
238 }
239}

241/* Turn solution V to a matrix.
242 FLAGS and TYPE determine the setup of the equations. If INVERSE is true we
243 are determining inverse matrix. TS and TD are normalization matrices for
244 source and destination coordinates. If KEEP_VECTOR is false, free vector V.
245 */
246static inline trans_4d_matrix
247solution_to_matrix (gsl_vector *v, int flags, enum scanner_type type,
248 bool inverse, trans_4d_matrix ts, trans_4d_matrix td,
249 bool keep_vector = false)
250{
251 trans_4d_matrix ret;
252 if (inverse)
253 std::swap (ts, td);
254 if (!(flags & homography::solve_vertical_strips))
255 {
256 ret(0, 0) = gsl_vector_get (v, 0);
257 ret(0, 1) = gsl_vector_get (v, 1);
258 ret(0, 2) = gsl_vector_get (v, 2);
259 ret(0, 3) = 0;
260
261 ret(1, 0) = gsl_vector_get (v, 3);
262 ret(1, 1) = gsl_vector_get (v, 4);
263 ret(1, 2) = 0;
264 ret(1, 3) = gsl_vector_get (v, 5);
265
266 if ((flags & (homography::solve_rotation) && type != lens_move_horizontally)
267 || (flags & homography::solve_free_rotation))
268 {
269 ret(2, 0) = gsl_vector_get (v, 6);
270 ret(2, 1) = gsl_vector_get (v, 7);
271 }
272 else
273 {
274 ret(2, 0) = 0;
275 ret(2, 1) = 0;
276 }
277 ret(2, 2) = 1;
278 ret(2, 3) = 0;
279
280 if (flags & homography::solve_free_rotation)
281 {
282 ret(3, 0) = gsl_vector_get (v, 8);
283 ret(3, 1) = gsl_vector_get (v, 9);
284 }
285 else if ((flags & homography::solve_rotation)
286 && type != lens_move_vertically)
287 {
288 ret(3, 0) = gsl_vector_get (v, 6);
289 ret(3, 1) = gsl_vector_get (v, 7);
290 }
291 else
292 {
293 ret(3, 0) = 0;
294 ret(3, 1) = 0;
295 }
296 ret(3, 2) = 0;
297 ret(3, 3) = 1;
298 }
299 else
300 {
301 ret(0, 0) = gsl_vector_get (v, 0);
302 ret(0, 1) = gsl_vector_get (v, 1);
303 ret(0, 2) = gsl_vector_get (v, 2);
304 ret(0, 3) = 0;
305
306 /* Make y coordiate orthogonal and 1/3th of length of x coordinate.
307 Length of this affect the size of "squares" used by interpolation
308 algorithms. Since there are 3 strips per period we want to have
309 y 1/3 of length. */
310 ret(1, 0) = -gsl_vector_get (v, 1) * 3;
311 ret(1, 1) = gsl_vector_get (v, 0) * 3;
312 ret(1, 2) = 0;
313 ret(1, 3) = 0;
314
315 if ((flags & homography::solve_free_rotation)
316 || (flags & homography::solve_rotation))
317 {
318 ret(2, 0) = gsl_vector_get (v, 3);
319 ret(2, 1) = gsl_vector_get (v, 4);
320 }
321 else
322 {
323 ret(2, 0) = 0;
324 ret(2, 1) = 0;
325 }
326 ret(2, 2) = 1;
327 ret(2, 3) = 0;
328
329 if ((flags & homography::solve_free_rotation)
330 || (flags & homography::solve_rotation))
331 {
332 ret(3, 0) = gsl_vector_get (v, 3);
333 ret(3, 1) = gsl_vector_get (v, 4);
334 }
335 else
336 {
337 ret(3, 0) = 0;
338 ret(3, 1) = 0;
339 }
340 ret(3, 2) = 0;
341 ret(3, 3) = 1;
342 //fprintf (stdout, "Inverse homography\n");
343 //ret.print (stdout);
344 //fprintf (stdout, "Homography\n");
345 ret = ret.invert ();
346 //ret.print (stdout);
347 }
348 if (!keep_vector)
349 gsl_vector_free (v);
350 if (debug_output)
351 ret.print (stdout);
352 td = td.invert ();
353 ret = td * ret;
354 ret = ret * ts;
355 //fprintf (stdout, "Normalized homography %i\n", flags & homography::solve_vertical_strips);
356 //ret.print (stdout);
357
358 /* Make things prettier. There is a redundancy between third and 4th column. */
359 ret(0, 2) += ret(0, 3);
360 ret(0, 3) = 0;
361 ret(1, 3) += ret(1, 2);
362 ret(1, 2) = 0;
363 ret(2, 2) += ret(2, 3);
364 ret(2, 3) = 0;
365 ret(3, 3) += ret(3, 2);
366 ret(3, 2) = 0;
367 for (int x = 0; x < 4; x++)
368 for (int y = 0; y < 4; y++)
369 ret(y, x) /= ret(3, 3);
370 //fprintf (stdout, "Pretty homography\n");
371 //ret.print (stdout);
372 return ret;
373}
374}

376/* Compute square of errors for homography matrix HOMOGRAPHY and POINTS.
377 FLAGS determines the setup of the equations. If TRANSFORMED is non-nullptr
378 initialize it to the set of source points transformed by the resulting
379 homography. */
380
381static double
382screen_compute_chisq (int flags, const std::vector <solver_parameters::solver_point_t> &points, trans_4d_matrix homography, std::vector <point_t> *transformed = nullptr)
383{
384 double chisq = 0;
385 int i = 0;
386 /* For vertical strips compute difference only in X direction. */
387 if (!(flags & homography::solve_vertical_strips))
388 for (auto point : points)
389 {
390 point_t t = homography.inverse_perspective_transform (point.img);
391 if (transformed)
392 (*transformed)[i]=t;
393 i++;
394 coord_t dist = point.scr.dist_sq2_from (t);
395 if (dist > 10000)
396 dist = 10000;
397 chisq += dist;
398 }
399 else
400 for (auto point : points)
401 {
402 point_t t = homography.inverse_perspective_transform (point.img);
403 coord_t dist = (point.scr.x - t.x) * (point.scr.x - t.x);
404 if (transformed)
405 (*transformed)[i]={t.x, point.scr.y};
406 i++;
407 if (dist > 10000)
408 dist = 10000;
409 chisq += dist;
410 }
411 return chisq;
412}

417/* Return homography matrix determined from POINTS using least squares
418 method. If MAP is non-null apply early corrections (such as lens correction).
419 If FLAGS is set to solve_screen_weights or solve_image_weights
420 then adjust weight according to distance from WCENTER_X and WCENTER_Y.
421 If CHISQ_RET is non-NULL initialize it to square of errors. */
422trans_4d_matrix
423get_matrix_ransac (const std::vector <solver_parameters::solver_point_t> &points, int flags,
424 enum scanner_type scanner_type, scr_to_img *map,
425 point_t wcenter, coord_t *chisq_ret,
426 bool final_run)
427{
428 unsigned int seed = 0;
429 int niterations = 500;
430 int nvariables = equation_variables (flags);
431 int eq_per_sample = equations_per_sample (flags);
432 int nsamples = nvariables / eq_per_sample;
433 trans_4d_matrix ret;
434 int max_inliers = 0;
435 double min_chisq = INT_MAX;
436 double min_inlier_chisq = INT_MAX;
437 coord_t dist = 1;
438 int n = points.size ();
439 coord_t scr_dist = 0.1;
440
441 /* Fix non-fixed lens the rotations are specified only by X or Y coordinates.
442 We need enough variables in that system, so just double number of samples.
443 ??? 5 values are enough */
444 if ((flags & homography::solve_rotation) && !is_fixed_lens (scanner_type))
445 nsamples++;
446 gsl_matrix *A = gsl_matrix_alloc (nsamples * eq_per_sample, nvariables);
447 gsl_vector *v = gsl_vector_alloc (nsamples * eq_per_sample);
448 std::vector <solver_parameters::solver_point_t> tpoints_vec;
449 /* Apply non-linear transformations. */
450 if (map)
451 {
452 tpoints_vec.resize (n);
453 for (int i = 0; i < n; i++)
454 {
455 tpoints_vec[i].img = map->apply_early_correction (points[i].img);
456 tpoints_vec[i].scr = points[i].scr;
457 }
458 }
459 const std::vector<solver_parameters::solver_point_t> &tpoints
460 = map ? tpoints_vec : points;
461 if (nsamples > n)
462 {
463 fprintf (stderr, "Too few samples in RANSAC\n");
464 abort ();
465 }
466
467 gsl_error_handler_t *old_handler = gsl_set_error_handler_off ();
468
469 gsl_vector *i_c = nullptr;
470 gsl_matrix *i_cov = nullptr;
471 gsl_multifit_linear_workspace *i_work = nullptr;
472
473 if (nsamples * eq_per_sample != nvariables)
474 {
475 i_c = gsl_vector_alloc (nvariables);
476 i_cov = gsl_matrix_alloc (nvariables, nvariables);
477 i_work = gsl_multifit_linear_alloc (nsamples * eq_per_sample, nvariables);
478 }
479
480 gsl_matrix_free (A);
481 gsl_vector_free (v);
482
483 #pragma omp parallel
484 {
485 unsigned int thread_seed = seed + 123 * omp_get_thread_num ();
486 gsl_matrix *A_local = gsl_matrix_alloc (nsamples * eq_per_sample, nvariables);
487 gsl_vector *v_local = gsl_vector_alloc (nsamples * eq_per_sample);
488 gsl_vector *i_c_local = nullptr;
489 gsl_matrix *i_cov_local = nullptr;
490 gsl_multifit_linear_workspace *i_work_local = nullptr;
491
492 if (nsamples * eq_per_sample != nvariables)
493 {
494 i_c_local = gsl_vector_alloc (nvariables);
495 i_cov_local = gsl_matrix_alloc (nvariables, nvariables);
496 i_work_local = gsl_multifit_linear_alloc (nsamples * eq_per_sample, nvariables);
497 }
498
499 #pragma omp for
500 for (int it = 0; it < niterations; it++)
501 {
502 const int maxsamples = 10;
503 int sample[maxsamples];
504 bool colinear = false;
505 int nattempts = 0;
506 trans_4d_matrix ts;
507 trans_4d_matrix td;
508 do
509 {
510 nattempts++;
511 /* Produce random sample. */
512 for (int i = 0; i < nsamples; i++)
513 {
514 bool ok;
515 do
516 {
517 sample[i] = fast_rand32 (&thread_seed) % n;
518 ok = true;
519 for (int j = 0; j < i; j++)
520 if (sample[i] == sample[j])
521 {
522 ok = false;
523 break;
524 }
525 }
526 while (!ok);
527 }
528
529 /* Normalize input. */
530 normalize_points scrnorm (nsamples), imgnorm (nsamples);
531 if (flags & homography::solve_vertical_strips)
532 for (int i = 0; i < nsamples; i++)
533 {
534 int p = sample[i];
535 scrnorm.account1_xonly (tpoints[p].scr, scanner_type);
536 imgnorm.account1 (tpoints[p].img, scanner_type);
537 }
538 else
539 for (int i = 0; i < nsamples; i++)
540 {
541 int p = sample[i];
542 scrnorm.account1 (tpoints[p].scr, scanner_type);
543 imgnorm.account1 (tpoints[p].img, scanner_type);
544 }
545 scrnorm.finish1 ();
546 imgnorm.finish1 ();
547 if (flags & homography::solve_vertical_strips)
548 for (int i = 0; i < nsamples; i++)
549 {
550 int p = sample[i];
551 scrnorm.account2_xonly (tpoints[p].scr);
552 imgnorm.account2 (tpoints[p].img);
553 }
554 else
555 for (int i = 0; i < nsamples; i++)
556 {
557 int p = sample[i];
558 scrnorm.account2 (tpoints[p].scr);
559 imgnorm.account2 (tpoints[p].img);
560 }
561
562 ts = scrnorm.get_matrix ();
563 td = imgnorm.get_matrix ();
564 /* Produce equations. */
565 for (int i = 0; i < nsamples; i++)
566 {
567 int p = sample[i];
568 init_equation (A_local, v_local, i, false, flags, scanner_type,
569 { tpoints[p].scr }, { tpoints[p].img }, ts, td);
570 }
571 if (!i_work_local)
572 colinear = (gsl_linalg_HH_svx (A_local, v_local) != GSL_SUCCESS);
573 else
574 {
575 double chisq;
576 colinear
577 = (gsl_multifit_linear (A_local, v_local, i_c_local, i_cov_local, &chisq, i_work_local))
578 != GSL_SUCCESS;
579 /* This can not be vector_memcpy since sizes of vectors does not
580 match. */
581 for (int i = 0; i < nvariables; i++)
582 gsl_vector_set (v_local, i, gsl_vector_get (i_c_local, i));
583 }
584 }
585 while (colinear && nattempts < 10000);
586
587 if (nattempts > 10000)
588 continue;
589
590 trans_4d_matrix cur
591 = solution_to_matrix (v_local, flags, scanner_type, false, ts, td, true);
592 int ninliers = 0;
593 double cur_chisq = 0, cur_inlier_chisq = 0;
594 if (!(flags & homography::solve_vertical_strips))
595 {
596 for (int i = 0; i < n; i++)
597 {
598 point_t t = cur.perspective_transform (tpoints[i].scr);
599 cur_chisq += t.dist_sq2_from (tpoints[i].img);
600 if (t.almost_eq (tpoints[i].img, dist))
601 {
602 ninliers++;
603 cur_inlier_chisq += t.dist_sq2_from (tpoints[i].img);
604 }
605 }
606 }
607 else
608 {
609 for (int i = 0; i < n; i++)
610 {
611 point_t s = cur.inverse_perspective_transform (tpoints[i].img);
612 cur_chisq += (tpoints[i].scr.x - s.x) * (tpoints[i].scr.x - s.x);
613 if (my_fabs (tpoints[i].scr.x - s.x) < scr_dist)
614 {
615 ninliers++;
616 cur_inlier_chisq += (tpoints[i].scr.x - s.x) * (tpoints[i].scr.x - s.x);
617 }
618 }
619 }
620 if (ninliers < nsamples)
621 continue;
622
623 #pragma omp critical
624 if ((ninliers > max_inliers)
625 || (ninliers == max_inliers
626 && cur_inlier_chisq < min_inlier_chisq))
627 {
628 ret = cur;
629 max_inliers = ninliers;
630 min_chisq = cur_chisq;
631 min_inlier_chisq = cur_inlier_chisq;
632 }
633 }
634 gsl_matrix_free (A_local);
635 gsl_vector_free (v_local);
636 if (i_c_local)
637 {
638 gsl_multifit_linear_free (i_work_local);
639 gsl_vector_free (i_c_local);
640 gsl_matrix_free (i_cov_local);
641 }
642 }
643
644 gsl_set_error_handler (old_handler);
645 if (i_c)
646 {
647 gsl_multifit_linear_free (i_work);
648 gsl_vector_free (i_c);
649 gsl_matrix_free (i_cov);
650 }
651
652 // if (final)
653 // printf ("Iteration %i inliers %i out of %i, chisq %f inlier chisq %f\n",
654 // iteration, max_inliers, n, min_chisq, min_inlier_chisq);
655 if (max_inliers > nsamples)
656 {

658 gsl_vector *y = gsl_vector_alloc (max_inliers * eq_per_sample);
659 gsl_vector *c = gsl_vector_alloc (nvariables);
660 gsl_matrix *cov = gsl_matrix_alloc (nvariables, nvariables);

663 trans_4d_matrix td;

670 point_t t = ret.perspective_transform (scr);
671 if (tpoints[i].img.almost_eq (t, dist))

675 }
676 }
677 }
678 else

680 for (int i = 0; i < n; i++)

682 point_t img = tpoints[i].img;
683 point_t scr = tpoints[i].scr;
684 point_t s = ret.inverse_perspective_transform (img);
685 if (fabs (scr.x - s.x) < scr_dist)

687 scrnorm.account1_xonly (scr, scanner_type);
688 imgnorm.account1 (tpoints[i].img, scanner_type);
689 }
690 }
691 }
692 scrnorm.finish1 ();
693 imgnorm.finish1 ();
694 if (!(flags & homography::solve_vertical_strips))

697 {
698 point_t scr = tpoints[i].scr;
699 point_t t = ret.perspective_transform (scr);
700 if (tpoints[i].img.almost_eq (t, dist))
701 {
702 scrnorm.account2 (scr);
703 imgnorm.account2 (tpoints[i].img);
704 }
705 }
706 }
707 else
708 {
709 for (int i = 0; i < n; i++)
710 {
711 point_t img = tpoints[i].img;
712 point_t scr = tpoints[i].scr;
713 point_t s = ret.inverse_perspective_transform (img);
714 if (fabs (scr.x - s.x) < scr_dist)
715 {
716 scrnorm.account2_xonly (scr);
717 imgnorm.account2 (tpoints[i].img);
718 }
719 }
720 }

722 td = imgnorm.get_matrix ();

725 if (!(flags & homography::solve_vertical_strips))
726 for (int i = 0; i < n; i++)
727 {
728 point_t scr = tpoints[i].scr;
729 point_t t = ret.perspective_transform (scr);
730 if (tpoints[i].img.almost_eq (t, dist))
731 {

733 tpoints[i].img, ts, td);
734 p++;
735 }
736 }
737 else
738 for (int i = 0; i < n; i++)
739 {
740 point_t img = tpoints[i].img;
741 point_t scr = tpoints[i].scr;
742 point_t s = ret.inverse_perspective_transform (img);
743 if (fabs (scr.x - s.x) < scr_dist)
744 {
745 init_equation (X, y, p, false, flags, scanner_type, scr,
746 tpoints[i].img, ts, td);
747 p++;
748 }
749 }
750 gsl_multifit_linear_workspace *work
751 = gsl_multifit_linear_alloc (n * eq_per_sample, nvariables);
752 gsl_multifit_linear (X, y, c, cov, &min_chisq, work);
753 gsl_multifit_linear_free (work);
754 gsl_matrix_free (X);
755 gsl_vector_free (y);
756 gsl_matrix_free (cov);
757 ret = solution_to_matrix (c, flags, scanner_type, false, ts, td);
758 /* Use screen so we do not get biass with lens correction. */
759 min_chisq = screen_compute_chisq (flags, tpoints, ret);
760 }
761 else if (final_run)
762 printf ("Failed to find inliers for ransac\n");
763 // if (final)
764 // printf ("chisq %f after least squares\n", min_chisq);
765
766 if (chisq_ret)
767 *chisq_ret = min_chisq;
768 return ret;

771/* Return homography matrix determined from POINTS using least squares
772 method. If MAP is non-null apply early corrections (such as lens correction).
773 If FLAGS is set to solve_screen_weights or solve_image_weights
774 then adjust weight according to distance from WCENTER_X and WCENTER_Y.
775 If CHISQ_RET is non-NULL initialize it to square of errors. */
776trans_4d_matrix
777get_matrix (const std::vector <solver_parameters::solver_point_t> &points, int flags,
778 enum scanner_type scanner_type, scr_to_img *map, point_t wcenter,
779 coord_t *chisq_ret,
780 std::vector <point_t> *transformed)
781{
782 int nvariables = equation_variables (flags);
783 int n = points.size ();
784 int eq_per_sample = equations_per_sample (flags);
785 int nequations = n * eq_per_sample;
786 gsl_matrix *X, *cov;
787 gsl_vector *y, *w = NULL, *c;
788 normalize_points scrnorm (n), imgnorm (n);
789 std::vector <solver_parameters::solver_point_t> tpoints_vec;
790 /* Apply non-linear transformations. */
791 if (map)
792 {
793 tpoints_vec.resize (n);
794 /* For lens solving it is faster to avoid precomputing lens inverse. */
795 if (map->early_correction_precomputed ())
796 for (int i = 0; i < n; i++)
797 {
798 tpoints_vec[i].img = map->apply_early_correction (points[i].img);
799 tpoints_vec[i].scr = points[i].scr;
800 }
801 else
802 for (int i = 0; i < n; i++)
803 {
804 tpoints_vec[i].img = map->nonprecomputed_apply_early_correction (points[i].img);
805 tpoints_vec[i].scr = points[i].scr;
806 }
807 }
808 const std::vector<solver_parameters::solver_point_t> &tpoints
809 = map ? tpoints_vec : points;
810 if (flags & homography::solve_vertical_strips)
811 for (int i = 0; i < n; i++)
812 {
813 scrnorm.account1_xonly (tpoints[i].scr, scanner_type);
814 imgnorm.account1 (tpoints[i].img, scanner_type);
815 }
816 else
817 for (int i = 0; i < n; i++)
818 {
819 scrnorm.account1 (tpoints[i].scr, scanner_type);
820 imgnorm.account1 (tpoints[i].img, scanner_type);
821 }
822 scrnorm.finish1 ();
823 imgnorm.finish1 ();
824 if (flags & homography::solve_vertical_strips)
825 for (int i = 0; i < n; i++)
826 {
827 scrnorm.account2_xonly (tpoints[i].scr);
828 imgnorm.account2 (tpoints[i].img);
829 }
830 else
831 for (int i = 0; i < n; i++)
832 {
833 scrnorm.account2 (tpoints[i].scr);
834 imgnorm.account2 (tpoints[i].img);
835 }
836 trans_4d_matrix ts = scrnorm.get_matrix ();
837 trans_4d_matrix td = imgnorm.get_matrix ();
838 coord_t xscale = 10;
839 coord_t yscale = 10;
840 const coord_t img_weight_pow = 2;
841 /* Do not scan screen weights according to scanner geometry:
842 screen coordinates are rotated. */
843 const coord_t screen_weight_scale = 100;
844 const coord_t screen_weight_pow = 2;
845 /* For moving lens, take into account that in one direction geometry changes
846 a lot more than in the other. */
847 if (scanner_type == lens_move_horizontally)
848 xscale = 100;
849 if (scanner_type == lens_move_vertically)
850 yscale = 100;
851 coord_t normscale = 0, sumscale = 0;
852 std::vector<coord_t> weights (
853 (flags & (solve_screen_weights | solve_image_weights)) ? n : 0);
854 if (flags & solve_image_weights)
855 for (int i = 0; i < n; i++)
856 {
857 coord_t dist = my_pow (tpoints[i].img.dist_sq2_from (wcenter, xscale, yscale), img_weight_pow);
858 dist *= dist;
859 double weight = 1 / (dist + 0.5);
860 weights[i] = weight;
861 normscale = std::max (normscale, weight);
862 sumscale += weight;
863 }
864 else if (flags & solve_screen_weights)
865 for (int i = 0; i < n; i++)
866 {
867 coord_t dist = my_pow (tpoints[i].scr.dist_from (wcenter, screen_weight_scale, screen_weight_scale), screen_weight_pow);
868 double weight = 1 / (dist + 0.5);
869 weights[i] = weight;
870 normscale = std::max (normscale, weight);
871 sumscale += weight;
872 }
873 if (flags & (solve_screen_weights | solve_image_weights))
874 {
875 coord_t minscale = sumscale / 100;
876 if (n > 500)
877 {
878 do
879 {
880 nequations = 0;
881 for (int i = 0; i < n; i++)
882 if (weights[i] >= minscale)
883 nequations += eq_per_sample;
884 if (nequations >= eq_per_sample * 100)
885 break;
886 minscale *= (coord_t)(1.0 / 8);
887 }
888 while (true);
889 }
890 else
891 minscale = 0;
892
893 if (normscale != 0)
894 normscale = 1 / normscale;
895 nequations = 0;
896 for (int i = 0; i < n; i++)
897 {
898 if (weights[i] >= minscale)
899 {
900 weights[i] *= normscale;
901 nequations += eq_per_sample;
902 if (!weights[i])
903 abort ();
904 }
905 else
906 weights[i] = 0;
907 }
908 if (nequations * eq_per_sample < nvariables)
909 abort ();
910 w = gsl_vector_alloc (nequations);
911 }
912
913 X = gsl_matrix_alloc (nequations, nvariables);
914 y = gsl_vector_alloc (nequations);
915 c = gsl_vector_alloc (nvariables);
916 cov = gsl_matrix_alloc (nvariables, nvariables);
917
918 for (int i = 0, eq = 0; i < n; i++)
919 {
920 if (w && !weights[i])
921 continue;
922 init_equation (X, y, eq, false, flags, scanner_type, tpoints[i].scr, tpoints[i].img,
923 ts, td);
924
925 if (w && eq_per_sample == 2)
926 {
927 gsl_vector_set (w, eq * 2, weights[i]);
928 gsl_vector_set (w, eq * 2 + 1, weights[i]);
929 }
930 else if (w && eq_per_sample == 1)
931 {
932 gsl_vector_set (w, eq, weights[i]);
933 }
934 eq++;
935 }
936 gsl_multifit_linear_workspace *work
937 = gsl_multifit_linear_alloc (nequations, nvariables);
938 double chisq;
939 if (w)
940 gsl_multifit_wlinear (X, w, y, c, cov, &chisq, work);
941 else
942 gsl_multifit_linear (X, y, c, cov, &chisq, work);
943 gsl_multifit_linear_free (work);
944 gsl_matrix_free (X);
945 gsl_vector_free (y);
946 if (w)
947 gsl_vector_free (w);
948 gsl_matrix_free (cov);
949 trans_4d_matrix ret
950 = solution_to_matrix (c, flags, scanner_type, false, ts, td);
951 /* Use screen so we do not get biass with lens correction. */
952 if (chisq_ret)
953 *chisq_ret = screen_compute_chisq (flags, tpoints, ret, transformed);
954 return ret;
955}

957/* Return homography matrix M.
958 Applying M.perspective_transofrm on (0,0) will yield to ZERO.
959 Applying M.perspective_transofrm on (1,0) will yield to X.
960 Applying M.perspective_transofrm on (0,1) will yield to Y.
961 Applying M.perspective_transofrm on (1,1) will yield to XPY (X+Y).
962 Applying M.perspective_transofrm on (2,3) will yield to TXPY (2*X+3*Y).
963 If INVERT is true then inverse of this transofrmation is computed.
964
965 This works for either scanner type. */
966trans_4d_matrix
967get_matrix_5points (bool invert, enum scanner_type scanner_type, point_t zero,
968 point_t x, point_t y, point_t xpy, point_t txpy) noexcept
969{
970 gsl_matrix *A = gsl_matrix_alloc (10, 10);
971 gsl_vector *v = gsl_vector_alloc (10);
972 normalize_points scrnorm (5), imgnorm (5);
973 scrnorm.account1 ({ 0, 0 }, scanner_type);
974 imgnorm.account1 (zero, scanner_type);
975 scrnorm.account1 ({ 1000, 0 }, scanner_type);
976 imgnorm.account1 (x, scanner_type);
977 scrnorm.account1 ({ 0, 1000 }, scanner_type);
978 imgnorm.account1 (y, scanner_type);
979 scrnorm.account1 ({ 1000, 1000 }, scanner_type);
980 imgnorm.account1 (xpy, scanner_type);
981 scrnorm.account1 ({ 2000, 3000 }, scanner_type);
982 imgnorm.account1 (txpy, scanner_type);
983 scrnorm.finish1 ();
984 imgnorm.finish1 ();
985 scrnorm.account2 ({ 0, 0 });
986 imgnorm.account2 (zero);
987 scrnorm.account2 ({ 1000, 0 });
988 imgnorm.account2 (x);
989 scrnorm.account2 ({ 0, 1000 });
990 imgnorm.account2 (y);
991 scrnorm.account2 ({ 1000, 1000 });
992 imgnorm.account2 (xpy);
993 scrnorm.account2 ({ 2000, 3000 });
994 imgnorm.account2 (txpy);
995 trans_4d_matrix ts = scrnorm.get_matrix ();
996 trans_4d_matrix td = imgnorm.get_matrix ();
997 init_equation (A, v, 0, invert, solve_free_rotation, fixed_lens, { 0, 0 },
998 zero, ts, td);
999 init_equation (A, v, 1, invert, solve_free_rotation, fixed_lens, { 1000, 0 },
1000 x, ts, td);
1001 init_equation (A, v, 2, invert, solve_free_rotation, fixed_lens, { 0, 1000 },
1002 y, ts, td);
1003 init_equation (A, v, 3, invert, solve_free_rotation, fixed_lens,
1004 { 1000, 1000 }, xpy, ts, td);
1005 init_equation (A, v, 4, invert, solve_free_rotation, fixed_lens,
1006 { 2000, 3000 }, txpy, ts, td);
1007
1008 if (debug_output)
1009 print_system (stdout, A, v);
1010 gsl_error_handler_t *old_handler = gsl_set_error_handler_off ();
1011 gsl_linalg_HH_svx (A, v);
1012 gsl_set_error_handler (old_handler);
1013 gsl_matrix_free (A);
1014 return solution_to_matrix (v, solve_free_rotation, fixed_lens, invert, ts,
1015 td);
1016}
