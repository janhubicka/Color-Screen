18/* Parameters for screen initialization. */
19struct screen_params
20{
21 enum scr_type t = Joly;
22 bool preview = false;
23 coord_t red_strip_width = (coord_t)0.0, green_strip_width = (coord_t)0.0;
24 bool anticipate_sharpening = false;
25 sharpen_parameters sharpen = {};
26
27 /* Return true if this structure is equal to O. */
28 bool
29 operator== (const screen_params &o) const
30 {
31 return t == o.t && preview == o.preview
32 && anticipate_sharpening == o.anticipate_sharpening
33 && sharpen == o.sharpen
34 /* We also blur, so we need to compare MTF if used. */
35 && sharpen.scanner_mtf_scale == o.sharpen.scanner_mtf_scale
36 && (!sharpen.scanner_mtf_scale || sharpen.scanner_mtf == o.sharpen.scanner_mtf)
37 && (!screen_with_varying_strips_p (t)
38 || (red_strip_width == o.red_strip_width
39 && green_strip_width == o.green_strip_width));
40 }
41};
42
43/* Return new screen for parameters P. Update PROGRESS. */
44std::unique_ptr<screen>
45get_new_screen (struct screen_params &p, progress_info *progress)
46{
47 auto s = std::make_unique<screen>();
48 if (progress)
49 progress->set_task ("initializing screen", 1);
50 if (p.preview)
51 s->initialize_preview (p.t, p.red_strip_width, p.green_strip_width);
52 else
53 s->initialize (p.t, p.red_strip_width, p.green_strip_width);
54 if (p.sharpen.get_mode () == sharpen_parameters::none && !p.sharpen.usm_radius)
55 return s;
56 auto blurred = std::make_unique<screen>();
57 if (progress)
58 progress->set_task ("blurring screen", 1);
59 if (p.sharpen.scanner_mtf_scale)
60 {
61 /* No need to adjust by screen::size. If p.screen_mtf_scale == screen::size
62 we should scale exactly by it. */
63 sharpen_parameters *vv[3] = {&p.sharpen, &p.sharpen, &p.sharpen};
64 blurred->empty ();
65 blurred->initialize_with_sharpen_parameters (*s, vv, p.anticipate_sharpening);
66 }
67 else
68 blurred->initialize_with_blur (*s, p.sharpen.usm_radius);
69 return blurred;
70}

162/* Initialize screen table for PARAM, TYPE, RED_STRIP_WIDTH, GREEN_STRIP_WIDTH and SHARPEN.
163 Update PROGRESS. */
164screen_table::screen_table (scanner_blur_correction_parameters *param,
165 scr_type type, luminosity_t red_strip_width,
166 luminosity_t green_strip_width,
167 const sharpen_parameters &sharpen,
168 progress_info *progress)
169 : m_id (lru_caches::get ()), m_width (param->get_width ()),
170 m_height (param->get_height ()), m_screen_table (m_width * m_height)
171{
172 screen s;
173 s.initialize (type, red_strip_width, green_strip_width);
174 if (progress)
175 progress->set_task ("computing screen table", m_width * m_height);
176#pragma omp parallel for default(none) shared(progress,sharpen) collapse(2) \
177 shared(param, s)
178 for (int y = 0; y < m_height; y++)
179 for (int x = 0; x < m_width; x++)
180 {
181 if (progress && progress->cancel_requested ())
182 continue;
183 switch (param->get_mode ())
184 {
185 case scanner_blur_correction_parameters::blur_radius:
186 m_screen_table[y * m_width + x].initialize_with_blur (
187 s, param->get_correction (x, y));
188 break;
189 case scanner_blur_correction_parameters::mtf_defocus:
190 {
191 sharpen_parameters sp = sharpen;
192 sp.scanner_mtf.defocus = param->get_correction (x, y);
193 sharpen_parameters *vv[3] = {&sp, &sp, &sp};
194 m_screen_table[y * m_width + x].initialize_with_sharpen_parameters (s, vv, false);
195 break;
196 }
197 case scanner_blur_correction_parameters::mtf_blur_diameter:
198 {
199 sharpen_parameters sp = sharpen;
200 sp.scanner_mtf.blur_diameter = param->get_correction (x, y);
201 sharpen_parameters *vv[3] = {&sp, &sp, &sp};
202 m_screen_table[y * m_width + x].initialize_with_sharpen_parameters (s, vv, false);
203 break;
204 }
205 case scanner_blur_correction_parameters::max_correction:
206 abort ();
207 }
208 if (progress)
209 progress->inc_progress ();
210 }
211}

213/* Initialize saturation loss table for SCREEN_TABLE, COLLECTION_SCREEN, IMG_WIDTH, IMG_HEIGHT, MAP, COLLECTION_THRESHOLD and SHARPEN.
214 Update PROGRESS. */
215saturation_loss_table::saturation_loss_table (
216 screen_table *screen_table, screen *collection_screen, int img_width,
217 int img_height, scr_to_img *map, luminosity_t collection_threshold,
218 const sharpen_parameters &sharpen,
219 progress_info *progress)
220 : m_id (lru_caches::get ()), m_width (screen_table->get_width ()),
221 m_height (screen_table->get_height ()), m_img_width (img_width),
222 m_img_height (img_height),
223 m_xstepinv ((coord_t)m_width / (coord_t)img_width),
224 m_ystepinv ((coord_t)m_height / (coord_t)img_height),
225 m_saturation_loss_table (m_width * m_height)
226{
227 if (progress)
228 progress->set_task ("computing saturation loss table", m_width * m_height);
229#pragma omp parallel for default(none) shared(progress) collapse(2) \
230 shared(screen_table, collection_screen, collection_threshold, map, sharpen)
231 for (int y = 0; y < m_height; y++)
232 for (int x = 0; x < m_width; x++)
233 {
234 if (progress && progress->cancel_requested ())
235 continue;
236 int idx = y * m_width + x;
237 /* No progress here since we compute in parallel. */
238 int xp = (int)((x + (coord_t)0.5) * (coord_t)m_img_width / (coord_t)m_width);
239 int yp = (int)((y + (coord_t)0.5) * (coord_t)m_img_height / (coord_t)m_height);
240 rgbdata cred, cgreen, cblue;
241 if (determine_color_loss (
242 &cred, &cgreen, &cblue, screen_table->get_screen (x, y),
243 /* TODO: No support for adaptive sharpening/blurring of simulated
244 screens yet. */
245 *collection_screen, NULL,
246 collection_threshold, sharpen, *map,
247 {xp - 100, yp - 100, 200, 200}))
248 {
249 color_matrix sat (cred.red, cgreen.red, cblue.red, (luminosity_t)0.0,
250 cred.green, cgreen.green, cblue.green, (luminosity_t)0.0,
251 cred.blue, cgreen.blue, cblue.blue, (luminosity_t)0.0,
252 (luminosity_t)0.0, (luminosity_t)0.0, (luminosity_t)0.0, (luminosity_t)1.0);
253 sat.transpose ();
254 m_saturation_loss_table[idx] = sat.invert ();
255 }
256 else
257 {
258 color_matrix id;
259 m_saturation_loss_table[idx] = id;
260 }
261 if (progress)
262 progress->inc_progress ();
263 }

405/* Initialize TILE to COLOR. */
406static void
407init_to_color (rgbdata color, tile_parameters &tile)
408{
409 color = color.clamp ();
410 int r = invert_gamma (color.red, -1) * 255 + (coord_t)0.5;
411 int g = invert_gamma (color.green, -1) * 255 + (coord_t)0.5;
412 int b = invert_gamma (color.blue, -1) * 255 + (coord_t)0.5;
413
414 for (int y = 0; y < tile.height; y++)
415 for (int x = 0; x < tile.width; x++)
416 {
417 tile.pixels[x * 3 + y * tile.rowstride] = r;
418 tile.pixels[x * 3 + y * tile.rowstride + 1] = g;
419 tile.pixels[x * 3 + y * tile.rowstride + 2] = b;
420 }
421}

423/* Render screen TILE of TYPE for RPARAM, PIXEL_SIZE and RST.
424 Update PROGRESS. */
425DLL_PUBLIC
426bool
427render_screen_tile (tile_parameters &tile, scr_type type,
428 const render_parameters &rparam, coord_t pixel_size,
429 enum render_screen_tile_type rst, progress_info *progress)
430{
431 color_matrix m;
432 sharpen_parameters sp;
433 bool avg = false;
434 if (progress)
435 progress->set_task ("rendering tile", 1);
436 if (rst == dot_spread)
437 {
438 std::shared_ptr<mtf> cur_mtf = mtf::get_mtf (rparam.sharpen.scanner_mtf, NULL);
439 return cur_mtf->render_dot_spread_tile (tile, progress);
440 }
441
442 if (rst >= (int)backlight_screen)
443 {
444 spectrum_dyes_to_xyz s;
445 s.set_backlight (spectrum_dyes_to_xyz::il_D,
446 rparam.backlight_temperature);
447 xyz backlight_white = s.whitepoint_xyz ();
448 xyz_srgb_matrix a;
449 rgbdata backlight;
450 a.apply_to_rgb (backlight_white.x, backlight_white.y, backlight_white.z,
451 &backlight.red, &backlight.green, &backlight.blue);
452 luminosity_t max = (luminosity_t)1.0 / std::max (std::max (backlight.red, backlight.green), backlight.blue);
453 if (type == Random)
454 type = Joly;
455 if (rst == backlight_screen || rst == corrected_backlight_screen)
456 {
457 init_to_color (backlight * max, tile);
458 return true;
459 }
460
461 bool spectrum_based;
462 bool optimized;
463 render_parameters r = rparam;
464 if (rst < (int)corrected_backlight_screen)
465 m = r.get_dyes_matrix (&spectrum_based, &optimized, NULL) * max;
466 else
467 m = r.get_rgb_to_xyz_matrix (NULL, false, patch_proportions (type, &rparam), d65_white) * max;
468 m = a * m;
469 if (rst == full_screen || rst == corrected_full_screen)
470 avg = true;
471 rst = original_screen;
472 }
473 if (type == Random)
474 return false;
475 if (rst != original_screen)
476 {
477 sp = rparam.sharpen;
478 sp.usm_radius *= pixel_size;
479 sp.scanner_mtf_scale *= pixel_size;
480 if (rst != sharpened_screen || sp.mode == sharpen_parameters::none)
481 sp.mode = sharpen_parameters::blur_deconvolution;
482 }
483 std::shared_ptr<screen> scr = render_to_scr::get_screen (
484 type, false, rst == sharpened_screen, sp, rparam.red_strip_width,
485 rparam.green_strip_width, progress);
486 if (!scr)
487 return false;
488 /* For small renders do just one period of screen. For bigger do multiple. */
489 int mult = tile.width > 100 ? 3 : 1;
490 if (!avg)
491 for (int y = 0; y < tile.height; y++)
492 for (int x = 0; x < tile.width; x++)
493 {
494 rgbdata wd
495 = scr->interpolated_mult ({ (coord_t)x * ((coord_t)mult / ((coord_t)tile.width)),
496 (coord_t)y * ((coord_t)mult / ((coord_t)tile.height)) });
497 m.apply_to_rgb (wd.red, wd.green, wd.blue, &wd.red, &wd.green,
498 &wd.blue);
499 wd = wd.clamp ();
500 tile.pixels[x * 3 + y * tile.rowstride]
501 = invert_gamma (wd.red, -1) * 255 + (coord_t)0.5;
502 tile.pixels[x * 3 + y * tile.rowstride + 1]
503 = invert_gamma (wd.green, -1) * 255 + (coord_t)0.5;
504 tile.pixels[x * 3 + y * tile.rowstride + 2]
505 = invert_gamma (wd.blue, -1) * 255 + (coord_t)0.5;
506 }
507 else
508 {
509 rgbdata sum = { (luminosity_t)0.0, (luminosity_t)0.0, (luminosity_t)0.0 };
510 for (int y = 0; y < screen::size; y++)
511 for (int x = 0; x < screen::size; x++)
512 sum += rgbdata{ scr->mult[y][x][0], scr->mult[y][x][1],
513 scr->mult[y][x][2] };
514 sum *= (luminosity_t)1.0 / (luminosity_t)(screen::size * screen::size);
515 m.apply_to_rgb (sum.red, sum.green, sum.blue, &sum.red, &sum.green,
516 &sum.blue);
517 init_to_color (sum, tile);
518 }
519
520 return true;
521}
