25/* Compute the inverse radial distortion for DIST by binary search.
26 PARAM are the lens parameters.
27 MAX is the maximum search radius.
28 INV_MAX_DIST_SQ2 is 1 / (max_dist^2) for normalization.
29 Returns the ratio R / DIST such that corrected_to_scan(R) == DIST. */
30pure_attr inline coord_t
31get_inverse (const lens_warp_correction_parameters &param, coord_t dist,
32 coord_t max, coord_t inv_max_dist_sq2)
33{
34 coord_t min = 0;
35 if (!dist)
36 return 1;
37 while (true)
38 {
39 coord_t r = (min + max) * (coord_t)0.5;
40 coord_t ra = r * param.get_ratio (r * r * inv_max_dist_sq2);
41 if (my_fabs (ra - dist) < 1.0 / (4 * 65536.0) || min == r || max == r)
42 {
43 if (lens_warp_correction::debug
44 && my_fabs (ra - dist) > lens_warp_correction::epsilon / 2)
45 printf ("Inexact lens inverse: %f:%f %f %f %ff\n", dist, ra, r,
46 r / dist, my_sqrt (r * r * inv_max_dist_sq2));
47 return r / dist;
48 }
49 else if (ra < dist)
50 min = r;
51 else
52 max = r;
53 }
54}
55/* Precompute the inverse radial distortion function for parameters P.
56 Provides progress updates via PROG. */
57std::unique_ptr<precomputed_function<coord_t>>
58get_new_inverse (struct lens_inverse_parameters &p, progress_info *prog)
59{
60 (void)prog;
61 coord_t inv_max_dist_sq2 = 1 / (p.max_dist * p.max_dist);
62
63 /* Determine the start of binary search for computing inverse.
64 For correction to make sense, get_ratio must be monotonously increasing
65 for whole image area. For negative correction coefficient the function
66 is not monotonously increasing for large values. So search carefully
67 for max element which inverts to m_max_dist.
68
69 In common case the lens parameters will be normalized so ratio of 1 is 1.
70 In this case max is max_dist. */
71 coord_t max = 1;
72 if (my_fabs (p.param.get_ratio (1) - 1)
73 < lens_warp_correction::epsilon / p.max_dist)
74 max = p.max_dist;
75 else
76 {
77 coord_t last = 0;
78 coord_t next;
79 while ((next = max * p.param.get_ratio (max * max * inv_max_dist_sq2))
80 < p.max_dist)
81 {
82 /* Did we reach point where function decreases now?
83 This means that parameters are broken, but we can cap search and
84 get somewhat sane results. */
85 if (last > next)
86 break;
87 max = 1.2 * max;
88 last = next;
89 }
90 }
91
92 /* Now precompute inverse. */
93 return std::make_unique<precomputed_function<coord_t>> (
94 0, p.max_dist, lens_warp_correction::size,
95 [&] (coord_t x) {
96 return get_inverse (p.param, x, max, inv_max_dist_sq2);
97 },
98 true);
99}
100static lru_cache<lens_inverse_parameters, precomputed_function<coord_t>,
101 get_new_inverse, 4>
102 lens_inverse_cache ("lens inverse functions");
103}

105/* Precompute everything needed to apply lens distortion.
106 CENTER is lens center in image coordinates.
107 C1, C2, C3 and C4 are corners of the scan.
108 Returns true on success. */
109bool
110lens_warp_correction::precompute (point_t center, point_t c1, point_t c2,
111 point_t c3, point_t c4)
112{
113 assert (!m_inverted_ratio);
114 if (m_params.is_noop ())
115 {
116 m_noop = true;
117 return true;
118 }
119 m_noop = false;
120 m_max_dist = std::max (
121 c1.dist_from (center),
122 std::max (c2.dist_from (center),
123 std::max (c3.dist_from (center), c4.dist_from (center))));
124 m_center = center;
125 m_inv_max_dist_sq2 = 1 / (m_max_dist * m_max_dist);
126 return true;
127}

145/* Transform point P from scan coordinates to corrected image coordinates
146 without using the precomputed table. Uses binary search for inversion. */
147pure_attr point_t
148lens_warp_correction::nonprecomputed_scan_to_corrected (point_t p) const
149{
150 if (m_noop)
151 return p;
152 bool too_far = false;
153 coord_t dist = p.dist_from (m_center);
154 coord_t max_search = m_max_dist;
155 if (dist > m_max_dist)
156 dist = m_max_dist, too_far = true;
157
158 point_t ret
159 = (p - m_center)
160 * get_inverse (m_params, dist, max_search, m_inv_max_dist_sq2)
161 + m_center;
162 if (debug && !too_far)
163 {
164 point_t orig = corrected_to_scan (ret);
165 if (!p.almost_eq (orig, epsilon))
166 fprintf (stderr,
167 "Lens correction inverse broken %f, %f -> %f, %f -> %f, "
168 "%f; dist %f ratio %f\n",
169 p.x - m_center.x, p.y - m_center.y, ret.x - m_center.x,
170 ret.y - m_center.y, orig.x - m_center.x, orig.y - m_center.y,
171 dist,
172 m_inverted_ratio
173 ? m_inverted_ratio->apply (p.dist_from (m_center))
174 : (coord_t)-1.0);
175 }
176 return ret;
177}
