25// Helper layout that ensures heightForWidth propagation
26class VerticalSlotLayout : public QVBoxLayout {
27public:
28 using QVBoxLayout::QVBoxLayout;
29 int heightForWidth(int w) const override {
30 // QVBoxLayout implementation of heightForWidth usually sums up its children
31 // but it needs to be explicitly enabled/overridden to work with our SlotWidget
32 if (count() == 0) return 0;
33 int h = spacing() * (count() - 1) + contentsMargins().top() + contentsMargins().bottom();
34 for (int i = 0; i < count(); ++i) {
35 QLayoutItem *item = itemAt(i);
36 if (item->widget()) {
37 if (item->widget()->hasHeightForWidth())
38 h += item->widget()->heightForWidth(w);
39 else
40 h += item->widget()->sizeHint().height();
41 } else if (item->layout()) {
42 if (item->layout()->hasHeightForWidth())
43 h += item->layout()->heightForWidth(w);
44 else
45 h += item->layout()->sizeHint().height();
46 } else {
47 h += item->sizeHint().height();
48 }
49 }
50 return h;
51 }

185QPixmap FinetuneImagesPanel::convertSimpleImageToQPixmap(const colorscreen::simple_image* img) {
186 if (!img || img->width <= 0 || img->height <= 0 || img->stride <= 0) {
187 return QPixmap();
188 }
189 if (img->width > 100000 || img->height > 100000) {
190 return QPixmap();
191 }
192 if (img->stride < img->width * 3) {
193 return QPixmap();
194 }
195
196 QImage qImg(img->width, img->height, QImage::Format_RGB888);
197 for (int y = 0; y < img->height; ++y) {
198 for (int x = 0; x < img->width; ++x) {
199 colorscreen::simple_image::rgb pixel = img->get_pixel(x, y);
200 int r = qBound(0, pixel.red, 255);
201 int g = qBound(0, pixel.green, 255);
202 int b = qBound(0, pixel.blue, 255);
203 qImg.setPixel(x, y, qRgb(r, g, b));
204 }
