101 /* Map screen coordinates to image coordinates. */
102 pure_attr inline point_t
103 to_img (point_t p) const noexcept
104 {
105 if (m_scr_to_img_mesh)
106 return m_scr_to_img_mesh->apply (p);
107 p = m_scr_to_img_homography_matrix.perspective_transform (p);
108 return inverse_early_correction (p);
109 }
110 pure_attr inline bool
111 to_img_in_mesh_range (point_t p) const noexcept
112 {
113 if (!m_scr_to_img_mesh)
114 return true;
115 return m_scr_to_img_mesh->in_range_p (p.x, p.y);
116 }
117 /* Map image coordinates to screen. */
118 pure_attr inline point_t
119 to_scr (point_t p) const noexcept
120 {
121 point_t inp = p;
122 if (m_img_to_scr_mesh)
123 return m_img_to_scr_mesh->apply (p);
124 //if (m_scr_to_img_mesh)
125 //return m_scr_to_img_mesh->invert (p);
126 else
127 {
128 p = apply_early_correction (p);
129 /* For scanners with moving lens the inverse transform may not
130 correspond to homography. */
131 if (m_do_homography)
132 p = m_img_to_scr_homography_matrix.perspective_transform (p);
133 else
134 {
135 p = m_perspective_matrix.inverse_perspective_transform (p);
136 m_inverse_matrix.apply (p.x, p.y, &p.x, &p.y);
137 }
138 }
139
140 /* Verify that inverse is working. */
141 if (debug)
142 {
143 point_t np = to_img (p);
144 if (!np.almost_eq (inp) && m_nwarnings < 10)
145 {
146 printf ("Warning: to_scr is not inverted by to_img %f %f turns to "
147 "%f %f\n",
148 inp.x, inp.y, np.x, np.y);
149 m_nwarnings++;
150 if (colorscreen_checking && 0)
151 abort ();
152 }
153 }
154 return p;
155 }

253 /* Return inverse lens correction for scan point SP. */
254 pure_attr point_t
255 inverse_lens_correction (point_t sp) const noexcept
256 {
257 point_t shift = { 0, 0 };
258 if (m_param.scanner_type == lens_move_horizontally)
259 shift.x = sp.x;
260 if (m_param.scanner_type == lens_move_vertically)
261 shift.y = sp.y;
262 return m_lens_correction.corrected_to_scan (sp - shift) + shift;
263 }

272 shift.y = sp.y;
273 return m_lens_correction.nonprecomputed_scan_to_corrected (sp - shift) + shift;
274 }
