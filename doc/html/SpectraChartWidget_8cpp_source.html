48// Precise Wavelength to RGB conversion using CIE CMFs
49static QColor wavelengthToRGB(double wavelength) {
50 if (wavelength < 380 || wavelength > 780)
51 return QColor(0, 0, 0);
52
53 // Interpolate CMFs
54 double pos = (wavelength - SPECTRUM_START) / (double)SPECTRUM_STEP;
55 int idx = (int)pos;
56 double t = pos - idx;
57
58 if (idx < 0 || idx >= SPECTRUM_SIZE - 1)
59 return QColor(0, 0, 0);
60
61 colorscreen::xyz c = {(colorscreen::luminosity_t)(colorscreen::cie_cmf_x[idx] * (1 - t) +
62 colorscreen::cie_cmf_x[idx + 1] * t),
63 (colorscreen::luminosity_t)(colorscreen::cie_cmf_y[idx] * (1 - t) +
64 colorscreen::cie_cmf_y[idx + 1] * t),
65 (colorscreen::luminosity_t)(colorscreen::cie_cmf_z[idx] * (1 - t) +
66 colorscreen::cie_cmf_z[idx + 1] * t)};
67 colorscreen::luminosity_t r, g, b;
68 (c * 0.15).to_srgb(&r, &g, &b);
69
70 return QColor(std::clamp((int)(r * 255), 0, 255),
71 std::clamp((int)(g * 255), 0, 255),
72 std::clamp((int)(b * 255), 0, 255));
73}
