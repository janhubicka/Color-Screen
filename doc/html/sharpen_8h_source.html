28{

143/* Sharpening worker. Sharpen DATA to OUT which both has dimensions WIDTH*HEIGHT.
144 DATA are accessed using getdata function and PARAM can be used to pass extra data around.
145 RADIUS and AMOUNT are usual parameters of unsharp masking.

147 For performance reasons do not use lambda function since it won't get inlined.
148 O is output type name, T is data type name, P is extra bookeeping parameter type. */
149template<typename O, typename mem_O, typename T,typename P, O (*getdata)(T data, int_point_t p, int width, P param)>
150bool
151sharpen(mem_O *out, T data, P param, int width, int height, luminosity_t radius, luminosity_t amount, progress_info *progress, bool parallel = true)
152{
153 luminosity_t *cmatrix = NULL;
154 int clen;
155 /* Fast path if we do no sharpening. */
156 if (!radius || !amount
157 || (clen = fir_blur::gen_convolve_matrix (radius, &cmatrix)) <= 1)
158 {
159 if (progress)
160 progress->set_task ("converting to linear HDR image", height);
161 if (cmatrix)
162 free (cmatrix);
163 non_sharpen<O,mem_O,T,P,getdata> (out, data, param, width, height, progress);
164 if (progress && progress->cancelled ())
165 return false;
166 return true;
167 }
168 if (progress)
