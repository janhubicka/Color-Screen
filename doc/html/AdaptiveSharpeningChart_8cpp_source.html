156void AdaptiveSharpeningChart::updatePreview()
157{
158 if (m_gridWidth <= 0 || m_gridHeight <= 0) {
159 m_preview = QImage();
160 m_dirty = false;
161 return;
162 }
163
164 QImage img(m_gridWidth, m_gridHeight, QImage::Format_RGB32);
165 img.fill(Qt::black); // Default background if needed, but we overwrite
166
167 double rangeRed = (m_maxRed > m_minRed) ? (m_maxRed - m_minRed) : 1.0;
168 double rangeGreen = (m_maxGreen > m_minGreen) ? (m_maxGreen - m_minGreen) : 1.0;
169 double rangeBlur = (m_maxBlur > m_minBlur) ? (m_maxBlur - m_minBlur) : 1.0;
170
171 for (int y = 0; y < m_gridHeight; y++) {
172 for (int x = 0; x < m_gridWidth; x++) {
173 QColor c = Qt::black; // Uncomputed color
174
175 if (m_mode == Mode_FinalCorrection && m_correction) {
176 double val = m_correction->get_correction(x, y);
177 // Heatmap
178 c = QColor::fromHsvF((1.0 - std::min(val, 2.0)/2.0) * 0.66, 1.0, 1.0);
179 } else if (m_mode == Mode_StripAnalysis || m_mode == Mode_BlurAnalysis) {
180 const Tile &t = m_liveData[y * m_gridWidth + x];
181 if (t.valid) {
182 if (m_mode == Mode_StripAnalysis && t.stripAnalyzed) {
183 // "red strips contributes red color and green strips green"
184 int r = 0;
185 if (m_maxRed > std::numeric_limits<double>::lowest()) {
186 double val = (t.red - m_minRed) / rangeRed;
187 r = std::clamp((int)(val * 255.0), 0, 255);
188 }
189
190 int g = 0;
191 if (m_maxGreen > std::numeric_limits<double>::lowest()) {
192 double val = (t.green - m_minGreen) / rangeGreen;
193 g = std::clamp((int)(val * 255.0), 0, 255);
194 }
195 c = QColor(r, g, 0);
196 } else if (m_mode == Mode_BlurAnalysis && t.blurAnalyzed) {
197 // Blur analysis: grayscale
198 int b = 0;
199 if (m_maxBlur > std::numeric_limits<double>::lowest()) {
200 double val = (t.blur - m_minBlur) / rangeBlur;
201 b = std::clamp((int)(val * 255.0), 0, 255);
202 }
203 c = QColor(b, b, b);
204 } else {
205 // Uncomputed
206 c = palette().color(QPalette::Window);
207 }
208 } else {
209 c = palette().color(QPalette::Window);
210 }
211 }
212 img.setPixelColor(x, y, c);
213 }
214 }
215
216 m_preview = img;
217 m_dirty = false;
218}
