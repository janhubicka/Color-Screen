23/* Matrix performing perspective transformation including tilts and shifting
24 to a given DISTANCE. */
25class rotation_distance_matrix : public matrix4x4<coord_t>
26{
27public:
28 rotation_distance_matrix (coord_t distance, coord_t tilt_x, coord_t tilt_y,
29 enum scanner_type type)
30 {
31 const coord_t rad_x = tilt_x * (coord_t)M_PI / 180;
32 const coord_t rad_y = tilt_y * (coord_t)M_PI / 180;
33 const coord_t sy = std::sin (rad_x); /*s*/
34 const coord_t cy = std::cos (rad_x); /*c*/
35 const coord_t sx = std::sin (rad_y); /*t*/
36 const coord_t cx = std::cos (rad_y); /*d*/
37 /* Rotation matrix is the following:
38 cy sy*sx cx*sy
39 0 cx -sx
40 -sy cy*sx cy*cx
41 We want to rotate plane with center in 0 and stretch the rest. */
42 (*this)(0, 0) = cy;
43 (*this)(0, 1) = sy * sx;
44 (*this)(0, 2) = 0;
45 (*this)(0, 3) = 0;
46 (*this)(1, 0) = 0;
47 (*this)(1, 1) = cx;
48 (*this)(1, 2) = 0;
49 (*this)(1, 3) = 0;
50 (*this)(2, 0) = -sy;
51 (*this)(2, 1) = cy * sx;
52 (*this)(2, 2) = distance;
53 (*this)(2, 3) = 0;
54 (*this)(3, 0) = -sy;
55 (*this)(3, 1) = cy * sx;
56 (*this)(3, 2) = 0;
57 (*this)(3, 3) = distance;
58 /* Disable perspective corrections along the lens movement axis. */
59 if (type == lens_move_horizontally)
60 (*this)(2, 0) = 0, (*this)(2, 1) = 0 /*, (*this)(2, 2)=1*/;
61 if (type == lens_move_vertically)
62 (*this)(3, 0) = 0, (*this)(3, 1) = 0 /*, (*this)(3, 3)=1*/;
63 }
64};

66/* Translate center to given coordinates (x,y). */
67class translation_3x3matrix : public matrix3x3<coord_t>
68{
69public:
70 /* Initialize translation to CENTER. */
71 translation_3x3matrix (point_t center)
72 {
73 (*this)(0, 2) = center.x;
74 (*this)(1, 2) = center.y;
75 }
76};

78/* Change basis to a given coordinate vectors. */
79class change_of_basis_3x3matrix : public matrix3x3<coord_t>
80{
81public:
82 /* Initialize change of basis to C1 and C2. */
83 change_of_basis_3x3matrix (point_t c1, point_t c2)
84 {
85 (*this)(0, 0) = c1.x;
86 (*this)(0, 1) = c2.x;
87 (*this)(1, 0) = c1.y;
88 (*this)(1, 1) = c2.y;
89 }
90};

92/* Rotate by given angle. */
93class rotation_2x2matrix : public matrix2x2<coord_t>
94{
95public:
96 /* Initialize rotation by ROTATION degrees. */
97 rotation_2x2matrix (coord_t rotation)
98 {
99 rotation *= (coord_t)M_PI / 180;
100 const coord_t s = std::sin (rotation);
101 const coord_t c = std::cos (rotation);
102 (*this)(0, 0) = c; (*this)(0, 1) = -s;
103 (*this)(1, 0) = s; (*this)(1, 1) = c;
104 }
105};

197/* Initialize matrices from mapping parameters. */
198void
199scr_to_img::initialize ()
200{
201 /* Now set up the projection matrix that combines tilts and shifting to a
202 * given distance. */
203 rotation_distance_matrix rd (m_param.projection_distance, m_param.tilt_x,
204 m_param.tilt_y, m_param.scanner_type);
205 m_perspective_matrix = rd;
206 /* Center and base vectors are in the scan coordinates. */
207 point_t corrected_center
208 = apply_early_correction (m_param.center);
209 point_t c1
210 = apply_early_correction (m_param.center + m_param.coordinate1);
211 point_t c2
212 = apply_early_correction (m_param.center + m_param.coordinate2);
213
214 corrected_center = m_perspective_matrix.inverse_perspective_transform (corrected_center);
215 c1 = m_perspective_matrix.inverse_perspective_transform (c1);
216 c2 = m_perspective_matrix.inverse_perspective_transform (c2);
217 c1 -= corrected_center;
218 c2 -= corrected_center;
219
220 /* Change-of-basis matrix. */
221 trans_3d_matrix mm;
222 change_of_basis_3x3matrix basis (c1, c2);
223 translation_3x3matrix trans (corrected_center);
224 mm = basis * mm;
225 mm = trans * mm;
226
227 m_matrix = mm;
228 m_inverse_matrix = m_matrix.invert ();
229
230 /* TODO: We support homography matrix only for true homography
231 transformations (which maps lines to lines). If scanner has moving lens,
232 straight lines are not preserved. However we probably still do 6 point
233 based matrix rather than manually doing direct and inverse transforms. */
234 {
235 m_do_homography = (m_param.scanner_type == fixed_lens
236 || (m_param.tilt_x == 0 && m_param.tilt_y == 0));
237 point_t zero;
238 point_t x;
239 point_t y;
240 point_t xpy;
241 point_t txpy;
242 coord_t tx, ty;
243 m_matrix.apply (0, 0, &tx, &ty);
244 zero = m_perspective_matrix.perspective_transform ({ tx, ty });
245 m_matrix.apply (1000, 0, &tx, &ty);
246 x = m_perspective_matrix.perspective_transform ({ tx, ty });
247 m_matrix.apply (0, 1000, &tx, &ty);
248 y = m_perspective_matrix.perspective_transform ({ tx, ty });
249 m_matrix.apply (1000, 1000, &tx, &ty);
250 xpy = m_perspective_matrix.perspective_transform ({ tx, ty });
251 m_matrix.apply (2000, 3000, &tx, &ty);
252 txpy = m_perspective_matrix.perspective_transform ({ tx, ty });
253 m_scr_to_img_homography_matrix = homography::get_matrix_5points (
254 false, m_param.scanner_type, zero, x, y, xpy, txpy);
255 if (m_do_homography)
256 m_img_to_scr_homography_matrix = homography::get_matrix_5points (
257 true, m_param.scanner_type, zero, x, y, xpy, txpy);
258#if 0
259 printf ("Matrix\n");
260 m_matrix.print (stdout);
261 printf ("Perspective\n");
262 m_perspective_matrix.print (stdout);
263 printf ("Homography\n");
264 m_scr_to_img_homography_matrix.print (stdout);
265#endif
266 if (debug)
267 {
268 bool found = false;
269 for (int x = 0; x <= 1000 && !found; x += 10)
270 for (int y = 0; y <= 1000 && !found; y += 10)
271 {
272 coord_t px, py;
273 coord_t xt, yt;
274 m_matrix.apply (x, y, &px, &py);
275 point_t pt = m_perspective_matrix.perspective_transform ({ px, py });
276 px = pt.x; py = pt.y;
277 point_t xtyt = m_scr_to_img_homography_matrix.perspective_transform ({ (coord_t) x, (coord_t) y });
278 xt = xtyt.x; yt = xtyt.y;
279 if (fabs (px - xt) > 1 || fabs (py - yt) > 1)
280 {
281 printf ("scr-to-img forward mismatch %i %i: %f %f should be "
282 "%f %f\n",
283 x, y, px, py, xt, yt);
284 found = true;
285 }
286 if (!m_do_homography)
287 continue;
288 coord_t bx, by;
289 point_t bxby = m_img_to_scr_homography_matrix.perspective_transform ({ px, py });
290 bx = bxby.x; by = bxby.y;
291 if (fabs (x - bx) > 1 || fabs (y - by) > 1)
292 {
293 printf ("scr-to-img backward mismatch %i %i: %f %f should "
294 "be %f %f\n",
295 x, y, px, py, bx, by);
296 found = true;
297 }
298 }
299 }
300 }
301
302 update_scr_to_final_parameters (m_param.final_ratio, m_param.final_angle);
303}

305/* Set parameters of the early correction for image of dimensions WIDTH x HEIGHT.
306 Return true on success. */
307bool
308scr_to_img::set_parameters_for_early_correction (
309 const scr_to_img_parameters &param,
310 int width, int height)
311{
312 m_param = param;
313
314 if (m_param.mesh_trans)
315 {
316 if (m_param.mesh_trans_is_scr_to_img)
317 m_scr_to_img_mesh = m_param.mesh_trans;
318 else
319 m_img_to_scr_mesh = m_param.mesh_trans;
320 return true;
321 }
322 else
323 m_scr_to_img_mesh = nullptr;
324 m_img_to_scr_mesh = nullptr;
325
326 m_inverted_projection_distance = (coord_t)1 / m_param.projection_distance;
327 m_nwarnings = 0;
328 assert (!debug || (width > 0 && height > 0));
329
330 /* Next initialize lens correction.
331 Lens center is specified in scan coordinates, so apply previous
332 corrections. */
333 point_t center = { m_param.lens_correction.center.x * width,
334 m_param.lens_correction.center.y * height};
335 point_t c1 = { (coord_t)0, (coord_t)0 };
336 point_t c2 = { (coord_t)width, (coord_t)0 };
337 point_t c3 = { (coord_t)0, (coord_t)height };
338 point_t c4 = { (coord_t)width, (coord_t)height };
339 m_lens_correction.set_parameters (m_param.lens_correction);
340 if (m_param.scanner_type == lens_move_horizontally)
341 c1.x = c2.x = c3.x = c4.x = center.x = m_param.lens_correction.center.x;
342 if (m_param.scanner_type == lens_move_vertically)
343 c1.y = c2.y = c3.y = c4.y = center.y = m_param.lens_correction.center.y;
344 if (!m_lens_correction.precompute (center, c1, c2, c3, c4))
345 return false;
346 /* For non-noop conversion we need to also precompute inverse. */
347 m_early_correction_precomputed = m_lens_correction.is_noop ();
348 return true;
349}

351/* Set all parameters for image of dimensions WIDTH x HEIGHT.
352 ROTATION_ADJUSTMENT can be used to adjust the rotation.
353 Return true on success. */
354bool
355scr_to_img::set_parameters (const scr_to_img_parameters &param,
356 int width, int height, coord_t rotation_adjustment)
357{
358 m_rotation_adjustment = rotation_adjustment;
359 if (!scr_to_img::set_parameters_for_early_correction (param, width, height))
360 return false;
361
362 if (m_scr_to_img_mesh)
363 {
364 int_image_area area = {0, 0, width, height};
365 m_img_to_scr_mesh = m_scr_to_img_mesh->compute_inverse (area);
366 if (!m_img_to_scr_mesh)
367 return false;
368 update_scr_to_final_parameters (m_param.final_ratio, m_param.final_angle);
369 return true;
370 }
371 if (m_img_to_scr_mesh)
372 {
373 m_scr_to_img_mesh = m_img_to_scr_mesh->compute_inverse ();
374 if (!m_scr_to_img_mesh)
375 return false;
376 update_scr_to_final_parameters (m_param.final_ratio, m_param.final_angle);
377 return true;
378 }
379
380 if (!m_lens_correction.precompute_inverse ())
381 return false;
382 m_early_correction_precomputed = true;
383 initialize ();
384 return true;
385}

396/* Determine rectangular section of the screen to which the whole image
397 with dimension img_width x img_height fits.
398
399 The section is returned as an int_image_area where x/y is the minimum
400 screen coordinate and width/height cover all screen coordinates mapped
401 from the image. */
402image_area
403scr_to_img::get_range (image_area area_in) const noexcept
404{
405 if (m_scr_to_img_mesh)
406 {
407 matrix2x2<coord_t> identity;
408 if (m_param.mesh_trans_is_scr_to_img)
409 return m_scr_to_img_mesh->get_range (identity, area_in);
410 else
411 return m_img_to_scr_mesh->get_src_range (area_in);
412 }
413 else
414 {
415 image_area area;
416 coord_t x1 = area_in.x;
417 coord_t y1 = area_in.y;
418 coord_t x2 = area_in.x + area_in.width;
419 coord_t y2 = area_in.y + area_in.height;
420
421 area.extend (to_scr ({ x1, y1 }));
422 area.extend (to_scr ({ x2, y1 }));
423 area.extend (to_scr ({ x1, y2 }));
424 area.extend (to_scr ({ x2, y2 }));
425
426 /* If we correct lens distortion the corners may not be extremes. */
427 if (!m_lens_correction.is_noop () || m_param.tilt_x || m_param.tilt_y)
428 {
429 const int steps = 16 * 1024;
430 for (int i = 1; i < steps; i++)
431 {
432 area.extend (to_scr ({ x1 + (x2 - x1) * i / steps, y1 }));
433 area.extend (to_scr ({ x1 + (x2 - x1) * i / steps, y2 }));
434 area.extend (to_scr ({ x1, y1 + (y2 - y1) * i / steps }));
435 area.extend (to_scr ({ x2, y1 + (y2 - y1) * i / steps }));
436 }
437 }
438 /* Adjust slightly to be safe. */
439 area.x -= 1;
440 area.y -= 1;
441 area.width += 2;
442 area.height += 2;
443 return area;
444 }
445}

458 if (m_scr_to_img_mesh)

462 image_area area;
463 coord_t x1 = area_in.x;
464 coord_t y1 = area_in.y;
465 coord_t x2 = area_in.x + area_in.width;
466 coord_t y2 = area_in.y + area_in.height;
467
468 area.extend (img_to_final ({ x1, y1 }));
469 area.extend (img_to_final ({ x2, y1 }));
470 area.extend (img_to_final ({ x1, y2 }));
471 area.extend (img_to_final ({ x2, y2 }));
472
473 /* If we correct lens distortion the corners may not be extremes. */
474 if (!m_lens_correction.is_noop () || m_param.tilt_x || m_param.tilt_y
475 || m_scr_to_img_mesh)
476 {
477 const int steps = 16 * 1024;
478 for (int i = 1; i < steps; i++)
479 {
480 area.extend (img_to_final ({ x1 + (x2 - x1) * i / steps, y1 }));
481 area.extend (img_to_final ({ x1 + (x2 - x1) * i / steps, y2 }));
482 area.extend (img_to_final ({ x1, y1 + (y2 - y1) * i / steps }));
483 area.extend (img_to_final ({ x2, y1 + (y2 - y1) * i / steps }));
484 }
485 }
486 return area;
487 }

548/* Estimate DPI for given pixel size. */
549coord_t
550scr_to_img_parameters::estimate_dpi (coord_t pixel_size) const
551{
552 switch (type)
553 {
554 case Paget:
555 case Finlay:
556 /* Screen size scanned by Epson scanner at 1800 DPI. */
557 /*return 1800 * ((1/sqrt (13.357325*13.357325 + 0.203263 * 0.203263)) / pixel_size);*/
558 /* Screen size scanned by Epson scanner at 2400 DPI. */
559 return 2400 * ((1/sqrt (19.776270*19.776270 + -0.119933*-0.119933)) / pixel_size);
560 case Dufay:
561 /* Screen size scanned by Nikon scanner at 4000 DPI. */
562 return 4000 * ((1/sqrt (3.045845*3.045845 + 7.194771*7.194771)) / pixel_size);
563 default:
564 return -1;
565 break;
566 }
567}
568/* Estimate pixel size for area. */
569pure_attr coord_t
570scr_to_img::pixel_size (int_image_area area) const noexcept
571{
572 histogram hist;
573 int steps = 11;
574 for (int y = 0; y < 5; y++)
575 for (int x = 0; x < 5; x++)
576 {
577 coord_t bx = area.x + (x+(coord_t)0.5) * area.width * (1 / (coord_t)steps);
578 coord_t by = area.y + (y+(coord_t)0.5) * area.height * (1 / (coord_t)steps);
579 coord_t sz = to_scr ({ bx + 0, by + 0 }).dist_from (to_scr ({ bx + 1, by + 0 }));
580 hist.pre_account (sz);
581 }
582 hist.finalize_range (256);
583 for (int y = 0; y < 5; y++)
584 for (int x = 0; x < 5; x++)
585 {
586 coord_t bx = area.x + (x+(coord_t)0.5) * area.width * (1 / (coord_t)steps);
587 coord_t by = area.y + (y+(coord_t)0.5) * area.height * (1 / (coord_t)steps);
588 coord_t sz = to_scr ({ bx + 0, by + 0 }).dist_from (to_scr ({ bx + 1, by + 0 }));
589 hist.account (sz);
590 }
591 hist.finalize ();
592 return hist.find_avg (0.2, 0.2);

597scr_to_img::get_img_range (int_image_area a) const noexcept
598{
599 coord_t x1 = a.x;
600 coord_t y1 = a.y;
601 coord_t x2 = a.x + a.width - 1;
602 coord_t y2 = a.y + a.height - 1;
603 if (!m_scr_to_img_mesh && !m_img_to_scr_mesh)
604 {
605 /* Compute all the corners. */
606 int_image_area area (int_point_t {(int64_t)my_floor (to_img ({ x1, y1 }).x), (int64_t)my_floor (to_img ({ x1, y1 }).y)});
607 area.extend (int_point_t {(int64_t)my_ceil (to_img ({ x1, y1 }).x), (int64_t)my_ceil (to_img ({ x1, y1 }).y)});
608 area.extend (int_point_t {(int64_t)my_floor (to_img ({ x2, y1 }).x), (int64_t)my_floor (to_img ({ x2, y1 }).y)});
609 area.extend (int_point_t {(int64_t)my_ceil (to_img ({ x2, y1 }).x), (int64_t)my_ceil (to_img ({ x2, y1 }).y)});
610 area.extend (int_point_t {(int64_t)my_floor (to_img ({ x1, y2 }).x), (int64_t)my_floor (to_img ({ x1, y2 }).y)});
611 area.extend (int_point_t {(int64_t)my_ceil (to_img ({ x1, y2 }).x), (int64_t)my_ceil (to_img ({ x1, y2 }).y)});
612 area.extend (int_point_t {(int64_t)my_floor (to_img ({ x2, y2 }).x), (int64_t)my_floor (to_img ({ x2, y2 }).y)});
613 area.extend (int_point_t {(int64_t)my_ceil (to_img ({ x2, y2 }).x), (int64_t)my_ceil (to_img ({ x2, y2 }).y)});
614
615 /* If we correct lens distortion the corners may not be extremes. */
616 if (!m_lens_correction.is_noop () || m_param.tilt_x || m_param.tilt_y)
617 {
618 const int steps = 16 * 1024;
619 for (int i = 1; i < steps; i++)
620 {
621 point_t p = to_img ({ x1 + (x2 - x1) * i / steps, y1 });
622 area.extend (int_point_t {(int64_t)my_floor (p.x), (int64_t)my_floor (p.y)});
623 area.extend (int_point_t {(int64_t)my_ceil (p.x), (int64_t)my_ceil (p.y)});
624 p = to_img ({ x1 + (x2 - x1) * i / steps, y2 });
625 area.extend (int_point_t {(int64_t)my_floor (p.x), (int64_t)my_floor (p.y)});
626 area.extend (int_point_t {(int64_t)my_ceil (p.x), (int64_t)my_ceil (p.y)});
627 p = to_img ({ x1, y1 + (y2 - y1) * i / steps });
628 area.extend (int_point_t {(int64_t)my_floor (p.x), (int64_t)my_floor (p.y)});
629 area.extend (int_point_t {(int64_t)my_ceil (p.x), (int64_t)my_ceil (p.y)});
630 p = to_img ({ x2, y1 + (y2 - y1) * i / steps });
631 area.extend (int_point_t {(int64_t)my_floor (p.x), (int64_t)my_floor (p.y)});
632 area.extend (int_point_t {(int64_t)my_ceil (p.x), (int64_t)my_ceil (p.y)});
633 }
634 }
635 /* Adjust slightly to be safe. */
636 area.x -= 1;
637 area.y -= 1;
638 area.width += 2;
639 area.height += 2;
640 return area;
641 }
642 else
643 {
644 matrix2x2<coord_t> identity;
645 if (m_param.mesh_trans_is_scr_to_img)
646 return int_image_area (m_scr_to_img_mesh->get_src_range (image_area (a)));
647 else
648 return int_image_area (m_img_to_scr_mesh->get_range (identity, image_area (a)));
649 }
650}
