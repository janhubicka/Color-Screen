113/* Return portion of screen occupied by red, green and blue patches for
114 screen of type T and rendering parameters RPARAM. The sum of individual
115 portions should be at most 1. */
116rgbdata
117patch_proportions (enum scr_type t, const render_parameters *rparam)
118{
119 switch (t)
120 {
121 case Paget:
122 case Thames:
123 /* The Paget Plate - This is believed was invention of C. Finlay
124 and consits of square elements, the blue being 0.063mm. with
125 0.085mm. for the red and green. It is stated by J. H. Pledge^20
126 to be roughly blue 8, red 7, green 7 there being two blue
127 elements in the unit. */
128 return {7/22.0,7/22.0,8/22.0};
129 case Finlay:
130 /* TODO: Measure actual portions. */
131 return {1/3.0,1/3.0,1/3.0};
132 case WarnerPowrie:
133 case Joly:
134 /* TODO: Measure actual portions. */
135 {
136 coord_t red_strip_width = rparam && rparam->red_strip_width ? rparam->red_strip_width : 1/3.0;
137 coord_t green_strip_width = rparam && rparam->green_strip_width ? rparam->green_strip_width : 1/3.0;
138 return {(luminosity_t)red_strip_width, (luminosity_t)green_strip_width, (luminosity_t)(1 - red_strip_width - green_strip_width)};
139 }
140
141 /* Red strips. */
142 case Dufay:
143 {
144 coord_t red_strip_width = rparam && rparam->red_strip_width ? rparam->red_strip_width : dufaycolor::red_strip_width;
145 coord_t green_strip_width = rparam && rparam->green_strip_width ? rparam->green_strip_width : dufaycolor::green_strip_width;
146 return {(luminosity_t)red_strip_width, (luminosity_t)(green_strip_width * (1 - red_strip_width)), (luminosity_t)((1 - green_strip_width) * (1 - red_strip_width))};
147 }
148 /* Green strips. */
149 case DioptichromeB:
150 {
151 /* On scan of separate DiopticrhomeB screen (possibly early test sample) we get approx
152 red_strip_width: 0.489965
153 green_strip_width: 0.499510
154 So it seems that both strips were 50%. */
155
156 luminosity_t red_strip_width = rparam && rparam->red_strip_width ? rparam->red_strip_width : 0.5;
157 luminosity_t green_strip_width = rparam && rparam->green_strip_width ? rparam->green_strip_width : 0.5;
158 return {(1 - green_strip_width) * red_strip_width, green_strip_width, (1 - green_strip_width) * (1 - red_strip_width)};
159 }
160 /* Blue strips. */
161 case ImprovedDioptichromeB:
162 {
163 luminosity_t red_strip_width = rparam && rparam->red_strip_width ? rparam->red_strip_width : 0.66;
164 luminosity_t green_strip_width = rparam && rparam->green_strip_width ? rparam->green_strip_width : 0.5;
165 return {red_strip_width * (1 - green_strip_width), green_strip_width * red_strip_width, 1 - red_strip_width};
166 }
167 /* Blue strips. */
168 case Omnicolore:
169 {
170 luminosity_t red_strip_width = rparam && rparam->red_strip_width ? rparam->red_strip_width : 0.69;
171 luminosity_t green_strip_width = rparam && rparam->green_strip_width ? rparam->green_strip_width : 0.55;
172 return {red_strip_width * (1 - green_strip_width), green_strip_width * red_strip_width, 1 - red_strip_width};
173 }
174 default:
175 abort ();
176 }
177}

644/* Return matrix which contains the color of dyes either as RGB or XYZ.
645 IMG is the image being rendered.
646 NORMALIZED_PATCHES is true if the RGB data represents patch intensities
647 regardless of their size.
648 PATCH_PROPORTIONS describes how much percent of screen is occupied by red,
649 green and blue patches respectively.
650 TARGET_WHITEPOINT is the whitepoint of the target color space. */
651
652color_matrix
653render_parameters::get_balanced_dyes_matrix (const image_data *img, bool normalized_patches, rgbdata patch_proportions, xyz target_whitepoint) const
654{
655 bool optimized;
656 bool spectrum_based;
657 bool correct_whitepoints = true;
658 color_matrix dyes = get_dyes_matrix (&spectrum_based, &optimized, img);
659
660 /* If dyes are normalised, we need to scale primaries to match their proportions in actual screen. */
661 if (normalized_patches)
662 dyes.scale_channels (patch_proportions.red, patch_proportions.green, patch_proportions.blue);
663 dyes.verify_last_row_0001 ();
664
665 if (apply_balance_to_model (color_model))
666 {
667 /* Determine whitepoint of the screen. For normalized patches it is {1, 1, 1} since color screens
668 should be neutral. */
669 rgbdata screen_whitepoint = {1, 1, 1};
670 if (!normalized_patches)
671 screen_whitepoint = patch_proportions;
672
673 /* Determine actual whitepoint of the screen. */
674 xyz dye_whitepoint;
675 dyes.apply_to_rgb (screen_whitepoint.red, screen_whitepoint.green, screen_whitepoint.blue, &dye_whitepoint.x, &dye_whitepoint.y, &dye_whitepoint.z);
676
677 /* Different dye balances. */
678 switch (dye_balance)
679 {
680 case render_parameters::dye_balance_none:
681 break;
682
683 /* Bradford correct the white of color screen into target whitepoint. */
684 case render_parameters::dye_balance_bradford:
685 dyes = bradford_whitepoint_adaptation_matrix (dye_whitepoint, target_whitepoint) * dyes;
686 /* Save one step and go directly to target white. */
687 correct_whitepoints = false;
688 break;
689
690 /* Scale so y of screen white is 1. */
691 case render_parameters::dye_balance_brightness:
692 if (dye_whitepoint.y > 0)
693 {
694 dyes = dyes * (1/dye_whitepoint.y);
695 dyes(3, 3) = 1;
696 }
697 break;
698
699 /* Scale intensity of dyes so they produce given whitepoint. This correspond to adjusting
700 sizes of color patches in the emulsion. */
701 case render_parameters::dye_balance_neutral:
702 {
703 xyz white = observer_whitepoint;
704 rgbdata scales;
705 dyes.invert ().apply_to_rgb (white.x, white.y, white.z, &scales.red, &scales.green, &scales.blue);
706 scales /= screen_whitepoint;
707 dyes.scale_channels (scales.red, scales.green, scales.blue);
708 }
709 break;
710
711 /* Scale final XYZ values to obtain whitepoint. This is probably always
712 worse than Bradford correction. */
713 case render_parameters::dye_balance_whitepoint:
714 for (int i = 0; i < 4; i++)
715 {
716 xyz white = observer_whitepoint;
717 dyes(0, i) *= white.x / dye_whitepoint.x;
718 dyes(1, i) *= white.y / dye_whitepoint.y;
719 dyes(2, i) *= white.z / dye_whitepoint.z;
720 }
721 break;
722 default:
723 abort ();
724 }
725 }
726
727 dyes.verify_last_row_0001 ();
728 /* After balancing to observer whitepoint bradford correct to target whitepoint. */
729 //printf (" Observer:");
730 //((xyz)observer_whitepoint).print (stdout);
731 //print_mid_white (dyes);
732 if (correct_whitepoints && (xyz)observer_whitepoint != target_whitepoint)
733 {
734 dyes = bradford_whitepoint_adaptation_matrix ((xyz)observer_whitepoint, target_whitepoint) * dyes;
735 dyes.verify_last_row_0001 ();
736 }
737 /* Don't do this as part of early adjustments, it confuses dark point of optimized matrices. */
738 color_matrix br (brightness, 0, 0, 0,
739 0, brightness, 0, 0,
740 0, 0, brightness, 0,
741 0, 0, 0, 1);
742 dyes = br * dyes;
743 //printf (" Target:");
744 //target_whitepoint.print (stdout);
745 //print_mid_white (dyes);
746 return dyes;
747}
