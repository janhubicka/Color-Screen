108 static int
109 f (const gsl_vector *v, void *params, gsl_vector *f_vec)
110 {
111 gsl_solver_proxy *proxy = static_cast<gsl_solver_proxy *> (params);
112 int p = proxy->temp_params.size ();
113 int n = f_vec->size;
114
115 for (int i = 0; i < p; i++)
116 proxy->temp_params[i] = gsl_vector_get (v, i);
117
118 proxy->c.constrain (proxy->temp_params.data ());
119
120 if (proxy->temp_residuals.size() != (size_t)n)
121 proxy->temp_residuals.resize(n);
122
123 auto res = proxy->c.residuals (proxy->temp_params.data (), proxy->temp_residuals.data ());
124
125 for (int i = 0; i < n; i++)
126 gsl_vector_set (f_vec, i, proxy->temp_residuals[i]);
127
128 if constexpr (std::is_same_v<decltype(res), int>)
129 return res;
130
131 return GSL_SUCCESS;
132 }
