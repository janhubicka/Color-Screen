74
75 /* Compensate saturation loss for pixel C at position P in image coordinates. */
76 pure_attr rgbdata
77 compensate_saturation_loss_img (point_t p, rgbdata c) const
78 {
79 int x, y;
80 coord_t rx = my_modf (p.x * m_xstepinv - (coord_t)0.5, &x);
81 coord_t ry = my_modf (p.y * m_ystepinv - (coord_t)0.5, &y);
82 if (x < 0)
83 x = 0, rx = (coord_t)0;
84 if (y < 0)
85 y = 0, ry = (coord_t)0;
86 if (x >= m_width - 1)
87 x = m_width - 2, rx = (coord_t)1;
88 if (y >= m_height - 1)
89 y = m_height - 2, ry = (coord_t)1;
90 rgbdata c00, c01, c10, c11;
91 m_saturation_loss_table [y * m_width + x].apply_to_rgb (c.red, c.green, c.blue, &c00.red, &c00.green, &c00.blue);
92 m_saturation_loss_table [y * m_width + x + 1].apply_to_rgb (c.red, c.green, c.blue, &c10.red, &c10.green, &c10.blue);
93 m_saturation_loss_table [(y + 1) * m_width + x].apply_to_rgb (c.red, c.green, c.blue, &c01.red, &c01.green, &c01.blue);
94 m_saturation_loss_table [(y + 1) * m_width + x + 1].apply_to_rgb (c.red, c.green, c.blue, &c11.red, &c11.green, &c11.blue);
95
96 return (c00 * ((coord_t)1.0 - rx) + c10 * rx) * ((coord_t)1.0 - ry)
97 + (c01 * ((coord_t)1.0 - rx) + c11 * rx) * ry;
98 }
99private:
100 /* Unique id of the image (used for caching). */
101 uint64_t m_id;
102 int m_width, m_height;
103 int m_img_width, m_img_height;
104 coord_t m_xstepinv, m_ystepinv;
105 std::vector <color_matrix> m_saturation_loss_table;
106};

487/* Determine image pixel X,Y in screen filter SCR using MAP.
488 Do antialiasing. Return screen position in RETP. */
489pure_attr inline rgbdata
490antialias_screen (const screen &scr, const scr_to_img &map,
491 int x, int y, point_t *retp = NULL) noexcept
492{
493 point_t p = map.to_scr ({ x + (coord_t)0.5, y + (coord_t)0.5 });
494 point_t px = map.to_scr ({ x + (coord_t)1.5, y + (coord_t)0.5 });
495 point_t py = map.to_scr ({ x + (coord_t)0.5, y + (coord_t)1.5 });
496 rgbdata am = { (luminosity_t)0.0, (luminosity_t)0.0, (luminosity_t)0.0 };
497 point_t pdx = (px - p) * ((coord_t)1.0 / (coord_t)6.0);
498 point_t pdy = (py - p) * ((coord_t)1.0 / (coord_t)6.0);
499 if (retp)
500 *retp = p;
501
502 for (int yy = -2; yy <= 2; yy++)
503 for (int xx = -2; xx <= 2; xx++)
504 am += scr.interpolated_mult (p + pdx * (coord_t)xx + pdy * (coord_t)yy);
505 return am * ((coord_t)1.0 / (coord_t)25);
506}

513{
514 point_t p = map.to_scr ({ x + (coord_t)0.5, y + (coord_t)0.5 });
