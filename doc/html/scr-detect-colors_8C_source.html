58 const bitmap_2d *known_patches, luminosity_t gamma,
59 progress_info *progress, FILE *report)

150/* Optimize screen colors using detected color patches in image IMG.
151 AREA is the area to optimize. */
152bool
153optimize_screen_colors (scr_detect_parameters *param, const image_data *img,
154 luminosity_t gamma, int_image_area area,
155 progress_info *progress, FILE *report)
156{
157 const double sharpen_amount = 0;
158 const double sharpen_radius = 3;
159 int clen = fir_blur::convolve_matrix_length (sharpen_radius);
160 mem_rgbdata *sharpened = (mem_rgbdata *)malloc (
161 (area.width + clen) * (area.height + clen) * sizeof (mem_rgbdata));
162 if (!sharpened)
163 return false;
164 std::shared_ptr<float[]> lookup_table[3];
165 if (!render::get_lookup_tables (lookup_table, gamma, img, progress))
166 return false;
167 struct imgtile section = { { lookup_table[0].get (), lookup_table[1].get (),
168 lookup_table[2].get () },
169 area.x - clen / 2,
170 area.y - clen / 2,
171 img };
172 sharpen<rgbdata, mem_rgbdata, imgtile *, int, &get_pixel> (
173 sharpened, &section, 0, area.width + clen, area.height + clen,
174 sharpen_radius, sharpen_amount, progress);
175 std::vector<entry> pixels;
176 for (int yy = area.y; yy < area.y + area.height; yy++)
177 for (int xx = area.x; xx < area.x + area.width; xx++)
178 {
179 struct entry e;
180 e.orig_color
181 = get_pixel (&section, { xx - area.x + clen / 2, yy - area.y + clen / 2 },
182 0, 0);
183 e.sharpened_color = sharpened[(yy - area.y + clen / 2) * (area.width + clen)
184 + xx - area.x + clen / 2];
185 e.priority = 3
186 - (e.orig_color.red + e.orig_color.green + e.orig_color.blue);
187 pixels.push_back (e);
188 }
189 free (sharpened);
190
191 sort (pixels.begin (), pixels.end (), compare_priorities);
192 int pos = pixels.size () / 2;
193 luminosity_t min_density = pixels[pos].orig_color.red + pixels[pos].orig_color.green + pixels[pos].orig_color.blue;
194 //printf ("\n min density %f\n", min_density);
195
196 for (entry &e : pixels)
197 e.priority = e.sharpened_color.red / std::max (e.sharpened_color.green + e.sharpened_color.blue, (luminosity_t)0.000001);
198 sort (pixels.begin (), pixels.end (), compare_priorities);
199
200 std::vector<rgbdata> reds;
201 for (entry &e : pixels)
202 {
203 if (e.orig_color.red + e.orig_color.green + e.orig_color.blue < min_density)
204 continue;
205 reds.push_back ((rgbdata){e.orig_color.red, e.orig_color.green, e.orig_color.blue});
206 //printf ("%f %f %f %f\n", e.orig_color.red, e.orig_color.green, e.orig_color.blue, e.priority);
207 if (reds.size () > pixels.size () / 1000)
208 break;
209 }
210
211
212 for (entry &e : pixels)
213 e.priority = e.sharpened_color.green / std::max (e.sharpened_color.red + e.sharpened_color.blue, (luminosity_t)0.000001);
214 sort (pixels.begin (), pixels.end (), compare_priorities);
215
216 std::vector<rgbdata> greens;
217 for (entry &e : pixels)
218 {
219 if (e.orig_color.red + e.orig_color.green + e.orig_color.blue < min_density)
220 continue;
221 greens.push_back ((rgbdata){e.orig_color.red, e.orig_color.green, e.orig_color.blue});
222 if (greens.size () > pixels.size () / 1000)
223 break;
224 }
225
226
227 for (entry &e : pixels)
228 e.priority = e.sharpened_color.blue / std::max (e.sharpened_color.red + e.sharpened_color.green, (luminosity_t)0.000001);
229 sort (pixels.begin (), pixels.end (), compare_priorities);
230
231 std::vector<rgbdata> blues;
232 for (entry &e : pixels)
233 {
234 if (e.orig_color.red + e.orig_color.green + e.orig_color.blue < min_density)
235 continue;
236 blues.push_back ((rgbdata){e.orig_color.red, e.orig_color.green, e.orig_color.blue});
237 if (blues.size () > pixels.size () / 1000)
238 break;
239 }
240 //printf ("%i %i %i\n",reds.size (), greens.size (), blues.size ());
241 if (!reds.size () || !greens.size () || !blues.size ()
242 || (reds.size () + greens.size () + blues.size ()) < 4 * 3)
243 return false;
244 optimize_screen_colors (param, reds.data (), reds.size (), greens.data (), greens.size (), blues.data (), blues.size (), progress, report);
245 return true;
246}

251static rgbdata
252optimize_chanel (rgbdata *colors, int n, int chanel)
253{
254 int matrixw = 2;
255 int matrixh = n * 2;
256
257 gsl_multifit_linear_workspace * work
258 = gsl_multifit_linear_alloc (matrixh, matrixw);
259 gsl_matrix *X = gsl_matrix_alloc (matrixh, matrixw);
260 gsl_vector *y = gsl_vector_alloc (matrixh);
261 gsl_vector *w = gsl_vector_alloc (matrixh);
262 gsl_vector *c = gsl_vector_alloc (matrixw);
263 gsl_matrix *cov = gsl_matrix_alloc (matrixw, matrixw);
264 int c1, c2;
265 if (chanel == 0)
266 c1 = 1, c2 = 2;
267 else if (chanel == 1)
268 c1 = 0, c2 = 2;
269 else
270 c1 = 0, c2 = 1;
271
272 /* Those are realy two independent linear regressions. We could simplify this. */
273 for (int i = 0; i < n; i++)
274 {
275 int e = i * 2;
276 gsl_matrix_set (X, e, 0, colors[i][chanel]);
277 gsl_matrix_set (X, e, 1, 0);
278 gsl_vector_set (y, e, colors[i][c1]);
279 gsl_matrix_set (X, e + 1, 0, 0);
280 gsl_matrix_set (X, e + 1, 1, colors[i][chanel]);
281 gsl_vector_set (y, e + 1, colors[i][c2]);
282 gsl_vector_set (w, e, 1);
283 gsl_vector_set (w, e + 1, 1);
284 }
285 double chisq;
286 gsl_multifit_wlinear (X, w, y, c, cov,
287 &chisq, work);
288 gsl_multifit_linear_free (work);
289
290 rgbdata ret;
291 ret[chanel] = 1;
292 ret[c1] = C(0);
293 ret[c2] = C(1);
294
295 gsl_matrix_free (X);
296 gsl_vector_free (y);
297 gsl_vector_free (w);
298 gsl_vector_free (c);
299 gsl_matrix_free (cov);
300 return ret;
301}

335 inline luminosity_t
336 objfunc_chanel (rgbdata *colors, int n, int chanel, color_matrix &mat)
337 {
338 luminosity_t sum = 0;
339 int c1, c2;
340 if (chanel == 0)
341 c1 = 1, c2 = 2;
342 else if (chanel == 1)
343 c1 = 0, c2 = 2;
344 else
345 c1 = 0, c2 = 1;
346 for (int i = 0; i < n; i++)
347 {
348 luminosity_t c[3];
349 mat.apply_to_rgb (colors[i].red, colors[i].green, colors[i].blue, &c[0], &c[1], &c[2]);
350 luminosity_t lsum = c[0]+c[1]+c[2];
351
352 // Negative value is bad.
353 if (c[chanel] < 0.00000001)
354 {
355 c[chanel] = 0.00000001;
356 sum += c[chanel] * c[chanel];
357 }
358 sum += c[c1] * c[c1] / (lsum * lsum);
359 sum += c[c2] * c[c2] / (lsum * lsum);
360 }
361 return sum / n;
362 }

390/* Optimize screen colors based on known red, green and blue samples
391 and store resulting black, red, green and blue colors to PARAM. */
392void
393optimize_screen_colors (scr_detect_parameters *param,
394 rgbdata *reds,
395 int nreds,
396 rgbdata *greens,
397 int ngreens,
398 rgbdata *blues,
399 int nblues, progress_info *progress, FILE *report)
400{
401 if (!nreds || !ngreens || !nblues)
402 abort ();
403 if (progress)
404 progress->set_task ("determining screen colors", 1);
405
406 param->black = {0, 0, 0};
407 param->red = optimize_chanel (reds, nreds, 0);
408 param->green = optimize_chanel (greens, ngreens, 1);
409 param->blue = optimize_chanel (blues, nblues, 2);
410 if (report)
411 {
412 fprintf (report, "Initial screen red: ");
413 param->red.print (report);
414 fprintf (report, "Initial screen green: ");
415 param->green.print (report);
416 fprintf (report, "Initial screen blue: ");
417 param->blue.print (report);
418 }
419
420 screen_color_solver solver;
421 solver.reds = reds;
422 solver.nreds = nreds;
423 solver.greens = greens;
424 solver.ngreens = ngreens;
425 solver.blues = blues;
426 solver.nblues = nblues;
427 /* Dark should be 0,0,0 */
428 solver.start[0] = solver.start[1] = solver.start[2] = 0;
429 /* If existing values makes sense use them as start point. */
430 if (param->red.red > 0.0000001
431 && param->green.green > 0.0000001
432 && param->blue.blue > 0.0000001)
433 {
434 solver.start[3] = param->red.green / param->red.red;
435 solver.start[4] = param->red.blue / param->red.red;
436 solver.start[5] = param->green.red / param->green.green;
437 solver.start[6] = param->green.blue / param->green.green;
438 solver.start[7] = param->blue.red / param->blue.blue;
439 solver.start[8] = param->blue.green / param->blue.blue;
440 }
441 else
442 {
443 solver.start[3] = 0;
444 solver.start[4] = 0;
445 solver.start[5] = 0;
446 solver.start[6] = 0;
447 solver.start[7] = 0;
448 solver.start[8] = 0;
449 }
450 simplex<luminosity_t, screen_color_solver>(solver, "optimizing colors", progress);
451 param->black = {solver.start[0], solver.start[1], solver.start[2]};
452 param->red = {1, solver.start[3], solver.start[4]};
453 param->green = {solver.start[5], 1, solver.start[6]};
454 param->blue = {solver.start[7], solver.start[8], 1};

456 {
457 fprintf (report, "optimized screen black: ");

459 fprintf (report, "optimized screen red: ");

464 param->blue.print (report);

473 }

477}
