758void ImageWidget::resizeEvent(QResizeEvent *event) {
759 QSize oldSize = m_lastSize;
760 QSize newSize = event->size();
761
762 if (oldSize.isValid() && oldSize.width() > 0 && oldSize.height() > 0 &&
763 newSize.width() > 0 && newSize.height() > 0 &&
764 oldSize != newSize && m_scale > 1e-9 && m_scan) {
765
766 // 1. Calculate current center in image coordinates
767 double centerX = m_viewX + (oldSize.width() / m_scale) / 2.0;
768 double centerY = m_viewY + (oldSize.height() / m_scale) / 2.0;
769
770 // 2. Calculate new fit scale (m_minScale)
771 CoordinateTransformer transformer(m_scan.get(), *m_rparams);
772 QSize transformedSize = transformer.getTransformedSize();
773 double imgW = transformedSize.width();
774 double imgH = transformedSize.height();
775
776 if (imgW > 0 && imgH > 0) {
777 double scaleX = (double)newSize.width() / imgW;
778 double scaleY = (double)newSize.height() / imgH;
779 double newMinScale = qMin(scaleX, scaleY);
780
781 // 3. Adjust m_scale to maintain relative zoom level
782 if (m_minScale > 1e-9) {
783 m_scale *= (newMinScale / m_minScale);
784 }
785 if (m_scale < 1e-9) m_scale = newMinScale; // fallback
786 m_minScale = newMinScale;
787 m_exploreTargetScale = m_scale;
788 }
789
790 // 4. Recalculate view position to keep the center point stable
791 m_viewX = centerX - (newSize.width() / m_scale) / 2.0;
792 m_viewY = centerY - (newSize.height() / m_scale) / 2.0;
793 if (!m_panAnimationActive) {
794 m_exploreTargetX = m_viewX;
795 m_exploreTargetY = m_viewY;
796 }
797 }
798
799 m_lastSize = newSize; // Update for next event
800 requestRender();
801 emit viewStateChanged(
802 QRectF(m_viewX, m_viewY, width() / m_scale, height() / m_scale), m_scale);
803}

954void ImageWidget::mouseMoveEvent(QMouseEvent *event) {
955 // Check for hover over registration points
956 if (!m_isDragging && m_showRegistrationPoints && m_solver) {
957 bool found = false;
958 const auto &points = m_solver->points;
959 for (size_t i = 0; i < points.size(); ++i) {
960 QPointF p_widget = imageToWidget(points[i].img);
961 if (QLineF(event->position(), p_widget).length() < 10) {
962 setCursor(Qt::PointingHandCursor);
963 found = true;
964 break;
965 }
966 }
967 if (!found) {
968 setCursor(Qt::ArrowCursor);
969 }
970 }
971
972 if (m_interactionMode == PanMode && m_isDragging) {
973 QPoint delta = event->pos() - m_lastMousePos;
974 m_lastMousePos = event->pos();
975
976 // Move view opposite to drag (m_viewX/Y are in transformed space already)
977 m_viewX -= delta.x() / m_scale;
978 m_viewY -= delta.y() / m_scale;
979 m_exploreTargetX = m_viewX;
980 m_exploreTargetY = m_viewY;
981
982 requestRender();
983 emit viewStateChanged(
984 QRectF(m_viewX, m_viewY, width() / m_scale, height() / m_scale),
985 m_scale);
986 } else if (m_interactionMode == SelectMode || m_interactionMode == AddPointMode || m_interactionMode == CropMode || m_interactionMode == GenericAreaMode) {
987 if (m_interactionMode == SelectMode && m_draggedPointIndex != -1) {
988 colorscreen::point_t imgPos = widgetToImage(event->position());
989 if (m_solver && (size_t)m_draggedPointIndex < m_solver->points.size()) {
990 m_solver->points[m_draggedPointIndex].img = imgPos;
991 m_pointsOverlayDirty = true;
992 update();
993 }
994 } else if (m_rubberBand && m_rubberBand->isVisible()) {
995 m_rubberBand->setGeometry(QRect(m_rubberBandOrigin, event->pos()).normalized());
996 }
997 } else if (m_interactionMode == SetCenterMode) {
998 if (m_dragTarget == DragTarget::Center && m_scrToImg) {
999 // Drag center: translate by offset
1000 colorscreen::point_t currentImg = widgetToImage(event->position());
1001 double xOffset = currentImg.x - m_dragStartImg.x;
1002 double yOffset = currentImg.y - m_dragStartImg.y;
1003 m_scrToImg->center.x = m_pressParams.center.x + xOffset;
1004 m_scrToImg->center.y = m_pressParams.center.y + yOffset;
1005 emit coordinateSystemChanged();
1006 update();
1007 } else if ((m_dragTarget == DragTarget::Axis1 || m_dragTarget == DragTarget::Axis2) && m_scrToImg) {
1008 // Drag axes: rotate and scale
1009 colorscreen::point_t currentImg = widgetToImage(event->position());
1010 double x1 = m_dragStartImg.x - m_pressParams.center.x;
1011 double y1 = m_dragStartImg.y - m_pressParams.center.y;
1012 double x2 = currentImg.x - m_pressParams.center.x; // Use press params center, not current
1013 double y2 = currentImg.y - m_pressParams.center.y;
1014
1015 double startDist = sqrt((x1 * x1) + (y1 * y1));
1016 double currentDist = sqrt((x2 * x2) + (y2 * y2));
1017 double scale = (startDist > 1e-9) ? currentDist / startDist : 1.0;
1018
1019 if (event->modifiers() & Qt::ShiftModifier) {
1020 scale = 1.0;
1021 }
1022
1023 // Enforce minimum length of 2.0
1024 double len1 = sqrt(m_pressParams.coordinate1.x*m_pressParams.coordinate1.x +
1025 m_pressParams.coordinate1.y*m_pressParams.coordinate1.y);
1026 double len2 = sqrt(m_pressParams.coordinate2.x*m_pressParams.coordinate2.x +
1027 m_pressParams.coordinate2.y*m_pressParams.coordinate2.y);
1028
1029 if (m_lockRelativeCoordinates) {
1030 // Both axes scale together, use stricter limit
1031 double minScale1 = (len1 > 1e-9) ? 2.0 / len1 : 0.0;
1032 double minScale2 = (len2 > 1e-9) ? 2.0 / len2 : 0.0;
1033 double minScale = std::max(minScale1, minScale2);
1034 if (scale < minScale) scale = minScale;
1035 } else {
1036 // Independent scaling
1037 if (m_dragTarget == DragTarget::Axis1) {
1038 double minScale1 = (len1 > 1e-9) ? 2.0 / len1 : 0.0;
1039 if (scale < minScale1) scale = minScale1;
1040 } else {
1041 double minScale2 = (len2 > 1e-9) ? 2.0 / len2 : 0.0;
1042 if (scale < minScale2) scale = minScale2;
1043 }
1044 }
1045
1046 double angle = atan2(y2, x2) - atan2(y1, x1);
1047
1048 if (angle != 0.0 || scale != 1.0) {
1049 double cosAngle = cos(angle);
1050 double sinAngle = sin(angle);
1051
1052 if (m_lockRelativeCoordinates) {
1053 // Standard behavior: Rotate/Scale BOTH axes
1054 m_scrToImg->coordinate1.x = (m_pressParams.coordinate1.x * cosAngle
1055 - m_pressParams.coordinate1.y * sinAngle) * scale;
1056 m_scrToImg->coordinate1.y = (m_pressParams.coordinate1.x * sinAngle
1057 + m_pressParams.coordinate1.y * cosAngle) * scale;
1058
1059 m_scrToImg->coordinate2.x = (m_pressParams.coordinate2.x * cosAngle
1060 - m_pressParams.coordinate2.y * sinAngle) * scale;
1061 m_scrToImg->coordinate2.y = (m_pressParams.coordinate2.x * sinAngle
1062 + m_pressParams.coordinate2.y * cosAngle) * scale;
1063 } else {
1064 // Independent behavior: Only affect the dragged axis
1065 // While we calculate 'angle' and 'scale' based on the drag, we only apply the transformation
1066 // to the dragged axis. The other axis should strictly remain at its CURRENT state (or initial state?).
1067 // If we use m_pressParams, we are essentially 'resetting' the non-dragged axis to its initial state
1068 // if we were to assign it. But if we DON'T touch it, it remains as is.
1069 // Since we might have moved it in a previous step?
1070 // Wait, m_scrToImg is the source of truth. If DragTarget::Axis1, we modify coordinate1.
1071 // We should NOT touch coordinate2.
1072
1073 if (m_dragTarget == DragTarget::Axis1) {
1074 m_scrToImg->coordinate1.x = (m_pressParams.coordinate1.x * cosAngle
1075 - m_pressParams.coordinate1.y * sinAngle) * scale;
1076 m_scrToImg->coordinate1.y = (m_pressParams.coordinate1.x * sinAngle
1077 + m_pressParams.coordinate1.y * cosAngle) * scale;
1078 } else {
1079 m_scrToImg->coordinate2.x = (m_pressParams.coordinate2.x * cosAngle
1080 - m_pressParams.coordinate2.y * sinAngle) * scale;
1081 m_scrToImg->coordinate2.y = (m_pressParams.coordinate2.x * sinAngle
1082 + m_pressParams.coordinate2.y * cosAngle) * scale;
1083 }
1084 }
1085
1086 emit coordinateSystemChanged();
1087 update();
1088 }
1089 }
1090 } else if (m_interactionMode == MeasureMode) {
1091 if (m_isMeasuring) {
1092 m_measureEnd = widgetToImage(event->position());
1093 update();
1094 }
1095 } else if (m_interactionMode == ExploreMode) {
1096 if (m_ignoreNextMouseMove) {
1097 m_ignoreNextMouseMove = false;
1098 return;
1099 }
1100
1101 QPoint center = rect().center();
1102 QPoint delta = event->pos() - center;
1103 if (delta.isNull()) return;
1104
1105 // Accumulate target distance
1106 m_exploreTargetX -= delta.x() / m_scale;
1107 m_exploreTargetY -= delta.y() / m_scale;
1108
1109 // Warp mouse back to center immediately
1110 m_ignoreNextMouseMove = true;
1111 QCursor::setPos(mapToGlobal(center));
1112 }

1207
1208 if (newLenSq > 4.0) {
1209 if (m_lockRelativeCoordinates) {
1210 // Update Coordinate1 and coordinate 2 to preserve relative
1211 double oldC1x = m_pressParams.coordinate1.x;
1212 double oldC1y = m_pressParams.coordinate1.y;
1213 double oldLen = sqrt(oldC1x*oldC1x + oldC1y*oldC1y);
1214 double newLen = sqrt(newLenSq);
1215
1216 double angleC1Old = atan2(oldC1y, oldC1x);
1217 double angleC2Old = atan2(m_pressParams.coordinate2.y, m_pressParams.coordinate2.x);
1218 double angleC1New = atan2(newC1y, newC1x);
1219
1220 double relAngle = angleC2Old - angleC1Old;
1221 double angleC2New = angleC1New + relAngle;
1222
1223 double oldLenC2 = sqrt(m_pressParams.coordinate2.x*m_pressParams.coordinate2.x +
1224 m_pressParams.coordinate2.y*m_pressParams.coordinate2.y);
1225 double scaleRatio = (oldLen > 1e-9) ? oldLenC2 / oldLen : 1.0;
1226 double newLenC2 = newLen * scaleRatio;
1227
1228 if (newLenC2 >= 2.0) {
1229 m_scrToImg->coordinate1.x = newC1x;
1230 m_scrToImg->coordinate1.y = newC1y;
1231 m_scrToImg->coordinate2.x = newLenC2 * cos(angleC2New);
1232 m_scrToImg->coordinate2.y = newLenC2 * sin(angleC2New);
1233 emit coordinateSystemChanged();
1234 update();
1235 }
1236 } else {
1237 // Independent update
1238 m_scrToImg->coordinate1.x = newC1x;
1239 m_scrToImg->coordinate1.y = newC1y;
1240 emit coordinateSystemChanged();
1241 update();
1242 }
1243 }
1244 } else if (m_dragTarget == DragTarget::Axis2 && isClick && m_scrToImg) {
1245 // Set Coordinate2
1246 colorscreen::point_t clickImg = widgetToImage(event->position());
1247 colorscreen::point_t center = m_scrToImg->center;
1248
1249 double newC2x = clickImg.x - center.x;
1250 double newC2y = clickImg.y - center.y;
1251
1252 double newLenC2Sq = newC2x*newC2x + newC2y*newC2y;
1253
1254 if (newLenC2Sq > 4.0) {
1255 if (m_lockRelativeCoordinates) {
1256 double oldC2x = m_pressParams.coordinate2.x;
1257 double oldC2y = m_pressParams.coordinate2.y;
1258 double oldLenC2 = sqrt(oldC2x*oldC2x + oldC2y*oldC2y);
1259 double newLenC2 = sqrt(newLenC2Sq);
1260
1261 double angleC1Old = atan2(m_pressParams.coordinate1.y, m_pressParams.coordinate1.x);
1262 double angleC2Old = atan2(oldC2y, oldC2x);
1263 double angleC2New = atan2(newC2y, newC2x);
1264
1265 double relAngle = angleC2Old - angleC1Old;
1266 double angleC1New = angleC2New - relAngle;
1267
1268 double oldLenC1 = sqrt(m_pressParams.coordinate1.x*m_pressParams.coordinate1.x +
1269 m_pressParams.coordinate1.y*m_pressParams.coordinate1.y);
1270 double scaleRatioNum = (oldLenC1 > 1e-9) ? oldLenC2 / oldLenC1 : 1.0;
1271 double newLenC1 = (scaleRatioNum > 1e-9) ? newLenC2 / scaleRatioNum : newLenC2;
1272
1273 if (newLenC1 >= 2.0) {
1274 m_scrToImg->coordinate2.x = newC2x;
1275 m_scrToImg->coordinate2.y = newC2y;
1276 m_scrToImg->coordinate1.x = newLenC1 * cos(angleC1New);
1277 m_scrToImg->coordinate1.y = newLenC1 * sin(angleC1New);
1278 emit coordinateSystemChanged();
1279 update();
1280 }
1281 } else {
1282 // Independent update
1283 m_scrToImg->coordinate2.x = newC2x;
1284 m_scrToImg->coordinate2.y = newC2y;
1285 emit coordinateSystemChanged();
1286 update();
1287 }
1288 }
1289 }
1290
1291 if (m_dragTarget != DragTarget::None) {
1292 emit coordinateSystemManipulationFinished();
1293 }
1294 // Reset drag target
1295 m_dragTarget = DragTarget::None;
1296 event->accept();
1297 }

1449void ImageWidget::pivotViewport(int oldRotIdx, int newRotIdx) {
1450 if (!m_scan) return;
1451
1452 // Viewport center in current transformed coordinates (relative to image)
1453 double centerX_tr = m_viewX + (width() / m_scale) / 2.0;
1454 double centerY_tr = m_viewY + (height() / m_scale) / 2.0;
1455
1456 // We use CoordinateTransformer twice: once with OLD rotation, once with NEW
1457 colorscreen::render_parameters oldParams = *m_rparams;
1458 oldParams.scan_rotation = oldRotIdx;
1459 CoordinateTransformer oldTrans(m_scan.get(), oldParams);
1460
1461 // Map to Scan
1462 colorscreen::point_t scanPt = oldTrans.transformedToScanCrop({centerX_tr, centerY_tr});
1463
1464 // Create params for NEW
1465 colorscreen::render_parameters newParams = *m_rparams;
1466 newParams.scan_rotation = newRotIdx;
1467 CoordinateTransformer newTrans(m_scan.get(), newParams);
1468
1469 // Map to Transformed (New)
1470 colorscreen::point_t newCenter_tr = newTrans.scanToTransformedCrop(scanPt);
1471
1472 // Set new viewX/Y to maintain center
1473 m_viewX = newCenter_tr.x - (width() / m_scale) / 2.0;
1474 m_viewY = newCenter_tr.y - (height() / m_scale) / 2.0;
1475 if (!m_panAnimationActive) {
1476 m_exploreTargetX = m_viewX;
1477 m_exploreTargetY = m_viewY;
1478 }
1479
1480 // Update m_minScale for the NEW orientation fit
1481 // This ensures subsequent resize events use the correct reference.
1482 double scaleX = (double)width() / newTrans.getTransformedCropSize().width();
1483 double scaleY = (double)height() / newTrans.getTransformedCropSize().height();
1484 m_minScale = qMin(scaleX, scaleY);
1485}

1603/* Schedule a background render of the registration-point overlay image.
1604 The resulting QImage is composited on top of the main pixmap in paintEvent.
1605 This allows the GUI to remain responsive while points are re-rendered
1606 after view changes or parameter updates. */
1607void ImageWidget::schedulePointsOverlayRender ()
1608{
1609 if (!m_solver || !m_scan || !m_rparams || !m_scrToImg || width () <= 0
1610 || height () <= 0)
1611 return;
1612
1613 if (m_pointsRenderPending) {
1614 m_pointsOverlayDirty = true;
1615 return;
1616 }
1617
1618 m_pointsRenderPending = true;
1619 m_pointsOverlayDirty = false;
1620
1621 /* Snapshot current view state and data for the background worker.
1622 We render the overlay at the current widget size with the current
1623 scale and pan. */
1624 double viewX = m_viewX;
1625 double viewY = m_viewY;
1626 double scale = m_scale;
1627 int w = width ();
1628 int h = height ();
1629 auto scrToImg = *m_scrToImg;
1630 auto scan = m_scan;
1631 auto rparams = *m_rparams; /* Snapshot rotation/mirror/crop state. */
1632 const auto points = m_solver->points; /* Shallow copy of registration points. */
1633 auto selected = m_selectedPoints; /* Copy selection set. */
1634 double heatmapTol = m_heatmapTolerance;
1635 double exaggerate = m_exaggerate;
1636 double maxArrowLen = m_maxArrowLength;
1637
1638 auto result = std::make_shared<QImage> ();
1639
1640 m_pointsQueue.runAsync (
1641 [=] (colorscreen::progress_info *) mutable
1642 {
1643 QImage overlay (w, h, QImage::Format_ARGB32_Premultiplied);
1644 overlay.fill (Qt::transparent);
1645
1646 if (points.empty ()) {
1647 *result = std::move (overlay);
1648 return;
1649 }
1650
1651 QPainter p (&overlay);
1652 p.setRenderHint (QPainter::Antialiasing);
1653
1654 colorscreen::scr_to_img map;
1655 map.set_parameters (scrToImg, *scan);
1656
1657 /* Background-safe equivalent of imageToWidget.
1658 Must account for rotation/mirror via CoordinateTransformer. */
1659 CoordinateTransformer transformer(scan.get(), rparams);
1660 auto toWidget = [viewX, viewY, scale, transformer] (const colorscreen::point_t &img)
1661 {
1662 colorscreen::point_t tr = transformer.scanToTransformedCrop(img);
1663 return QPointF ((tr.x - viewX) * scale, (tr.y - viewY) * scale);
1664 };
1665
1666 colorscreen::point_t c1 = scrToImg.coordinate1;
1667 double dot_period = sqrt (c1.x * c1.x + c1.y * c1.y);
1668 if (dot_period < 0.1) dot_period = 10.0;
1669 double threshold = dot_period * heatmapTol;
1670 double amp_scale = (scale > 1.0) ? 1.0 : (exaggerate / (dot_period / 2.0));
1671 double marginPixels = maxArrowLen + 20.0 * scale;
1672
1673 bool isVerticalStrips
1674 = colorscreen::screen_with_vertical_strips_p (scrToImg.type);
1675
1676 struct DrawPoint
1677 {
1678 QPointF start;
1679 QPointF simulated;
1680 QPointF arrowEnd;
1681 double arrowAngle;
1682 QColor color;
1683 bool hasArrow;
1684 bool isSelected;
1685 bool isTargetVisible;
1686 };
1687
1688 std::vector<DrawPoint> visible;
1689 visible.reserve (std::min (points.size (), (size_t)8192));
1690
1691 const auto &pointsVec = points.read ();
1692 for (size_t i = 0; i < pointsVec.size (); ++i) {
1693 const auto &xi = pointsVec[i].img;
1694 QPointF start = toWidget (xi);
1695
1696 /* Culling. */
1697 if (start.x () < -marginPixels || start.x () > w + marginPixels
1698 || start.y () < -marginPixels || start.y () > h + marginPixels)
1699 continue;
1700
1701 /* Compute simulated location. */
1702 colorscreen::point_t scr_p = pointsVec[i].scr;
1703 if (isVerticalStrips) {
1704 colorscreen::point_t scr2 = map.to_scr (xi);
1705 scr_p.y = scr2.y;
1706 }
1707 colorscreen::point_t p_sim_img = map.to_img (scr_p);
1708 QPointF simulated = toWidget (p_sim_img);
1709
1710 double dx_img = p_sim_img.x - xi.x;
1711 double dy_img = p_sim_img.y - xi.y;
1712 double dist_img = sqrt (dx_img * dx_img + dy_img * dy_img);
1713
1714 /* Heatmap color calculation. */
1715 QColor color;
1716 double t = (threshold > 0) ? dist_img / threshold : 1.0;
1717 if (t <= 0) color = Qt::green;
1718 else if (t >= 1.0) color = Qt::red;
1719 else {
1720 int r_val, g_val;
1721 if (t < 0.5) {
1722 r_val = static_cast<int> (255 * (t / 0.5));
1723 g_val = 255;
1724 } else {
1725 r_val = 255;
1726 g_val = static_cast<int> (255 * (1.0 - (t - 0.5) / 0.5));
1727 }
1728 color = QColor (r_val, g_val, 0);
1729 }
1730
1731 /* Arrow geometry. */
1732 colorscreen::point_t xi_d = { xi.x + dx_img * amp_scale,
1733 xi.y + dy_img * amp_scale };
1734 QPointF arrowEnd = toWidget (xi_d);
1735 double dx_w = arrowEnd.x () - start.x ();
1736 double dy_w = arrowEnd.y () - start.y ();
1737 double dist_w2 = dx_w * dx_w + dy_w * dy_w;
1738 bool hasArrow = dist_w2 > 25.0;
1739 double arrowAngle = 0.0;
1740
1741 if (hasArrow) {
1742 if (dist_w2 > maxArrowLen * maxArrowLen) {
1743 double dist_w = sqrt (dist_w2);
1744 dx_w *= maxArrowLen / dist_w;
1745 dy_w *= maxArrowLen / dist_w;
1746 arrowEnd = QPointF (start.x () + dx_w, start.y () + dy_w);
1747 }
1748 arrowAngle = std::atan2 (dy_w, dx_w);
1749 }
1750
1751 /* Only draw the simulated target if it's shifted from the source. */
1752 double dx_w_sim = simulated.x () - start.x ();
1753 double dy_w_sim = simulated.y () - start.y ();
1754 bool isTargetVisible = (dx_w_sim * dx_w_sim + dy_w_sim * dy_w_sim) > 0.25;
1755
1756 bool isSelected = selected.count (
1757 SelectedPoint{ i, SelectedPoint::RegistrationPoint });
1758 visible.push_back ({ start, simulated, arrowEnd, arrowAngle, color,
1759 hasArrow, isSelected, isTargetVisible });
1760 }
1761
1762 /* Batched rendering passes. */
1763
1764 /* Pass 2: Selected halos. */
1765 {
1766 bool any = false;
1767 for (const auto &d : visible) if (d.isSelected) { any = true; break; }
1768 if (any) {
1769 p.setPen (QPen (Qt::cyan, 6, Qt::SolidLine, Qt::RoundCap));
1770 p.setBrush (Qt::NoBrush);
1771 for (const auto &d : visible) if (d.isSelected) p.drawEllipse (d.start, 8, 8);
1772 }
1773 }
1774
1775 /* Pass 3: Actual locations (black circle outline). */
1776 {
1777 p.setPen (QPen (Qt::black, 4));
1778 p.setBrush (Qt::NoBrush);
1779 for (const auto &d : visible) p.drawEllipse (d.start, 4, 4);
1780 }
1781
1782 /* Pass 4: Actual locations (colored circle outline). */
1783 {
1784 QHash<QRgb, QList<QPointF>> buckets;
1785 for (const auto &d : visible) buckets[d.color.rgba ()] << d.start;
1786 for (auto it = buckets.cbegin (); it != buckets.cend (); ++it) {
1787 p.setPen (QPen (QColor::fromRgba (it.key ()), 2));
1788 for (const auto &pt : it.value()) p.drawEllipse (pt, 4, 4);
1789 }
1790 }
1791
1792 /* Pass 5: Arrows. */
1793 {
1794 QList<QLineF> bLines, bH1, bH2;
1795 QHash<QRgb, QList<QLineF>> cLines, cH1, cH2;
1796 for (const auto &d : visible) {
1797 if (!d.hasArrow) continue;
1798 bLines << QLineF (d.start, d.arrowEnd);
1799 double headLen = 8.0;
1800 double cosA = std::cos (d.arrowAngle - M_PI / 6);
1801 double sinA = std::sin (d.arrowAngle - M_PI / 6);

1885void ImageWidget::exploreTick() {
1886 bool needsUpdate = false;
1887
1888 if (m_plusHeld) {
1889 m_keyboardZoomVelocity += 0.00016; // accelerate
1890 if (m_keyboardZoomVelocity > 0.005) m_keyboardZoomVelocity = 0.005; // 1/3rd of 0.015
1891 m_exploreTargetScale *= (1.0 + m_keyboardZoomVelocity);
1892 m_exploreZoomSpeed = 0.15;
1893 needsUpdate = true;
1894 } else if (m_minusHeld) {
1895 m_keyboardZoomVelocity += 0.00016; // accelerate
1896 if (m_keyboardZoomVelocity > 0.005) m_keyboardZoomVelocity = 0.005; // 1/3rd of 0.015
1897 m_exploreTargetScale /= (1.0 + m_keyboardZoomVelocity);
1898 m_exploreZoomSpeed = 0.15;
1899 needsUpdate = true;
1900 } else {
1901 m_keyboardZoomVelocity = 0;
1902 }
1903
1904 double ds = m_exploreTargetScale - m_scale;
1905 if (std::abs(ds / (m_scale + 1e-9)) > 0.001) {
1906 double oldScale = m_scale;
1907 m_scale += ds * m_exploreZoomSpeed;
1908
1909 double focusX, focusY;
1910 if (m_zoomFocusCenter || m_interactionMode == ExploreMode) {
1911 focusX = width() / 2.0;
1912 focusY = height() / 2.0;
1913 } else {
1914 QPoint p = mapFromGlobal(QCursor::pos());
1915 focusX = p.x();
1916 focusY = p.y();
1917 }
1918
1919 double centerImgX = m_viewX + focusX / oldScale;
1920 double centerImgY = m_viewY + focusY / oldScale;
1921
1922 double shiftX = (centerImgX - focusX / m_scale) - m_viewX;
1923 double shiftY = (centerImgY - focusY / m_scale) - m_viewY;
1924
1925 m_viewX += shiftX;
1926 m_viewY += shiftY;
1927 needsUpdate = true;
1928 } else {
1929 m_scale = m_exploreTargetScale;
1930 }
1931
1932 // Perform pan animation if requested or in ExploreMode
1933 if (!m_isDragging && (m_interactionMode == ExploreMode || m_panAnimationActive)) {
1934 double dx = m_exploreTargetX - m_viewX;
1935 double dy = m_exploreTargetY - m_viewY;
1936
1937 double distSq = dx*dx + dy*dy;
1938 if (distSq > 0.0001) {
1939 // Smooth factor
1940 double moveX = dx * 0.15;
1941 double moveY = dy * 0.15;
1942
1943 // Limited maximal speed
1944 double maxSpeed = 20.0 / m_scale;
1945 double moveDist = std::sqrt(moveX*moveX + moveY*moveY);
1946 if (moveDist > maxSpeed) {
1947 moveX = (moveX / moveDist) * maxSpeed;
1948 moveY = (moveY / moveDist) * maxSpeed;
1949 }
1950
1951 m_viewX += moveX;
1952 m_viewY += moveY;
1953 needsUpdate = true;
1954 } else {
1955 m_viewX = m_exploreTargetX;
1956 m_viewY = m_exploreTargetY;
1957 m_panAnimationActive = false;
1958 needsUpdate = true;
1959 }
1960 }
1961
1962 if (m_interactionMode != ExploreMode && !needsUpdate) {
1963 m_exploreTimer->stop();
1964 }
1965
1966 if (needsUpdate) {
1967 requestRender();
1968 if (m_renderQueue.hasActiveTasks() && !m_refreshTimer->isActive()) {
1969 m_refreshTimer->start();
1970 }
1971 emit viewStateChanged(
1972 QRectF(m_viewX, m_viewY, width() / m_scale, height() / m_scale), m_scale);
1973 }
