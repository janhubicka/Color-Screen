73class MeasurementContainer : public QWidget {
74public:
75 using ReorderCallback = std::function<void(int, int)>;
76 MeasurementContainer(ReorderCallback onReorder, QWidget *parent = nullptr)
77 : QWidget(parent), m_onReorder(onReorder) {
78 setAcceptDrops(true);
79 }
80
81protected:
82 void dragEnterEvent(QDragEnterEvent *event) override {
83 if (event->mimeData()->hasFormat("application/x-mtf-measurement-index"))
84 event->acceptProposedAction();
85 }
86
87 void dragMoveEvent(QDragMoveEvent *event) override {
88 event->acceptProposedAction();
89 }
90
91 void dropEvent(QDropEvent *event) override {
92 bool ok;
93 int fromIndex = event->mimeData()->data("application/x-mtf-measurement-index").toInt(&ok);
94 if (ok) {
95 // Find which row we dropped on. We can iterate through the layout.
96 QVBoxLayout *layout = qobject_cast<QVBoxLayout*>(this->layout());
97 if (layout) {
98 int toIndex = -1;
99 for (int i = 0; i < layout->count(); ++i) {
100 QWidget *w = layout->itemAt(i)->widget();
101 if (w && event->position().toPoint().y() < w->geometry().bottom()) {
102 toIndex = i;
103 break;
104 }
105 }
106
107 // Adjustment for header row? The current implementation adds a header at index 0.
108 if (toIndex != -1) {
109 // Header is at 0, measurements start at 1
110 int actualFrom = fromIndex;
111 int actualTo = std::max(0, toIndex - 1); // -1 because of header
112
113 // Cap at measurement count
114 // We don't have the count here easily, but let the callback handle it.
115 if (actualFrom != actualTo) {
116 m_onReorder(actualFrom, actualTo);
117 }
118 }
119 }
120 }
121 event->acceptProposedAction();
122 }
123
124 ReorderCallback m_onReorder;
125};

519void SharpnessPanel::updateMTFChart() {
520 if (!m_mtfChart)
521 return;
522
523 ParameterState state = m_stateGetter();
524
525 // Compute MTF curves with 100 steps
526 mtf_parameters::computed_mtf curves =
527 state.rparams.sharpen.scanner_mtf.compute_curves(100);
528
529 // Pass simulation flag to chart
530 bool canSimulateDifraction =
531 state.rparams.sharpen.scanner_mtf.simulate_diffraction_p();
532 // Calculate screen frequency if applicable
533 double screenFreq = -1;
534 auto img = m_imageGetter();
535 if (img && state.scrToImg.type != colorscreen::Random) {
536 colorscreen::scr_to_img scrToImgObj;
537 scrToImgObj.set_parameters(state.scrToImg, *img);
538 double pixel_size = scrToImgObj.pixel_size({0, 0, img->width, img->height});
539 screenFreq = colorscreen::scr_names[(int)state.scrToImg.type].frequency * pixel_size;
540 }
541
542 m_mtfChart->setMTFData(curves, canSimulateDifraction,
543 state.rparams.sharpen.scanner_mtf.scan_dpi,
544 screenFreq);
545
546 // Pass all measured MTF data if available
547 const auto &scanner_mtf = state.rparams.sharpen.scanner_mtf;
548 if (!scanner_mtf.measurements.empty()) {
549 m_mtfChart->setMeasuredMTF(scanner_mtf.measurements, {scanner_mtf.get_channel_wavelength (0), scanner_mtf.get_channel_wavelength (1), scanner_mtf.get_channel_wavelength (2), scanner_mtf.get_channel_wavelength (3)});
550 } else {
551 // No measured data, clear it
552 m_mtfChart->setMeasuredMTF({}, {});
553 }
554
555 // Update diffraction notice
556 bool canSimulate = state.rparams.sharpen.scanner_mtf.can_simulate_diffraction_p();
557 if (canSimulate) {
558 m_diffractionNotice->hide();
559 } else {
560 m_diffractionNotice->show();
561 QStringList missing;
562 const auto &mtf = state.rparams.sharpen.scanner_mtf;
563 if (mtf.pixel_pitch <= 0) missing << "<b>Sensor pixel pitch</b>";
564 if (mtf.f_stop <= 0) missing << "<b>Nominal f-stop</b>";
565 if (mtf.wavelength <= 0) missing << "<b>Wavelength</b>";
566 if (mtf.scan_dpi <= 0) missing << "<b>Resolution</b>";
567
568 m_diffractionNotice->setText(QString("To enable diffraction simulation, please set missing data: %1.")
569 .arg(missing.join(", ")));
570 }

684 this, tr("Load QuickMTF measurements"), "",
685 tr("QuickMTF files (*.csv *.txt);;All Files (*)"));

703 if (state.rparams.sharpen.scanner_mtf.load_csv(

705 QMessageBox::warning(
706 this, tr("Warning"),
707 tr("Error loading MTF measurement from %1: %2")
708 .arg(fileName)
709 .arg(error ? QString::fromUtf8(error) : tr("Unknown error")));
710 fclose(f);
711 continue;
712 }
713 fclose(f);
714 anySuccess = true;
715 }
716
717 if (anySuccess) {
718 // Now apply the change
719 applyChange([state](ParameterState &s) {
720 s = state;
721 }, tr("Load MTF measurements"));
722 updateMeasurementList();
723 }
