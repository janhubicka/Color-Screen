75/* Precompute indices for inverse lookup. This speeds up invert() by mapping
76 output coordinates back to mesh cells. */
77void
78mesh::precompute_inverse () const
79{
80 if (m_data.empty () || !m_invdata.empty ())
81 return;
82
83 mesh_coord_t minx = std::numeric_limits<mesh_coord_t>::max ();
84 mesh_coord_t maxx = std::numeric_limits<mesh_coord_t>::lowest ();
85 mesh_coord_t miny = std::numeric_limits<mesh_coord_t>::max ();
86 mesh_coord_t maxy = std::numeric_limits<mesh_coord_t>::lowest ();
87
88 /* Find bounding box of the mesh. */
89#pragma omp parallel for reduction(min:minx, miny) reduction(max:maxx, maxy)
90 for (int i = 0; i < (int)m_data.size (); i++)
91 {
92 minx = std::min (m_data[i].x, minx);
93 maxx = std::max (m_data[i].x, maxx);
94 miny = std::min (m_data[i].y, miny);
95 maxy = std::max (m_data[i].y, maxy);
96 }
97
98 m_invxshift = -minx;
99 m_invyshift = -miny;
100 m_invwidth = m_width * 2;
101 m_invheight = m_height * 2;
102
103 /* Handle meshes with single point in a dimension to avoid division by zero. */
104 m_invxstep = (m_width > 1) ? (maxx - minx) / (m_width - 1) : 1.0f;
105 m_invystep = (m_height > 1) ? (maxy - miny) / (m_height - 1) : 1.0f;
106
107 /* Avoid division by zero when mesh has zero width or height in image area. */
108 m_invxstepinv = (m_invxstep > 1e-9f) ? 1.0f / m_invxstep : 0.0f;
109 m_invystepinv = (m_invystep > 1e-9f) ? 1.0f / m_invystep : 0.0f;
110
111 m_invdata.resize (m_invwidth * m_invheight);
112
113 /* Initialize inverse lookup table. */
114#pragma omp parallel for
115 for (int i = 0; i < (int)m_invdata.size (); i++)
116 {
117 m_invdata[i].minx = m_width;
118 m_invdata[i].maxx = 0;
119 m_invdata[i].miny = m_height;
120 m_invdata[i].maxy = 0;
121 }
122
123 /* Fill inverse lookup table with cells covering each output region. */
124#pragma omp parallel for collapse(2)
125 for (int y = 0; y < m_height - 1; y++)
126 for (int x = 0; x < m_width - 1; x++)
127 {
128 mesh_coord_t cell_minx = m_data[y * m_width + x].x;
129 mesh_coord_t cell_maxx = m_data[y * m_width + x].x;
130 mesh_coord_t cell_miny = m_data[y * m_width + x].y;
131 mesh_coord_t cell_maxy = m_data[y * m_width + x].y;
132
133 /* Check all 4 corners of the cell. */
134 for (int dy = 0; dy <= 1; dy++)
135 for (int dx = 0; dx <= 1; dx++)
136 {
137 cell_minx = std::min (m_data[(y + dy) * m_width + x + dx].x, cell_minx);
138 cell_maxx = std::max (m_data[(y + dy) * m_width + x + dx].x, cell_maxx);
139 cell_miny = std::min (m_data[(y + dy) * m_width + x + dx].y, cell_miny);
140 cell_maxy = std::max (m_data[(y + dy) * m_width + x + dx].y, cell_maxy);
141 }
142
143 int iminx = floor ((cell_minx + m_invxshift) * m_invxstepinv);
144 int imaxx = floor ((cell_maxx + m_invxshift) * m_invxstepinv);
145 int iminy = floor ((cell_miny + m_invyshift) * m_invystepinv);
146 int imaxy = floor ((cell_maxy + m_invyshift) * m_invystepinv);
147
148 iminx = std::clamp (iminx, 0, m_invwidth - 1);
149 imaxx = std::clamp (imaxx, 0, m_invwidth - 1);
150 iminy = std::clamp (iminy, 0, m_invheight - 1);
151 imaxy = std::clamp (imaxy, 0, m_invheight - 1);
152
153 /* Update coverage info for each overlapping lookup bucket. */
154 for (int yy = iminy; yy <= imaxy; yy++)
155 for (int xx = iminx; xx <= imaxx; xx++)
156 {
157 int idx = yy * m_invwidth + xx;
158#pragma omp critical(mesh_inv_update)
159 {
160 m_invdata[idx].minx = std::min (m_invdata[idx].minx, (unsigned int)x);
161 m_invdata[idx].maxx = std::max (m_invdata[idx].maxx, (unsigned int)x);
162 m_invdata[idx].miny = std::min (m_invdata[idx].miny, (unsigned int)y);
163 m_invdata[idx].maxy = std::max (m_invdata[idx].maxy, (unsigned int)y);
164 }
165 }
166 }
167}

169/* Find source point corresponding to image point IP using the inverse lookup table. */
170point_t
171mesh::invert (point_t ip) const
172{
173 mesh_point p = { (mesh_coord_t)ip.x, (mesh_coord_t)ip.y };
174 int ix = std::clamp ((int)((ip.x + m_invxshift) * m_invxstepinv), 0, m_invwidth - 1);
175 int iy = std::clamp ((int)((ip.y + m_invyshift) * m_invystepinv), 0, m_invheight - 1);
176 if (ix >= 0 && iy >= 0 && ix < m_invwidth && iy < m_invheight && !m_invdata.empty ())
177 {
178 int pp = iy * m_invwidth + ix;
179 for (int y = m_invdata[pp].miny; y <= (int)m_invdata[pp].maxy; y++)
180 for (int x = m_invdata[pp].minx; x <= (int)m_invdata[pp].maxx; x++)
181 {
182 /* Determine cell corners. */
183 mesh_point p1 = m_data[y * m_width + x];
184 mesh_point p2 = m_data[y * m_width + x + 1];
185 mesh_point p3 = m_data[(y + 1) * m_width + x];
186 mesh_point p4 = m_data[(y + 1) * m_width + x + 1];
187
188 /* Check if point is above or below diagonal. */
189 mesh_coord_t sgn1 = sign (p, p1, p4);
190 if (sgn1 > 0)
191 {
192 /* Check if point is inside of the triangle. */
193 if (sign (p, p4, p3) < 0 || sign (p, p3, p1) < 0)
194 continue;
195 mesh_coord_t rx, ry;
196 intersect_vectors (p1.x, p1.y, p.x - p1.x, p.y - p1.y, p3.x,
197 p3.y, p4.x - p3.x, p4.y - p3.y, &rx, &ry);
198 rx = 1.0f / rx;
199 return { (ry * rx + x) * m_xstep - m_xshift,
200 (rx + y) * m_ystep - m_yshift };
201 }
202 else
203 {
204 /* Check if point is inside of the triangle. */
205 if (sign (p, p4, p2) > 0 || sign (p, p2, p1) > 0)
206 continue;
207 mesh_coord_t rx, ry;
208 intersect_vectors (p1.x, p1.y, p.x - p1.x, p.y - p1.y, p2.x,
209 p2.y, p4.x - p2.x, p4.y - p2.y, &rx, &ry);
210 rx = 1.0f / rx;
211 return { (rx + x) * m_xstep - m_xshift,
212 (ry * rx + y) * m_ystep - m_yshift };
213 }
214 }
215 }
216
217 /* Fallback: return mesh boundaries. */
218 point_t ret;
219 if (ix < m_invwidth / 2)
220 ret.x = -m_xshift;
221 else
222 ret.x = -m_xshift + m_xstep * (m_width - 1);
223 if (iy < m_invheight / 2)
224 ret.y = -m_yshift;
225 else
226 ret.y = -m_yshift + m_ystep * (m_height - 1);
227 return ret;
228}

230/* Small helper to push mesh entry (X, Y) to image range [X1, Y1]..[X2, Y2]. */
231point_t
232mesh::push_to_range (int x, int y, image_area area) const
233{
234 coord_t x1 = area.x;
235 coord_t y1 = area.y;
236 coord_t x2 = area.x + area.width;
237 coord_t y2 = area.y + area.height;
238 coord_t xx = m_data[y * m_width + x].x;
239 coord_t yy = m_data[y * m_width + x].y;
240 if (xx >= x1 && xx <= x2 && yy >= y1 && yy <= y2)
241 return { x * m_xstep - m_xshift, y * m_ystep - m_yshift };
242 if (xx < x1)
243 xx = x1;
244 if (xx > x2)
245 xx = x2;
246 if (yy < y1)
247 yy = y1;
248 if (yy > y2)
249 yy = y2;
250 return invert ({ xx, yy });
251}

253/* Determine range in target coordinates covering AREA transformed by TRANS. */
254image_area
255mesh::get_range (matrix2x2<coord_t> trans, image_area area_in) const noexcept
256{
257 image_area area;
258
259 for (int y = 0; y < m_height - 1; y++)
260 for (int x = 0; x < m_width - 1; x++)
261 {
262 image_area mesh_cell_area;
263 for (int dy = 0; dy <= 1; dy++)
264 for (int dx = 0; dx <= 1; dx++)
265 mesh_cell_area.extend (m_data[(y + dy) * m_width + x + dx]);
266
267 if (!mesh_cell_area.overlap_p (area_in))
268 continue;
269
270 /* For overlapping cells, transform corners to source coordinates. */
271 for (int dy = 0; dy <= 1; dy++)
272 for (int dx = 0; dx <= 1; dx++)
273 {
274 point_t p = push_to_range (x + dx, y + dy, area_in);
275 area.extend (trans.apply_to_vector (p));
276 }
277 }
278 return area;
279}

281/* Determine range in target coordinates covering AREA_IN of source. */
282image_area
283mesh::get_src_range (image_area area_in) const noexcept
284{
285 image_area area;
286
287 if (m_width < 1 || m_height < 1)
288 return area;
289
290 /* 1. Account for the 4 corners of AREA_IN. */
291 area.extend (apply ({area_in.x, area_in.y}));
292 area.extend (apply ({area_in.x + area_in.width, area_in.y}));
293 area.extend (apply ({area_in.x, area_in.y + area_in.height}));
294 area.extend (apply ({area_in.x + area_in.width, area_in.y + area_in.height}));
295
296 /* Find the range of grid indices. */
297 int ix1 = floor ((area_in.x + m_xshift) * m_xstepinv);
298 int ix2 = floor ((area_in.x + area_in.width + m_xshift) * m_xstepinv);
299 int iy1 = floor ((area_in.y + m_yshift) * m_ystepinv);
300 int iy2 = floor ((area_in.y + area_in.height + m_yshift) * m_ystepinv);
301
302 ix1 = std::clamp (ix1, -1, m_width - 1);
303 ix2 = std::clamp (ix2, -1, m_width - 1);
304 iy1 = std::clamp (iy1, -1, m_height - 1);
305 iy2 = std::clamp (iy2, -1, m_height - 1);
306
307 /* 2. Account for intersections of AREA_IN boundaries with grid lines. */
308 /* Top and bottom boundaries. */
309 for (int ix = ix1 + 1; ix <= ix2; ix++)
310 {
311 area.extend (apply ({ix * m_xstep - m_xshift, area_in.y}));
312 area.extend (apply ({ix * m_xstep - m_xshift, area_in.y + area_in.height}));
313 }
314 /* Left and right boundaries. */
315 for (int iy = iy1 + 1; iy <= iy2; iy++)
316 {
317 area.extend (apply ({area_in.x, iy * m_ystep - m_yshift}));
318 area.extend (apply ({area_in.x + area_in.width, iy * m_ystep - m_yshift}));
319 }
320
321 /* 3. Account for interior grid points. */
322 int gx1 = ceil ((area_in.x + m_xshift) * m_xstepinv);
323 int gx2 = floor ((area_in.x + area_in.width + m_xshift) * m_xstepinv);
324 int gy1 = ceil ((area_in.y + m_yshift) * m_ystepinv);
325 int gy2 = floor ((area_in.y + area_in.height + m_yshift) * m_ystepinv);
326
327 gx1 = std::clamp (gx1, 0, m_width - 1);
328 gx2 = std::clamp (gx2, 0, m_width - 1);
329 gy1 = std::clamp (gy1, 0, m_height - 1);
330 gy2 = std::clamp (gy2, 0, m_height - 1);
331
332 for (int ix = gx1; ix <= gx2; ix++)
333 for (int iy = gy1; iy <= gy2; iy++)
334 area.extend (m_data[iy * m_width + ix]);
335
336 return area;
337}

367std::unique_ptr<mesh>
368mesh::load (FILE *f, const char **error)
369{
370 if (!expect_keyword (f, "mesh_dimensions:"))
371 {
372 *error = "expected mesh_dimensions";
373 return nullptr;
374 }
375 int width, height;
376 if (fscanf (f, "%i %i", &width, &height) != 2)
377 {
378 *error = "failed to parse mesh_dimensions";
379 return nullptr;
380 }
381 float xshift, yshift;
382 if (!expect_keyword (f, "mesh_shifts:"))
383 {
384 *error = "expected mesh_shifts";
385 return nullptr;
386 }
387 if (fscanf (f, "%f %f", &xshift, &yshift) != 2)
388 {
389 *error = "failed to parse mesh_shifts";
390 return nullptr;
391 }
392 float xstep, ystep;
393 if (!expect_keyword (f, "mesh_steps:"))
394 {
395 *error = "expected mesh_steps";
396 return nullptr;
397 }
398 if (fscanf (f, "%f %f", &xstep, &ystep) != 2)
399 {
400 *error = "failed to parse mesh_steps";
401 return nullptr;
402 }
403 if (!expect_keyword (f, "mesh_points:"))
404 {
405 *error = "expected mesh_points";
406 return nullptr;
407 }
408 auto m = std::make_unique<mesh> (xshift, yshift, xstep, ystep, width, height);
409 for (int y = 0; y < height; y++)
410 {
411 for (int x = 0; x < width; x++)
412 {
413 float sx, sy;
414 if (!expect_keyword (f, "(") || fscanf (f, "%f", &sx) != 1
415 || !expect_keyword (f, ",") || fscanf (f, "%f", &sy) != 1
416 || !expect_keyword (f, ")"))
417 {
418 *error = "failed to parse mesh points";
419 return nullptr;
420 }
421 m->set_point ({ x, y }, { sx, sy });
422 }
423 }
424 if (!expect_keyword (f, "mesh_end"))
425 {
426 *error = "expected mesh_end";
427 return nullptr;
428 }
429 return m;

432/* Grow mesh dimensions by given number of points to LEFT, RIGHT, TOP and BOTTOM. */
433bool
434mesh::grow (int left, int right, int top, int bottom)
435{
436 int new_width = m_width + left + right;
437 int new_height = m_height + top + bottom;
438
439 if (!m_invdata.empty ())
440 abort (); // Growing mesh after inverse precomputation is not supported.
441
442 std::vector<mesh_point> new_data (new_width * new_height);
443 for (int y = 0; y < m_height; y++)
444 std::copy (m_data.begin () + y * m_width,
445 m_data.begin () + (y + 1) * m_width,
446 new_data.begin () + (y + top) * new_width + left);
447
448 m_data = std::move (new_data);
449 m_xshift += left * m_xstep;
450 m_yshift += top * m_ystep;
451 m_width = new_width;
452 m_height = new_height;
453 return true;
454}

508/* Compute an inverse mesh covering the given AREA or the full bounding box of original mesh target coordinates. */
509std::unique_ptr<mesh>
510mesh::compute_inverse_uncached (int_optional_image_area area, progress_info *progress) const
511{
512 if (m_width == 0 || m_height == 0 || (area.set && area.empty_p ()))
513 return std::make_unique<mesh> (0, 0, 1.0f, 1.0f, 0, 0);
514
515 mesh_coord_t minx = std::numeric_limits<mesh_coord_t>::max ();
516 mesh_coord_t maxx = std::numeric_limits<mesh_coord_t>::lowest ();
517 mesh_coord_t miny = std::numeric_limits<mesh_coord_t>::max ();
518 mesh_coord_t maxy = std::numeric_limits<mesh_coord_t>::lowest ();
519
520 if (area.set)
521 {
522 minx = area.x;
523 maxx = area.x + area.width - 1;
524 miny = area.y;
525 maxy = area.y + area.height - 1;
526 }
527 else
528 {
529 for (int i = 0; i < (int)m_data.size (); i++)
530 {
531 minx = std::min (m_data[i].x, minx);
532 maxx = std::max (m_data[i].x, maxx);
533 miny = std::min (m_data[i].y, miny);
534 maxy = std::max (m_data[i].y, maxy);
535 }
536 }
537
538 /* Handle meshes with single point in a dimension to avoid division by zero. */
539 mesh_coord_t invxstep = (m_width > 1) ? (maxx - minx) / (m_width - 1) : 1.0f;
540 mesh_coord_t invystep = (m_height > 1) ? (maxy - miny) / (m_height - 1) : 1.0f;
541
542 if (invxstep <= 0.0f)
543 invxstep = 1.0f;
544 if (invystep <= 0.0f)
545 invystep = 1.0f;
546
547 int invwidth = my_ceil ((maxx - minx) / invxstep) + 1;
548 int invheight = my_ceil ((maxy - miny) / invystep) + 1;
549
550 int64_t max_size = (int64_t)m_width * m_height * 32;
551 if ((int64_t)invwidth * invheight > max_size)
552 {
553 double factor = std::sqrt ((double)((int64_t)invwidth * invheight) / max_size);
554 invxstep *= factor;
555 invystep *= factor;
556 invwidth = my_ceil ((maxx - minx) / invxstep) + 1;
557 invheight = my_ceil ((maxy - miny) / invystep) + 1;
558 }
559
560 if (invwidth < 2)
561 invwidth = 2;
562 if (invheight < 2)
563 invheight = 2;
564
565 precompute_inverse ();
566
567 auto inv_mesh = std::make_unique<mesh> (-minx, -miny, invxstep, invystep, invwidth, invheight);
568
569#pragma omp parallel for collapse(2)
570 for (int y = 0; y < invheight; y++)
571 for (int x = 0; x < invwidth; x++)
572 {
573 if (progress && progress->cancel_requested ())
574 continue;
575 point_t ip = { (coord_t)(minx + x * invxstep), (coord_t)(miny + y * invystep) };
576 point_t src = invert (ip);
577 inv_mesh->set_point ({(int64_t)x, (int64_t)y}, src);
578 }
579
580 if (progress && progress->cancel_requested ())
581 return nullptr;
582
583 return inv_mesh;
584}

589{

591#pragma omp parallel for

593 {
594 point_t p = { (coord_t)m_data[i].x, (coord_t)m_data[i].y };

597 }
