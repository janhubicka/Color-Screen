160/* Create new image layer histogram using parameters P. Report progress
161 to PROGRESS. */
162std::unique_ptr<histogram>
163get_new_image_layer_histogram (image_layer_histogram_params &p,
164 progress_info *)
165{
166 histogram hist;
167
168 /* First determine the global range of values. */
169#pragma omp parallel for reduction(histogram_range : hist)
170 for (int y = p.crop.y; y < p.crop.y + p.crop.height; y++)
171 for (int x = p.crop.x; x < p.crop.x + p.crop.width; x++)
172 hist.pre_account (p.r->get_unadjusted_data ({x, y}));
173
174 hist.finalize_range (65536);
175
176 /* Now account values into the finalized range. */
177#pragma omp parallel for reduction(histogram_entries : hist)
178 for (int y = p.crop.y; y < p.crop.y + p.crop.height; y++)
179 for (int x = p.crop.x; x < p.crop.x + p.crop.width; x++)
180 hist.account (p.r->get_unadjusted_data ({x, y}));
181
182 hist.finalize ();
183 return std::make_unique<histogram> (std::move (hist));
184}

186/* Create new input lookup table using parameters P. Report progress
187 to PROGRESS. */
188std::unique_ptr<luminosity_t[]>
189get_new_lookup_table (lookup_table_params &p, progress_info *)
190{
191 bool use_table = (p.gamma == 0) && !p.gamma_table.empty ();
192 /* Use some sane data if table is missing. */
193 if (p.gamma == 0 && !use_table)
194 p.gamma = 1;
195 auto lookup_table = std::make_unique<luminosity_t[]> (p.maxval + 1);
196 luminosity_t gamma = p.gamma;
197 if (gamma != -1)
198 gamma = std::clamp (gamma, (luminosity_t)0.0001, (luminosity_t)100.0);
199 luminosity_t mul = 1 / (luminosity_t)(p.maxval);
200
201 luminosity_t dark_point = p.dark_point;
202 luminosity_t scan_exposure = p.scan_exposure;
203
204 if (!use_table)
205 for (int i = 0; i <= p.maxval; i++)
206 lookup_table[i] = (apply_gamma (i * mul, gamma) - dark_point)
207 * scan_exposure;
208 else
209 for (int i = 0; i <= p.maxval; i++)
210 lookup_table[i] = (p.gamma_table[i] - dark_point) * scan_exposure;
211 return lookup_table;
212}

234/* Tables used during gray data computation. */
235struct gray_data_tables
236{
237 std::shared_ptr<luminosity_t[]> rtable = nullptr;
238 std::shared_ptr<luminosity_t[]> gtable = nullptr;
239 std::shared_ptr<luminosity_t[]> btable = nullptr;

241 luminosity_t red = 1, green = 1, blue = 1;
242 backlight_correction *correction = nullptr;

245/* Compute lookup tables for grayscale conversion using parameters P.
246 If CORRECTION is true, backlight correction is enabled. Report progress
247 to PROGRESS. */
248inline gray_data_tables
249compute_gray_data_tables (graydata_params &p, bool correction,
250 progress_info *progress)
251{
252 gray_data_tables ret = {};
253 luminosity_t red = p.red;
254 luminosity_t green = p.green;
255 luminosity_t blue = p.blue;
256 rgbdata dark = p.dark;
257
258 ret.red = red;
259 ret.green = green;
260 ret.blue = blue;
261 ret.dark = dark;
262
263 lookup_table_params par;
264 par.gamma = p.gamma;
265 par.maxval = p.img->maxval;
266 par.scan_exposure = correction ? 1 : red;
267 par.dark_point = correction ? 0 : dark.red;
268 par.gamma_table = p.gamma_table[0];
269 ret.rtable = lookup_table_cache.get (par, progress);
270 if (!ret.rtable)
271 return ret;
272 par.scan_exposure = correction ? 1 : green;
273 par.dark_point = correction ? 0 : dark.green;
274 par.gamma_table = p.gamma_table[1];
275 ret.gtable = lookup_table_cache.get (par, progress);
276 if (!ret.gtable)
277 {
278 ret.rtable = nullptr;
279 return ret;
280 }
281 par.scan_exposure = correction ? 1 : blue;
282 par.dark_point = correction ? 0 : dark.blue;
283 par.gamma_table = p.gamma_table[2];
284 ret.btable = lookup_table_cache.get (par, progress);
285 if (!ret.btable)
286 {
287 ret.rtable = nullptr;
288 ret.gtable = nullptr;
289 return ret;
290 }
291 return ret;
292}

294/* Compute grayscale value for source pixel R, G, B at index X, Y
295 using tables T. */
296inline luminosity_t
297compute_gray_data (gray_data_tables &t, int x, int y, int r, int g, int b)
298{
299 luminosity_t l1 = t.rtable[r];
300 luminosity_t l2 = t.gtable[g];
301 luminosity_t l3 = t.btable[b];
302 if (t.correction)
303 {
304 l1 = (t.correction->apply (l1, x, y,
305 backlight_correction_parameters::red)
306 - t.dark.red)
307 * t.red;
308 l2 = (t.correction->apply (l2, x, y,
309 backlight_correction_parameters::green)
310 - t.dark.green)
311 * t.green;
312 l3 = (t.correction->apply (l3, x, y,
313 backlight_correction_parameters::blue)
314 - t.dark.blue)
315 * t.blue;
316 }
317 return l1 + l2 + l3;
318}

363/* Create new grayscale and sharpened data using parameters P.
364 Report progress to PROGRESS. */
365std::unique_ptr<sharpened_data>
366get_new_gray_sharpened_data (gray_and_sharpen_params &p,
367 progress_info *progress)
368{
369 auto ret = std::make_unique<sharpened_data> (p.gp.img->width, p.gp.img->height);
370 if (!ret || !ret->m_data)
371 return nullptr;
372 mem_luminosity_t *out = ret->m_data;
373
374 bool ok;
375 if (p.gp.img->has_grayscale_or_ir () && !p.gp.ignore_infrared)
376 {
377 lookup_table_params par;
378 getdata_params d;
379 par.maxval = p.gp.img->maxval;
380 par.gamma = p.gp.gamma;
381 d.table = lookup_table_cache.get (par, progress);
382 d.correction = p.gp.backlight;
383 d.width = p.gp.img->width;
384 d.height = p.gp.img->height;
385 if (!d.table)
386 return nullptr;
387 if (d.correction)
388 {
389 if (p.sp.deconvolution_p ())
390 {
391 ok = deconvolve<luminosity_t, mem_luminosity_t,
392 uint16_t *, getdata_params &,
393 getdata_helper_correction> (
394 out, (uint16_t *)p.gp.img->get_data_ptr (), d, p.gp.img->width, p.gp.img->height,
395 p.sp, progress, true);
396 }
397 else
398 ok = sharpen<luminosity_t, mem_luminosity_t, uint16_t *,
399 getdata_params &, getdata_helper_correction> (
400 out, (uint16_t *)p.gp.img->get_data_ptr (), d, p.gp.img->width, p.gp.img->height,
401 p.sp.get_mode () == sharpen_parameters::none ? 0 : p.sp.usm_radius,
402 p.sp.usm_amount, progress);
403 }
404 else if (p.sp.deconvolution_p ())
405 {
406 ok = deconvolve<luminosity_t, mem_luminosity_t, uint16_t *,
407 getdata_params &, getdata_helper_no_correction> (
408 out, p.gp.img->get_data_ptr (), d, p.gp.img->width, p.gp.img->height,
409 p.sp, progress, true);
410 }
411 else
412 ok = sharpen<luminosity_t, mem_luminosity_t, uint16_t *,
413 getdata_params &, getdata_helper_no_correction> (
414 out, p.gp.img->get_data_ptr (), d, p.gp.img->width, p.gp.img->height,
415 p.sp.get_mode () == sharpen_parameters::none ? 0 : p.sp.usm_radius,
416 p.sp.usm_amount, progress);
417 }
418 else
419 {
420 gray_data_tables t
421 = compute_gray_data_tables (p.gp, p.gp.backlight != nullptr, progress);
422 if (!t.rtable)
423 ok = false;
424 else
425 {
426 t.correction = p.gp.backlight;
427 if (p.sp.deconvolution_p ())
428 {
429 ok = deconvolve<luminosity_t, mem_luminosity_t,
430 const image_data *, gray_data_tables &,
431 getdata_helper2> (
432 out, p.gp.img, t, p.gp.img->width, p.gp.img->height,
433 p.sp, progress, true);
434 }
435 else
436 ok = sharpen<luminosity_t, mem_luminosity_t, const image_data *,
437 gray_data_tables &, getdata_helper2> (
438 out, p.gp.img, t, p.gp.img->width, p.gp.img->height,
439 p.sp.get_mode () == sharpen_parameters::none ? 0 : p.sp.usm_radius,
440 p.sp.usm_amount, progress);
441 }
442 }
443 if (!ok)
444 return nullptr;
445 return ret;
446}

464/* Precompute all data needed for rendering. */
465bool
466render::precompute_all (bool grayscale_needed, bool normalized_patches,
467 rgbdata patch_proportions, progress_info *progress)
468{
469 if (m_params.backlight_correction)
470 {
471 backlight_correction_cache_params p
472 = { m_params.backlight_correction.get (),
473 m_params.backlight_correction->id,
474 m_img.width,
475 m_img.height,
476 m_params.backlight_correction_black,
477 true };
478 m_backlight_correction = backlight_correction_cache.get (
479 p, progress, &m_backlight_correction_id);
480 if (!m_backlight_correction)
481 return false;
482 }
483 if (m_img.has_rgb ())
484 {
485 lookup_table_params par;
486 par.maxval = m_img.maxval;
487 par.gamma = m_params.gamma;
488 if (par.gamma == 0)
489 par.gamma_table = m_img.to_linear[0];
490 m_rgb_lookup_table[0] = lookup_table_cache.get (par, progress);
491 if (!m_rgb_lookup_table[0])
492 return false;
493 if (par.gamma == 0)
494 {
495 par.gamma_table = m_img.to_linear[1];
496 m_rgb_lookup_table[1] = lookup_table_cache.get (par, progress);
497 par.gamma_table = m_img.to_linear[2];
498 m_rgb_lookup_table[2] = lookup_table_cache.get (par, progress);
499 }
500 else
501 {
502 m_rgb_lookup_table[1] = lookup_table_cache.get (par, progress);
503 m_rgb_lookup_table[2] = lookup_table_cache.get (par, progress);
504 }
505 }
506
507 if (grayscale_needed)
508 {
509 bool ir_simulation = !m_img.has_grayscale_or_ir ()
510 || (m_img.has_rgb () && m_params.ignore_infrared);
511 gray_and_sharpen_params p
512 = { { m_img.id,
513 &m_img,
514 m_params.gamma,
515 { m_img.to_linear[0], m_img.to_linear[1], m_img.to_linear[2] },
516 ir_simulation ? m_params.mix_dark : rgbdata{1.0f, 1.0f, 1.0f},
517 ir_simulation ? m_params.mix_red : 1.0f,
518 ir_simulation ? m_params.mix_green : 1.0f,
519 ir_simulation ? m_params.mix_blue : 1.0f,
520 m_backlight_correction.get (),
521 m_backlight_correction_id,
522 ir_simulation ? m_params.ignore_infrared : false }, m_params.sharpen };
523 m_sharpened_data_holder
524 = gray_and_sharpened_data_cache.get (p, progress, &m_gray_data_id);
525 if (!m_sharpened_data_holder)
526 return false;
527 m_sharpened_data = m_sharpened_data_holder->m_data;
528 }
529 if (m_params.contact_copy.simulate)
530 {
531 m_sensitivity_hd_curve = std::make_unique <richards_hd_curve> (100, m_params.contact_copy.emulsion_characteristic_curve);
532 m_sensitivity = std::make_unique <film_sensitivity> (m_sensitivity_hd_curve.get (), m_params.contact_copy.preflash, m_params.contact_copy.exposure, m_params.contact_copy.boost);
533 m_sensitivity->precompute ();
534 }
535 if (m_sensitivity)
536 {
537 luminosity_t lmin=-0.1;
538 luminosity_t lmax=2;
539 const int steps = 65536;
540 luminosity_t yvals[steps];
541 for (int i = 0; i < steps; i++)
542 yvals[i] = adjust_luminosity_ir (i * (lmax - lmin) / (steps - 1) + lmin);
543 m_adjust_luminosity = std::make_unique <precomputed_function <luminosity_t>> (lmin, lmax, yvals, steps);
544 }
545 return out_color.precompute (m_params, &m_img, normalized_patches, patch_proportions, progress);
546}

572 return false;

621/* Sample square patch with center C and corner offsets P1, P2. */
622luminosity_t
623render::sample_img_square (point_t c, point_t p1, point_t p2) const
624{
625 luminosity_t acc = 0, weights = 0;
626 coord_t x_min_val = std::min ({c.x - p1.x, c.x + p1.x, c.x - p2.x, c.x + p2.x});
627 coord_t x_max_val = std::max ({c.x - p1.x, c.x + p1.x, c.x - p2.x, c.x + p2.x});
628 int xmin = std::max ((int)(x_min_val - (coord_t)0.5), 0);
629 int xmax = std::min ((int)ceil (x_max_val + (coord_t)0.5), m_img.width - 1);
630
631 /* If the resolution is too small, just sample given point. */
632 if (xmax - xmin < 2)
633 return get_img_pixel ({c.x, c.y});
634
635 /* For bigger resolution we can sample few points in the square. */
636 if (xmax - xmin < 6)
637 {
638 int samples = (int)my_floor (my_sqrt (p1.x * p1.x + p1.y * p1.y) + (coord_t)0.5) * 2;
639 if (!samples)
640 return get_img_pixel ({c.x, c.y});
641 luminosity_t rec = (luminosity_t)1.0 / samples;
642 for (int y = -samples; y <= samples; y++)
643 for (int x = -samples; x <= samples; x++)
644 {
645 int w = 1 + (samples - abs (x) - abs (y));
646 if (w < 0)
647 continue;
648 acc += (luminosity_t)w * get_img_pixel ({c.x + (p1.x * x + p2.x * y) * rec,
649 c.y + (p1.y * x + p2.y * y) * rec});
650 weights += (luminosity_t)w;
651 }
652 }
653 /* Faster version of the above which does not need multiple calls to get_img_pixel.
654 It however may suffer from banding when spots are too small. */
655 else
656 {
657 coord_t y_min_val = std::min ({c.y - p1.y, c.y + p1.y, c.y - p2.y, c.y + p2.y});
658 coord_t y_max_val = std::max ({c.y - p1.y, c.y + p1.y, c.y - p2.y, c.y + p2.y});
659 int ymin = std::max ((int)(y_min_val - (coord_t)0.5), 0);
660 int ymax = std::min ((int)ceil (y_max_val + (coord_t)0.5), m_img.height - 1);
661 matrix2x2<coord_t> base (p1.x, p2.x, p1.y, p2.y);
662 matrix2x2<coord_t> inv = base.invert ();
663 for (int y = ymin; y <= ymax; y++)
664 {
665 for (int x = xmin; x <= xmax; x++)
666 {
667 coord_t cx = (coord_t)x + (coord_t)0.5 - c.x;
668 coord_t cy = (coord_t)y + (coord_t)0.5 - c.y;
669 coord_t ccx, ccy;
670 inv.apply_to_vector (cx, cy, &ccx, &ccy);
671 luminosity_t w = (luminosity_t)(fabs (ccx) + fabs (ccy));
672
673 if (w < 1)
674 {
675 w = (luminosity_t)1.0 - w;
676 acc += w * get_data ({x, y});
677 weights += w;
678 }
679 }
680 }
681 }
682 if (weights > 0)
683 return acc / weights;
684 return 0;
685}

694/* Get linearized pixel at XX, YY with given RANGE. */
695rgbdata
696get_linearized_pixel (const image_data &img, render_parameters &rparam, int xx,
697 int yy, int range, progress_info *progress)
698{
699 rgbdata color = { 0, 0, 0 };
700 int n = 0;
701 const image_data *imgp = &img;
702 if (img.stitch)
703 {
704 int tx, ty;
705 point_t scr = img.stitch->common_scr_to_img.final_to_scr (
706 { (coord_t)(xx + img.xmin), (coord_t)(yy + img.ymin) });
707 if (!img.stitch->tile_for_scr (&rparam, scr.x, scr.y, &tx, &ty, true))
708 return color;
709 point_t p = img.stitch->images[ty][tx].common_scr_to_img (scr);
710 xx = nearest_int (p.x);
711 yy = nearest_int (p.y);
712 imgp = img.stitch->images[ty][tx].img.get ();
713 }
714 render r (*imgp, rparam, 255);
715 if (!r.precompute_all (!img.has_rgb (), false, { 1.0f / 3.0f, 1.0f / 3.0f, 1.0f / 3.0f }, progress))
716 return rgbdata(0, 0, 0);
717 for (int y = yy - range; y < yy + range; y++)
718 for (int x = xx - range; x < xx + range; x++)
719 if (x >= 0 && x < img.width && y >= 0 && y < img.height)
720 {
721 if (img.has_rgb ())
722 color += r.get_linearized_rgb_pixel ({x, y});
723 else
724 {
725 rgbdata color2 = { 1, 1, 1 };
726 color += color2 * r.get_unadjusted_data ({x, y});
727 }
728 n++;
729 }
730 return n ? color / (float)n : color;
731}

734std::vector <rgbdata>
735hd_y_to_rgb (render_parameters &rparam, int steps, luminosity_t miny, luminosity_t maxy, rgbdata patch_proportions, hd_axis_type axis_type)
736{
737 out_color_adjustments a (256);
738 if (!a.precompute (rparam, nullptr, false, patch_proportions, nullptr))
739 return {};
740 std::vector <rgbdata> data (steps);
741 for (int i = 0 ; i < steps; i++)
742 {
743 luminosity_t y = i * (maxy - miny) / (steps - 1) + miny;
744 y = hd_axis_y_to_linear (y, rparam.contact_copy.boost, axis_type);
745 int_rgbdata out = a.final_color ({y, y, y});
746 data[i].red = out.red;
747 data[i].green = out.green;
748 data[i].blue = out.blue;
749 }
750 return data;
751}

762std::vector<uint64_t>
763hd_x_histogram (render_parameters &rparam, image_data &img, int steps, luminosity_t minx, luminosity_t maxx, hd_axis_type axis_type, progress_info *progress)
764{
765 /* TODO: Handle stitched projects. */
766 if (img.stitch)
767 return {};
768 render r (img, rparam, 256);
769 if (!r.precompute_all (true, false, {1, 1, 1}, progress))
770 return {};
771 auto hist = r.get_image_layer_histogram (progress);
772 if (!hist)
773 return {};
774 std::vector<uint64_t> data (steps);
775 for (size_t i = 0 ; i < hist->n_entries (); i++)
776 {
777 luminosity_t xv = hist->index_to_val (i);
778 xv = (xv - rparam.dark_point) * rparam.scan_exposure;
779 luminosity_t x = hd_linear_to_axis_x (xv, axis_type, rparam.contact_copy.preflash, rparam.contact_copy.exposure);
780 if (x < minx || x > maxx)
781 continue;
782 int idx = nearest_int ((x - minx) * (steps-1) / (maxx - minx));
783 if (idx == steps)
784 idx = steps-1;
785 data[idx] += hist->entry (i);
786 }
787 return data;
